   152→      l.genres.includes(smartMatchGenre)
   153→    );

   155→      .filter((l) => l.rankByGenre?.[smartMatchGenre] && l.rankByGenre[smartMatchGenre] <= 20)
   156→      .sort((a, b) => (a.rankByGenre?.[smartMatchGenre] || 999) - (b.rankByGenre?.[smartMatchGenre] || 999));
   157→    const mid = genreLabels
   158→      .filter((l) => l.rankByGenre?.[smartMatchGenre] && l.rankByGenre[smartMatchGenre] > 20 && l.rankByGenre[smartMatchGenre] <= 50)
   159→      .sort((a, b) => (a.rankByGenre?.[smartMatchGenre] || 999) - (b.rankByGenre?.[smartMatchGenre] || 999));
   160→    const emerging = genreLabels
   161→      .filter((l) => l.trending && !top.includes(l) && !mid.includes(l))
   162→      .sort((a, b) => (b.trendingPointsByGenre?.[smartMatchGenre] || 0) - (a.trendingPointsByGenre?.[smartMatchGenre] || 0))

   475→            </Button>
   476→          </PopoverTrigger>
   477→          <PopoverContent className="w-72 p-0 bg-card border-border/50" align="start">

   504→            </div>
   505→          </PopoverContent>
   506→        </Popover>
   507→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   508→          <SelectTrigger className="w-full sm:w-[150px] bg-secondary/50 border-border/50">

   510→            <SelectValue placeholder={t(locale, "labels.allStatus")} />
