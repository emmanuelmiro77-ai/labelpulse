    88→// ==================== ROBUST STORAGE ====================
    89→// Custom storage with data protection:
    90→// 1. Primary localStorage key
    91→// 2. Backup localStorage key (debounced — not on every save)

    99→// Debounced backup writing
   100→let _backupTimer: ReturnType<typeof setTimeout> | null = null;
   101→const BACKUP_DEBOUNCE_MS = 60000; // 60 seconds

   103→function safeJsonParse(str: string | null): any | null {
   104→  if (!str) return null;
   105→  try {
   106→    return JSON.parse(str);
   107→  } catch {
   108→    return null;
   109→  }
   110→}
   111→
   112→function safeLocalStorageGet(key: string): string | null {
   113→  try {
   114→    return localStorage.getItem(key);
   115→  } catch {
   116→    return null;
   117→  }
   118→}
   119→
   120→function safeLocalStorageSet(key: string, value: string): boolean {
   121→  try {
   122→    localStorage.setItem(key, value);
   123→    return true;
   124→  } catch (e) {
   125→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   126→    return false;
   127→  }
   128→}
   129→
   130→function safeLocalStorageRemove(key: string): void {
   131→  try {
   132→    localStorage.removeItem(key);
   133→  } catch {
   134→    // Ignore
   135→  }
   136→}

   138→/**
   139→ * Check if persisted data has user edits (emails, notes, links, etc.)
   140→ * This helps us detect if the store was accidentally reset to seed data.
   141→ */
   142→function countUserEditedLabels(labels: any[]): number {
   143→  if (!Array.isArray(labels)) return 0;
   144→  return labels.filter((l) =>
   145→    (l.emails && l.emails.length > 0) ||
   146→    (l.notes && l.notes.trim() !== "") ||
   147→    (l.website && l.website.trim() !== "") ||
   148→    (l.demoLink && l.demoLink.trim() !== "") ||
   149→    (l.socialLink && l.socialLink.trim() !== "") ||
   150→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   151→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   152→    l.isCustom === true
   153→  ).length;
   154→}
   155→
   156→function getTotalLabelCount(data: any): number {
   157→  if (!data) return 0;
   158→  if (Array.isArray(data.labels)) return data.labels.length;
   159→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   160→  return 0;
   161→}
   162→
   163→/**
   164→ * Extract user-edited label count from raw localStorage string.
   165→ * Used by the write guard to check if a write would cause data loss.
   166→ */
   167→function getUserEditCountFromRaw(raw: string | null): number {
   168→  if (!raw) return 0;
   169→  const data = safeJsonParse(raw);
   170→  if (!data) return 0;
   171→  const labels = data.state?.labels || data.labels || [];
   172→  return countUserEditedLabels(labels);
   173→}

   175→function debouncedBackupWrite(value: string): void {
   176→  if (_backupTimer) {
   177→    clearTimeout(_backupTimer);
   178→  }
   179→  _backupTimer = setTimeout(() => {
   180→    safeLocalStorageSet(BACKUP_KEY, value);
   181→    _backupTimer = null;
   182→  }, BACKUP_DEBOUNCE_MS);
   183→}
   184→
   185→const robustStorage: StateStorage = {
   186→  getItem: (name: string): string | null => {
   187→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   188→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   189→
   190→    const primaryData = safeJsonParse(primaryRaw);
   191→    const backupData = safeJsonParse(backupRaw);
   192→
   193→    // If primary is missing/corrupt, try backup
   194→    if (!primaryData) {
   195→      if (backupData) {
   196→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   197→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   198→        return backupRaw;
   199→      }
   200→      // Both empty — first visit
   201→      return null;
   202→    }
   203→
   204→    // Both exist — check for data loss
   205→    if (backupData) {
   206→      const primaryLabelCount = getTotalLabelCount(primaryData);
   207→      const backupLabelCount = getTotalLabelCount(backupData);
   208→      const primaryUserEdits = primaryData.state?.labels
   209→        ? countUserEditedLabels(primaryData.state.labels)
   210→        : countUserEditedLabels(primaryData.labels || []);
   211→      const backupUserEdits = backupData.state?.labels
   212→        ? countUserEditedLabels(backupData.state.labels)
   213→        : countUserEditedLabels(backupData.labels || []);
   214→
   215→      // If primary has fewer labels than backup AND backup has more or equal user data
   216→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   217→        console.warn(
   218→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   219→        );
   220→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   221→        return backupRaw;
   222→      }
   223→
   224→      // If primary has fewer user edits than backup, use backup
   225→      if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   226→        console.warn(
   227→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   228→        );
   229→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   230→        return backupRaw;
   231→      }
   232→    }
   233→
   234→    return primaryRaw;
   235→  },
   236→
   237→  setItem: (name: string, value: string): void => {
   238→    // SMART WRITE GUARD: Before writing, check if this write would cause data loss.
   239→    // Compare the new value's user-edit count with what's currently in localStorage.
   240→    // If the new value has FEWER user edits than what's stored, it's likely a
   241→    // seed-data overwrite — block it to protect user data.
   242→    const currentRaw = safeLocalStorageGet(PRIMARY_KEY);
   243→    const currentEdits = getUserEditCountFromRaw(currentRaw);
   244→    const newEdits = getUserEditCountFromRaw(value);
   245→
   246→    if (currentEdits > 0 && newEdits < currentEdits) {
   247→      // We're about to write data with FEWER user edits than what's stored.
   248→      // This is likely the seed data being written before rehydration completes.
   249→      console.warn(
   250→        `[LabelPulse Storage] BLOCKED write: would reduce user edits from ${currentEdits} to ${newEdits}. Protecting user data.`
   251→      );
   252→      return;
   253→    }
   254→
   255→    // Write to primary immediately
   256→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   257→
   258→    // Write to backup with debounce
   259→    debouncedBackupWrite(value);
   260→
   261→    if (!primaryOk) {
   262→      console.error("[LabelPulse Storage] Primary write failed!");
   263→    }
   264→  },
   265→
   266→  removeItem: (name: string): void => {
   267→    safeLocalStorageRemove(PRIMARY_KEY);
   268→    safeLocalStorageRemove(BACKUP_KEY);
   269→  },
   270→};

   272→/**
   273→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   274→ */
   275→export function forceBackupNow(): void {
   276→  if (typeof window === "undefined") return;
   277→  if (_backupTimer) {
   278→    clearTimeout(_backupTimer);
   279→    _backupTimer = null;
   280→  }
   281→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   282→  if (primaryRaw) {
   283→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   284→  }
   285→}

   364→    // Beatport/ranking data — prefer imported (it's the fresh data)
   365→    genres: imported.genres?.length ? imported.genres : existing.genres,
   366→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,

   370→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,

   397→/**
   398→ * Repair corrupted label data.
   399→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   400→ */
   401→function repairLabelData(labels: Label[]): Label[] {
   402→  const emailLabelCount = new Map<string, number>();
   403→  for (const l of labels) {
   404→    if (l.emails && Array.isArray(l.emails)) {
   405→      for (const e of l.emails) {
   406→        if (e && e.trim()) {
   407→          const key = e.toLowerCase().trim();
   408→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   409→        }
   410→      }
   411→    }
   412→  }
   413→
   414→  const corruptedEmails = new Set<string>();
   415→  for (const [email, count] of emailLabelCount) {
   416→    if (count > 5) {
   417→      corruptedEmails.add(email);
   418→      console.warn(
   419→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   420→      );
   421→    }
   422→  }
   423→
   424→  if (corruptedEmails.size === 0) {
   425→    return labels;
   426→  }
   427→
   428→  const emailOwner = new Map<string, string>();
   429→  for (const email of corruptedEmails) {
   430→    for (const l of labels) {
   431→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   432→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   433→        if (domain && l.name.toLowerCase().includes(domain)) {
   434→          emailOwner.set(email, l.id);
   435→          break;
   436→        }
   437→      }
   438→    }
   439→    if (!emailOwner.has(email)) {
   440→      for (const l of labels) {
   441→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   442→          emailOwner.set(email, l.id);
   443→          break;
   444→        }
   445→      }
   446→    }
   447→  }
   448→
   449→  return labels.map(l => {
   450→    if (!l.emails || !Array.isArray(l.emails)) return l;
   451→    const cleanedEmails = l.emails.filter(e => {
   452→      const key = e?.toLowerCase().trim();
   453→      if (!key) return false;
   454→      if (!corruptedEmails.has(key)) return true;
   455→      return emailOwner.get(key) === l.id;
   456→    });
   457→    const cleanedContactInfo = cleanedEmails[0] || "";
   458→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   459→      console.log(
   460→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   461→      );
   462→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   463→    }
   464→    return l;
   465→  });
   466→}

   854→// Helper: get tier for a label in a given genre
   855→export function getLabelTier(
   856→  label: Label,
   857→  genre?: string
   858→): "top" | "mid" | "emerging" | null {
   859→  if (!genre) genre = label.genres?.[0];
   860→  if (!genre) return null;
   861→  const rank = label.rankByGenre?.[genre];
   862→  if (!rank) return label.trending ? "emerging" : null;
   863→  if (rank <= 20) return "top";
   864→  if (rank <= 50) return "mid";
   865→  return label.trending ? "emerging" : null;
   866→}
