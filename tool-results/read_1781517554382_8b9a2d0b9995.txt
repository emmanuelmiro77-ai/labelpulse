   109→// ==================== ROBUST STORAGE ====================
   110→// Custom storage with data protection:
   111→// 1. Primary localStorage key
   112→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)
   113→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
   114→// 4. Rehydration guard: blocks ALL writes until Zustand persist loads data
   115→
   116→const PRIMARY_KEY = "labelpulse-storage";
   117→const BACKUP_KEY = "labelpulse-storage-backup";
   118→const SEED_LABEL_COUNT = labelData.labels.length;
   119→
   120→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   121→// Before rehydration, the store contains seed data which would overwrite user data.
   122→let _rehydrated = false;
   123→
   124→function safeJsonParse(str: string | null): any | null {
   125→  if (!str) return null;
   126→  try {
   127→    return JSON.parse(str);
   128→  } catch {
   129→    return null;
   130→  }
   131→}
   132→
   133→function safeLocalStorageGet(key: string): string | null {
   134→  try {
   135→    return localStorage.getItem(key);
   136→  } catch {
   137→    return null;
   138→  }
   139→}
   140→
   141→function safeLocalStorageSet(key: string, value: string): boolean {
   142→  try {
   143→    localStorage.setItem(key, value);
   144→    return true;
   145→  } catch (e) {
   146→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   147→    return false;
   148→  }
   149→}
   150→
   151→function safeLocalStorageRemove(key: string): void {
   152→  try {
   153→    localStorage.removeItem(key);
   154→  } catch {
   155→    // Ignore
   156→  }
   157→}

   159→/**
   160→ * Check if persisted data has user edits (emails, notes, links, etc.)
   161→ * This helps us detect if the store was accidentally reset to seed data.
   162→ */
   163→function countUserEditedLabels(labels: any[]): number {
   164→  if (!Array.isArray(labels)) return 0;
   165→  return labels.filter((l) =>
   166→    (l.emails && l.emails.length > 0) ||
   167→    (l.notes && l.notes.trim() !== "") ||
   168→    (l.website && l.website.trim() !== "") ||
   169→    (l.demoLink && l.demoLink.trim() !== "") ||
   170→    (l.socialLink && l.socialLink.trim() !== "") ||
   171→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   172→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   173→    l.isCustom === true
   174→  ).length;
   175→}
   176→
   177→function getTotalLabelCount(data: any): number {
   178→  if (!data) return 0;
   179→  if (Array.isArray(data.labels)) return data.labels.length;
   180→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   181→  return 0;
   182→}
   183→
   184→/**
   185→ * Extract user-edited label count from raw localStorage string.
   186→ * Used by the write guard to check if a write would cause data loss.
   187→ */
   188→function getUserEditCountFromRaw(raw: string | null): number {
   189→  if (!raw) return 0;
   190→  const data = safeJsonParse(raw);
   191→  if (!data) return 0;
   192→  const labels = data.state?.labels || data.labels || [];
   193→  return countUserEditedLabels(labels);
   194→}

   196→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   197→
   198→const robustStorage: StateStorage = {
   199→  getItem: (name: string): string | null => {
   200→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   201→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   202→
   203→    const primaryData = safeJsonParse(primaryRaw);
   204→    const backupData = safeJsonParse(backupRaw);
   205→
   206→    let result: string | null = null;
   207→
   208→    // If primary is missing/corrupt, try backup
   209→    if (!primaryData) {
   210→      if (backupData) {
   211→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   212→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   213→        result = backupRaw;
   214→      }
   215→      // Both empty — first visit, result stays null
   216→    } else if (backupData) {
   217→      // Both exist — check for data loss
   218→      const primaryLabelCount = getTotalLabelCount(primaryData);
   219→      const backupLabelCount = getTotalLabelCount(backupData);
   220→      const primaryUserEdits = primaryData.state?.labels
   221→        ? countUserEditedLabels(primaryData.state.labels)
   222→        : countUserEditedLabels(primaryData.labels || []);
   223→      const backupUserEdits = backupData.state?.labels
   224→        ? countUserEditedLabels(backupData.state.labels)
   225→        : countUserEditedLabels(backupData.labels || []);
   226→
   227→      // If primary has fewer labels than backup AND backup has more or equal user data
   228→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   229→        console.warn(
   230→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   231→        );
   232→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   233→        result = backupRaw;
   234→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   235→        // If primary has fewer user edits than backup, use backup
   236→        console.warn(
   237→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   238→        );
   239→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   240→        result = backupRaw;
   241→      } else {
   242→        result = primaryRaw;
   243→      }
   244→    } else {
   245→      // Primary exists, no backup — normal case
   246→      result = primaryRaw;
   247→    }
   248→
   249→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   250→    // merged data immediately after this getItem returns, and we MUST allow that write.
   251→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   252→    _rehydrated = true;
   253→
   254→    return result;
   255→  },
   256→
   257→  setItem: (name: string, value: string): void => {
   258→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   259→    // Before rehydration, the store only has seed data — writing it would overwrite
   260→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   261→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   262→    if (!_rehydrated) {
   263→      console.warn(
   264→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   265→      );
   266→      return;
   267→    }
   268→
   269→    // Write to primary immediately
   270→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   271→
   272→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   273→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   274→    safeLocalStorageSet(BACKUP_KEY, value);
   275→
   276→    if (!primaryOk) {
   277→      console.error("[LabelPulse Storage] Primary write failed!");
   278→    }
   279→  },
   280→
   281→  removeItem: (name: string): void => {
   282→    safeLocalStorageRemove(PRIMARY_KEY);
   283→    safeLocalStorageRemove(BACKUP_KEY);
   284→  },
   285→};
   286→
   287→/**
   288→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   289→ * Since backup is now written on every setItem, this is a safety net.
   290→ */
   291→export function forceBackupNow(): void {
   292→  if (typeof window === "undefined") return;
   293→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   294→  if (primaryRaw) {
   295→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   296→  }
   297→}

   366→/**
   367→ * Merge helper: preserves ALL user-editable fields from existing label
   368→ * when imported label has empty/default values for those fields.
   369→ */
   370→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
   371→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
   372→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
   373→  const hasNewRankData = Object.keys(imported.rankByGenre || {}).length > 0;
   374→  const currentRank = hasNewRankData && Object.keys(existing.rankByGenre || {}).length > 0
   375→    ? { ...existing.rankByGenre }
   376→    : existing.prevRankByGenre || {};
   377→
   378→  return {
   379→    // User-editable data — ALWAYS prefer existing if it has real data
   380→    emails: imported.emails?.length ? imported.emails : existing.emails,
   381→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
   382→    website: imported.website?.trim() || existing.website,
   383→    demoLink: imported.demoLink?.trim() || existing.demoLink,
   384→    socialLink: imported.socialLink?.trim() || existing.socialLink,
   385→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   386→    notes: imported.notes?.trim() || existing.notes,
   387→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
   388→
   389→    // Beatport/ranking data — prefer imported (it's the fresh data)
   390→    genres: imported.genres?.length ? imported.genres : existing.genres,
   391→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
   392→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
   393→    trending: imported.trending || existing.trending,
   394→    trendingRankByGenre: Object.keys(imported.trendingRankByGenre || {}).length ? imported.trendingRankByGenre : existing.trendingRankByGenre,
   395→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
   396→
   397→    // Ranking history — save current rank as "previous" when new data arrives
   398→    prevRankByGenre: currentRank,
   399→  };
   400→}

   428→/**
   429→ * Repair corrupted label data.
   430→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   431→ */
   432→function repairLabelData(labels: Label[]): Label[] {
   433→  const emailLabelCount = new Map<string, number>();
   434→  for (const l of labels) {
   435→    if (l.emails && Array.isArray(l.emails)) {
   436→      for (const e of l.emails) {
   437→        if (e && e.trim()) {
   438→          const key = e.toLowerCase().trim();
   439→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   440→        }
   441→      }
   442→    }
   443→  }
   444→
   445→  const corruptedEmails = new Set<string>();
   446→  for (const [email, count] of emailLabelCount) {
   447→    if (count > 5) {
   448→      corruptedEmails.add(email);
   449→      console.warn(
   450→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   451→      );
   452→    }
   453→  }
   454→
   455→  if (corruptedEmails.size === 0) {
   456→    return labels;
   457→  }
   458→
   459→  const emailOwner = new Map<string, string>();
   460→  for (const email of corruptedEmails) {
   461→    for (const l of labels) {
   462→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   463→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   464→        if (domain && l.name.toLowerCase().includes(domain)) {
   465→          emailOwner.set(email, l.id);
   466→          break;
   467→        }
   468→      }
   469→    }
   470→    if (!emailOwner.has(email)) {
   471→      for (const l of labels) {
   472→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   473→          emailOwner.set(email, l.id);
   474→          break;
   475→        }
   476→      }
   477→    }
   478→  }
   479→
   480→  return labels.map(l => {
   481→    if (!l.emails || !Array.isArray(l.emails)) return l;
   482→    const cleanedEmails = l.emails.filter(e => {
   483→      const key = e?.toLowerCase().trim();
   484→      if (!key) return false;
   485→      if (!corruptedEmails.has(key)) return true;
   486→      return emailOwner.get(key) === l.id;
   487→    });
   488→    const cleanedContactInfo = cleanedEmails[0] || "";
   489→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   490→      console.log(
   491→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   492→      );
   493→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   494→    }
   495→    return l;
   496→  });
   497→}

   990→// Helper: get tier for a label in a given genre
   991→export function getLabelTier(
   992→  label: Label,
   993→  genre?: string
   994→): "top" | "mid" | "emerging" | null {
   995→  if (!genre) genre = label.genres?.[0];
   996→  if (!genre) return null;
   997→  const rank = label.rankByGenre?.[genre];
   998→  if (!rank) return label.trending ? "emerging" : null;
   999→  if (rank <= 20) return "top";
  1000→  if (rank <= 50) return "mid";
  1001→  return label.trending ? "emerging" : null;
  1002→}

  1004→// ==================== CLOUD SYNC ====================

  1013→/**
  1014→ * Sincronizza lo stato corrente con Supabase (debounced).
  1015→ * Chiamata dopo ogni azione utente che modifica i dati.
  1016→ */
  1017→export function syncToCloud(): void {
  1018→  if (!isSupabaseConfigured) return;
  1019→
  1020→  if (_cloudSyncTimer) {
  1021→    clearTimeout(_cloudSyncTimer);

  1024→  _cloudSyncTimer = setTimeout(async () => {
  1025→    _cloudSyncTimer = null;
  1026→    const state = useAppStore.getState();

  1028→      labels: state.labels,
  1029→      demos: state.demos,
  1030→      activeTab: state.activeTab,

  1034→      rankingsUpdatedAt: state.rankingsUpdatedAt,
  1035→      lastSavedAt: state.lastSavedAt,
  1036→      rankingSnapshots: state.rankingSnapshots,
  1037→    };
  1038→    await saveStateToCloud(dataToSync);
  1039→  }, CLOUD_SYNC_DEBOUNCE_MS);
  1040→}

  1070→ * Carica i dati dal cloud e aggiorna lo store se i dati cloud sono più recenti.
  1071→ * Chiamata una volta all'avvio dell'app, dopo la reidratazione da localStorage.
  1072→ */
  1073→export async function loadFromCloud(): Promise<void> {
  1074→  if (!isSupabaseConfigured) {
  1075→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
  1076→    useAppStore.setState({ hasCloudSynced: true });
  1077→    return;
  1078→  }
  1079→
  1080→  try {
  1081→    const cloudData = await loadStateFromCloud();

  1083→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
  1084→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
  1085→      await forceCloudSync();
  1086→      useAppStore.setState({ hasCloudSynced: true });
  1087→      return;
  1088→    }
  1089→
  1090→    const cloud = cloudData as any;
  1091→    const localState = useAppStore.getState();
  1092→
  1093→    // Confronta i timestamp per decidere quali dati usare
  1094→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
  1095→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
  1096→
  1097→    if (cloudSavedAt > localSavedAt) {
  1098→      // I dati cloud sono più recenti — aggiorna lo store locale
  1099→      console.log(
  1100→        `[LabelPulse Cloud] Cloud data is newer (cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}). Updating local store.`
  1101→      );
  1102→
  1103→      // Usa la stessa logica di merge della reidratazione
  1104→      const merged = mergeCloudData(cloud, localState);
  1105→      useAppStore.setState({
  1106→        ...merged,
  1107→        hasCloudSynced: true,
  1108→      });
  1109→    } else {
  1110→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1111→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1112→      useAppStore.setState({ hasCloudSynced: true });
  1113→    }
  1114→  } catch (err) {
  1115→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1116→    useAppStore.setState({ hasCloudSynced: true });
  1117→  }
  1118→}
