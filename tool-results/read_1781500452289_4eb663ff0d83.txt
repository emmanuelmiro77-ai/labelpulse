    95→/**
    96→ * Get the time cutoff for a period (in milliseconds from now).
    97→ */
    98→function getPeriodCutoff(period: RankingTimePeriod): number {
    99→  const now = Date.now();
   100→  switch (period) {
   101→    case "current": return now; // Only current data
   102→    case "1m": return now - 30 * 24 * 60 * 60 * 1000;
   103→    case "3m": return now - 90 * 24 * 60 * 60 * 1000;
   104→    case "1y": return now - 365 * 24 * 60 * 60 * 1000;
   105→    case "all": return 0; // All time
   106→  }
   107→}

   109→// ==================== AGGREGATION LOGIC ====================
   110→
   111→/**
   112→ * Aggregate ranking snapshots for a given genre and time period.
   113→ * Returns a map of labelName -> { totalPoints, bestRank, snapshotCount, averageRank }
   114→ */
   115→function aggregateSnapshots(
   116→  snapshots: RankingSnapshot[],
   117→  genre: string,
   118→  period: RankingTimePeriod
   119→): Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number }> {
   120→  const cutoff = getPeriodCutoff(period);
   121→  const result = new Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }>();
   122→
   123→  for (const snapshot of snapshots) {
   124→    if (new Date(snapshot.timestamp).getTime() < cutoff) continue;
   125→    if (period === "current") continue; // Current period doesn't use snapshots
   126→
   127→    const genreData = snapshot.genres[genre];
   128→    if (!genreData) continue;
   129→
   130→    for (const [labelName, data] of Object.entries(genreData)) {
   131→      const existing = result.get(labelName);
   132→      if (existing) {
   133→        existing.totalPoints += data.points;
   134→        existing.rankSum += data.rank;
   135→        existing.snapshotCount++;
   136→        if (data.rank < existing.bestRank) existing.bestRank = data.rank;
   137→      } else {
   138→        result.set(labelName, {
   139→          totalPoints: data.points,
   140→          bestRank: data.rank,
   141→          snapshotCount: 1,
   142→          rankSum: data.rank,
   143→        });
   144→      }
   145→    }
   146→  }
   147→
   148→  // Calculate average rank
   149→  for (const [, val] of result) {
   150→    (val as any).averageRank = Math.round(val.rankSum / val.snapshotCount);
   151→  }
   152→
   153→  return result;
   154→}

   156→/**
   157→ * Build ranked list for a genre, considering the time period.
   158→ * - "current": uses live label data (rankByGenre)
   159→ * - Other periods: aggregates snapshots and re-ranks by cumulative points
   160→ */
   161→function buildRankedList(
   162→  labels: Label[],
   163→  snapshots: RankingSnapshot[],
   164→  genre: string,
   165→  period: RankingTimePeriod
   166→): RankedLabel[] {
   167→  if (period === "current") {
   168→    // Current view: use live label data
   169→    const ranked: RankedLabel[] = [];
   170→    for (const label of labels) {
   171→      const rank = label.rankByGenre?.[genre];
   172→      if (rank === undefined || rank === null) continue;
   173→
   174→      const prevRank = label.prevRankByGenre?.[genre] ?? null;
   175→      const points = label.pointsByGenre?.[genre] ?? 0;
   176→      let movement: number | null = null;
   177→
   178→      if (prevRank !== null && prevRank !== undefined) {
   179→        movement = prevRank - rank;
   180→      } else if (Object.keys(label.prevRankByGenre || {}).length > 0) {
   181→        movement = null;
   182→      }
   183→
   184→      ranked.push({ label, rank, prevRank, points, movement, snapshotCount: 1 });
   185→    }
   186→    return ranked;
   187→  }
   188→
   189→  // Historical/cumulative view: aggregate snapshots
   190→  const aggregated = aggregateSnapshots(snapshots, genre, period);
   191→  if (aggregated.size === 0) return [];
   192→
   193→  // Sort by total points descending to determine cumulative rank
   194→  const sorted = Array.from(aggregated.entries()).sort((a, b) => b[1].totalPoints - a[1].totalPoints);
   195→
   196→  // For movement in cumulative view, compare with the current live data
   197→  const ranked: RankedLabel[] = [];
   198→  for (let i = 0; i < sorted.length; i++) {
   199→    const [labelName, data] = sorted[i];
   200→    const cumulativeRank = i + 1;
   201→
   202→    // Find matching label in current data
   203→    const matchingLabel = labels.find(l => l.name === labelName);
   204→    const currentRank = matchingLabel?.rankByGenre?.[genre] ?? null;
   205→
   206→    // Movement: compare current live rank with cumulative rank
   207→    let movement: number | null = null;
   208→    if (currentRank !== null) {
   209→      movement = currentRank - cumulativeRank; // positive = label is better live than cumulative (hot right now)
   210→    }
   211→
   212→    // Use the label if found, otherwise create a minimal reference
   213→    const label = matchingLabel || {
   214→      id: `snapshot_${labelName.toLowerCase().replace(/[^a-z0-9]+/g, '_')}`,
   215→      name: labelName,
   216→      genre: genre,
   217→      submissionType: "email" as const,
   218→      contactInfo: "",
   219→      status: "open" as const,
   220→      notes: "",
   221→      createdAt: new Date().toISOString(),
   222→      emails: [],
   223→      website: "",
   224→      demoLink: "",
   225→      socialLink: "",
   226→      soundcloudLink: "",
   227→      genres: [genre],
   228→      rankByGenre: { [genre]: currentRank || 999 },
   229→      pointsByGenre: {},
   230→      trending: false,
   231→      trendingRankByGenre: {},
   232→      trendingPointsByGenre: {},
   233→      prevRankByGenre: {},
   234→    };
   235→
   236→    ranked.push({
   237→      label,
   238→      rank: cumulativeRank,
   239→      prevRank: currentRank,
   240→      points: data.totalPoints,
   241→      movement,
   242→      snapshotCount: data.snapshotCount,
   243→    });
   244→  }
   245→
   246→  return ranked;
   247→}

   251→export function RankingsPage() {
   252→  const { labels, locale, rankingsUpdatedAt, rankingSnapshots } = useAppStore();
   253→  const snapshots = rankingSnapshots || []; // defensive: might be undefined for existing users
   254→  const [selectedGenre, setSelectedGenre] = useState<string | null>(null);
   255→  const [sortMode, setSortMode] = useState<SortMode>("rank");
   256→  const [movementFilter, setMovementFilter] = useState<MovementFilter>("all");
   257→  const [timePeriod, setTimePeriod] = useState<RankingTimePeriod>("current");
   258→  const [showGenreDropdown, setShowGenreDropdown] = useState(false);
   259→
   260→  // Get all genres from labels, sorted alphabetically
   261→  const allGenres = useMemo(() => {
   262→    const genreSet = new Set<string>();
   263→    for (const label of labels) {
   264→      if (label.genres) {
   265→        for (const g of label.genres) {
   266→          genreSet.add(g);
   267→        }
   268→      }
   269→    }
   270→    return Array.from(genreSet).sort((a, b) => a.localeCompare(b));
   271→  }, [labels]);
   272→
   273→  // Build ranked list for selected genre and period
   274→  const rankedList = useMemo((): RankedLabel[] => {
   275→    if (!selectedGenre) return [];
   276→    return buildRankedList(labels, snapshots, selectedGenre, timePeriod);
   277→  }, [labels, snapshots, selectedGenre, timePeriod]);
   278→
   279→  // Apply filters
   280→  const filteredList = useMemo(() => {
   281→    let result = [...rankedList];
   282→
   283→    // Movement filter
   284→    switch (movementFilter) {
   285→      case "rising":
   286→        result = result.filter((l) => l.movement !== null && l.movement > 0);
   287→        break;
   288→      case "falling":
   289→        result = result.filter((l) => l.movement !== null && l.movement < 0);
   290→        break;
   291→      case "new":
   292→        result = result.filter((l) => l.movement === null);
   293→        break;
   294→      case "stable":
   295→        result = result.filter((l) => l.movement === 0);
   296→        break;
   297→    }
   298→
   299→    // Sort
   300→    switch (sortMode) {
   301→      case "rank":
   302→        result.sort((a, b) => a.rank - b.rank);
   303→        break;
   304→      case "movement":
   305→        result.sort((a, b) => {
   306→          if (a.movement === null && b.movement === null) return a.rank - b.rank;
   307→          if (a.movement === null) return -1;
   308→          if (b.movement === null) return 1;
   309→          return b.movement - a.movement;
   310→        });
   311→        break;
   312→      case "points":
   313→        result.sort((a, b) => b.points - a.points);
   314→        break;
   315→    }
   316→
   317→    return result;
   318→  }, [rankedList, movementFilter, sortMode]);
   319→
   320→  // Stats summary
   321→  const stats = useMemo(() => {
   322→    const total = rankedList.length;
   323→    const rising = rankedList.filter((l) => l.movement !== null && l.movement > 0).length;
   324→    const falling = rankedList.filter((l) => l.movement !== null && l.movement < 0).length;
   325→    const newEntries = rankedList.filter((l) => l.movement === null).length;
   326→
   327→    return { total, rising, falling, newEntries };
   328→  }, [rankedList]);
   329→
   330→  // Top rising labels across ALL genres (for the spotlight section)
   331→  const topRisers = useMemo(() => {
   332→    const risers: { label: Label; genre: string; movement: number; rank: number }[] = [];
   333→
   334→    for (const label of labels) {
   335→      if (!label.prevRankByGenre || Object.keys(label.prevRankByGenre).length === 0) continue;
   336→      for (const genre of label.genres) {
   337→        const currentRank = label.rankByGenre?.[genre];
   338→        const prevRank = label.prevRankByGenre?.[genre];
   339→        if (currentRank && prevRank) {
   340→          const movement = prevRank - currentRank;
   341→          if (movement > 0) {
   342→            risers.push({ label, genre, movement, rank: currentRank });
   343→          }
   344→        }
   345→      }
   346→    }
   347→
   348→    return risers.sort((a, b) => b.movement - a.movement).slice(0, 10);
   349→  }, [labels]);
   350→
   351→  const hasPreviousData = useMemo(() => {
   352→    return labels.some((l) => l.prevRankByGenre && Object.keys(l.prevRankByGenre).length > 0);
   353→  }, [labels]);
   354→
   355→  // Snapshot stats for info display
   356→  const snapshotStats = useMemo(() => {
   357→    if (!snapshots || snapshots.length === 0) return { count: 0, oldest: null as string | null, newest: null as string | null };
   358→    const sorted = [...snapshots].sort((a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime());
   359→    return {
   360→      count: snapshots.length,
   361→      oldest: sorted[0].timestamp,
   362→      newest: sorted[sorted.length - 1].timestamp,
   363→    };
   364→  }, [snapshots]);
   365→
   366→  const hasSnapshots = snapshots.length > 0;
   367→
   368→  return (
   369→    <div className="space-y-6">
   370→      {/* Alert if no previous data */}
   371→      {!hasPreviousData && timePeriod === "current" && (
   372→        <div className="flex items-start gap-3 p-4 rounded-lg bg-amber-500/10 border border-amber-500/30">
   373→          <AlertTriangle className="h-5 w-5 text-amber-400 shrink-0 mt-0.5" />
   374→          <div>
   375→            <p className="text-sm text-amber-400 font-medium">{t(locale, "rankings.noHistory")}</p>
   376→            <p className="text-xs text-muted-foreground mt-1">{t(locale, "rankings.noHistoryDesc")}</p>
   377→          </div>
   378→        </div>
   379→      )}
   380→
   381→      {/* Snapshot info banner */}
   382→      {timePeriod !== "current" && !hasSnapshots && (
   383→        <div className="flex items-start gap-3 p-4 rounded-lg bg-cyan-500/10 border border-cyan-500/30">
   384→          <History className="h-5 w-5 text-cyan-400 shrink-0 mt-0.5" />
   385→          <div>
   386→            <p className="text-sm text-cyan-400 font-medium">{t(locale, "rankings.noSnapshots")}</p>
   387→            <p className="text-xs text-muted-foreground mt-1">{t(locale, "rankings.noSnapshotsDesc")}</p>
   388→          </div>
   389→        </div>
   390→      )}
   391→
   392→      {/* Snapshot stats (when viewing historical data) */}
   393→      {timePeriod !== "current" && hasSnapshots && (
   394→        <div className="flex items-center gap-3 p-3 rounded-lg bg-muted/30 border border-border/50">
   395→          <History className="h-4 w-4 text-muted-foreground shrink-0" />
   396→          <div className="flex flex-wrap gap-4 text-xs text-muted-foreground">
   397→            <span>{snapshotStats.count} {t(locale, "rankings.snapshots")}</span>
   398→            {snapshotStats.oldest && (
   399→              <span>{t(locale, "rankings.from")}: {new Date(snapshotStats.oldest).toLocaleDateString(locale === "it" ? "it-IT" : "en-US")}</span>
   400→            )}
   401→            {snapshotStats.newest && (
   402→              <span>{t(locale, "rankings.to")}: {new Date(snapshotStats.newest).toLocaleDateString(locale === "it" ? "it-IT" : "en-US")}</span>
   403→            )}
   404→          </div>
   405→        </div>
   406→      )}
   407→
   408→      {/* Spotlight: Top Risers (only if we have previous data and viewing current) */}
   409→      {timePeriod === "current" && hasPreviousData && topRisers.length > 0 && (
   410→        <div className="space-y-3">
   411→          <div className="flex items-center gap-2">
   412→            <Flame className="h-5 w-5 text-orange-400" />
   413→            <h3 className="text-sm font-semibold text-foreground">{t(locale, "rankings.spotlight")}</h3>
   414→          </div>
   415→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-2">
   416→            {topRisers.map((item, idx) => (
   417→              <div
   418→                key={`${item.label.id}-${item.genre}`}
   419→                className="flex items-center gap-2 p-2.5 rounded-lg bg-gradient-to-r from-emerald-500/10 to-transparent border border-emerald-500/20"
   420→              >
   421→                <span className="text-lg font-bold text-emerald-400 shrink-0">+{item.movement}</span>
   422→                <div className="min-w-0">
   423→                  <p className="text-xs font-medium text-foreground truncate">{item.label.name}</p>
   424→                  <p className="text-[10px] text-muted-foreground truncate">{item.genre}</p>
   425→                </div>
   426→              </div>
   427→            ))}
   428→          </div>
   429→        </div>
   430→      )}
   431→
   432→      {/* Genre selector + Filters */}
   433→      <div className="space-y-4">
   434→        {/* Genre dropdown */}
   435→        <div className="flex flex-col sm:flex-row gap-3">
   436→          <div className="relative flex-1">
   437→            <Button
   438→              variant="outline"
   439→              className="w-full justify-between text-sm"
   440→              onClick={() => setShowGenreDropdown(!showGenreDropdown)}
   441→            >
   442→              <span className="flex items-center gap-2">
   443→                <BarChart3 className="h-4 w-4 text-primary" />
   444→                {selectedGenre || t(locale, "rankings.selectGenre")}
   445→              </span>
   446→              <ChevronDown className={`h-4 w-4 transition-transform ${showGenreDropdown ? "rotate-180" : ""}`} />
   447→            </Button>
   448→            {showGenreDropdown && (
   449→              <div className="absolute z-50 mt-1 w-full max-h-64 overflow-y-auto rounded-lg border border-border bg-popover shadow-lg">
   450→                {allGenres.map((genre) => {
   451→                  const count = labels.filter((l) => l.rankByGenre?.[genre]).length;
   452→                  return (
   453→                    <button
   454→                      key={genre}
   455→                      onClick={() => {
   456→                        setSelectedGenre(genre);
   457→                        setShowGenreDropdown(false);
   458→                      }}
   459→                      className={`w-full flex items-center justify-between px-3 py-2 text-sm hover:bg-accent/50 transition-colors ${
   460→                        selectedGenre === genre ? "bg-primary/10 text-primary font-medium" : "text-foreground"
   461→                      }`}
   462→                    >
   463→                      <span className="truncate">{genre}</span>
   464→                      <span className="text-xs text-muted-foreground shrink-0 ml-2">{count}</span>

   466→                  );
   467→                })}
   468→              </div>
   469→            )}
   470→          </div>
   471→
   472→          {/* Sort & Filter controls */}
   473→          {selectedGenre && (
   474→            <div className="flex gap-2 flex-wrap">
   475→              {/* Sort */}
   476→              <div className="flex rounded-lg border border-border overflow-hidden">
   477→                {([["rank", Trophy], ["movement", TrendingUp], ["points", BarChart3]] as const).map(([mode, Icon]) => (

   479→                    key={mode}
   480→                    onClick={() => setSortMode(mode)}
   481→                    className={`flex items-center gap-1.5 px-3 py-1.5 text-xs font-medium transition-colors ${
   482→                      sortMode === mode
   483→                        ? "bg-primary/15 text-primary"
   484→                        : "text-muted-foreground hover:text-foreground hover:bg-secondary/50"
   485→                    }`}
   486→                  >
   487→                    <Icon className="h-3.5 w-3.5" />
   488→                    {t(locale, `rankings.sort.${mode}`)}

   493→              {/* Movement filter */}

   496→                  <button

   498→                    onClick={() => setMovementFilter(filter)}

   501→                        ? "bg-primary/15 text-primary"

   504→                    title={t(locale, `rankings.filter.${filter}`)}
   505→                  >
   506→                    <Icon className="h-3.5 w-3.5" />
   507→                  </button>
   508→                ))}
   509→              </div>
   510→            </div>
   511→          )}
   512→        </div>
   513→
   514→        {/* Time Period Selector */}
   515→        {selectedGenre && (
   516→          <div className="flex flex-col sm:flex-row items-start sm:items-center gap-2">
   517→            <span className="text-xs text-muted-foreground font-medium shrink-0">{t(locale, "rankings.periodLabel")}:</span>
   518→            <div className="flex rounded-lg border border-border overflow-hidden">
   519→              {(["current", "1m", "3m", "1y", "all"] as RankingTimePeriod[]).map((period) => {
   520→                const Icon = getPeriodIcon(period);
   521→                const isDisabled = period !== "current" && !hasSnapshots;
   522→                return (
   523→                  <button
   524→                    key={period}
   525→                    onClick={() => !isDisabled && setTimePeriod(period)}
   526→                    className={`flex items-center gap-1.5 px-3 py-1.5 text-xs font-medium transition-colors ${
   527→                      timePeriod === period
   528→                        ? "bg-cyan-500/15 text-cyan-400"
   529→                        : isDisabled
   530→                          ? "text-muted-foreground/40 cursor-not-allowed"
   531→                          : "text-muted-foreground hover:text-foreground hover:bg-secondary/50"
   532→                    }`}
   533→                    title={isDisabled ? t(locale, "rankings.periodDisabled") : getPeriodLabel(period, locale)}
   534→                  >
   535→                    <Icon className="h-3.5 w-3.5" />
   536→                    {getPeriodLabel(period, locale)}
   537→                  </button>
   538→                );
   539→              })}
   540→            </div>
   541→          </div>
   542→        )}
   543→
   544→        {/* Stats bar */}
   545→        {selectedGenre && stats.total > 0 && (
   546→          <div className="flex flex-wrap gap-3 text-xs">
   547→            <span className="flex items-center gap-1 text-muted-foreground">
   548→              <Eye className="h-3 w-3" /> {stats.total} {t(locale, "rankings.labels")}
   549→            </span>
   550→            {timePeriod === "current" && (
   551→              <>
   552→                <span className="flex items-center gap-1 text-emerald-400">
   553→                  <TrendingUp className="h-3 w-3" /> {stats.rising} {t(locale, "rankings.rising")}
   554→                </span>
   555→                <span className="flex items-center gap-1 text-red-400">
   556→                  <TrendingDown className="h-3 w-3" /> {stats.falling} {t(locale, "rankings.falling")}
   557→                </span>
   558→                <span className="flex items-center gap-1 text-cyan-400">
   559→                  <ArrowUpRight className="h-3 w-3" /> {stats.newEntries} {t(locale, "rankings.newEntries")}
   560→                </span>
   561→              </>
   562→            )}
   563→            {timePeriod !== "current" && (
   564→              <span className="flex items-center gap-1 text-cyan-400">
   565→                <History className="h-3 w-3" /> {t(locale, "rankings.cumulativeView")}
   566→              </span>
   567→            )}
   568→            {rankingsUpdatedAt && (
   569→              <span className="flex items-center gap-1 text-muted-foreground ml-auto">
   570→                {t(locale, "rankings.lastUpdate")}: {new Date(rankingsUpdatedAt).toLocaleDateString(locale === "it" ? "it-IT" : "en-US")}
   571→              </span>
   572→            )}
   573→          </div>
   574→        )}
   575→
   576→        {/* Rankings table */}
   577→        {selectedGenre && filteredList.length > 0 && (
   578→          <div className="rounded-lg border border-border/50 overflow-hidden">
   579→            {/* Table header */}
   580→            <div className={`grid items-center gap-1 px-3 py-2 bg-secondary/30 text-xs font-medium text-muted-foreground border-b border-border/50 ${
   581→              timePeriod !== "current"
   582→                ? "grid-cols-[3rem_1fr_4.5rem_4rem_4.5rem_3.5rem] sm:grid-cols-[3rem_1fr_5rem_5rem_5.5rem_4rem]"
   583→                : "grid-cols-[3rem_1fr_4.5rem_4rem_4rem] sm:grid-cols-[3rem_1fr_5rem_5rem_5rem]"
   584→            }`}>
   585→              <span>#</span>
   586→              <span>{t(locale, "rankings.colLabel")}</span>
   587→              <span className="text-right">{t(locale, "rankings.colRank")}</span>
   588→              <span className="text-right">{t(locale, "rankings.colMovement")}</span>
   589→              <span className="text-right">{timePeriod !== "current" ? t(locale, "rankings.colCumPoints") : t(locale, "rankings.colPoints")}</span>
   590→              {timePeriod !== "current" && (
   591→                <span className="text-right">{t(locale, "rankings.colUpdates")}</span>
   592→              )}
   593→            </div>
   594→
   595→            {/* Table body */}
   596→            <div className="divide-y divide-border/30">
   597→              {filteredList.map((item) => {
   598→                const tier = getLabelTier(item.label, selectedGenre!);
   599→                const tierBadge = getTierBadge(tier);
   600→
   601→                return (
   602→                  <div
   603→                    key={`${item.label.id}-${item.rank}`}
   604→                    className={`grid items-center gap-1 px-3 py-2.5 text-sm transition-colors hover:bg-secondary/20 ${
   605→                      timePeriod !== "current"
   606→                        ? "grid-cols-[3rem_1fr_4.5rem_4rem_4.5rem_3.5rem] sm:grid-cols-[3rem_1fr_5rem_5rem_5.5rem_4rem]"
   607→                        : "grid-cols-[3rem_1fr_4.5rem_4rem_4rem] sm:grid-cols-[3rem_1fr_5rem_5rem_5rem]"
   608→                    } ${item.rank <= 3 ? "bg-primary/5" : ""}`}
   609→                  >
   610→                    {/* Rank number */}
   611→                    <span className={`font-mono font-bold ${item.rank <= 3 ? "text-primary" : "text-foreground"}`}>
   612→                      {item.rank}
   613→                    </span>
   614→
   615→                    {/* Label name + tier */}
   616→                    <div className="flex items-center gap-2 min-w-0">
   617→                      <span className="font-medium text-foreground truncate">{item.label.name}</span>
   618→                      {timePeriod === "current" && (
   619→                        <>
   620→                          <span className={`shrink-0 flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[10px] border ${tierBadge.color}`}>
   621→                            {tierBadge.icon}
   622→                            {tier === "top" ? "T" : tier === "mid" ? "M" : tier === "emerging" ? "E" : ""}
   623→                          </span>
   624→                          {item.label.trending && <Flame className="h-3 w-3 text-orange-400 shrink-0" />}
   625→                        </>
   626→                      )}
   627→                    </div>
   628→
   629→                    {/* Current rank */}
   630→                    <span className="text-right font-mono text-foreground">
   631→                      #{item.rank}
   632→                    </span>
   633→
   634→                    {/* Movement */}
   635→                    <div className="flex items-center justify-end gap-1">
   636→                      {timePeriod === "current" ? (
   637→                        <>
   638→                          {getMovementIcon(item.movement)}
   639→                          <span className={`font-mono text-xs ${getMovementColor(item.movement)}`}>
   640→                            {getMovementText(item.movement, locale)}
   641→                          </span>
   642→                        </>
   643→                      ) : (
   644→                        <span className="font-mono text-xs text-muted-foreground">—</span>
   645→                      )}
   646→                    </div>
   647→
   648→                    {/* Points */}
   649→                    <span className="text-right font-mono text-xs text-muted-foreground">
   650→                      {item.points.toLocaleString(locale === "it" ? "it-IT" : "en-US")}
   651→                    </span>
   652→
   653→                    {/* Snapshot count (only for cumulative views) */}
   654→                    {timePeriod !== "current" && (
   655→                      <span className="text-right font-mono text-[10px] text-muted-foreground">
   656→                        {item.snapshotCount}×
   657→                      </span>
   658→                    )}
   659→                  </div>
   660→                );
   661→              })}
   662→            </div>
   663→          </div>
   664→        )}
   665→
   666→        {/* Empty state */}
   667→        {selectedGenre && filteredList.length === 0 && (
   668→          <div className="flex flex-col items-center justify-center py-12 text-center">
   669→            <BarChart3 className="h-10 w-10 text-muted-foreground/30 mb-3" />
   670→            <p className="text-sm text-muted-foreground">
   671→              {timePeriod !== "current" && !hasSnapshots
   672→                ? t(locale, "rankings.noSnapshots")
   673→                : movementFilter !== "all"
   674→                  ? t(locale, "rankings.noResultsFilter")
   675→                  : t(locale, "rankings.noResults")}
   676→            </p>
   677→          </div>
   678→        )}
   679→
   680→        {/* No genre selected */}
   681→        {!selectedGenre && (
   682→          <div className="flex flex-col items-center justify-center py-16 text-center">
   683→            <Trophy className="h-14 w-14 text-primary/30 mb-4" />
   684→            <h3 className="text-lg font-semibold text-foreground mb-2">{t(locale, "rankings.title")}</h3>
   685→            <p className="text-sm text-muted-foreground max-w-md">
   686→              {t(locale, "rankings.selectGenrePrompt")}
   687→            </p>
   688→          </div>
   689→        )}
   690→      </div>
   691→    </div>
   692→  );
