   212→   212→function safeJsonParse(str: string | null): any | null {
   213→   213→  if (!str) return null;
   214→   214→  try {
   215→   215→    return JSON.parse(str);
   216→   216→  } catch {
   217→   217→    return null;
   218→   218→  }
   219→   219→}

   221→   221→function safeLocalStorageGet(key: string): string | null {
   222→   222→  try {
   223→   223→    return localStorage.getItem(key);
   224→   224→  } catch {
   225→   225→    return null;
   226→   226→  }
   227→   227→}
   228→   228→
   229→   229→function safeLocalStorageSet(key: string, value: string): boolean {
   230→   230→  try {
   231→   231→    localStorage.setItem(key, value);
   232→   232→    return true;
   233→   233→  } catch (e) {
   234→   234→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   235→   235→    return false;
   236→   236→  }
   237→   237→}

   247→   247→/**
   248→   248→ * Extract rankingSnapshots array from a raw localStorage JSON string.
   249→   249→ * Returns [] if not found / invalid.
   250→   250→ */
   251→   251→function extractSnapshotsFromRaw(raw: string | null): RankingSnapshot[] {
   252→   252→  if (!raw) return [];
   253→   253→  const data = safeJsonParse(raw);
   254→   254→  if (!data) return [];
   255→   255→  const snaps = data.state?.rankingSnapshots ?? data.rankingSnapshots;
   256→   256→  return Array.isArray(snaps) ? snaps : [];
   257→   257→}
   258→   258→
   259→   259→/**
   260→   260→ * Merge two snapshot arrays, deduping by `id` first, then by `timestamp+source`.
   261→   261→ * Always returns ascending-by-timestamp order.
   262→   262→ * Used by: importData (backup file), robustStorage.getItem (safety-net restore),
   263→   263→ * and loadFromCloud (cloud merge).
   264→   264→ */
   265→   265→function mergeSnapshots(
   266→   266→  base: RankingSnapshot[] | null | undefined,
   267→   267→  incoming: RankingSnapshot[] | null | undefined
   268→   268→): RankingSnapshot[] {
   269→   269→  const a = Array.isArray(base) ? base : [];
   270→   270→  const b = Array.isArray(incoming) ? incoming : [];
   271→   271→  const seen = new Set<string>();
   272→   272→  const out: RankingSnapshot[] = [];
   273→   273→  for (const s of [...a, ...b]) {
   274→   274→    if (!s || typeof s !== "object") continue;
   275→   275→    const key = s.id || `${s.timestamp}|${s.source}`;
   276→   276→    if (seen.has(key)) continue;
   277→   277→    seen.add(key);
   278→   278→    out.push(s);
   279→   279→  }
   280→   280→  out.sort(
   281→   281→    (x, y) =>
   282→   282→      new Date(x.timestamp || 0).getTime() - new Date(y.timestamp || 0).getTime()
   283→   283→  );
   284→   284→  return out;
   285→   285→}

   302→   302→  ).length;
   303→   303→}

   305→   305→function getTotalLabelCount(data: any): number {
   306→   306→  if (!data) return 0;
   307→   307→  if (Array.isArray(data.labels)) return data.labels.length;
   308→   308→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   309→   309→  return 0;

   313→   313→ * Extract user-edited label count from raw localStorage string.

   316→   316→function getUserEditCountFromRaw(raw: string | null): number {
   317→   317→  if (!raw) return 0;
   318→   318→  const data = safeJsonParse(raw);
   319→   319→  if (!data) return 0;
   320→   320→  const labels = data.state?.labels || data.labels || [];
   321→   321→  return countUserEditedLabels(labels);

   324→   324→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   325→   325→
   326→   326→const robustStorage: StateStorage = {
   327→   327→  getItem: (name: string): string | null => {
   328→   328→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   329→   329→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   330→   330→
   331→   331→    const primaryData = safeJsonParse(primaryRaw);
   332→   332→    const backupData = safeJsonParse(backupRaw);
   333→   333→
   334→   334→    let result: string | null = null;
   335→   335→    let usedSidecarSnapshots = false;
   336→   336→
   337→   337→    // If primary is missing/corrupt, try backup
   338→   338→    if (!primaryData) {
   339→   339→      if (backupData) {
   340→   340→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   341→   341→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   342→   342→        result = backupRaw;
   343→   343→      }
   344→   344→      // Both empty — first visit, result stays null
   345→   345→    } else if (backupData) {
   346→   346→      // Both exist — check for data loss
   347→   347→      const primaryLabelCount = getTotalLabelCount(primaryData);
   348→   348→      const backupLabelCount = getTotalLabelCount(backupData);
   349→   349→      const primaryUserEdits = primaryData.state?.labels
   350→   350→        ? countUserEditedLabels(primaryData.state.labels)
   351→   351→        : countUserEditedLabels(primaryData.labels || []);
   352→   352→      const backupUserEdits = backupData.state?.labels
   353→   353→        ? countUserEditedLabels(backupData.state.labels)
   354→   354→        : countUserEditedLabels(backupData.labels || []);
   355→   355→
   356→   356→      // If primary has fewer labels than backup AND backup has more or equal user data
   357→   357→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   358→   358→        console.warn(
   359→   359→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   360→   360→        );
   361→   361→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   362→   362→        result = backupRaw;
   363→   363→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   364→   364→        // If primary has fewer user edits than backup, use backup
   365→   365→        console.warn(
   366→   366→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   367→   367→        );
   368→   368→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   369→   369→        result = backupRaw;
   370→   370→      } else {
   371→   371→        result = primaryRaw;
   372→   372→      }
   373→   373→    } else {
   374→   374→      // Primary exists, no backup — normal case
   375→   375→      result = primaryRaw;
   376→   376→    }
   377→   377→
   378→   378→    // CRITICAL SAFETY NET for rankingSnapshots:
   379→   379→    // If the chosen result has 0 snapshots but the dedicated SNAPSHOTS_BACKUP_KEY
   380→   380→    // has snapshots, splice them in. This handles the scenario where the main
   381→   381→    // store got wiped (bad cloud sync, migration bug, partial overwrite) but the
   382→   382→    // snapshots sidecar survived. Without this, the user loses months of
   383→   383→    // "storico caricamenti" and the Classifiche page resets to "first time".
   384→   384→    try {
   385→   385→      const currentSnaps = extractSnapshotsFromRaw(result);
   386→   386→      const sidecarRaw = safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY);
   387→   387→      const sidecarSnaps = extractSnapshotsFromRaw(sidecarRaw);
   388→   388→      if (sidecarSnaps.length > 0 && sidecarSnaps.length > currentSnaps.length) {
   389→   389→        console.warn(
   390→   390→          `[LabelPulse Storage] SNAPSHOTS RECOVERY: result has ${currentSnaps.length} snapshots, sidecar has ${sidecarSnaps.length}. Merging sidecar into result.`
   391→   391→        );
   392→   392→        const mergedSnaps = mergeSnapshots(currentSnaps, sidecarSnaps);
   393→   393→        const parsed = result ? safeJsonParse(result) : {};
   394→   394→        if (parsed && typeof parsed === "object") {
   395→   395→          if (parsed.state && typeof parsed.state === "object") {
   396→   396→            parsed.state.rankingSnapshots = mergedSnaps;
   397→   397→          } else {
   398→   398→            parsed.rankingSnapshots = mergedSnaps;
   399→   399→          }
   400→   400→          result = JSON.stringify(parsed);
   401→   401→          usedSidecarSnapshots = true;
   402→   402→        }
   403→   403→      }
   404→   404→    } catch (e) {
   405→   405→      console.warn("[LabelPulse Storage] Snapshots sidecar recovery failed:", e);
   406→   406→    }
   407→   407→
   408→   408→    // CRITICAL SAFETY NET for userProfile:
   409→   409→    // Same logic as snapshots. If the chosen result has an empty/seed profile
   410→   410→    // but PROFILE_BACKUP_KEY has real data, splice it in. This is the user's
   411→   411→    // identity card — losing artistName, bio, email, links is catastrophic.
   412→   412→    try {
   413→   413→      const currentProfile = extractProfileFromRaw(result);
   414→   414→      const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   415→   415→      if (sidecarProfile && !currentProfile) {
   416→   416→        console.warn(
   417→   417→          `[LabelPulse Storage] PROFILE RECOVERY: result has empty profile, sidecar has data (artistName=${!!sidecarProfile.artistName}, bio=${!!sidecarProfile.bio}). Merging sidecar into result.`
   418→   418→        );
   419→   419→        const parsed = result ? safeJsonParse(result) : {};
   420→   420→        if (parsed && typeof parsed === "object") {
   421→   421→          if (parsed.state && typeof parsed.state === "object") {
   422→   422→            parsed.state.userProfile = sidecarProfile;
   423→   423→          } else {
   424→   424→            parsed.userProfile = sidecarProfile;
   425→   425→          }
   426→   426→          result = JSON.stringify(parsed);
   427→   427→        }
   428→   428→      } else if (sidecarProfile && currentProfile) {
   429→   429→        // Both have data — merge (sidecar fills empty fields in current)
   430→   430→        const mergedProfile = mergeProfiles(sidecarProfile, currentProfile);
   431→   431→        const parsed = result ? safeJsonParse(result) : {};
   432→   432→        if (parsed && typeof parsed === "object" && mergedProfile) {
   433→   433→          if (parsed.state && typeof parsed.state === "object") {
   434→   434→            parsed.state.userProfile = mergedProfile;
   435→   435→          } else {
   436→   436→            parsed.userProfile = mergedProfile;
   437→   437→          }
   438→   438→          result = JSON.stringify(parsed);
   439→   439→        }
   440→   440→      }
   441→   441→    } catch (e) {
   442→   442→      console.warn("[LabelPulse Storage] Profile sidecar recovery failed:", e);
   443→   443→    }
   444→   444→
   445→   445→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   446→   446→    // merged data immediately after this getItem returns, and we MUST allow that write.
   447→   447→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   448→   448→    _rehydrated = true;
   449→   449→
   450→   450→    if (usedSidecarSnapshots) {
   451→   451→      console.info("[LabelPulse Storage] rankingSnapshots restored from sidecar backup.");
   452→   452→    }
   453→   453→
   454→   454→    return result;
   455→   455→  },
   456→   456→
   457→   457→  setItem: (name: string, value: string): void => {
   458→   458→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   459→   459→    // Before rehydration, the store only has seed data — writing it would overwrite
   460→   460→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   461→   461→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   462→   462→    if (!_rehydrated) {
   463→   463→      console.warn(
   464→   464→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   465→   465→      );
   466→   466→      return;
   467→   467→    }
   468→   468→
   469→   469→    // Write to primary immediately
   470→   470→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   471→   471→
   472→   472→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   473→   473→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   474→   474→    safeLocalStorageSet(BACKUP_KEY, value);
   475→   475→
   476→   476→    // CRITICAL: also persist rankingSnapshots to a dedicated slot.
   477→   477→    // This is the user's "storico caricamenti" — months of imports that cannot
   478→   478→    // be regenerated. Keeping it in a separate key means even a catastrophic
   479→   479→    // main-store wipe (bad cloud sync, migration bug, quota error) leaves the
   480→   480→    // snapshots recoverable.
   481→   481→    try {
   482→   482→      const snaps = extractSnapshotsFromRaw(value);
   483→   483→      if (snaps.length > 0) {
   484→   484→        // Merge with whatever's already in the snapshots slot — NEVER overwrite
   485→   485→        // existing snapshots with fewer snapshots (would happen if the current
   486→   486→        // write had a partial/empty snapshot array due to some upstream bug).
   487→   487→        const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   488→   488→        const merged = mergeSnapshots(existing, snaps);
   489→   489→        safeLocalStorageSet(
   490→   490→          SNAPSHOTS_BACKUP_KEY,
   491→   491→          JSON.stringify(merged)
   492→   492→        );
   493→   493→      }
   494→   494→    } catch (e) {
   495→   495→      console.warn("[LabelPulse Storage] Snapshots-sidecar write failed:", e);
   496→   496→    }
   497→   497→
   498→   498→    // CRITICAL: also persist userProfile to a dedicated slot.
   499→   499→    // Same pattern as snapshots. The profile is the user's identity card —
   500→   500→    // artistName, bio, email, social links, photo. We can NEVER lose this,
   501→   501→    // even if cloud sync wipes the main store.
   502→   502→    try {
   503→   503→      const profile = extractProfileFromRaw(value);
   504→   504→      if (profile) {
   505→   505→        const existing = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   506→   506→        const merged = mergeProfiles(existing, profile);
   507→   507→        if (merged) {
   508→   508→          safeLocalStorageSet(
   509→   509→            PROFILE_BACKUP_KEY,
   510→   510→            JSON.stringify({ userProfile: merged })
   511→   511→          );
   512→   512→        }
   513→   513→      }
   514→   514→    } catch (e) {
   515→   515→      console.warn("[LabelPulse Storage] Profile-sidecar write failed:", e);
   516→   516→    }
   517→   517→
   518→   518→    if (!primaryOk) {
   519→   519→      console.error("[LabelPulse Storage] Primary write failed!");
   520→   520→    }
   521→   521→  },
   522→   522→
   523→   523→  removeItem: (name: string): void => {
   524→   524→    safeLocalStorageRemove(PRIMARY_KEY);
   525→   525→    safeLocalStorageRemove(BACKUP_KEY);
   526→   526→    // NOTE: do NOT remove SNAPSHOTS_BACKUP_KEY or PROFILE_BACKUP_KEY here.
   527→   527→    // These are the user's permanent records (months of ranking history +
   528→   528→    // artist identity) — clearing the main store should NEVER nuke them.
   529→   529→    // The only way to clear them is an explicit user action ("reset all data"
   530→   530→    // button) which calls safeLocalStorageRemove directly.
   531→   531→  },
   532→   532→};

   534→   534→/**
   535→   535→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   536→   536→ * Since backup is now written on every setItem, this is a safety net.
   537→   537→ */
   538→   538→export function forceBackupNow(): void {
   539→   539→  if (typeof window === "undefined") return;
   540→   540→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   541→   541→  if (primaryRaw) {
   542→   542→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   543→   543→  }
   544→   544→  // Also touch the snapshots sidecar so it stays warm.
   545→   545→  const snaps = extractSnapshotsFromRaw(primaryRaw);
   546→   546→  if (snaps.length > 0) {
   547→   547→    const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   548→   548→    const merged = mergeSnapshots(existing, snaps);
   549→   549→    safeLocalStorageSet(SNAPSHOTS_BACKUP_KEY, JSON.stringify(merged));
   550→   550→  }
   551→   551→}

   568→   568→export function restoreSnapshotsFromSidecar(): number {

   579→   579→    );
   580→   580→  }
   581→   581→  return Math.max(added, 0);
   582→   582→}

   586→   586→/**
   587→   587→ * Extract userProfile from a raw JSON string (main store JSON, backup JSON,
   588→   588→ * or sidecar JSON). Returns null if missing/invalid.
   589→   589→ */
   590→   590→function extractProfileFromRaw(raw: string | null): any | null {
   591→   591→  if (!raw) return null;
   592→   592→  try {
   593→   593→    const parsed = JSON.parse(raw);
   594→   594→    const profile = parsed?.state?.userProfile ?? parsed?.userProfile ?? null;
   595→   595→    if (profile && typeof profile === "object") {
   596→   596→      // Must have at least one non-empty field to be considered "real"
   597→   597→      const hasData =
   598→   598→        !!profile.artistName ||
   599→   599→        !!profile.bio ||
   600→   600→        !!profile.email ||
   601→   601→        !!profile.scLink ||
   602→   602→        !!profile.photoUrl ||
   603→   603→        (Array.isArray(profile.links) && profile.links.length > 0) ||
   604→   604→        !!profile.cyaniteApiToken ||
   605→   605→        !!profile.supabaseUrl;
   606→   606→      if (hasData) return profile;
   607→   607→    }
   608→   608→  } catch {
   609→   609→    /* ignore */
   610→   610→  }
   611→   611→  return null;
   612→   612→}

   614→   614→/**
   615→   615→ * Merge two user profiles — non-empty fields win.
   616→   616→ * This is the safety net: if the cloud ever sends a profile with empty
   617→   617→ * artistName, but the sidecar has the real artistName, we keep the real one.
   618→   618→ */
   619→   619→export function mergeProfiles(base: any | null, incoming: any | null): any | null {
   620→   620→  if (!base && !incoming) return null;
   621→   621→  if (!base) return incoming;
   622→   622→  if (!incoming) return base;
   623→   623→  const merged: any = { ...base };
   624→   624→  for (const k of Object.keys(incoming)) {
   625→   625→    const v = incoming[k];
   626→   626→    if (k === "links") {
   627→   627→      // Merge link arrays (dedupe by type|value)
   628→   628→      const a = Array.isArray(base.links) ? base.links : [];
   629→   629→      const b = Array.isArray(v) ? v : [];
   630→   630→      const seen = new Set<string>();
   631→   631→      const out: { type: string; value: string }[] = [];
   632→   632→      for (const link of [...a, ...b]) {
   633→   633→        if (!link || typeof link !== "object") continue;
   634→   634→        const key = `${link.type}|${link.value}`;
   635→   635→        if (seen.has(key)) continue;
   636→   636→        seen.add(key);
   637→   637→        out.push(link);
   638→   638→      }
   639→   639→      merged.links = out;
   640→   640→    } else if (typeof v === "string") {
   641→   641→      // Non-empty string wins over empty
   642→   642→      if (v && !merged[k]) merged[k] = v;
   643→   643→    } else {
   644→   644→      // For any other type, prefer defined over undefined
   645→   645→      if (v !== undefined && v !== null && !merged[k]) merged[k] = v;
   646→   646→    }
   647→   647→  }
   648→   648→  return merged;
   649→   649→}

   777→   777→/**
   778→   778→ * Merge incoming artists (from a fresh scrape) with existing artists
   779→   779→ * (already in the store from previous scrapes).
   780→   780→ *
   781→   781→ * Strategy:
   782→   782→ * - Key by artist.id (e.g. "bp_6824"). Same Beatport id = same artist.
   783→   783→ * - For each artist that exists: union genres, union labelsPublishedOn,
   784→   784→ *   MERGE tracksByGenre by track.id (preserve tracks from older scrapes,
   785→   785→ *   update with new position info), sum totalPoints, min bestPosition.
   786→   786→ * - firstSeenAt preserved from existing; lastSeenAt = now.
   787→   787→ * - Trending recomputed fresh from current data.
   788→   788→ * - New artists: appended with firstSeenAt = lastSeenAt = now.
   789→   789→ *
   790→   790→ * This is what makes historical data accumulate: each scrape adds new
   791→   791→ * tracks to artists' tracksByGenre without losing previously-seen tracks.
   792→   792→ */
   793→   793→function mergeArtists(existing: Artist[], incoming: Artist[], now: string): Artist[] {
   794→   794→  const map = new Map<string, Artist>();
   795→   795→
   796→   796→  // Seed with existing
   797→   797→  for (const a of existing) {
   798→   798→    map.set(a.id, { ...a });
   799→   799→  }
   800→   800→
   801→   801→  for (const inc of incoming) {
   802→   802→    const ex = map.get(inc.id);
   803→   803→    if (!ex) {
   804→   804→      // New artist — first time we see them
   805→   805→      map.set(inc.id, {
   806→   806→        ...inc,
   807→   807→        firstSeenAt: now,
   808→   808→        lastSeenAt: now,
   809→   809→      });
   810→   810→      continue;
   811→   811→    }
   812→   812→
   813→   813→    // Existing artist — merge fields
   814→   814→    // Union genres
   815→   815→    const genres = Array.from(new Set([...(ex.genres || []), ...(inc.genres || [])]));
   816→   816→
   817→   817→    // Union labelsPublishedOn
   818→   818→    const labelsPublishedOn = Array.from(
   819→   819→      new Set([...(ex.labelsPublishedOn || []), ...(inc.labelsPublishedOn || [])])
   820→   820→    );
   821→   821→
   822→   822→    // Merge tracksByGenre — preserve old tracks, add/update from incoming
   823→   823→    const tracksByGenre: Record<string, ArtistTrack[]> = { ...(ex.tracksByGenre || {}) };
   824→   824→    for (const genre of Object.keys(inc.tracksByGenre || {})) {
   825→   825→      const incTracks = inc.tracksByGenre[genre] || [];
   826→   826→      const exTracks = tracksByGenre[genre] || [];
   827→   827→      const byId = new Map<number, ArtistTrack>();
   828→   828→      // Seed with existing
   829→   829→      for (const t of exTracks) byId.set(t.id, t);
   830→   830→      // Update/add with incoming (incoming overrides — fresh positions)
   831→   831→      for (const t of incTracks) byId.set(t.id, t);
   832→   832→      tracksByGenre[genre] = Array.from(byId.values()).sort((a, b) => a.position - b.position);
   833→   833→    }
   834→   834→
   835→   835→    // Recompute totalPoints & bestPosition from current incoming data
   836→   836→    // (the existing totals were from older scrapes — we want the snapshot
   837→   837→    // from THIS import to drive the points, otherwise the totals just
   838→   838→    // accumulate forever and "points" loses meaning).
   839→   839→    // BUT: keep max totalPoints seen (artists who peaked before should
   840→   840→    // keep credit). Actually no — let's use the LATEST totalPoints,
   841→   841→    // since that's the current ranking signal. Old tracks are still in
   842→   842→    // tracksByGenre for browsing, but the points reflect "right now".
   843→   843→    const totalPoints = inc.totalPoints;
   844→   844→    const bestPosition = Math.min(ex.bestPosition, inc.bestPosition);
   845→   845→
   846→   846→    // Trending: use incoming (fresh computation)
   847→   847→    const trending = inc.trending;
   848→   848→    const trendingRankByGenre = inc.trendingRankByGenre;
   849→   849→    const trendingPointsByGenre = inc.trendingPointsByGenre;
   850→   850→
   851→   851→    // isRemixerOnly: false if EITHER snapshot has them as primary
   852→   852→    const isRemixerOnly = ex.isRemixerOnly && inc.isRemixerOnly;
   853→   853→
   854→   854→    map.set(inc.id, {
   855→   855→      ...ex,
   856→   856→      ...inc,
   857→   857→      genres,
   858→   858→      labelsPublishedOn,
   859→   859→      tracksByGenre,
   860→   860→      totalPoints,
   861→   861→      bestPosition,
   862→   862→      trending,
   863→   863→      trendingRankByGenre,
   864→   864→      trendingPointsByGenre,
   865→   865→      isRemixerOnly,
   866→   866→      firstSeenAt: ex.firstSeenAt || now,
   867→   867→      lastSeenAt: now,
   868→   868→    });
   869→   869→  }
   870→   870→
   871→   871→  return Array.from(map.values());
   872→   872→}

   883→   883→    ? { ...existing.rankByGenre }
   884→   884→    : existing.prevRankByGenre || {};
   885→   885→
   886→   886→  return {
   887→   887→    // User-editable data — ALWAYS prefer existing if it has real data
   888→   888→    emails: imported.emails?.length ? imported.emails : existing.emails,

   896→   896→

   936→   936→/**
   937→   937→ * Repair corrupted label data.
   938→   938→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   939→   939→ */
   940→   940→function repairLabelData(labels: Label[]): Label[] {
   941→   941→  const emailLabelCount = new Map<string, number>();
   942→   942→  for (const l of labels) {
   943→   943→    if (l.emails && Array.isArray(l.emails)) {
   944→   944→      for (const e of l.emails) {
   945→   945→        if (e && e.trim()) {
   946→   946→          const key = e.toLowerCase().trim();
   947→   947→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   948→   948→        }
   949→   949→      }
   950→   950→    }
   951→   951→  }
   952→   952→
   953→   953→  const corruptedEmails = new Set<string>();
   954→   954→  for (const [email, count] of emailLabelCount) {
   955→   955→    if (count > 5) {
   956→   956→      corruptedEmails.add(email);
   957→   957→      console.warn(
   958→   958→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   959→   959→      );
   960→   960→    }
   961→   961→  }
   962→   962→
   963→   963→  if (corruptedEmails.size === 0) {
   964→   964→    return labels;
   965→   965→  }
   966→   966→
   967→   967→  const emailOwner = new Map<string, string>();
   968→   968→  for (const email of corruptedEmails) {
   969→   969→    for (const l of labels) {
   970→   970→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   971→   971→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   972→   972→        if (domain && l.name.toLowerCase().includes(domain)) {
   973→   973→          emailOwner.set(email, l.id);
   974→   974→          break;
   975→   975→        }
   976→   976→      }
   977→   977→    }
   978→   978→    if (!emailOwner.has(email)) {
   979→   979→      for (const l of labels) {
   980→   980→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   981→   981→          emailOwner.set(email, l.id);
   982→   982→          break;
   983→   983→        }
   984→   984→      }
   985→   985→    }
   986→   986→  }
   987→   987→
   988→   988→  return labels.map(l => {
   989→   989→    if (!l.emails || !Array.isArray(l.emails)) return l;
   990→   990→    const cleanedEmails = l.emails.filter(e => {
   991→   991→      const key = e?.toLowerCase().trim();
   992→   992→      if (!key) return false;
   993→   993→      if (!corruptedEmails.has(key)) return true;
   994→   994→      return emailOwner.get(key) === l.id;
   995→   995→    });
   996→   996→    const cleanedContactInfo = cleanedEmails[0] || "";
   997→   997→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   998→   998→      console.log(
   999→   999→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
  1000→  1000→      );
  1001→  1001→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
  1002→  1002→    }
  1003→  1003→    return l;
  1004→  1004→  });
  1005→  1005→}

  1794→  1794→    //      equals cloud timestamp → SKIP merge → if local was wiped (e.g., fresh incognito
  1795→  1795→    //      after a force-sync), cloud data is also ignored.
  1796→  1796→    //   3. After the skip, useAuthEffect's forceCloudSync() would push the EMPTY local state
  1797→  1797→    //      back to cloud, OVERWRITING the cloud's good data.

  1800→  1800→    // timestamps. Timestamps are only used as a tiebreaker when both have data.

  1802→  1802→      !!cloud.userProfile?.artistName ||
  1803→  1803→      !!cloud.userProfile?.bio ||
  1804→  1804→      !!cloud.userProfile?.email ||

  1809→  1809→      !!localState.userProfile?.artistName ||
  1810→  1810→      !!localState.userProfile?.bio ||
  1811→  1811→      !!localState.userProfile?.email ||
  1812→  1812→      !!localState.userProfile?.scLink ||
  1813→  1813→      !!localState.userProfile?.photoUrl ||

  1817→  1817→    const cloudHasSnapshots = Array.isArray(cloud.rankingSnapshots) && cloud.rankingSnapshots.length > 0;
  1818→  1818→    const cloudHasDemos = Array.isArray(cloud.demos) && cloud.demos.length > 0;
  1819→  1819→    const localHasLabels = (localState.labels?.length ?? 0) > 0;
  1820→  1820→    const localHasSnapshots = (localState.rankingSnapshots?.length ?? 0) > 0;

  1832→  1832→      cloudBringsNewProfile ||
  1833→  1833→      cloudBringsNewLabels ||
  1834→  1834→      cloudBringsNewSnapshots ||
  1835→  1835→      cloudBringsNewDemos ||

  1851→  1851→      // il locale ha dati, NON fidarti — è probabile che il cloud sia stato
  1852→  1852→      // resettato / corrotto. Mantieni i dati locali e basta.

  1913→  1913→/**

  1916→  1916→ *
  1917→  1917→ * CRITICAL SAFETY RULE: mai sovrascrivere array locali non-vuoti con

  1922→  1922→function mergeCloudData(cloudData: any, localState: any): Partial<AppState> {
  1923→  1923→  const merged: Partial<AppState> = {};
  1924→  1924→
  1925→  1925→  // Labels: applica defaults e merge con seed data, come nella reidratazione

  1931→  1931→
  1932→  1932→    // Aggiungi seed labels mancanti

  1934→  1934→    const cloudNames = new Set(
  1935→  1935→      cloudData.labels.map((l: any) => l.name?.toLowerCase().trim()).filter(Boolean)
  1936→  1936→    );
  1937→  1937→    const seedLabels = buildLabelsFromData();
  1938→  1938→    for (const seed of seedLabels) {
  1939→  1939→      if (!cloudIds.has(seed.id) && !cloudNames.has(seed.name.toLowerCase().trim())) {
  1940→  1940→        (merged.labels as Label[]).push(seed);
  1941→  1941→      }
  1942→  1942→    }
  1943→  1943→
  1944→  1944→    // Repair corrupted data
  1945→  1945→    merged.labels = repairLabelData(merged.labels);
  1946→  1946→  } else {
  1947→  1947→    // Cloud non ha labels (o array vuoto) → mantieni locali
  1948→  1948→    merged.labels = localState.labels;
  1949→  1949→  }
  1950→  1950→
  1951→  1951→  // Demos: SOLO se cloud ha effettivamente demos, mantieni locali altrimenti
  1952→  1952→  // (before: `Array.isArray(cloudData.demos) ? cloudData.demos : localState.demos`
  1953→  1953→  //  sovrascriveva demos locali con [] se cloud aveva demos:[])
  1954→  1954→  if (Array.isArray(cloudData.demos) && cloudData.demos.length > 0) {
  1955→  1955→    merged.demos = cloudData.demos;
  1956→  1956→  } else if (Array.isArray(cloudData.demos) && (!localState.demos || localState.demos.length === 0)) {
  1957→  1957→    // Cloud ha demos:[] e locale è vuoto → ok a usare [] (no-op)
  1958→  1958→    merged.demos = cloudData.demos;

  1960→  1960→    merged.demos = localState.demos;
  1961→  1961→  }

  1981→  1981→  // storiche, perderli significa perdere mesi di import Beatport/Beatstats.
  1982→  1982→  if (Array.isArray(cloudData.rankingSnapshots) && cloudData.rankingSnapshots.length > 0) {

  1984→  1984→  } else if (Array.isArray(cloudData.rankingSnapshots) && (!localState.rankingSnapshots || localState.rankingSnapshots.length === 0)) {
  1985→  1985→    // Cloud vuoto e locale vuoto → ok a usare []
  1986→  1986→    merged.rankingSnapshots = cloudData.rankingSnapshots;
  1987→  1987→  } else {
