    88→// ==================== ROBUST STORAGE ====================
    89→// Custom storage with bulletproof data protection:
    90→// 1. Primary localStorage key
    91→// 2. Backup localStorage key (debounced — NOT written on every save)
    92→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
    93→// 4. Write-blocking until rehydration completes (prevents race condition)
    94→
    95→const PRIMARY_KEY = "labelpulse-storage";
    96→const BACKUP_KEY = "labelpulse-storage-backup";
    97→const SEED_LABEL_COUNT = labelData.labels.length;
    98→
    99→// CRITICAL: Flag to prevent writes before Zustand rehydration completes.
   100→// Without this, the store initializes with seed data (empty user fields),
   101→// and any state change during the pre-rehydration window would write
   102→// seed data to localStorage, permanently destroying user edits.
   103→let _rehydrated = false;
   104→
   105→// Debounced backup writing — don't corrupt backup alongside primary
   106→let _backupTimer: ReturnType<typeof setTimeout> | null = null;
   107→const BACKUP_DEBOUNCE_MS = 60000; // 60 seconds
   108→
   109→function safeJsonParse(str: string | null): any | null {
   110→  if (!str) return null;
   111→  try {
   112→    return JSON.parse(str);
   113→  } catch {
   114→    return null;
   115→  }
   116→}
   117→
   118→function safeLocalStorageGet(key: string): string | null {
   119→  try {
   120→    return localStorage.getItem(key);
   121→  } catch {
   122→    return null;
   123→  }
   124→}
   125→
   126→function safeLocalStorageSet(key: string, value: string): boolean {
   127→  try {
   128→    localStorage.setItem(key, value);
   129→    return true;
   130→  } catch (e) {
   131→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   132→    return false;
   133→  }
   134→}
   135→
   136→function safeLocalStorageRemove(key: string): void {
   137→  try {
   138→    localStorage.removeItem(key);
   139→  } catch {
   140→    // Ignore
   141→  }
   142→}

   144→/**
   145→ * Check if persisted data has user edits (emails, notes, links, etc.)
   146→ * This helps us detect if the store was accidentally reset to seed data.
   147→ */
   148→function countUserEditedLabels(labels: any[]): number {
   149→  if (!Array.isArray(labels)) return 0;
   150→  return labels.filter((l) =>
   151→    (l.emails && l.emails.length > 0) ||
   152→    (l.notes && l.notes.trim() !== "") ||
   153→    (l.website && l.website.trim() !== "") ||
   154→    (l.demoLink && l.demoLink.trim() !== "") ||
   155→    (l.socialLink && l.socialLink.trim() !== "") ||
   156→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   157→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   158→    l.isCustom === true
   159→  ).length;
   160→}
   161→
   162→function getTotalLabelCount(data: any): number {
   163→  if (!data) return 0;
   164→  if (Array.isArray(data.labels)) return data.labels.length;
   165→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   166→  return 0;
   167→}

   169→function debouncedBackupWrite(value: string): void {
   170→  // Cancel any pending backup write
   171→  if (_backupTimer) {
   172→    clearTimeout(_backupTimer);
   173→  }
   174→  // Schedule a new backup write after debounce period
   175→  _backupTimer = setTimeout(() => {
   176→    safeLocalStorageSet(BACKUP_KEY, value);
   177→    _backupTimer = null;
   178→  }, BACKUP_DEBOUNCE_MS);
   179→}
   180→
   181→const robustStorage: StateStorage = {
   182→  getItem: (name: string): string | null => {
   183→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   184→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   185→
   186→    const primaryData = safeJsonParse(primaryRaw);
   187→    const backupData = safeJsonParse(backupRaw);
   188→
   189→    // If primary is missing/corrupt, try backup
   190→    if (!primaryData) {
   191→      if (backupData) {
   192→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   193→        // Restore backup to primary
   194→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   195→        return backupRaw;
   196→      }
   197→      // Both empty — first visit
   198→      return null;
   199→    }
   200→
   201→    // Both exist — check for data loss
   202→    if (backupData) {
   203→      const primaryLabelCount = getTotalLabelCount(primaryData);
   204→      const backupLabelCount = getTotalLabelCount(backupData);
   205→      const primaryUserEdits = primaryData.state?.labels
   206→        ? countUserEditedLabels(primaryData.state.labels)
   207→        : countUserEditedLabels(primaryData.labels || []);
   208→      const backupUserEdits = backupData.state?.labels
   209→        ? countUserEditedLabels(backupData.state.labels)
   210→        : countUserEditedLabels(backupData.labels || []);
   211→
   212→      // If primary has fewer labels than backup AND backup has more or equal user data
   213→      // this indicates a data loss event (threshold: any decrease, was +50)
   214→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   215→        console.warn(
   216→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   217→        );
   218→        // Restore from backup
   219→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   220→        return backupRaw;
   221→      }
   222→
   223→      // If primary has fewer user edits than backup, use backup
   224→      if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   225→        console.warn(
   226→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   227→        );
   228→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   229→        return backupRaw;
   230→      }
   231→    }
   232→
   233→    return primaryRaw;
   234→  },
   235→
   236→  setItem: (name: string, value: string): void => {
   237→    // CRITICAL FIX: Block ALL writes until rehydration is complete.
   238→    // Before rehydration, the store contains seed data with empty user fields.
   239→    // Writing this to localStorage would destroy user edits.
   240→    if (!_rehydrated) {
   241→      console.warn("[LabelPulse Storage] Blocked write before rehydration — protecting user data");
   242→      return;
   243→    }
   244→
   245→    // Write to primary immediately
   246→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   247→
   248→    // Write to backup with debounce — so if primary gets corrupted,
   249→    // the backup still holds the last good state
   250→    debouncedBackupWrite(value);
   251→
   252→    if (!primaryOk) {
   253→      console.error("[LabelPulse Storage] Primary write failed! Backup will be saved on debounce.");
   254→    }
   255→  },
   256→
   257→  removeItem: (name: string): void => {
   258→    // Only remove when explicitly requested (e.g., logout/clear)
   259→    safeLocalStorageRemove(PRIMARY_KEY);
   260→    safeLocalStorageRemove(BACKUP_KEY);
   261→  },
   262→};

   281→/**
   282→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   283→ * This ensures the backup is up-to-date even with debouncing.
   284→ */
   285→export function forceBackupNow(): void {
   286→  if (!_rehydrated) return;
   287→  if (typeof window === "undefined") return; // SSR guard
   288→  if (_backupTimer) {
   289→    clearTimeout(_backupTimer);
   290→    _backupTimer = null;
   291→  }
   292→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   293→  if (primaryRaw) {
   294→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   295→  }
   296→}

   410→/**
   411→ * Repair corrupted label data.
   412→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   413→ * Returns repaired labels array.
   414→ */
   415→function repairLabelData(labels: Label[]): Label[] {
   416→  // Count how many labels each email appears on
   417→  const emailLabelCount = new Map<string, number>();
   418→  for (const l of labels) {
   419→    if (l.emails && Array.isArray(l.emails)) {
   420→      for (const e of l.emails) {
   421→        if (e && e.trim()) {
   422→          const key = e.toLowerCase().trim();
   423→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   424→        }
   425→      }
   426→    }
   427→  }
   428→
   429→  // If any email appears on more than 5 labels, it's likely corrupted
   430→  // (no legitimate email belongs to 5+ different labels)
   431→  const corruptedEmails = new Set<string>();
   432→  for (const [email, count] of emailLabelCount) {
   433→    if (count > 5) {
   434→      corruptedEmails.add(email);
   435→      console.warn(
   436→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the first label.`
   437→      );
   438→    }
   439→  }
   440→
   441→  if (corruptedEmails.size === 0) {
   442→    return labels; // No corruption detected
   443→  }
   444→
   445→  // For each corrupted email, find the FIRST label that should keep it
   446→  // (try to match by name, otherwise keep the first one chronologically)
   447→  const emailOwner = new Map<string, string>(); // email → label ID that keeps it
   448→  for (const email of corruptedEmails) {
   449→    // Try to find a label whose name contains part of the email domain
   450→    for (const l of labels) {
   451→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   452→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   453→        if (domain && l.name.toLowerCase().includes(domain)) {
   454→          emailOwner.set(email, l.id);
   455→          break;
   456→        }
   457→      }
   458→    }
   459→    // If no name match, keep the first label with this email
   460→    if (!emailOwner.has(email)) {
   461→      for (const l of labels) {
   462→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   463→          emailOwner.set(email, l.id);
   464→          break;
   465→        }
   466→      }
   467→    }
   468→  }
   469→
   470→  // Remove corrupted emails from labels that don't own them
   471→  return labels.map(l => {
   472→    if (!l.emails || !Array.isArray(l.emails)) return l;
   473→    const cleanedEmails = l.emails.filter(e => {
   474→      const key = e?.toLowerCase().trim();
   475→      if (!key) return false;
   476→      if (!corruptedEmails.has(key)) return true; // Not corrupted, keep it
   477→      // Corrupted email: only keep if this label owns it
   478→      return emailOwner.get(key) === l.id;
   479→    });
   480→    const cleanedContactInfo = cleanedEmails[0] || "";
   481→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   482→      console.log(
   483→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   484→      );
   485→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   486→    }
   487→    return l;
   488→  });
   489→}

   586→          set((state) => ({
   587→            demos: state.demos.map((d) =>
   588→              d.id === id ? { ...d, ...updates } : d
   589→            ),
   590→            lastSavedAt: new Date().toISOString(),
   591→          }));
   592→        }
   593→      },
   594→
   595→      setActiveTab: (tab) => set({ activeTab: tab }),
   596→      setLocale: (locale) => set({ locale }),
   597→      setUserProfile: (profile) => set((state) => ({ userProfile: { ...state.userProfile, ...profile }, lastSavedAt: new Date().toISOString() })),
   598→
   599→      getGenres: () => labelData.genres,
   600→
   601→      setGmailAuth: (auth) => set({ gmailAuth: auth }),
   602→      clearGmailAuth: () => set({ gmailAuth: { isConnected: false, email: "", accessToken: "", expiresAt: 0 } }),
   603→
   604→      setRankingsUpdatedAt: (date) => set({ rankingsUpdatedAt: date }),
   605→
   606→      exportData: () => {
   607→        const state = get();

   617→          }
   618→        };

   620→      },
   621→
   622→      importData: (jsonString: string) => {
   623→        try {
   624→          const parsed = JSON.parse(jsonString);
   625→          if (parsed.app !== "labelpulse" || !parsed.data) return false;
   626→          
   627→          const { labels: importedLabels = [], demos: importedDemos = [], userProfile, locale: importedLocale } = parsed.data;
   628→          const currentLabels = get().labels;
   629→          const currentDemos = get().demos;
   630→
   631→          // === LABEL MERGE ===
   632→          const currentLabelById = new Map(currentLabels.map(l => [l.id, l]));
   633→          const currentLabelByName = new Map(currentLabels.map(l => [l.name.toLowerCase().trim(), l]));
   634→
   635→          const newLabels: Label[] = [];
   636→
   637→          for (const imported of importedLabels as Label[]) {
   638→            // 1) Same ID → merge, ALWAYS preserving user data
   639→            if (currentLabelById.has(imported.id)) {
   640→              const existing = currentLabelById.get(imported.id)!;
   641→              const merged = {
   642→                ...existing,
   643→                ...mergePreservingUserData(existing, imported),
   644→              };
   645→              currentLabelById.set(imported.id, merged);
   646→              continue;
   647→            }
   648→
   649→            // 2) Same name (case-insensitive) → merge and keep existing ID
   650→            const nameKey = imported.name.toLowerCase().trim();
   651→            if (currentLabelByName.has(nameKey)) {
   652→              const existing = currentLabelByName.get(nameKey)!;
   653→              const mergedLabel: Label = {
   654→                ...existing,
   655→                ...mergePreservingUserData(existing, imported),
   656→              };
   657→              currentLabelById.set(existing.id, mergedLabel);
   658→              currentLabelByName.set(nameKey, mergedLabel);
   659→              continue;
   660→            }
   661→
   662→            // 3) Brand new label → add it
   663→            newLabels.push({
   664→              ...imported,
   665→              id: imported.id || genId(),
   666→              createdAt: imported.createdAt || new Date().toISOString(),
   667→              isCustom: imported.isCustom ?? true,
   668→              genres: imported.genres || [],
   669→              rankByGenre: imported.rankByGenre || {},
   670→              pointsByGenre: imported.pointsByGenre || {},
   671→              trending: imported.trending || false,
   672→              trendingRankByGenre: imported.trendingRankByGenre || {},
   673→              trendingPointsByGenre: imported.trendingPointsByGenre || {},
   674→              emails: imported.emails || [],
   675→              website: imported.website || "",
   676→              demoLink: imported.demoLink || "",
   677→              socialLink: imported.socialLink || "",
   678→              soundcloudLink: imported.soundcloudLink || "",
   679→              contactInfo: imported.contactInfo || "",
   680→              notes: imported.notes || "",
   681→              status: imported.status || "open",
   682→              submissionType: imported.submissionType || "email",
   683→            });
   684→          }
   685→
   686→          const mergedLabels = [...currentLabelById.values(), ...newLabels];
   687→
   688→          // === DEMO MERGE ===
   689→          const currentDemoById = new Map(currentDemos.map(d => [d.id, d]));
   690→          const currentDemoByKey = new Map(
   691→            currentDemos.map(d => [`${d.trackName.toLowerCase().trim()}||${d.labelId}`, d])
   692→          );
   693→
   694→          const newDemos: Demo[] = [];
   695→
   696→          for (const imported of importedDemos as Demo[]) {
   697→            if (currentDemoById.has(imported.id)) {
   698→              const existing = currentDemoById.get(imported.id)!;
   699→              currentDemoById.set(imported.id, {
   700→                ...existing,
   701→                ...imported,
   702→                id: existing.id,
   703→              });
   704→              continue;
   705→            }
   706→
   707→            const key = `${imported.trackName?.toLowerCase().trim()}||${imported.labelId}`;
   708→            if (currentDemoByKey.has(key)) {
   709→              const existing = currentDemoByKey.get(key)!;
   710→              const merged: Demo = {
   711→                ...existing,
   712→                ...imported,
   713→                id: existing.id,
   714→              };
   715→              currentDemoById.set(existing.id, merged);
   716→              currentDemoByKey.set(key, merged);
   717→              continue;
   718→            }
   719→
   720→            newDemos.push({
   721→              ...imported,
   722→              id: imported.id || genId(),
   723→              createdAt: imported.createdAt || new Date().toISOString(),
   724→            });
   725→          }
   726→
   727→          const mergedDemos = [...currentDemoById.values(), ...newDemos];
   728→
   729→          // Check if this import contains rankings data
   730→          const isRankingsImport = parsed._meta?.source === 'beatport' || parsed._meta?.source === 'beatstats';
   731→          const hasGenresAndLabels = parsed.genres && parsed.labels && Array.isArray(parsed.labels);
   732→          
   733→          set({
   734→            labels: mergedLabels,
   735→            demos: mergedDemos,
   736→            userProfile: userProfile || get().userProfile,
   737→            locale: importedLocale || get().locale,
   738→            ...(isRankingsImport || hasGenresAndLabels ? { rankingsUpdatedAt: new Date().toISOString() } : {}),
   739→            lastSavedAt: new Date().toISOString(),
   740→          });
   741→          return true;
   742→        } catch {
   743→          return false;
   744→        }
   745→      },
   746→    }),
   747→    {
   748→      name: PRIMARY_KEY,
   749→      version: 9,
   750→      storage: createJSONStorage(() => robustStorage),
   751→      migrate: (persisted: any, version: number) => {
   752→        // Migrations run from oldest to newest (proper order)
   753→        if (version < 5) {
   754→          if (persisted.demos) {
   755→            const seedIds = ["demo_1", "demo_2", "demo_3", "demo_4", "demo_5", "demo_6"];
   756→            persisted.demos = persisted.demos.filter(
   757→              (d: any) => !seedIds.includes(d.id)
   758→            );
   759→          }
   760→          if (persisted.labels) {
   761→            persisted.labels = persisted.labels.map((l: any) => ({
   762→              ...l,
   763→              genres: l.genres || (l.genre ? [l.genre] : []),
   764→              rankByGenre: l.rankByGenre || {},
   765→              pointsByGenre: l.pointsByGenre || {},
   766→              trending: l.trending || false,
   767→              trendingRankByGenre: l.trendingRankByGenre || {},
   768→              trendingPointsByGenre: l.trendingPointsByGenre || {},
   769→              isCustom: l.isCustom ?? false,
   770→              website: l.website || "",
   771→              demoLink: l.demoLink || "",
   772→              socialLink: l.socialLink || "",
   773→              soundcloudLink: l.soundcloudLink || "",
   774→              emails: Array.isArray(l.emails) ? l.emails : (l.contactInfo ? [l.contactInfo] : []),
   775→            }));
   776→          }
   777→          if (persisted.demos) {
   778→            persisted.demos = persisted.demos.map((d: any) => ({
   779→              ...d,
   780→              sentDate: d.sentDate ?? null,
   781→              link: d.link || "",
   782→              notes: d.notes || "",
   783→              pitchText: d.pitchText || "",

   788→            persisted.locale = "it";
   789→          }
   790→          if (!persisted.userProfile) {
   791→            persisted.userProfile = { artistName: "", scLink: "" };

   793→        }
   794→        if (version < 6) {
   795→          if (!persisted.gmailAuth) {

   797→          }
   798→        }
   799→        if (version < 7) {
   800→          if (!persisted.rankingsUpdatedAt) {
   801→            persisted.rankingsUpdatedAt = null;
   802→          }
   803→        }
   804→        if (version < 8) {

   806→            persisted.lastSavedAt = null;
   807→          }
   808→          if (persisted.labels) {
   809→            persisted.labels = persisted.labels.map((l: any) => {

   812→              }

   814→            });
   815→          }
   816→        }
   817→        // v9: No data structure changes — just fixing the rehydration race condition
   818→        // The fix is in robustStorage.setItem blocking writes before rehydration

   821→      partialize: (state) => ({

   831→      merge: (persistedState: any, currentState: any) => {
   832→        // SIMPLE & SAFE merge: preserve persisted data, add defaults, append new seed labels
   833→        // CRITICAL: We must NEVER modify persisted user data during merge.
   834→        // The old complex mergeLabelsWithSeed function was causing data corruption.
   835→
   836→        // Guard: if persistedState is null/undefined (first visit or SSR), use currentState
   837→        if (!persistedState) {
   838→          return currentState;
   839→        }
   840→
   841→        const merged = { ...currentState, ...persistedState };
   842→
   843→        if (persistedState.labels && Array.isArray(persistedState.labels) && persistedState.labels.length > 0) {
   844→          // Step 1: Add safe defaults to ALL persisted labels (never overwrite existing values)
   845→          merged.labels = persistedState.labels.map((l: any) => ({
   846→            ...LABEL_DEFAULTS,
   847→            ...l,
   848→          }));
   849→
   850→          // Step 2: Append NEW seed labels that don't exist in persisted data
   851→          // (e.g., labels added to labels-data.json in app updates)
   852→          const persistedIds = new Set(persistedState.labels.map((l: any) => l.id));
   853→          const persistedNames = new Set(
   854→            persistedState.labels
   855→              .map((l: any) => l.name?.toLowerCase().trim())
   856→              .filter(Boolean)
   857→          );
   858→          const seedLabels = buildLabelsFromData();
   859→          for (const seed of seedLabels) {
   860→            if (!persistedIds.has(seed.id) && !persistedNames.has(seed.name.toLowerCase().trim())) {
   861→              merged.labels.push(seed);
   862→            }
   863→          }
   864→
   865→          // Step 3: Repair corrupted data (e.g., same email on too many labels)
   866→          merged.labels = repairLabelData(merged.labels);
   867→        }
   868→        // If persistedState.labels is missing/empty, we fall through to
   869→        // currentState.labels which is buildLabelsFromData() — correct for first visit
   870→
   871→        return merged;
   872→      },
   873→      onRehydrateStorage: () => (state, error) => {

   876→          // Even on error, mark as rehydrated so the app can function

   883→          );
   884→          // Mark as rehydrated — NOW writes are allowed
   885→          markRehydrated();
   886→          // Force an immediate backup to ensure backup is up-to-date
   887→          forceBackupNow();
   888→        } else {
   889→          // state is null (first visit or storage empty) — still mark rehydrated
   890→          markRehydrated();
   891→        }
   892→      },
   893→    }
   894→  )

   897→// Helper: get tier for a label in a given genre
   898→export function getLabelTier(
   899→  label: Label,
   900→  genre?: string
   901→): "top" | "mid" | "emerging" | null {
   902→  if (!genre) genre = label.genres?.[0];
   903→  if (!genre) return null;
   904→  const rank = label.rankByGenre?.[genre];
   905→  if (!rank) return label.trending ? "emerging" : null;
   906→  if (rank <= 20) return "top";
   907→  if (rank <= 50) return "mid";
   908→  return label.trending ? "emerging" : null;
   909→}
