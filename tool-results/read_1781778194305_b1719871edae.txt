   134→// ==================== ROBUST STORAGE ====================
   135→// Custom storage with data protection:
   136→// 1. Primary localStorage key
   137→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)
   138→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
   139→// 4. Rehydration guard: blocks ALL writes until Zustand persist loads data
   140→
   141→const PRIMARY_KEY = "labelpulse-storage";
   142→const BACKUP_KEY = "labelpulse-storage-backup";
   143→const SEED_LABEL_COUNT = labelData.labels.length;
   144→
   145→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   146→// Before rehydration, the store contains seed data which would overwrite user data.
   147→let _rehydrated = false;
   148→
   149→function safeJsonParse(str: string | null): any | null {
   150→  if (!str) return null;
   151→  try {
   152→    return JSON.parse(str);
   153→  } catch {
   154→    return null;
   155→  }
   156→}
   157→
   158→function safeLocalStorageGet(key: string): string | null {
   159→  try {
   160→    return localStorage.getItem(key);
   161→  } catch {
   162→    return null;
   163→  }
   164→}
   165→
   166→function safeLocalStorageSet(key: string, value: string): boolean {
   167→  try {
   168→    localStorage.setItem(key, value);
   169→    return true;
   170→  } catch (e) {
   171→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   172→    return false;
   173→  }
   174→}
   175→
   176→function safeLocalStorageRemove(key: string): void {
   177→  try {
   178→    localStorage.removeItem(key);
   179→  } catch {
   180→    // Ignore
   181→  }
   182→}

   184→/**
   185→ * Check if persisted data has user edits (emails, notes, links, etc.)
   186→ * This helps us detect if the store was accidentally reset to seed data.
   187→ */
   188→function countUserEditedLabels(labels: any[]): number {
   189→  if (!Array.isArray(labels)) return 0;
   190→  return labels.filter((l) =>
   191→    (l.emails && l.emails.length > 0) ||
   192→    (l.notes && l.notes.trim() !== "") ||
   193→    (l.website && l.website.trim() !== "") ||
   194→    (l.demoLink && l.demoLink.trim() !== "") ||
   195→    (l.socialLink && l.socialLink.trim() !== "") ||
   196→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   197→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   198→    l.isCustom === true
   199→  ).length;
   200→}
   201→
   202→function getTotalLabelCount(data: any): number {
   203→  if (!data) return 0;
   204→  if (Array.isArray(data.labels)) return data.labels.length;
   205→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   206→  return 0;
   207→}

   209→/**
   210→ * Extract user-edited label count from raw localStorage string.
   211→ * Used by the write guard to check if a write would cause data loss.
   212→ */
   213→function getUserEditCountFromRaw(raw: string | null): number {
   214→  if (!raw) return 0;
   215→  const data = safeJsonParse(raw);
   216→  if (!data) return 0;
   217→  const labels = data.state?.labels || data.labels || [];
   218→  return countUserEditedLabels(labels);
   219→}

   221→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   222→
   223→const robustStorage: StateStorage = {
   224→  getItem: (name: string): string | null => {
   225→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   226→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   227→
   228→    const primaryData = safeJsonParse(primaryRaw);
   229→    const backupData = safeJsonParse(backupRaw);
   230→
   231→    let result: string | null = null;
   232→
   233→    // If primary is missing/corrupt, try backup
   234→    if (!primaryData) {
   235→      if (backupData) {
   236→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   237→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   238→        result = backupRaw;
   239→      }
   240→      // Both empty — first visit, result stays null
   241→    } else if (backupData) {
   242→      // Both exist — check for data loss
   243→      const primaryLabelCount = getTotalLabelCount(primaryData);
   244→      const backupLabelCount = getTotalLabelCount(backupData);
   245→      const primaryUserEdits = primaryData.state?.labels
   246→        ? countUserEditedLabels(primaryData.state.labels)
   247→        : countUserEditedLabels(primaryData.labels || []);
   248→      const backupUserEdits = backupData.state?.labels
   249→        ? countUserEditedLabels(backupData.state.labels)
   250→        : countUserEditedLabels(backupData.labels || []);
   251→
   252→      // If primary has fewer labels than backup AND backup has more or equal user data
   253→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   254→        console.warn(
   255→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   256→        );
   257→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   258→        result = backupRaw;
   259→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   260→        // If primary has fewer user edits than backup, use backup
   261→        console.warn(
   262→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   263→        );
   264→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   265→        result = backupRaw;
   266→      } else {
   267→        result = primaryRaw;
   268→      }
   269→    } else {
   270→      // Primary exists, no backup — normal case
   271→      result = primaryRaw;
   272→    }
   273→
   274→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   275→    // merged data immediately after this getItem returns, and we MUST allow that write.
   276→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   277→    _rehydrated = true;
   278→
   279→    return result;
   280→  },
   281→
   282→  setItem: (name: string, value: string): void => {
   283→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   284→    // Before rehydration, the store only has seed data — writing it would overwrite
   285→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   286→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   287→    if (!_rehydrated) {
   288→      console.warn(
   289→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   290→      );
   291→      return;
   292→    }
   293→
   294→    // Write to primary immediately
   295→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   296→
   297→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   298→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   299→    safeLocalStorageSet(BACKUP_KEY, value);
   300→
   301→    if (!primaryOk) {
   302→      console.error("[LabelPulse Storage] Primary write failed!");
   303→    }
   304→  },
   305→
   306→  removeItem: (name: string): void => {
   307→    safeLocalStorageRemove(PRIMARY_KEY);
   308→    safeLocalStorageRemove(BACKUP_KEY);
   309→  },
   310→};
   311→
   312→/**
   313→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   314→ * Since backup is now written on every setItem, this is a safety net.
   315→ */
   316→export function forceBackupNow(): void {
   317→  if (typeof window === "undefined") return;
   318→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   319→  if (primaryRaw) {
   320→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   321→  }
   322→}

   398→/**
   399→ * Merge helper: preserves ALL user-editable fields from existing label
   400→ * when imported label has empty/default values for those fields.
   401→ */
   402→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
   403→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
   404→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
   405→  const hasNewRankData = Object.keys(imported.rankByGenre || {}).length > 0;
   406→  const currentRank = hasNewRankData && Object.keys(existing.rankByGenre || {}).length > 0
   407→    ? { ...existing.rankByGenre }
   408→    : existing.prevRankByGenre || {};
   409→
   410→  return {
   411→    // User-editable data — ALWAYS prefer existing if it has real data
   412→    emails: imported.emails?.length ? imported.emails : existing.emails,
   413→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
   414→    website: imported.website?.trim() || existing.website,
   415→    demoLink: imported.demoLink?.trim() || existing.demoLink,
   416→    socialLink: imported.socialLink?.trim() || existing.socialLink,
   417→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   418→    notes: imported.notes?.trim() || existing.notes,
   419→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
   420→
   421→    // Beatport/ranking data — prefer imported (it's the fresh data)
   422→    genres: imported.genres?.length ? imported.genres : existing.genres,
   423→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
   424→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
   425→    trending: imported.trending || existing.trending,
   426→    trendingRankByGenre: Object.keys(imported.trendingRankByGenre || {}).length ? imported.trendingRankByGenre : existing.trendingRankByGenre,
   427→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
   428→
   429→    // Ranking history — save current rank as "previous" when new data arrives
   430→    prevRankByGenre: currentRank,
   431→  };
   432→}

   460→/**
   461→ * Repair corrupted label data.
   462→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   463→ */
   464→function repairLabelData(labels: Label[]): Label[] {
   465→  const emailLabelCount = new Map<string, number>();
   466→  for (const l of labels) {
   467→    if (l.emails && Array.isArray(l.emails)) {
   468→      for (const e of l.emails) {
   469→        if (e && e.trim()) {
   470→          const key = e.toLowerCase().trim();
   471→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   472→        }
   473→      }
   474→    }
   475→  }
   476→
   477→  const corruptedEmails = new Set<string>();
   478→  for (const [email, count] of emailLabelCount) {
   479→    if (count > 5) {
   480→      corruptedEmails.add(email);
   481→      console.warn(
   482→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   483→      );
   484→    }
   485→  }
   486→
   487→  if (corruptedEmails.size === 0) {
   488→    return labels;
   489→  }
   490→
   491→  const emailOwner = new Map<string, string>();
   492→  for (const email of corruptedEmails) {
   493→    for (const l of labels) {
   494→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   495→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   496→        if (domain && l.name.toLowerCase().includes(domain)) {
   497→          emailOwner.set(email, l.id);
   498→          break;
   499→        }
   500→      }
   501→    }
   502→    if (!emailOwner.has(email)) {
   503→      for (const l of labels) {
   504→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   505→          emailOwner.set(email, l.id);
   506→          break;
   507→        }
   508→      }
   509→    }
   510→  }
   511→
   512→  return labels.map(l => {
   513→    if (!l.emails || !Array.isArray(l.emails)) return l;
   514→    const cleanedEmails = l.emails.filter(e => {
   515→      const key = e?.toLowerCase().trim();
   516→      if (!key) return false;
   517→      if (!corruptedEmails.has(key)) return true;
   518→      return emailOwner.get(key) === l.id;
   519→    });
   520→    const cleanedContactInfo = cleanedEmails[0] || "";
   521→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   522→      console.log(
   523→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   524→      );
   525→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   526→    }
   527→    return l;
   528→  });
   529→}

  1044→  label: Label,
  1045→  genre?: string
  1046→): "top" | "mid" | "emerging" | null {
  1047→  if (!genre) genre = label.genres?.[0];
  1048→  if (!genre) return null;
  1049→  const rank = label.rankByGenre?.[genre];
  1050→  if (!rank) return label.trending ? "emerging" : null;
  1051→  if (rank <= 20) return "top";
  1052→  if (rank <= 50) return "mid";
  1053→  return label.trending ? "emerging" : null;

  1056→// ==================== CLOUD SYNC ====================
  1057→// Sistema di sincronizzazione con Supabase.
  1058→// - Ogni modifica ai dati viene salvata su Supabase (debounced)
  1059→// - All'avvio, i dati vengono caricati dal cloud se più recenti
  1060→// - Se Supabase non è configurato, funziona solo con localStorage
  1061→
  1062→let _cloudSyncTimer: ReturnType<typeof setTimeout> | null = null;
  1063→const CLOUD_SYNC_DEBOUNCE_MS = 3000; // 3 secondi di debounce
  1064→
  1065→/**
  1066→ * Sincronizza lo stato corrente con Supabase (debounced).
  1067→ * Chiamata dopo ogni azione utente che modifica i dati.
  1068→ */
  1069→export function syncToCloud(): void {
  1070→  if (!isSupabaseConfigured()) return;
  1071→  // Skip if we're applying a remote update (avoids feedback loop)
  1072→  if (isApplyingRemoteUpdate()) return;
  1073→
  1074→  if (_cloudSyncTimer) {
  1075→    clearTimeout(_cloudSyncTimer);
  1076→  }
  1077→
  1078→  _cloudSyncTimer = setTimeout(async () => {
  1079→    _cloudSyncTimer = null;
  1080→    if (isApplyingRemoteUpdate()) return;
  1081→    const state = useAppStore.getState();
  1082→    // Update lastSavedAt to track which device last wrote
  1083→    const lastSavedAt = new Date().toISOString();
  1084→    useAppStore.setState({ lastSavedAt });
  1085→    const dataToSync = {
  1086→      labels: state.labels,
  1087→      demos: state.demos,
  1088→      activeTab: state.activeTab,
  1089→      locale: state.locale,
  1090→      userProfile: state.userProfile,
  1091→      gmailAuth: state.gmailAuth,
  1092→      rankingsUpdatedAt: state.rankingsUpdatedAt,
  1093→      lastSavedAt,
  1094→      rankingSnapshots: state.rankingSnapshots,
  1095→    };
  1096→    await saveStateToCloud(dataToSync);
  1097→  }, CLOUD_SYNC_DEBOUNCE_MS);
  1098→}

  1130→/**
  1131→ * Carica i dati dal cloud e aggiorna lo store se i dati cloud sono più recenti.
  1132→ * Chiamata una volta all'avvio dell'app, dopo la reidratazione da localStorage.
  1133→ */
  1134→export async function loadFromCloud(): Promise<void> {
  1135→  if (!isSupabaseConfigured()) {
  1136→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
  1137→    useAppStore.setState({ hasCloudSynced: true });
  1138→    return;
  1139→  }
  1140→
  1141→  try {
  1142→    const cloudData = await loadStateFromCloud();
  1143→    if (!cloudData) {
  1144→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
  1145→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
  1146→      await forceCloudSync();
  1147→      useAppStore.setState({ hasCloudSynced: true });
  1148→      // Setup realtime subscription for future updates
  1149→      setupRealtimeSubscriptionSafe();
  1150→      return;
  1151→    }
  1152→
  1153→    const cloud = cloudData as any;
  1154→    const localState = useAppStore.getState();
  1155→
  1156→    // Confronta i timestamp per decidere quali dati usare
  1157→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
  1158→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
  1159→
  1160→    if (cloudSavedAt > localSavedAt) {
  1161→      // I dati cloud sono più recenti — aggiorna lo store locale
  1162→      console.log(
  1163→        `[LabelPulse Cloud] Cloud data is newer (cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}). Updating local store.`
  1164→      );
  1165→
  1166→      // SAFETY CHECK: se il cloud è "più recente" ma ha array vuoti dove
  1167→      // il locale ha dati, NON fidarti — è probabile che il cloud sia stato
  1168→      // resettato / corrotto. Mantieni i dati locali e basta.
  1169→      const cloudHasLabels = Array.isArray(cloud.labels) && cloud.labels.length > 0;
  1170→      const cloudHasSnapshots = Array.isArray(cloud.rankingSnapshots) && cloud.rankingSnapshots.length > 0;
  1171→      const cloudHasDemos = Array.isArray(cloud.demos) && cloud.demos.length > 0;
  1172→      const localHasLabels = (localState.labels?.length ?? 0) > 0;
  1173→      const localHasSnapshots = (localState.rankingSnapshots?.length ?? 0) > 0;
  1174→      const localHasDemos = (localState.demos?.length ?? 0) > 0;
  1175→
  1176→      if (
  1177→        (localHasLabels && !cloudHasLabels) ||
  1178→        (localHasSnapshots && !cloudHasSnapshots) ||
  1179→        (localHasDemos && !cloudHasDemos)
  1180→      ) {
  1181→        console.warn(
  1182→          "[LabelPulse Cloud] CLOUD DATA LOSS DETECTED — cloud is newer but has empty arrays where local has data. Skipping merge to preserve local data. Run a manual sync to upload local to cloud."
  1183→        );
  1184→        useAppStore.setState({ hasCloudSynced: true });
  1185→        // Trigger upload of local data so cloud gets repopulated
  1186→        setTimeout(() => forceCloudSync(), 500);
  1187→        setupRealtimeSubscriptionSafe();
  1188→        return;
  1189→      }
  1190→
  1191→      // Usa la stessa logica di merge della reidratazione
  1192→      const merged = mergeCloudData(cloud, localState);
  1193→      useAppStore.setState({
  1194→        ...merged,
  1195→        hasCloudSynced: true,
  1196→      });
  1197→    } else {
  1198→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1199→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1200→      useAppStore.setState({ hasCloudSynced: true });
  1201→    }
  1202→
  1203→    // Setup realtime subscription to receive future updates from other devices
  1204→    setupRealtimeSubscriptionSafe();
  1205→  } catch (err) {
  1206→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1207→    useAppStore.setState({ hasCloudSynced: true });
  1208→  }
  1209→}

  1211→/**
  1212→ * Setup the realtime subscription safely (only on client, only if configured).
  1213→ * Re-imported lazily to avoid circular deps at module load.
  1214→ */
  1215→function setupRealtimeSubscriptionSafe(): void {
  1216→  if (typeof window === "undefined") return;
  1217→  if (!isSupabaseConfigured()) return;
  1218→  // Defer to next tick so we don't block the initial render
  1219→  setTimeout(() => {
  1220→    try {
  1221→      // Re-import to avoid circular dependency
  1222→      import("./supabase").then(({ setupRealtimeSubscription }) => {
  1223→        setupRealtimeSubscription();
  1224→      });
  1225→    } catch (err) {
  1226→      console.warn("[LabelPulse Cloud] Realtime subscription setup failed:", err);
  1227→    }
  1228→  }, 100);
  1229→}

  1231→/**
  1232→ * Merge dei dati cloud con quelli locali, preservando i dati utente.
  1233→ * Usa la stessa logica della funzione merge di Zustand persist.
  1234→ *
  1235→ * CRITICAL SAFETY RULE: mai sovrascrivere array locali non-vuoti con
  1236→ * array cloud vuoti. Se il cloud ha perso dati (es. tabella corrotta,
  1237→ * sync parziale, progetto Supabase resettato), non vogliamo propagare
  1238→ * la perdita ai dispositivi che hanno ancora i dati.
  1239→ */
  1240→function mergeCloudData(cloudData: any, localState: any): Partial<AppState> {
  1241→  const merged: Partial<AppState> = {};
  1242→
  1243→  // Labels: applica defaults e merge con seed data, come nella reidratazione
  1244→  if (cloudData.labels && Array.isArray(cloudData.labels) && cloudData.labels.length > 0) {
  1245→    merged.labels = cloudData.labels.map((l: any) => ({
  1246→      ...LABEL_DEFAULTS,
  1247→      ...l,
  1248→    }));
  1249→
  1250→    // Aggiungi seed labels mancanti
  1251→    const cloudIds = new Set(cloudData.labels.map((l: any) => l.id));
  1252→    const cloudNames = new Set(
  1253→      cloudData.labels.map((l: any) => l.name?.toLowerCase().trim()).filter(Boolean)
  1254→    );
  1255→    const seedLabels = buildLabelsFromData();
  1256→    for (const seed of seedLabels) {
  1257→      if (!cloudIds.has(seed.id) && !cloudNames.has(seed.name.toLowerCase().trim())) {
  1258→        (merged.labels as Label[]).push(seed);
  1259→      }
  1260→    }
  1261→
  1262→    // Repair corrupted data
  1263→    merged.labels = repairLabelData(merged.labels);
  1264→  } else {
  1265→    // Cloud non ha labels (o array vuoto) → mantieni locali
  1266→    merged.labels = localState.labels;
  1267→  }
  1268→
  1269→  // Demos: SOLO se cloud ha effettivamente demos, mantieni locali altrimenti
  1270→  // (before: `Array.isArray(cloudData.demos) ? cloudData.demos : localState.demos`
  1271→  //  sovrascriveva demos locali con [] se cloud aveva demos:[])
  1272→  if (Array.isArray(cloudData.demos) && cloudData.demos.length > 0) {
  1273→    merged.demos = cloudData.demos;
  1274→  } else if (Array.isArray(cloudData.demos) && (!localState.demos || localState.demos.length === 0)) {
  1275→    // Cloud ha demos:[] e locale è vuoto → ok a usare [] (no-op)
  1276→    merged.demos = cloudData.demos;
  1277→  } else {
  1278→    merged.demos = localState.demos;
  1279→  }
  1280→
  1281→  // Simple fields
  1282→  merged.activeTab = cloudData.activeTab || localState.activeTab;
  1283→  merged.locale = cloudData.locale || localState.locale;
  1284→  merged.userProfile = cloudData.userProfile || localState.userProfile;
  1285→  merged.gmailAuth = cloudData.gmailAuth || localState.gmailAuth;
  1286→  merged.rankingsUpdatedAt = cloudData.rankingsUpdatedAt ?? localState.rankingsUpdatedAt;
  1287→  merged.lastSavedAt = cloudData.lastSavedAt ?? localState.lastSavedAt;
  1288→
  1289→  // rankingSnapshots: STesso principio — non sovrascrivere snapshots locali
  1290→  // con array vuoto dal cloud. Gli snapshots sono il cuore delle classifiche
  1291→  // storiche, perderli significa perdere mesi di import Beatport/Beatstats.
  1292→  if (Array.isArray(cloudData.rankingSnapshots) && cloudData.rankingSnapshots.length > 0) {
  1293→    merged.rankingSnapshots = cloudData.rankingSnapshots;
  1294→  } else if (Array.isArray(cloudData.rankingSnapshots) && (!localState.rankingSnapshots || localState.rankingSnapshots.length === 0)) {
  1295→    // Cloud vuoto e locale vuoto → ok a usare []
  1296→    merged.rankingSnapshots = cloudData.rankingSnapshots;
  1297→  } else {
  1298→    // Locale ha dati, cloud vuoto → mantieni locali
  1299→    merged.rankingSnapshots = localState.rankingSnapshots || [];
  1300→  }
  1301→
  1302→  return merged;
  1303→}
