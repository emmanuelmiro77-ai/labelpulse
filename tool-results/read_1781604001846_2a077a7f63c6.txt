   113→// ==================== ROBUST STORAGE ====================
   114→// Custom storage with data protection:
   115→// 1. Primary localStorage key
   116→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)
   117→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
   118→// 4. Rehydration guard: blocks ALL writes until Zustand persist loads data
   119→
   120→const PRIMARY_KEY = "labelpulse-storage";
   121→const BACKUP_KEY = "labelpulse-storage-backup";
   122→const SEED_LABEL_COUNT = labelData.labels.length;
   123→
   124→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   125→// Before rehydration, the store contains seed data which would overwrite user data.
   126→let _rehydrated = false;
   127→
   128→function safeJsonParse(str: string | null): any | null {
   129→  if (!str) return null;
   130→  try {
   131→    return JSON.parse(str);
   132→  } catch {
   133→    return null;
   134→  }
   135→}
   136→
   137→function safeLocalStorageGet(key: string): string | null {
   138→  try {
   139→    return localStorage.getItem(key);
   140→  } catch {
   141→    return null;
   142→  }
   143→}
   144→
   145→function safeLocalStorageSet(key: string, value: string): boolean {
   146→  try {
   147→    localStorage.setItem(key, value);
   148→    return true;
   149→  } catch (e) {
   150→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   151→    return false;
   152→  }
   153→}
   154→
   155→function safeLocalStorageRemove(key: string): void {
   156→  try {
   157→    localStorage.removeItem(key);
   158→  } catch {
   159→    // Ignore
   160→  }
   161→}

   163→/**
   164→ * Check if persisted data has user edits (emails, notes, links, etc.)
   165→ * This helps us detect if the store was accidentally reset to seed data.
   166→ */
   167→function countUserEditedLabels(labels: any[]): number {
   168→  if (!Array.isArray(labels)) return 0;
   169→  return labels.filter((l) =>
   170→    (l.emails && l.emails.length > 0) ||
   171→    (l.notes && l.notes.trim() !== "") ||
   172→    (l.website && l.website.trim() !== "") ||
   173→    (l.demoLink && l.demoLink.trim() !== "") ||
   174→    (l.socialLink && l.socialLink.trim() !== "") ||
   175→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   176→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   177→    l.isCustom === true
   178→  ).length;
   179→}
   180→
   181→function getTotalLabelCount(data: any): number {
   182→  if (!data) return 0;
   183→  if (Array.isArray(data.labels)) return data.labels.length;
   184→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   185→  return 0;
   186→}
   187→
   188→/**
   189→ * Extract user-edited label count from raw localStorage string.
   190→ * Used by the write guard to check if a write would cause data loss.
   191→ */
   192→function getUserEditCountFromRaw(raw: string | null): number {
   193→  if (!raw) return 0;
   194→  const data = safeJsonParse(raw);
   195→  if (!data) return 0;
   196→  const labels = data.state?.labels || data.labels || [];
   197→  return countUserEditedLabels(labels);
   198→}

   200→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   201→
   202→const robustStorage: StateStorage = {
   203→  getItem: (name: string): string | null => {
   204→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   205→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   206→
   207→    const primaryData = safeJsonParse(primaryRaw);
   208→    const backupData = safeJsonParse(backupRaw);
   209→
   210→    let result: string | null = null;
   211→
   212→    // If primary is missing/corrupt, try backup
   213→    if (!primaryData) {
   214→      if (backupData) {
   215→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   216→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   217→        result = backupRaw;
   218→      }
   219→      // Both empty — first visit, result stays null
   220→    } else if (backupData) {
   221→      // Both exist — check for data loss
   222→      const primaryLabelCount = getTotalLabelCount(primaryData);
   223→      const backupLabelCount = getTotalLabelCount(backupData);
   224→      const primaryUserEdits = primaryData.state?.labels
   225→        ? countUserEditedLabels(primaryData.state.labels)
   226→        : countUserEditedLabels(primaryData.labels || []);
   227→      const backupUserEdits = backupData.state?.labels
   228→        ? countUserEditedLabels(backupData.state.labels)
   229→        : countUserEditedLabels(backupData.labels || []);
   230→
   231→      // If primary has fewer labels than backup AND backup has more or equal user data
   232→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   233→        console.warn(
   234→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   235→        );
   236→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   237→        result = backupRaw;
   238→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   239→        // If primary has fewer user edits than backup, use backup
   240→        console.warn(
   241→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   242→        );
   243→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   244→        result = backupRaw;
   245→      } else {
   246→        result = primaryRaw;
   247→      }
   248→    } else {
   249→      // Primary exists, no backup — normal case
   250→      result = primaryRaw;
   251→    }
   252→
   253→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   254→    // merged data immediately after this getItem returns, and we MUST allow that write.
   255→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   256→    _rehydrated = true;
   257→
   258→    return result;
   259→  },
   260→
   261→  setItem: (name: string, value: string): void => {
   262→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   263→    // Before rehydration, the store only has seed data — writing it would overwrite
   264→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   265→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   266→    if (!_rehydrated) {
   267→      console.warn(
   268→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   269→      );
   270→      return;
   271→    }
   272→
   273→    // Write to primary immediately
   274→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   275→
   276→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   277→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   278→    safeLocalStorageSet(BACKUP_KEY, value);
   279→
   280→    if (!primaryOk) {
   281→      console.error("[LabelPulse Storage] Primary write failed!");
   282→    }
   283→  },
   284→
   285→  removeItem: (name: string): void => {
   286→    safeLocalStorageRemove(PRIMARY_KEY);
   287→    safeLocalStorageRemove(BACKUP_KEY);
   288→  },
   289→};
   290→
   291→/**
   292→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   293→ * Since backup is now written on every setItem, this is a safety net.
   294→ */
   295→export function forceBackupNow(): void {
   296→  if (typeof window === "undefined") return;
   297→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   298→  if (primaryRaw) {
   299→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   300→  }
   301→}

   382→/**
   383→ * Merge helper: preserves ALL user-editable fields from existing label
   384→ * when imported label has empty/default values for those fields.
   385→ */
   386→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
   387→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
   388→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
   389→  const hasNewRankData = Object.keys(imported.rankByGenre || {}).length > 0;
   390→  const currentRank = hasNewRankData && Object.keys(existing.rankByGenre || {}).length > 0
   391→    ? { ...existing.rankByGenre }
   392→    : existing.prevRankByGenre || {};
   393→
   394→  return {
   395→    // User-editable data — ALWAYS prefer existing if it has real data
   396→    emails: imported.emails?.length ? imported.emails : existing.emails,
   397→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
   398→    website: imported.website?.trim() || existing.website,
   399→    demoLink: imported.demoLink?.trim() || existing.demoLink,
   400→    socialLink: imported.socialLink?.trim() || existing.socialLink,
   401→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   402→    notes: imported.notes?.trim() || existing.notes,
   403→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
   404→
   405→    // Beatport/ranking data — prefer imported (it's the fresh data)
   406→    genres: imported.genres?.length ? imported.genres : existing.genres,
   407→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
   408→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
   409→    trending: imported.trending || existing.trending,
   410→    trendingRankByGenre: Object.keys(imported.trendingRankByGenre || {}).length ? imported.trendingRankByGenre : existing.trendingRankByGenre,
   411→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
   412→
   413→    // Ranking history — save current rank as "previous" when new data arrives
   414→    prevRankByGenre: currentRank,
   415→  };
   416→}

   444→/**
   445→ * Repair corrupted label data.
   446→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   447→ */
   448→function repairLabelData(labels: Label[]): Label[] {
   449→  const emailLabelCount = new Map<string, number>();
   450→  for (const l of labels) {
   451→    if (l.emails && Array.isArray(l.emails)) {
   452→      for (const e of l.emails) {
   453→        if (e && e.trim()) {
   454→          const key = e.toLowerCase().trim();
   455→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   456→        }
   457→      }
   458→    }
   459→  }
   460→
   461→  const corruptedEmails = new Set<string>();
   462→  for (const [email, count] of emailLabelCount) {
   463→    if (count > 5) {
   464→      corruptedEmails.add(email);
   465→      console.warn(
   466→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   467→      );
   468→    }
   469→  }
   470→
   471→  if (corruptedEmails.size === 0) {
   472→    return labels;
   473→  }
   474→
   475→  const emailOwner = new Map<string, string>();
   476→  for (const email of corruptedEmails) {
   477→    for (const l of labels) {
   478→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   479→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   480→        if (domain && l.name.toLowerCase().includes(domain)) {
   481→          emailOwner.set(email, l.id);
   482→          break;
   483→        }
   484→      }
   485→    }
   486→    if (!emailOwner.has(email)) {
   487→      for (const l of labels) {
   488→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   489→          emailOwner.set(email, l.id);
   490→          break;
   491→        }
   492→      }
   493→    }
   494→  }
   495→
   496→  return labels.map(l => {
   497→    if (!l.emails || !Array.isArray(l.emails)) return l;
   498→    const cleanedEmails = l.emails.filter(e => {
   499→      const key = e?.toLowerCase().trim();
   500→      if (!key) return false;
   501→      if (!corruptedEmails.has(key)) return true;
   502→      return emailOwner.get(key) === l.id;
   503→    });
   504→    const cleanedContactInfo = cleanedEmails[0] || "";
   505→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   506→      console.log(
   507→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   508→      );
   509→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   510→    }
   511→    return l;
   512→  });
   513→}

  1018→// Helper: get tier for a label in a given genre
  1019→export function getLabelTier(
  1020→  label: Label,
  1021→  genre?: string
  1022→): "top" | "mid" | "emerging" | null {
  1023→  if (!genre) genre = label.genres?.[0];
  1024→  if (!genre) return null;
  1025→  const rank = label.rankByGenre?.[genre];
  1026→  if (!rank) return label.trending ? "emerging" : null;
  1027→  if (rank <= 20) return "top";
  1028→  if (rank <= 50) return "mid";
  1029→  return label.trending ? "emerging" : null;
  1030→}

  1032→// ==================== CLOUD SYNC ====================
  1033→// Sistema di sincronizzazione con Supabase.
  1034→// - Ogni modifica ai dati viene salvata su Supabase (debounced)
  1035→// - All'avvio, i dati vengono caricati dal cloud se più recenti
  1036→// - Quando l'utente torna nell'app (visibility change), si pulla dal cloud
  1037→// - Pull periodico ogni 2 minuti per sincronizzare tra dispositivi
  1038→// - Merge intelligente per-label: preserva le modifiche locali non presenti nel cloud

  1041→let _cloudSyncTimer: ReturnType<typeof setTimeout> | null = null;
  1042→let _cloudPullTimer: ReturnType<typeof setInterval> | null = null;
  1043→const CLOUD_SYNC_DEBOUNCE_MS = 3000; // 3 secondi di debounce
  1044→const CLOUD_PULL_INTERVAL_MS = 2 * 60 * 1000; // 2 minuti di pull periodico

  1047→/**
  1048→ * Sincronizza lo stato corrente con Supabase (debounced).
  1049→ * Chiamata dopo ogni azione utente che modifica i dati.
  1050→ */
  1051→export function syncToCloud(): void {
  1052→  if (!isSupabaseConfigured) return;
  1053→  if (!useAppStore.getState().isAuthenticated) return; // Skip se non autenticato
  1054→
  1055→  if (_cloudSyncTimer) {
  1056→    clearTimeout(_cloudSyncTimer);
  1057→  }
  1058→
  1059→  _cloudSyncTimer = setTimeout(async () => {
  1060→    _cloudSyncTimer = null;
  1061→    const state = useAppStore.getState();
  1062→    const now = new Date().toISOString();
  1063→    const dataToSync = {
  1064→      labels: state.labels,
  1065→      demos: state.demos,
  1066→      activeTab: state.activeTab,
  1067→      locale: state.locale,
  1068→      userProfile: state.userProfile,

  1070→      rankingsUpdatedAt: state.rankingsUpdatedAt,
  1071→      lastSavedAt: now,
  1072→      rankingSnapshots: state.rankingSnapshots,
  1073→    };
  1074→    const success = await saveStateToCloud(dataToSync);
  1075→    if (success) {
  1076→      useAppStore.setState({ lastSavedAt: now });
  1077→    }
  1078→  }, CLOUD_SYNC_DEBOUNCE_MS);
  1079→}

  1128→    useAppStore.setState({ hasCloudSynced: true });
  1129→    return;
  1130→  }
  1131→
  1132→  try {
  1133→    const cloudData = await loadStateFromCloud();
  1134→    if (!cloudData) {
  1135→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
  1136→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
  1137→      await forceCloudSync();
  1138→      useAppStore.setState({ hasCloudSynced: true });
  1139→      startPeriodicCloudPull();
  1140→      return;
  1141→    }
  1142→
  1143→    const cloud = cloudData as any;
  1144→    const localState = useAppStore.getState();
  1145→
  1146→    // ALWAYS do smart merge — field-by-field, non-empty values win.
  1147→    // We no longer skip merge based on timestamps because:
  1148→    // - Device A might have a newer timestamp but missing data that Device B has
  1149→    // - The "non-empty wins" merge is safe: it only ADDS data, never removes it
  1150→    console.log("[LabelPulse Cloud] Merging cloud data with local data...");
  1151→    const merged = mergeCloudData(cloud, localState);
  1152→    useAppStore.setState({
  1153→      ...merged,
  1154→      hasCloudSynced: true,
  1155→    });
  1156→
  1157→    // After merge, push the combined data back to cloud
  1158→    // so the cloud always has the most complete version
  1159→    syncToCloud();
  1160→
  1161→    // Avvia il pull periodico dal cloud
  1162→    startPeriodicCloudPull();
  1163→  } catch (err) {
  1164→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1165→    useAppStore.setState({ hasCloudSynced: true });
  1166→    startPeriodicCloudPull();
  1167→  }
  1168→}
  1169→
  1170→/**
  1171→ * Pull dal cloud quando l'utente torna nell'app (visibility change to visible).
  1172→ * Questo è il momento chiave per sincronizzare tra dispositivi:
  1173→ * se hai fatto modifiche sul PC e poi apri il telefono, qui si sincronizzano.
  1174→ */
  1175→export async function pullFromCloud(): Promise<void> {
  1176→  if (!isSupabaseConfigured) return;
  1177→  if (!useAppStore.getState().isAuthenticated) return; // Skip se non autenticato
  1178→  if (_isSyncing) return; // Previene sync concorrenti
  1179→
  1180→  _isSyncing = true;
  1181→  try {
  1182→    const cloudData = await loadStateFromCloud();
  1183→    if (!cloudData) {
  1184→      _isSyncing = false;
  1185→      return;
  1186→    }
  1187→
  1188→    const cloud = cloudData as any;
  1189→    const localState = useAppStore.getState();
  1190→
  1191→    // ALWAYS do smart merge — same as loadFromCloud
  1192→    // Non-empty values win, so this only adds missing data, never removes it
  1193→    console.log("[LabelPulse Cloud] Pull: merging cloud with local...");
  1194→    const merged = mergeCloudData(cloud, localState);
  1195→    useAppStore.setState(merged);
  1196→
  1197→    // Push the merged data back so cloud stays complete
  1198→    syncToCloud();
  1199→  } catch (err) {
  1200→    console.error("[LabelPulse Cloud] Pull from cloud failed:", err);
  1201→  } finally {
  1202→    _isSyncing = false;
  1203→  }
  1204→}
  1205→
  1206→/**
  1207→ * Avvia il pull periodico dal cloud (ogni 2 minuti).
  1208→ * Permette di sincronizzare automaticamente tra dispositivi
  1209→ * anche se l'app rimane aperta su entrambi.
  1210→ */
  1211→function startPeriodicCloudPull(): void {
  1212→  if (_cloudPullTimer) return; // Già avviato
  1213→  if (!isSupabaseConfigured) return;
  1214→
  1215→  _cloudPullTimer = setInterval(() => {
  1216→    pullFromCloud();
  1217→  }, CLOUD_PULL_INTERVAL_MS);
  1218→
  1219→  console.log(`[LabelPulse Cloud] Periodic pull started (every ${CLOUD_PULL_INTERVAL_MS / 1000}s)`);
  1220→}
  1221→
  1222→/**
  1223→ * Ferma il pull periodico (usato per cleanup).
  1224→ */
  1225→export function stopPeriodicCloudPull(): void {
  1226→  if (_cloudPullTimer) {
  1227→    clearInterval(_cloudPullTimer);
  1228→    _cloudPullTimer = null;
  1229→    console.log("[LabelPulse Cloud] Periodic pull stopped");
  1230→  }
  1231→}
  1232→
  1233→/**
  1234→ * Merge intelligente dei dati cloud con quelli locali.
  1235→ * - Per le labels: merge per-ID, preservando le modifiche locali che il cloud non ha
  1236→ * - Per le demos: cloud vince se più recente (sono transazioni, non editing continuo)
  1237→ * - Per i campi semplici: cloud vince se più recente
  1238→ * - Aggiunge sempre le seed labels mancanti
  1239→ */
  1240→function mergeCloudData(cloudData: any, localState: any): Partial<AppState> {
  1241→  const merged: Partial<AppState> = {};
  1242→
  1243→  // Labels: merge per-ID intelligente
  1244→  if (cloudData.labels && Array.isArray(cloudData.labels) && cloudData.labels.length > 0) {
  1245→    // Crea una mappa delle label locali per ID
  1246→    const localLabelMap = new Map<string, Label>();
  1247→    for (const l of localState.labels || []) {
  1248→      if (l.id) localLabelMap.set(l.id, l);
  1249→    }
  1250→
  1251→    // Per ogni label cloud, fai merge con quella locale
  1252→    const mergedLabels: Label[] = cloudData.labels.map((cloudLabel: any) => {
  1253→      const localLabel = localLabelMap.get(cloudLabel.id);
  1254→      const cloudWithDefaults: Label = {
  1255→        ...LABEL_DEFAULTS,
  1256→        ...cloudLabel,
  1257→      };
  1258→
  1259→      if (!localLabel) {
  1260→        // Label nuova dal cloud (non esiste localmente)
  1261→        return cloudWithDefaults;
  1262→      }
  1263→
  1264→      // Label esistente sia in cloud che localmente: merge intelligente
  1265→      // Per i campi utente-editabili: se il cloud ha un valore non-vuoto, usalo;
  1266→      // se il cloud ha un valore vuoto ma il locale ha dati, mantieni il locale.
  1267→      // Questo preserva le modifiche locali che il cloud non ha ancora.
  1268→      return {
  1269→        ...cloudWithDefaults,
  1270→        // Merge intelligente dei campi utente-editabili
  1271→        emails: (cloudLabel.emails?.length > 0) ? cloudLabel.emails : localLabel.emails,
  1272→        contactInfo: cloudLabel.contactInfo?.trim() || localLabel.contactInfo,
  1273→        website: cloudLabel.website?.trim() || localLabel.website,
  1274→        demoLink: cloudLabel.demoLink?.trim() || localLabel.demoLink,
  1275→        socialLink: cloudLabel.socialLink?.trim() || localLabel.socialLink,
  1276→        soundcloudLink: cloudLabel.soundcloudLink?.trim() || localLabel.soundcloudLink,
  1277→        beatportLink: cloudLabel.beatportLink?.trim() || localLabel.beatportLink,
  1278→        customLinks: (cloudLabel.customLinks?.length > 0) ? cloudLabel.customLinks : localLabel.customLinks,
  1279→        notes: cloudLabel.notes?.trim() || localLabel.notes,
  1280→        // Ranking data: always prefer cloud (it's the authoritative source)
  1281→        rankByGenre: Object.keys(cloudLabel.rankByGenre || {}).length > 0 ? cloudLabel.rankByGenre : localLabel.rankByGenre,
  1282→        pointsByGenre: Object.keys(cloudLabel.pointsByGenre || {}).length > 0 ? cloudLabel.pointsByGenre : localLabel.pointsByGenre,
  1283→        prevRankByGenre: Object.keys(cloudLabel.prevRankByGenre || {}).length > 0 ? cloudLabel.prevRankByGenre : localLabel.prevRankByGenre,
  1284→      };
  1285→    });
  1286→
  1287→    // Aggiungi label locali che non esistono nel cloud (custom labels create localmente)
  1288→    const cloudIds = new Set(cloudData.labels.map((l: any) => l.id));
  1289→    for (const localLabel of localState.labels || []) {
  1290→      if (localLabel.id && !cloudIds.has(localLabel.id) && localLabel.isCustom) {
  1291→        mergedLabels.push(localLabel);
  1292→      }
  1293→    }
  1294→
  1295→    // Aggiungi seed labels mancanti (né in cloud né in locale)
  1296→    const allIds = new Set(mergedLabels.map((l: any) => l.id));
  1297→    const allNames = new Set(
  1298→      mergedLabels.map((l: any) => l.name?.toLowerCase().trim()).filter(Boolean)
  1299→    );
  1300→    const seedLabels = buildLabelsFromData();
  1301→    for (const seed of seedLabels) {
  1302→      if (!allIds.has(seed.id) && !allNames.has(seed.name.toLowerCase().trim())) {
  1303→        mergedLabels.push(seed);
  1304→      }
  1305→    }
  1306→
  1307→    // Repair corrupted data
  1308→    merged.labels = repairLabelData(mergedLabels);
  1309→  } else {
  1310→    merged.labels = localState.labels;
  1311→  }
  1312→
  1313→  // Demos: cloud wins if available (demos are transactional)
  1314→  merged.demos = Array.isArray(cloudData.demos) ? cloudData.demos : localState.demos;
  1315→
  1316→  // Simple fields: prefer cloud data (cloud is newer)
  1317→  merged.activeTab = cloudData.activeTab || localState.activeTab;
  1318→  merged.locale = cloudData.locale || localState.locale;
  1319→
  1320→  // UserProfile: field-by-field merge — non-empty value wins (from either side)
  1321→  // This prevents empty cloud strings from overwriting populated local values
  1322→  const cloudProfile = cloudData.userProfile || {};
  1323→  const localProfile = localState.userProfile || {};
  1324→  merged.userProfile = {
  1325→    artistName: cloudProfile.artistName?.trim() || localProfile.artistName || "",
  1326→    scLink: cloudProfile.scLink?.trim() || localProfile.scLink || "",
  1327→    bio: cloudProfile.bio?.trim() || localProfile.bio || "",
  1328→    email: cloudProfile.email?.trim() || localProfile.email || "",
  1329→    photoUrl: cloudProfile.photoUrl?.trim() || localProfile.photoUrl || "",
  1330→    links: (cloudProfile.links?.length > 0) ? cloudProfile.links : (localProfile.links || []),
  1331→  };
  1332→
  1333→  // GmailAuth: only use cloud if actually connected; otherwise keep local
  1334→  merged.gmailAuth = cloudData.gmailAuth && cloudData.gmailAuth.isConnected
  1335→    ? cloudData.gmailAuth
  1336→    : (localState.gmailAuth && localState.gmailAuth.isConnected
  1337→      ? localState.gmailAuth
  1338→      : { isConnected: false, email: "", accessToken: "", expiresAt: 0 });
  1339→
  1340→  merged.rankingsUpdatedAt = cloudData.rankingsUpdatedAt ?? localState.rankingsUpdatedAt;
  1341→  merged.lastSavedAt = cloudData.lastSavedAt ?? localState.lastSavedAt;
  1342→  merged.rankingSnapshots = Array.isArray(cloudData.rankingSnapshots) && cloudData.rankingSnapshots.length > 0
  1343→    ? cloudData.rankingSnapshots
  1344→    : (localState.rankingSnapshots || []);
  1345→
  1346→  return merged;
  1347→}
