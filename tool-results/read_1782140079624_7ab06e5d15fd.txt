   212→function safeJsonParse(str: string | null): any | null {
   213→  if (!str) return null;
   214→  try {
   215→    return JSON.parse(str);
   216→  } catch {
   217→    return null;
   218→  }
   219→}
   220→
   221→function safeLocalStorageGet(key: string): string | null {
   222→  try {
   223→    return localStorage.getItem(key);
   224→  } catch {
   225→    return null;
   226→  }
   227→}
   228→
   229→function safeLocalStorageSet(key: string, value: string): boolean {
   230→  try {
   231→    localStorage.setItem(key, value);
   232→    return true;
   233→  } catch (e) {
   234→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   235→    return false;
   236→  }
   237→}

   239→function safeLocalStorageRemove(key: string): void {
   240→  try {
   241→    localStorage.removeItem(key);
   242→  } catch {
   243→    // Ignore
   244→  }

   247→/**
   248→ * Extract rankingSnapshots array from a raw localStorage JSON string.
   249→ * Returns [] if not found / invalid.
   250→ */
   251→function extractSnapshotsFromRaw(raw: string | null): RankingSnapshot[] {
   252→  if (!raw) return [];
   253→  const data = safeJsonParse(raw);
   254→  if (!data) return [];
   255→  const snaps = data.state?.rankingSnapshots ?? data.rankingSnapshots;
   256→  return Array.isArray(snaps) ? snaps : [];
   257→}

   259→/**
   260→ * Merge two snapshot arrays, deduping by `id` first, then by `timestamp+source`.
   261→ * Always returns ascending-by-timestamp order.
   262→ * Used by: importData (backup file), robustStorage.getItem (safety-net restore),
   263→ * and loadFromCloud (cloud merge).
   264→ */
   265→function mergeSnapshots(
   266→  base: RankingSnapshot[] | null | undefined,
   267→  incoming: RankingSnapshot[] | null | undefined
   268→): RankingSnapshot[] {
   269→  const a = Array.isArray(base) ? base : [];
   270→  const b = Array.isArray(incoming) ? incoming : [];
   271→  const seen = new Set<string>();
   272→  const out: RankingSnapshot[] = [];
   273→  for (const s of [...a, ...b]) {
   274→    if (!s || typeof s !== "object") continue;
   275→    const key = s.id || `${s.timestamp}|${s.source}`;
   276→    if (seen.has(key)) continue;
   277→    seen.add(key);
   278→    out.push(s);
   279→  }
   280→  out.sort(
   281→    (x, y) =>
   282→      new Date(x.timestamp || 0).getTime() - new Date(y.timestamp || 0).getTime()
   283→  );
   284→  return out;
   285→}

   293→/**
   294→ * Check if persisted data has user edits (emails, notes, links, etc.)
   295→ * This helps us detect if the store was accidentally reset to seed data.
   296→ */
   297→function countUserEditedLabels(labels: any[]): number {
   298→  if (!Array.isArray(labels)) return 0;
   299→  return labels.filter((l) =>
   300→    (l.emails && l.emails.length > 0) ||
   301→    (l.notes && l.notes.trim() !== "") ||
   302→    (l.website && l.website.trim() !== "") ||
   303→    (l.demoLink && l.demoLink.trim() !== "") ||
   304→    (l.socialLink && l.socialLink.trim() !== "") ||
   305→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   306→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   307→    l.isCustom === true
   308→  ).length;
   309→}

   311→function getTotalLabelCount(data: any): number {
   312→  if (!data) return 0;
   313→  if (Array.isArray(data.labels)) return data.labels.length;
   314→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   315→  return 0;
   316→}

   318→/**
   319→ * Extract user-edited label count from raw localStorage string.
   320→ * Used by the write guard to check if a write would cause data loss.
   321→ */
   322→function getUserEditCountFromRaw(raw: string | null): number {
   323→  if (!raw) return 0;
   324→  const data = safeJsonParse(raw);
   325→  if (!data) return 0;
   326→  const labels = data.state?.labels || data.labels || [];
   327→  return countUserEditedLabels(labels);

   330→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   331→
   332→const robustStorage: StateStorage = {
   333→  getItem: (name: string): string | null => {
   334→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   335→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   336→
   337→    const primaryData = safeJsonParse(primaryRaw);
   338→    const backupData = safeJsonParse(backupRaw);
   339→
   340→    let result: string | null = null;
   341→    let usedSidecarSnapshots = false;
   342→
   343→    // If primary is missing/corrupt, try backup
   344→    if (!primaryData) {
   345→      if (backupData) {
   346→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   347→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   348→        result = backupRaw;
   349→      }
   350→      // Both empty — first visit, result stays null
   351→    } else if (backupData) {
   352→      // Both exist — check for data loss
   353→      const primaryLabelCount = getTotalLabelCount(primaryData);
   354→      const backupLabelCount = getTotalLabelCount(backupData);
   355→      const primaryUserEdits = primaryData.state?.labels
   356→        ? countUserEditedLabels(primaryData.state.labels)
   357→        : countUserEditedLabels(primaryData.labels || []);
   358→      const backupUserEdits = backupData.state?.labels
   359→        ? countUserEditedLabels(backupData.state.labels)
   360→        : countUserEditedLabels(backupData.labels || []);
   361→
   362→      // If primary has fewer labels than backup AND backup has more or equal user data
   363→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   364→        console.warn(
   365→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   366→        );
   367→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   368→        result = backupRaw;
   369→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   370→        // If primary has fewer user edits than backup, use backup
   371→        console.warn(
   372→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   373→        );
   374→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   375→        result = backupRaw;
   376→      } else {
   377→        result = primaryRaw;
   378→      }
   379→    } else {
   380→      // Primary exists, no backup — normal case
   381→      result = primaryRaw;
   382→    }
   383→
   384→    // CRITICAL SAFETY NET for rankingSnapshots:
   385→    // If the chosen result has 0 snapshots but the dedicated SNAPSHOTS_BACKUP_KEY
   386→    // has snapshots, splice them in. This handles the scenario where the main
   387→    // store got wiped (bad cloud sync, migration bug, partial overwrite) but the
   388→    // snapshots sidecar survived. Without this, the user loses months of
   389→    // "storico caricamenti" and the Classifiche page resets to "first time".
   390→    try {
   391→      const currentSnaps = extractSnapshotsFromRaw(result);
   392→      const sidecarRaw = safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY);
   393→      const sidecarSnaps = extractSnapshotsFromRaw(sidecarRaw);
   394→      if (sidecarSnaps.length > 0 && sidecarSnaps.length > currentSnaps.length) {
   395→        console.warn(
   396→          `[LabelPulse Storage] SNAPSHOTS RECOVERY: result has ${currentSnaps.length} snapshots, sidecar has ${sidecarSnaps.length}. Merging sidecar into result.`
   397→        );
   398→        const mergedSnaps = mergeSnapshots(currentSnaps, sidecarSnaps);
   399→        const parsed = result ? safeJsonParse(result) : {};
   400→        if (parsed && typeof parsed === "object") {
   401→          if (parsed.state && typeof parsed.state === "object") {
   402→            parsed.state.rankingSnapshots = mergedSnaps;
   403→          } else {
   404→            parsed.rankingSnapshots = mergedSnaps;
   405→          }
   406→          result = JSON.stringify(parsed);
   407→          usedSidecarSnapshots = true;
   408→        }
   409→      }
   410→    } catch (e) {
   411→      console.warn("[LabelPulse Storage] Snapshots sidecar recovery failed:", e);
   412→    }
   413→
   414→    // CRITICAL SAFETY NET for userProfile:
   415→    // Same logic as snapshots. If the chosen result has an empty/seed profile
   416→    // but PROFILE_BACKUP_KEY has real data, splice it in. This is the user's
   417→    // identity card — losing artistName, bio, email, links is catastrophic.
   418→    try {
   419→      const currentProfile = extractProfileFromRaw(result);
   420→      const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   421→      if (sidecarProfile && !currentProfile) {
   422→        console.warn(
   423→          `[LabelPulse Storage] PROFILE RECOVERY: result has empty profile, sidecar has data (artistName=${!!sidecarProfile.artistName}, bio=${!!sidecarProfile.bio}). Merging sidecar into result.`
   424→        );
   425→        const parsed = result ? safeJsonParse(result) : {};
   426→        if (parsed && typeof parsed === "object") {
   427→          if (parsed.state && typeof parsed.state === "object") {
   428→            parsed.state.userProfile = sidecarProfile;
   429→          } else {
   430→            parsed.userProfile = sidecarProfile;
   431→          }
   432→          result = JSON.stringify(parsed);
   433→        }
   434→      } else if (sidecarProfile && currentProfile) {
   435→        // Both have data — merge (sidecar fills empty fields in current)
   436→        const mergedProfile = mergeProfiles(sidecarProfile, currentProfile);
   437→        const parsed = result ? safeJsonParse(result) : {};
   438→        if (parsed && typeof parsed === "object" && mergedProfile) {
   439→          if (parsed.state && typeof parsed.state === "object") {
   440→            parsed.state.userProfile = mergedProfile;
   441→          } else {
   442→            parsed.userProfile = mergedProfile;
   443→          }
   444→          result = JSON.stringify(parsed);
   445→        }
   446→      }
   447→    } catch (e) {
   448→      console.warn("[LabelPulse Storage] Profile sidecar recovery failed:", e);
   449→    }
   450→
   451→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   452→    // merged data immediately after this getItem returns, and we MUST allow that write.
   453→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   454→    _rehydrated = true;
   455→
   456→    if (usedSidecarSnapshots) {
   457→      console.info("[LabelPulse Storage] rankingSnapshots restored from sidecar backup.");
   458→    }
   459→
   460→    return result;
   461→  },
   462→
   463→  setItem: (name: string, value: string): void => {
   464→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   465→    // Before rehydration, the store only has seed data — writing it would overwrite
   466→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   467→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   468→    if (!_rehydrated) {
   469→      console.warn(
   470→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   471→      );
   472→      return;
   473→    }
   474→
   475→    // Write to primary immediately
   476→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   477→
   478→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   479→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   480→    safeLocalStorageSet(BACKUP_KEY, value);
   481→
   482→    // CRITICAL: also persist rankingSnapshots to a dedicated slot.
   483→    // This is the user's "storico caricamenti" — months of imports that cannot
   484→    // be regenerated. Keeping it in a separate key means even a catastrophic
   485→    // main-store wipe (bad cloud sync, migration bug, quota error) leaves the
   486→    // snapshots recoverable.
   487→    try {
   488→      const snaps = extractSnapshotsFromRaw(value);
   489→      if (snaps.length > 0) {
   490→        // Merge with whatever's already in the snapshots slot — NEVER overwrite
   491→        // existing snapshots with fewer snapshots (would happen if the current
   492→        // write had a partial/empty snapshot array due to some upstream bug).
   493→        const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   494→        const merged = mergeSnapshots(existing, snaps);
   495→        safeLocalStorageSet(
   496→          SNAPSHOTS_BACKUP_KEY,
   497→          JSON.stringify(merged)
   498→        );
   499→      }
   500→    } catch (e) {
   501→      console.warn("[LabelPulse Storage] Snapshots-sidecar write failed:", e);
   502→    }
   503→
   504→    // CRITICAL: also persist userProfile to a dedicated slot.
   505→    // Same pattern as snapshots. The profile is the user's identity card —
   506→    // artistName, bio, email, social links, photo. We can NEVER lose this,
   507→    // even if cloud sync wipes the main store.
   508→    try {
   509→      const profile = extractProfileFromRaw(value);
   510→      if (profile) {
   511→        const existing = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   512→        const merged = mergeProfiles(existing, profile);
   513→        if (merged) {
   514→          safeLocalStorageSet(
   515→            PROFILE_BACKUP_KEY,
   516→            JSON.stringify({ userProfile: merged })
   517→          );
   518→        }
   519→      }
   520→    } catch (e) {
   521→      console.warn("[LabelPulse Storage] Profile-sidecar write failed:", e);
   522→    }
   523→
   524→    if (!primaryOk) {
   525→      console.error("[LabelPulse Storage] Primary write failed!");
   526→    }
   527→  },
   528→
   529→  removeItem: (name: string): void => {
   530→    safeLocalStorageRemove(PRIMARY_KEY);
   531→    safeLocalStorageRemove(BACKUP_KEY);
   532→    // NOTE: do NOT remove SNAPSHOTS_BACKUP_KEY or PROFILE_BACKUP_KEY here.
   533→    // These are the user's permanent records (months of ranking history +
   534→    // artist identity) — clearing the main store should NEVER nuke them.
   535→    // The only way to clear them is an explicit user action ("reset all data"
   536→    // button) which calls safeLocalStorageRemove directly.
   537→  },
   538→};

   541→ * Force-write a backup immediately (e.g., on visibility change or before unload).

   546→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   547→  if (primaryRaw) {
   548→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);

   555→    safeLocalStorageSet(SNAPSHOTS_BACKUP_KEY, JSON.stringify(merged));
   556→  }

   592→/**
   593→ * Extract userProfile from a raw JSON string (main store JSON, backup JSON,
   594→ * or sidecar JSON). Returns null if missing/invalid.
   595→ */
   596→function extractProfileFromRaw(raw: string | null): any | null {
   597→  if (!raw) return null;
   598→  try {
   599→    const parsed = JSON.parse(raw);
   600→    const profile = parsed?.state?.userProfile ?? parsed?.userProfile ?? null;
   601→    if (profile && typeof profile === "object") {
   602→      // Must have at least one non-empty field to be considered "real"
   603→      const hasData =
   604→        !!profile.artistName ||
   605→        !!profile.bio ||
   606→        !!profile.email ||
   607→        !!profile.scLink ||
   608→        !!profile.photoUrl ||
   609→        (Array.isArray(profile.links) && profile.links.length > 0) ||
   610→        !!profile.cyaniteApiToken ||
   611→        !!profile.supabaseUrl;
   612→      if (hasData) return profile;
   613→    }
   614→  } catch {
   615→    /* ignore */
   616→  }
   617→  return null;
   618→}
   619→
   620→/**
   621→ * Merge two user profiles — non-empty fields win.
   622→ * This is the safety net: if the cloud ever sends a profile with empty
   623→ * artistName, but the sidecar has the real artistName, we keep the real one.
   624→ */
   625→export function mergeProfiles(base: any | null, incoming: any | null): any | null {
   626→  if (!base && !incoming) return null;
   627→  if (!base) return incoming;
   628→  if (!incoming) return base;
   629→  const merged: any = { ...base };
   630→  for (const k of Object.keys(incoming)) {
   631→    const v = incoming[k];
   632→    if (k === "links") {
   633→      // Merge link arrays (dedupe by type|value)
   634→      const a = Array.isArray(base.links) ? base.links : [];
   635→      const b = Array.isArray(v) ? v : [];
   636→      const seen = new Set<string>();
   637→      const out: { type: string; value: string }[] = [];
   638→      for (const link of [...a, ...b]) {
   639→        if (!link || typeof link !== "object") continue;
   640→        const key = `${link.type}|${link.value}`;
   641→        if (seen.has(key)) continue;
   642→        seen.add(key);
   643→        out.push(link);
   644→      }
   645→      merged.links = out;
   646→    } else if (typeof v === "string") {
   647→      // Non-empty string wins over empty
   648→      if (v && !merged[k]) merged[k] = v;
   649→    } else {
   650→      // For any other type, prefer defined over undefined
   651→      if (v !== undefined && v !== null && !merged[k]) merged[k] = v;
   652→    }
   653→  }
   654→  return merged;
   655→}

   787→/**
   788→ * Merge incoming artists (from a fresh scrape) with existing artists
   789→ * (already in the store from previous scrapes).
   790→ *
   791→ * Strategy:
   792→ * - Key by artist.id (e.g. "bp_6824"). Same Beatport id = same artist.
   793→ * - For each artist that exists: union genres, union labelsPublishedOn,
   794→ *   MERGE tracksByGenre by track.id (preserve tracks from older scrapes,
   795→ *   update with new position info), sum totalPoints, min bestPosition.
   796→ * - firstSeenAt preserved from existing; lastSeenAt = now.
   797→ * - Trending recomputed fresh from current data.
   798→ * - New artists: appended with firstSeenAt = lastSeenAt = now.
   799→ *
   800→ * This is what makes historical data accumulate: each scrape adds new
   801→ * tracks to artists' tracksByGenre without losing previously-seen tracks.
   802→ */
   803→function mergeArtists(existing: Artist[], incoming: Artist[], now: string): Artist[] {
   804→  const map = new Map<string, Artist>();
   805→
   806→  // Seed with existing
   807→  for (const a of existing) {
   808→    map.set(a.id, { ...a });
   809→  }
   810→
   811→  for (const inc of incoming) {
   812→    const ex = map.get(inc.id);
   813→    if (!ex) {
   814→      // New artist — first time we see them
   815→      map.set(inc.id, {
   816→        ...inc,
   817→        firstSeenAt: now,
   818→        lastSeenAt: now,
   819→      });
   820→      continue;
   821→    }
   822→
   823→    // Existing artist — merge fields
   824→    // Union genres
   825→    const genres = Array.from(new Set([...(ex.genres || []), ...(inc.genres || [])]));
   826→
   827→    // Union labelsPublishedOn
   828→    const labelsPublishedOn = Array.from(
   829→      new Set([...(ex.labelsPublishedOn || []), ...(inc.labelsPublishedOn || [])])
   830→    );
   831→
   832→    // Merge tracksByGenre — preserve old tracks, add/update from incoming
   833→    const tracksByGenre: Record<string, ArtistTrack[]> = { ...(ex.tracksByGenre || {}) };
   834→    for (const genre of Object.keys(inc.tracksByGenre || {})) {
   835→      const incTracks = inc.tracksByGenre[genre] || [];
   836→      const exTracks = tracksByGenre[genre] || [];
   837→      const byId = new Map<number, ArtistTrack>();
   838→      // Seed with existing
   839→      for (const t of exTracks) byId.set(t.id, t);
   840→      // Update/add with incoming (incoming overrides — fresh positions)
   841→      for (const t of incTracks) byId.set(t.id, t);
   842→      tracksByGenre[genre] = Array.from(byId.values()).sort((a, b) => a.position - b.position);
   843→    }
   844→
   845→    // Recompute totalPoints & bestPosition from current incoming data
   846→    // (the existing totals were from older scrapes — we want the snapshot
   847→    // from THIS import to drive the points, otherwise the totals just
   848→    // accumulate forever and "points" loses meaning).
   849→    // BUT: keep max totalPoints seen (artists who peaked before should
   850→    // keep credit). Actually no — let's use the LATEST totalPoints,
   851→    // since that's the current ranking signal. Old tracks are still in
   852→    // tracksByGenre for browsing, but the points reflect "right now".
   853→    const totalPoints = inc.totalPoints;
   854→    const bestPosition = Math.min(ex.bestPosition, inc.bestPosition);
   855→
   856→    // Trending: use incoming (fresh computation)
   857→    const trending = inc.trending;
   858→    const trendingRankByGenre = inc.trendingRankByGenre;
   859→    const trendingPointsByGenre = inc.trendingPointsByGenre;
   860→
   861→    // isRemixerOnly: false if EITHER snapshot has them as primary
   862→    const isRemixerOnly = ex.isRemixerOnly && inc.isRemixerOnly;
   863→
   864→    map.set(inc.id, {
   865→      ...ex,
   866→      ...inc,
   867→      genres,
   868→      labelsPublishedOn,
   869→      tracksByGenre,
   870→      totalPoints,
   871→      bestPosition,
   872→      trending,
   873→      trendingRankByGenre,
   874→      trendingPointsByGenre,
   875→      isRemixerOnly,
   876→      firstSeenAt: ex.firstSeenAt || now,
   877→      lastSeenAt: now,
   878→    });
   879→  }
   880→
   881→  return Array.from(map.values());
   882→}

   888→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
   889→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
   890→  // This allows the Rankings page to show movement arrows (↑↓) between imports.

   892→  const currentRank = hasNewRankData && Object.keys(existing.rankByGenre || {}).length > 0
   893→    ? { ...existing.rankByGenre }

   895→
   896→  return {
   897→    // User-editable data — ALWAYS prefer existing if it has real data
   898→    emails: imported.emails?.length ? imported.emails : existing.emails,
   899→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
   900→    website: imported.website?.trim() || existing.website,

   902→    socialLink: imported.socialLink?.trim() || existing.socialLink,
   903→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   904→    notes: imported.notes?.trim() || existing.notes,
   905→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
   906→
   907→    // Beatport/ranking data — prefer imported (it's the fresh data)
   908→    genres: imported.genres?.length ? imported.genres : existing.genres,
   909→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
   910→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
   911→    trending: imported.trending || existing.trending,

   913→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
   914→
   915→    // Ranking history — save current rank as "previous" when new data arrives
   916→    prevRankByGenre: currentRank,
   917→  };

   946→/**
   947→ * Repair corrupted label data.
   948→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   949→ */
   950→function repairLabelData(labels: Label[]): Label[] {
   951→  const emailLabelCount = new Map<string, number>();
   952→  for (const l of labels) {
   953→    if (l.emails && Array.isArray(l.emails)) {
   954→      for (const e of l.emails) {
   955→        if (e && e.trim()) {
   956→          const key = e.toLowerCase().trim();
   957→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   958→        }
   959→      }
   960→    }
   961→  }
   962→
   963→  const corruptedEmails = new Set<string>();
   964→  for (const [email, count] of emailLabelCount) {
   965→    if (count > 5) {
   966→      corruptedEmails.add(email);
   967→      console.warn(
   968→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   969→      );
   970→    }
   971→  }
   972→
   973→  if (corruptedEmails.size === 0) {
   974→    return labels;
   975→  }
   976→
   977→  const emailOwner = new Map<string, string>();
   978→  for (const email of corruptedEmails) {
   979→    for (const l of labels) {
   980→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   981→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   982→        if (domain && l.name.toLowerCase().includes(domain)) {
   983→          emailOwner.set(email, l.id);
   984→          break;
   985→        }
   986→      }
   987→    }
   988→    if (!emailOwner.has(email)) {
   989→      for (const l of labels) {
   990→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   991→          emailOwner.set(email, l.id);
   992→          break;
   993→        }
   994→      }
   995→    }
   996→  }
   997→
   998→  return labels.map(l => {
   999→    if (!l.emails || !Array.isArray(l.emails)) return l;
  1000→    const cleanedEmails = l.emails.filter(e => {
  1001→      const key = e?.toLowerCase().trim();
  1002→      if (!key) return false;
  1003→      if (!corruptedEmails.has(key)) return true;
  1004→      return emailOwner.get(key) === l.id;
  1005→    });
  1006→    const cleanedContactInfo = cleanedEmails[0] || "";
  1007→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
  1008→      console.log(
  1009→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
  1010→      );
  1011→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
  1012→    }
  1013→    return l;
  1014→  });
  1015→}
