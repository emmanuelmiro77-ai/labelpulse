   113→// ==================== ROBUST STORAGE ====================
   114→// Custom storage with data protection:
   115→// 1. Primary localStorage key
   116→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)
   117→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
   118→// 4. Rehydration guard: blocks ALL writes until Zustand persist loads data
   119→
   120→const PRIMARY_KEY = "labelpulse-storage";
   121→const BACKUP_KEY = "labelpulse-storage-backup";
   122→const SEED_LABEL_COUNT = labelData.labels.length;
   123→
   124→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   125→// Before rehydration, the store contains seed data which would overwrite user data.
   126→let _rehydrated = false;
   127→
   128→function safeJsonParse(str: string | null): any | null {
   129→  if (!str) return null;
   130→  try {
   131→    return JSON.parse(str);
   132→  } catch {
   133→    return null;
   134→  }
   135→}
   136→
   137→function safeLocalStorageGet(key: string): string | null {
   138→  try {
   139→    return localStorage.getItem(key);
   140→  } catch {
   141→    return null;
   142→  }
   143→}
   144→
   145→function safeLocalStorageSet(key: string, value: string): boolean {
   146→  try {
   147→    localStorage.setItem(key, value);
   148→    return true;
   149→  } catch (e) {
   150→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   151→    return false;
   152→  }
   153→}
   154→
   155→function safeLocalStorageRemove(key: string): void {
   156→  try {
   157→    localStorage.removeItem(key);
   158→  } catch {
   159→    // Ignore
   160→  }
   161→}

   163→/**
   164→ * Check if persisted data has user edits (emails, notes, links, etc.)
   165→ * This helps us detect if the store was accidentally reset to seed data.
   166→ */
   167→function countUserEditedLabels(labels: any[]): number {
   168→  if (!Array.isArray(labels)) return 0;
   169→  return labels.filter((l) =>
   170→    (l.emails && l.emails.length > 0) ||
   171→    (l.notes && l.notes.trim() !== "") ||
   172→    (l.website && l.website.trim() !== "") ||
   173→    (l.demoLink && l.demoLink.trim() !== "") ||
   174→    (l.socialLink && l.socialLink.trim() !== "") ||
   175→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   176→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   177→    l.isCustom === true
   178→  ).length;
   179→}
   180→
   181→function getTotalLabelCount(data: any): number {
   182→  if (!data) return 0;
   183→  if (Array.isArray(data.labels)) return data.labels.length;
   184→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   185→  return 0;
   186→}
   187→
   188→/**
   189→ * Extract user-edited label count from raw localStorage string.
   190→ * Used by the write guard to check if a write would cause data loss.
   191→ */
   192→function getUserEditCountFromRaw(raw: string | null): number {
   193→  if (!raw) return 0;
   194→  const data = safeJsonParse(raw);
   195→  if (!data) return 0;
   196→  const labels = data.state?.labels || data.labels || [];
   197→  return countUserEditedLabels(labels);
   198→}

   200→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   201→
   202→const robustStorage: StateStorage = {
   203→  getItem: (name: string): string | null => {
   204→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   205→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   206→
   207→    const primaryData = safeJsonParse(primaryRaw);
   208→    const backupData = safeJsonParse(backupRaw);
   209→
   210→    let result: string | null = null;
   211→
   212→    // If primary is missing/corrupt, try backup
   213→    if (!primaryData) {
   214→      if (backupData) {
   215→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   216→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   217→        result = backupRaw;
   218→      }
   219→      // Both empty — first visit, result stays null
   220→    } else if (backupData) {
   221→      // Both exist — check for data loss
   222→      const primaryLabelCount = getTotalLabelCount(primaryData);
   223→      const backupLabelCount = getTotalLabelCount(backupData);
   224→      const primaryUserEdits = primaryData.state?.labels
   225→        ? countUserEditedLabels(primaryData.state.labels)
   226→        : countUserEditedLabels(primaryData.labels || []);
   227→      const backupUserEdits = backupData.state?.labels
   228→        ? countUserEditedLabels(backupData.state.labels)
   229→        : countUserEditedLabels(backupData.labels || []);
   230→
   231→      // If primary has fewer labels than backup AND backup has more or equal user data
   232→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   233→        console.warn(
   234→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   235→        );
   236→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   237→        result = backupRaw;
   238→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   239→        // If primary has fewer user edits than backup, use backup
   240→        console.warn(
   241→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   242→        );
   243→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   244→        result = backupRaw;
   245→      } else {
   246→        result = primaryRaw;
   247→      }
   248→    } else {
   249→      // Primary exists, no backup — normal case
   250→      result = primaryRaw;
   251→    }
   252→
   253→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   254→    // merged data immediately after this getItem returns, and we MUST allow that write.
   255→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   256→    _rehydrated = true;
   257→
   258→    return result;
   259→  },
   260→
   261→  setItem: (name: string, value: string): void => {
   262→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   263→    // Before rehydration, the store only has seed data — writing it would overwrite
   264→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   265→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   266→    if (!_rehydrated) {
   267→      console.warn(
   268→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   269→      );
   270→      return;
   271→    }
   272→
   273→    // Write to primary immediately
   274→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   275→
   276→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   277→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   278→    safeLocalStorageSet(BACKUP_KEY, value);
   279→
   280→    if (!primaryOk) {
   281→      console.error("[LabelPulse Storage] Primary write failed!");
   282→    }
   283→  },
   284→
   285→  removeItem: (name: string): void => {
   286→    safeLocalStorageRemove(PRIMARY_KEY);
   287→    safeLocalStorageRemove(BACKUP_KEY);
   288→  },
   289→};
   290→
   291→/**
   292→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   293→ * Since backup is now written on every setItem, this is a safety net.
   294→ */
   295→export function forceBackupNow(): void {
   296→  if (typeof window === "undefined") return;
   297→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   298→  if (primaryRaw) {
   299→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   300→  }
   301→}

   374→/**
   375→ * Merge helper: preserves ALL user-editable fields from existing label
   376→ * when imported label has empty/default values for those fields.
   377→ */
   378→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
   379→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
   380→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
   381→  const hasNewRankData = Object.keys(imported.rankByGenre || {}).length > 0;
   382→  const currentRank = hasNewRankData && Object.keys(existing.rankByGenre || {}).length > 0
   383→    ? { ...existing.rankByGenre }
   384→    : existing.prevRankByGenre || {};
   385→
   386→  return {
   387→    // User-editable data — ALWAYS prefer existing if it has real data
   388→    emails: imported.emails?.length ? imported.emails : existing.emails,
   389→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
   390→    website: imported.website?.trim() || existing.website,
   391→    demoLink: imported.demoLink?.trim() || existing.demoLink,
   392→    socialLink: imported.socialLink?.trim() || existing.socialLink,
   393→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   394→    notes: imported.notes?.trim() || existing.notes,
   395→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
   396→
   397→    // Beatport/ranking data — prefer imported (it's the fresh data)
   398→    genres: imported.genres?.length ? imported.genres : existing.genres,
   399→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
   400→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
   401→    trending: imported.trending || existing.trending,
   402→    trendingRankByGenre: Object.keys(imported.trendingRankByGenre || {}).length ? imported.trendingRankByGenre : existing.trendingRankByGenre,
   403→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
   404→
   405→    // Ranking history — save current rank as "previous" when new data arrives
   406→    prevRankByGenre: currentRank,
   407→  };
   408→}

   436→/**
   437→ * Repair corrupted label data.
   438→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   439→ */
   440→function repairLabelData(labels: Label[]): Label[] {
   441→  const emailLabelCount = new Map<string, number>();
   442→  for (const l of labels) {
   443→    if (l.emails && Array.isArray(l.emails)) {
   444→      for (const e of l.emails) {
   445→        if (e && e.trim()) {
   446→          const key = e.toLowerCase().trim();
   447→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   448→        }
   449→      }
   450→    }
   451→  }
   452→
   453→  const corruptedEmails = new Set<string>();
   454→  for (const [email, count] of emailLabelCount) {
   455→    if (count > 5) {
   456→      corruptedEmails.add(email);
   457→      console.warn(
   458→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   459→      );
   460→    }
   461→  }
   462→
   463→  if (corruptedEmails.size === 0) {
   464→    return labels;
   465→  }
   466→
   467→  const emailOwner = new Map<string, string>();
   468→  for (const email of corruptedEmails) {
   469→    for (const l of labels) {
   470→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   471→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   472→        if (domain && l.name.toLowerCase().includes(domain)) {
   473→          emailOwner.set(email, l.id);
   474→          break;
   475→        }
   476→      }
   477→    }
   478→    if (!emailOwner.has(email)) {
   479→      for (const l of labels) {
   480→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   481→          emailOwner.set(email, l.id);
   482→          break;
   483→        }
   484→      }
   485→    }
   486→  }
   487→
   488→  return labels.map(l => {
   489→    if (!l.emails || !Array.isArray(l.emails)) return l;
   490→    const cleanedEmails = l.emails.filter(e => {
   491→      const key = e?.toLowerCase().trim();
   492→      if (!key) return false;
   493→      if (!corruptedEmails.has(key)) return true;
   494→      return emailOwner.get(key) === l.id;
   495→    });
   496→    const cleanedContactInfo = cleanedEmails[0] || "";
   497→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   498→      console.log(
   499→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   500→      );
   501→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   502→    }
   503→    return l;
   504→  });
   505→}

   998→// Helper: get tier for a label in a given genre
   999→export function getLabelTier(
  1000→  label: Label,
  1001→  genre?: string
  1002→): "top" | "mid" | "emerging" | null {
  1003→  if (!genre) genre = label.genres?.[0];
  1004→  if (!genre) return null;
  1005→  const rank = label.rankByGenre?.[genre];
  1006→  if (!rank) return label.trending ? "emerging" : null;
  1007→  if (rank <= 20) return "top";
  1008→  if (rank <= 50) return "mid";
  1009→  return label.trending ? "emerging" : null;
  1010→}

  1012→// ==================== CLOUD SYNC ====================
  1013→// Sistema di sincronizzazione con Supabase.

  1021→/**
  1022→ * Sincronizza lo stato corrente con Supabase (debounced).
  1023→ * Chiamata dopo ogni azione utente che modifica i dati.
  1024→ */

  1026→  if (!isSupabaseConfigured) return;
  1027→
  1028→  if (_cloudSyncTimer) {
  1029→    clearTimeout(_cloudSyncTimer);

  1032→  _cloudSyncTimer = setTimeout(async () => {
  1033→    _cloudSyncTimer = null;
  1034→    const state = useAppStore.getState();
  1035→    const dataToSync = {
  1036→      labels: state.labels,
  1037→      demos: state.demos,
  1038→      activeTab: state.activeTab,
  1039→      locale: state.locale,
  1040→      userProfile: state.userProfile,
  1041→      gmailAuth: state.gmailAuth,
  1042→      rankingsUpdatedAt: state.rankingsUpdatedAt,
  1043→      lastSavedAt: state.lastSavedAt,
  1044→      rankingSnapshots: state.rankingSnapshots,
  1045→    };
  1046→    await saveStateToCloud(dataToSync);
  1047→  }, CLOUD_SYNC_DEBOUNCE_MS);
  1048→}

  1077→/**
  1078→ * Carica i dati dal cloud e aggiorna lo store se i dati cloud sono più recenti.
  1079→ * Chiamata una volta all'avvio dell'app, dopo la reidratazione da localStorage.
  1080→ */
  1081→export async function loadFromCloud(): Promise<void> {
  1082→  if (!isSupabaseConfigured) {
  1083→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
  1084→    useAppStore.setState({ hasCloudSynced: true });
  1085→    return;
  1086→  }
  1087→
  1088→  try {
  1089→    const cloudData = await loadStateFromCloud();
  1090→    if (!cloudData) {
  1091→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
  1092→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
  1093→      await forceCloudSync();
  1094→      useAppStore.setState({ hasCloudSynced: true });
  1095→      return;
  1096→    }
  1097→
  1098→    const cloud = cloudData as any;
  1099→    const localState = useAppStore.getState();
  1100→
  1101→    // Confronta i timestamp per decidere quali dati usare
  1102→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
  1103→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
  1104→
  1105→    if (cloudSavedAt > localSavedAt) {
  1106→      // I dati cloud sono più recenti — aggiorna lo store locale
  1107→      console.log(
  1108→        `[LabelPulse Cloud] Cloud data is newer (cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}). Updating local store.`
  1109→      );
  1110→
  1111→      // Usa la stessa logica di merge della reidratazione
  1112→      const merged = mergeCloudData(cloud, localState);
  1113→      useAppStore.setState({
  1114→        ...merged,
  1115→        hasCloudSynced: true,
  1116→      });
  1117→    } else {
  1118→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1119→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1120→      useAppStore.setState({ hasCloudSynced: true });
  1121→    }
  1122→  } catch (err) {
  1123→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1124→    useAppStore.setState({ hasCloudSynced: true });
  1125→  }
  1126→}
