   221→/**
   222→ * Get the time cutoff for a period (in milliseconds from now).
   223→ */
   224→function getPeriodCutoff(period: RankingTimePeriod): number {
   225→  const now = Date.now();
   226→  switch (period) {
   227→    case "current": return now; // Only current data
   228→    case "1m": return now - 30 * 24 * 60 * 60 * 1000;
   229→    case "3m": return now - 90 * 24 * 60 * 60 * 1000;
   230→    case "1y": return now - 365 * 24 * 60 * 60 * 1000;
   231→    case "all": return 0; // All time
   232→  }
   233→}

   235→// ==================== AGGREGATION LOGIC ====================
   236→
   237→/**
   238→ * Aggregate ranking snapshots for a given genre and time period.
   239→ * Returns a map of labelName -> { totalPoints, bestRank, snapshotCount, averageRank }
   240→ */
   241→function aggregateSnapshots(
   242→  snapshots: RankingSnapshot[],
   243→  genre: string,
   244→  period: RankingTimePeriod
   245→): Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }> {
   246→  const cutoff = getPeriodCutoff(period);
   247→  const result = new Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }>();
   248→
   249→  // Helper to get or create entry with defaults
   250→  function getOrCreate(name: string): { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number } {
   251→    let entry = result.get(name);
   252→    if (!entry) {
   253→      entry = { totalPoints: 0, bestRank: 999, snapshotCount: 0, averageRank: 0, rankSum: 0 };
   254→      result.set(name, entry);
   255→    }
   256→    return entry;
   257→  }
   258→
   259→  for (const snapshot of snapshots) {
   260→    if (new Date(snapshot.timestamp).getTime() < cutoff) continue;
   261→    if (period === "current") continue; // Current period doesn't use snapshots
   262→
   263→    const genreData = snapshot.genres[genre];
   264→    if (!genreData) continue;
   265→
   266→    for (const [labelName, data] of Object.entries(genreData)) {
   267→      const existing = getOrCreate(labelName);
   268→      existing.totalPoints += data.points;
   269→      existing.rankSum += data.rank;
   270→      existing.snapshotCount++;
   271→      if (data.rank < existing.bestRank) existing.bestRank = data.rank;
   272→    }
   273→  }
   274→
   275→  // Calculate average rank
   276→  for (const [, val] of result) {
   277→    (val as any).averageRank = Math.round(val.rankSum / val.snapshotCount);
   278→  }
   279→
   280→  return result;
   281→}

   283→/**
   284→ * Build ranked list for a genre, considering the time period.
   285→ * - "current": uses live label data (rankByGenre)
   286→ * - Other periods: aggregates snapshots and re-ranks by cumulative points
   287→ */
   288→function buildRankedList(
   289→  labels: Label[],
   290→  snapshots: RankingSnapshot[],
   291→  genre: string,
   292→  period: RankingTimePeriod
   293→): RankedLabel[] {
   294→  if (period === "current") {
   295→    // Current view: use live label data
   296→    const ranked: RankedLabel[] = [];
   297→    for (const label of labels) {
   298→      const rank = label.rankByGenre?.[genre];
   299→      if (rank === undefined || rank === null) continue;
   300→
   301→      const prevRank = label.prevRankByGenre?.[genre] ?? null;
   302→      const points = label.pointsByGenre?.[genre] ?? 0;
   303→      let movement: number | null = null;
   304→
   305→      if (prevRank !== null && prevRank !== undefined) {
   306→        movement = prevRank - rank;
   307→      } else if (Object.keys(label.prevRankByGenre || {}).length > 0) {
   308→        movement = null;
   309→      }
   310→
   311→      ranked.push({ label, rank, prevRank, points, movement, snapshotCount: 1 });
   312→    }
   313→    return ranked;
   314→  }
   315→
   316→  // Historical/cumulative view: aggregate snapshots
   317→  const aggregated = aggregateSnapshots(snapshots, genre, period);
   318→  if (aggregated.size === 0) return [];
   319→
   320→  // Sort by total points descending to determine cumulative rank
   321→  const sorted = Array.from(aggregated.entries()).sort((a, b) => b[1].totalPoints - a[1].totalPoints);
   322→
   323→  // For movement in cumulative view, compare with the current live data
   324→  const ranked: RankedLabel[] = [];
   325→  for (let i = 0; i < sorted.length; i++) {
   326→    const [labelName, data] = sorted[i];
   327→    const cumulativeRank = i + 1;
   328→
   329→    // Find matching label in current data
   330→    const matchingLabel = labels.find(l => l.name === labelName);
   331→    const currentRank = matchingLabel?.rankByGenre?.[genre] ?? null;
   332→
   333→    // Movement: compare current live rank with cumulative rank
   334→    let movement: number | null = null;
   335→    if (currentRank !== null) {
   336→      movement = currentRank - cumulativeRank; // positive = label is better live than cumulative (hot right now)
   337→    }
   338→
   339→    // Use the label if found, otherwise create a minimal reference
   340→    const label = matchingLabel || {
   341→      id: `snapshot_${labelName.toLowerCase().replace(/[^a-z0-9]+/g, '_')}`,
   342→      name: labelName,
   343→      genre: genre,
   344→      submissionType: "email" as const,
   345→      contactInfo: "",
   346→      status: "open" as const,
   347→      notes: "",
   348→      createdAt: new Date().toISOString(),
   349→      emails: [],
   350→      website: "",
   351→      demoLink: "",
   352→      socialLink: "",
   353→      soundcloudLink: "",
   354→      beatportLink: "",
   355→      customLinks: [],
   356→      genres: [genre],
   357→      rankByGenre: { [genre]: currentRank || 999 },
   358→      pointsByGenre: {},
   359→      trending: false,
   360→      trendingRankByGenre: {},
   361→      trendingPointsByGenre: {},
   362→      prevRankByGenre: {},
   363→    };
   364→
   365→    ranked.push({
   366→      label,
   367→      rank: cumulativeRank,
   368→      prevRank: currentRank,
   369→      points: data.totalPoints,
   370→      movement,
   371→      snapshotCount: data.snapshotCount,
   372→    });
   373→  }
   374→
   375→  return ranked;
   376→}

   380→export function RankingsPage() {
   381→  const { labels, locale, rankingsUpdatedAt, rankingSnapshots } = useAppStore();
   382→  const snapshots = rankingSnapshots || []; // defensive: might be undefined for existing users
   383→  const [selectedGenre, setSelectedGenre] = useState<string | null>(null);
   384→  const [sortMode, setSortMode] = useState<SortMode>("rank");
   385→  const [movementFilter, setMovementFilter] = useState<MovementFilter>("all");
   386→  const [timePeriod, setTimePeriod] = useState<RankingTimePeriod>("current");
   387→  const [showGenreDropdown, setShowGenreDropdown] = useState(false);
   388→  const [spotlightShowAll, setSpotlightShowAll] = useState(false); // when true + genre selected, show global risers instead of genre-filtered
   389→
   390→  // Get all genres from labels, sorted alphabetically
   391→  const allGenres = useMemo(() => {
   392→    const genreSet = new Set<string>();
   393→    for (const label of labels) {
   394→      if (label.genres) {
   395→        for (const g of label.genres) {
   396→          genreSet.add(g);
   397→        }
   398→      }
   399→    }
   400→    return Array.from(genreSet).sort((a, b) => a.localeCompare(b));
   401→  }, [labels]);
   402→
   403→  // Build ranked list for selected genre and period
   404→  const rankedList = useMemo((): RankedLabel[] => {
   405→    if (!selectedGenre) return [];
   406→    return buildRankedList(labels, snapshots, selectedGenre, timePeriod);
   407→  }, [labels, snapshots, selectedGenre, timePeriod]);
   408→
   409→  // Apply filters
   410→  const filteredList = useMemo(() => {
   411→    let result = [...rankedList];
   412→
   413→    // Movement filter
   414→    switch (movementFilter) {
   415→      case "rising":
   416→        result = result.filter((l) => l.movement !== null && l.movement > 0);
   417→        break;
   418→      case "falling":
   419→        result = result.filter((l) => l.movement !== null && l.movement < 0);
   420→        break;
   421→      case "new":
   422→        result = result.filter((l) => l.movement === null);
   423→        break;
   424→      case "stable":
   425→        result = result.filter((l) => l.movement === 0);
   426→        break;
   427→    }
   428→
   429→    // Sort
   430→    switch (sortMode) {
   431→      case "rank":
   432→        result.sort((a, b) => a.rank - b.rank);
   433→        break;
   434→      case "movement":
   435→        result.sort((a, b) => {
   436→          if (a.movement === null && b.movement === null) return a.rank - b.rank;
   437→          if (a.movement === null) return -1;
   438→          if (b.movement === null) return 1;
   439→          return b.movement - a.movement;
   440→        });
   441→        break;
   442→      case "points":
   443→        result.sort((a, b) => b.points - a.points);
   444→        break;
   445→    }
   446→
   447→    return result;
   448→  }, [rankedList, movementFilter, sortMode]);
   449→
   450→  // Stats summary
   451→  const stats = useMemo(() => {
   452→    const total = rankedList.length;
   453→    const rising = rankedList.filter((l) => l.movement !== null && l.movement > 0).length;
   454→    const falling = rankedList.filter((l) => l.movement !== null && l.movement < 0).length;
   455→    const newEntries = rankedList.filter((l) => l.movement === null).length;
   456→
   457→    return { total, rising, falling, newEntries };
   458→  }, [rankedList]);
   459→
   460→  // Top rising labels across ALL genres (for the global spotlight section)
   461→  const topRisers = useMemo(() => {
   462→    const risers: { label: Label; genre: string; movement: number; rank: number }[] = [];
   463→
   464→    for (const label of labels) {
   465→      if (!label.prevRankByGenre || Object.keys(label.prevRankByGenre).length === 0) continue;
   466→      for (const genre of label.genres) {
   467→        const currentRank = label.rankByGenre?.[genre];
   468→        const prevRank = label.prevRankByGenre?.[genre];
   469→        if (currentRank && prevRank) {
   470→          const movement = prevRank - currentRank;
   471→          if (movement > 0) {
   472→            risers.push({ label, genre, movement, rank: currentRank });
   473→          }
   474→        }
   475→      }
   476→    }
   477→
   478→    return risers.sort((a, b) => b.movement - a.movement).slice(0, 10);
   479→  }, [labels]);
   480→
   481→  // Top rising labels for the SELECTED genre (genre-specific spotlight)
   482→  const topRisersForGenre = useMemo(() => {
   483→    if (!selectedGenre) return [];
   484→    const risers: { label: Label; genre: string; movement: number; rank: number }[] = [];
   485→
   486→    for (const label of labels) {
   487→      const currentRank = label.rankByGenre?.[selectedGenre];
   488→      const prevRank = label.prevRankByGenre?.[selectedGenre];
   489→      if (currentRank && prevRank) {
   490→        const movement = prevRank - currentRank;
   491→        if (movement > 0) {
   492→          risers.push({ label, genre: selectedGenre, movement, rank: currentRank });
   493→        }
   494→      }
   495→    }
   496→
   497→    return risers.sort((a, b) => b.movement - a.movement).slice(0, 10);
   498→  }, [labels, selectedGenre]);
   499→
   500→  const hasPreviousData = useMemo(() => {
   501→    return labels.some((l) => l.prevRankByGenre && Object.keys(l.prevRankByGenre).length > 0);
   502→  }, [labels]);
   503→
   504→  // Snapshot stats for info display
   505→  const snapshotStats = useMemo(() => {
   506→    if (!snapshots || snapshots.length === 0) return { count: 0, oldest: null as string | null, newest: null as string | null };
   507→    const sorted = [...snapshots].sort((a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime());
   508→    return {
   509→      count: snapshots.length,
   510→      oldest: sorted[0].timestamp,
   511→      newest: sorted[sorted.length - 1].timestamp,
   512→    };
   513→  }, [snapshots]);
   514→
   515→  // Spotlight risers: genre-filtered or global, based on toggle
   516→  const spotlightRisers = useMemo(() => {
   517→    if (!hasPreviousData) return [];
   518→    const isGenreFiltered = selectedGenre && !spotlightShowAll;
   519→    return isGenreFiltered ? topRisersForGenre : topRisers;
   520→  }, [hasPreviousData, selectedGenre, spotlightShowAll, topRisers, topRisersForGenre]);
   521→
   522→  const hasSnapshots = snapshots.length > 0;
   523→
   524→  // === EMERGENCY HISTORY RECOVERY ===
   525→  // If the live store has 0 snapshots but the dedicated sidecar backup slot
   526→  // (labelpulse-snapshots-backup) has snapshots, show a one-click "restore history"
   527→  // button. This is the safety-net UI for the scenario where a bad cloud sync,
   528→  // a partial import, or a localStorage hiccup wipes the live rankingSnapshots
   529→  // array but the sidecar survived.
   530→  const [sidecarCount, setSidecarCount] = useState(0);
   531→  const [restoredMsg, setRestoredMsg] = useState<string | null>(null);
   532→  useEffect(() => {
   533→    setSidecarCount(readSnapshotsSidecar().length);
   534→  }, [snapshots.length]);
   535→  const showRestoreHistoryBtn =
   536→    !hasSnapshots && sidecarCount > 0 && labels.length > 0;
   537→  const handleRestoreHistory = () => {
   538→    const added = restoreSnapshotsFromSidecar();
   539→    if (added > 0) {
   540→      setRestoredMsg(
   541→        locale === "it"
   542→          ? `Storico recuperato: ${added} snapshot importati dal backup di emergenza.`
   543→          : `History recovered: ${added} snapshot(s) imported from emergency backup.`
   544→      );
   545→    } else {
   546→      setRestoredMsg(
   547→        locale === "it"
   548→          ? "Nessuno snapshot nuovo da recuperare (già sincronizzati)."
   549→          : "No new snapshots to recover (already in sync)."
   550→      );
   551→    }
