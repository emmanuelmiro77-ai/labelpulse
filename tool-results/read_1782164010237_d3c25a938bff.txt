   212→function safeJsonParse(str: string | null): any | null {
   213→  if (!str) return null;
   214→  try {
   215→    return JSON.parse(str);
   216→  } catch {
   217→    return null;
   218→  }
   219→}

   221→function safeLocalStorageGet(key: string): string | null {
   222→  try {
   223→    return localStorage.getItem(key);
   224→  } catch {
   225→    return null;
   226→  }
   227→}

   229→function safeLocalStorageSet(key: string, value: string): boolean {
   230→  try {
   231→    localStorage.setItem(key, value);
   232→    return true;
   233→  } catch (e) {
   234→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   235→    return false;
   236→  }
   237→}

   247→/**
   248→ * Extract rankingSnapshots array from a raw localStorage JSON string.
   249→ * Returns [] if not found / invalid.
   250→ */
   251→function extractSnapshotsFromRaw(raw: string | null): RankingSnapshot[] {
   252→  if (!raw) return [];
   253→  const data = safeJsonParse(raw);
   254→  if (!data) return [];
   255→  const snaps = data.state?.rankingSnapshots ?? data.rankingSnapshots;
   256→  return Array.isArray(snaps) ? snaps : [];
   257→}

   259→/**
   260→ * Merge two snapshot arrays, deduping by `id` first, then by `timestamp+source`.
   261→ * Always returns ascending-by-timestamp order.
   262→ * Used by: importData (backup file), robustStorage.getItem (safety-net restore),
   263→ * and loadFromCloud (cloud merge).
   264→ */
   265→function mergeSnapshots(
   266→  base: RankingSnapshot[] | null | undefined,
   267→  incoming: RankingSnapshot[] | null | undefined
   268→): RankingSnapshot[] {
   269→  const a = Array.isArray(base) ? base : [];
   270→  const b = Array.isArray(incoming) ? incoming : [];
   271→  const seen = new Set<string>();
   272→  const out: RankingSnapshot[] = [];
   273→  for (const s of [...a, ...b]) {
   274→    if (!s || typeof s !== "object") continue;
   275→    const key = s.id || `${s.timestamp}|${s.source}`;
   276→    if (seen.has(key)) continue;
   277→    seen.add(key);
   278→    out.push(s);
   279→  }
   280→  out.sort(
   281→    (x, y) =>
   282→      new Date(x.timestamp || 0).getTime() - new Date(y.timestamp || 0).getTime()
   283→  );
   284→  return out;
   285→}

   305→  if (!Array.isArray(labels)) return 0;
   306→  return labels.filter((l) =>
   307→    (l.emails && l.emails.length > 0) ||
   308→    (l.notes && l.notes.trim() !== "") ||
   309→    (l.website && l.website.trim() !== "") ||
   310→    (l.demoLink && l.demoLink.trim() !== "") ||
   311→    (l.socialLink && l.socialLink.trim() !== "") ||
   312→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   313→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   314→    l.isCustom === true
   315→  ).length;

   318→function getTotalLabelCount(data: any): number {
   319→  if (!data) return 0;
   320→  if (Array.isArray(data.labels)) return data.labels.length;
   321→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   322→  return 0;

   339→const robustStorage: StateStorage = {
   340→  getItem: (name: string): string | null => {
   341→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   342→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   343→
   344→    const primaryData = safeJsonParse(primaryRaw);
   345→    const backupData = safeJsonParse(backupRaw);
   346→
   347→    let result: string | null = null;
   348→    let usedSidecarSnapshots = false;
   349→
   350→    // If primary is missing/corrupt, try backup
   351→    if (!primaryData) {
   352→      if (backupData) {
   353→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   354→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   355→        result = backupRaw;
   356→      }
   357→      // Both empty — first visit, result stays null
   358→    } else if (backupData) {
   359→      // Both exist — check for data loss
   360→      const primaryLabelCount = getTotalLabelCount(primaryData);
   361→      const backupLabelCount = getTotalLabelCount(backupData);
   362→      const primaryUserEdits = primaryData.state?.labels
   363→        ? countUserEditedLabels(primaryData.state.labels)
   364→        : countUserEditedLabels(primaryData.labels || []);
   365→      const backupUserEdits = backupData.state?.labels
   366→        ? countUserEditedLabels(backupData.state.labels)
   367→        : countUserEditedLabels(backupData.labels || []);
   368→
   369→      // If primary has fewer labels than backup AND backup has more or equal user data
   370→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   371→        console.warn(
   372→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   373→        );
   374→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   375→        result = backupRaw;
   376→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   377→        // If primary has fewer user edits than backup, use backup
   378→        console.warn(
   379→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   380→        );
   381→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   382→        result = backupRaw;
   383→      } else {
   384→        result = primaryRaw;
   385→      }
   386→    } else {
   387→      // Primary exists, no backup — normal case
   388→      result = primaryRaw;
   389→    }
   390→
   391→    // CRITICAL SAFETY NET for rankingSnapshots:
   392→    // If the chosen result has 0 snapshots but the dedicated SNAPSHOTS_BACKUP_KEY
   393→    // has snapshots, splice them in. This handles the scenario where the main
   394→    // store got wiped (bad cloud sync, migration bug, partial overwrite) but the
   395→    // snapshots sidecar survived. Without this, the user loses months of
   396→    // "storico caricamenti" and the Classifiche page resets to "first time".
   397→    try {
   398→      const currentSnaps = extractSnapshotsFromRaw(result);
   399→      const sidecarRaw = safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY);
   400→      const sidecarSnaps = extractSnapshotsFromRaw(sidecarRaw);
   401→      if (sidecarSnaps.length > 0 && sidecarSnaps.length > currentSnaps.length) {
   402→        console.warn(
   403→          `[LabelPulse Storage] SNAPSHOTS RECOVERY: result has ${currentSnaps.length} snapshots, sidecar has ${sidecarSnaps.length}. Merging sidecar into result.`
   404→        );
   405→        const mergedSnaps = mergeSnapshots(currentSnaps, sidecarSnaps);
   406→        const parsed = result ? safeJsonParse(result) : {};
   407→        if (parsed && typeof parsed === "object") {
   408→          if (parsed.state && typeof parsed.state === "object") {
   409→            parsed.state.rankingSnapshots = mergedSnaps;
   410→          } else {
   411→            parsed.rankingSnapshots = mergedSnaps;
   412→          }
   413→          result = JSON.stringify(parsed);
   414→          usedSidecarSnapshots = true;
   415→        }
   416→      }
   417→    } catch (e) {
   418→      console.warn("[LabelPulse Storage] Snapshots sidecar recovery failed:", e);
   419→    }
   420→
   421→    // CRITICAL SAFETY NET for userProfile:
   422→    // Same logic as snapshots. If the chosen result has an empty/seed profile
   423→    // but PROFILE_BACKUP_KEY has real data, splice it in. This is the user's
   424→    // identity card — losing artistName, bio, email, links is catastrophic.
   425→    try {
   426→      const currentProfile = extractProfileFromRaw(result);
   427→      const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   428→      if (sidecarProfile && !currentProfile) {
   429→        console.warn(
   430→          `[LabelPulse Storage] PROFILE RECOVERY: result has empty profile, sidecar has data (artistName=${!!sidecarProfile.artistName}, bio=${!!sidecarProfile.bio}). Merging sidecar into result.`
   431→        );
   432→        const parsed = result ? safeJsonParse(result) : {};
   433→        if (parsed && typeof parsed === "object") {
   434→          if (parsed.state && typeof parsed.state === "object") {
   435→            parsed.state.userProfile = sidecarProfile;
   436→          } else {
   437→            parsed.userProfile = sidecarProfile;
   438→          }
   439→          result = JSON.stringify(parsed);
   440→        }
   441→      } else if (sidecarProfile && currentProfile) {
   442→        // Both have data — merge (sidecar fills empty fields in current)
   443→        const mergedProfile = mergeProfiles(sidecarProfile, currentProfile);
   444→        const parsed = result ? safeJsonParse(result) : {};
   445→        if (parsed && typeof parsed === "object" && mergedProfile) {
   446→          if (parsed.state && typeof parsed.state === "object") {
   447→            parsed.state.userProfile = mergedProfile;
   448→          } else {
   449→            parsed.userProfile = mergedProfile;
   450→          }
   451→          result = JSON.stringify(parsed);
   452→        }
   453→      }
   454→    } catch (e) {
   455→      console.warn("[LabelPulse Storage] Profile sidecar recovery failed:", e);
   456→    }
   457→
   458→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   459→    // merged data immediately after this getItem returns, and we MUST allow that write.
   460→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   461→    _rehydrated = true;
   462→
   463→    if (usedSidecarSnapshots) {
   464→      console.info("[LabelPulse Storage] rankingSnapshots restored from sidecar backup.");
   465→    }
   466→
   467→    return result;
   468→  },
   469→
   470→  setItem: (name: string, value: string): void => {
   471→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   472→    // Before rehydration, the store only has seed data — writing it would overwrite
   473→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   474→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   475→    if (!_rehydrated) {
   476→      console.warn(
   477→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   478→      );
   479→      return;
   480→    }
   481→
   482→    // Write to primary immediately
   483→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   484→
   485→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   486→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   487→    safeLocalStorageSet(BACKUP_KEY, value);
   488→
   489→    // CRITICAL: also persist rankingSnapshots to a dedicated slot.
   490→    // This is the user's "storico caricamenti" — months of imports that cannot
   491→    // be regenerated. Keeping it in a separate key means even a catastrophic
   492→    // main-store wipe (bad cloud sync, migration bug, quota error) leaves the
   493→    // snapshots recoverable.
   494→    try {
   495→      const snaps = extractSnapshotsFromRaw(value);
   496→      if (snaps.length > 0) {
   497→        // Merge with whatever's already in the snapshots slot — NEVER overwrite
   498→        // existing snapshots with fewer snapshots (would happen if the current
   499→        // write had a partial/empty snapshot array due to some upstream bug).
   500→        const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   501→        const merged = mergeSnapshots(existing, snaps);
   502→        safeLocalStorageSet(
   503→          SNAPSHOTS_BACKUP_KEY,
   504→          JSON.stringify(merged)
   505→        );
   506→      }
   507→    } catch (e) {
   508→      console.warn("[LabelPulse Storage] Snapshots-sidecar write failed:", e);
   509→    }
   510→
   511→    // CRITICAL: also persist userProfile to a dedicated slot.
   512→    // Same pattern as snapshots. The profile is the user's identity card —
   513→    // artistName, bio, email, social links, photo. We can NEVER lose this,
   514→    // even if cloud sync wipes the main store.
   515→    try {
   516→      const profile = extractProfileFromRaw(value);
   517→      if (profile) {
   518→        const existing = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   519→        const merged = mergeProfiles(existing, profile);
   520→        if (merged) {
   521→          safeLocalStorageSet(
   522→            PROFILE_BACKUP_KEY,
   523→            JSON.stringify({ userProfile: merged })
   524→          );
   525→        }
   526→      }
   527→    } catch (e) {
   528→      console.warn("[LabelPulse Storage] Profile-sidecar write failed:", e);
   529→    }
   530→
   531→    if (!primaryOk) {
   532→      console.error("[LabelPulse Storage] Primary write failed!");
   533→    }
   534→  },
   535→
   536→  removeItem: (name: string): void => {
   537→    safeLocalStorageRemove(PRIMARY_KEY);
   538→    safeLocalStorageRemove(BACKUP_KEY);
   539→    // NOTE: do NOT remove SNAPSHOTS_BACKUP_KEY or PROFILE_BACKUP_KEY here.
   540→    // These are the user's permanent records (months of ranking history +
   541→    // artist identity) — clearing the main store should NEVER nuke them.
   542→    // The only way to clear them is an explicit user action ("reset all data"
   543→    // button) which calls safeLocalStorageRemove directly.
   544→  },

   547→/**
   548→ * Force-write a backup immediately (e.g., on visibility change or before unload).

   599→/**
   600→ * Extract userProfile from a raw JSON string (main store JSON, backup JSON,
   601→ * or sidecar JSON). Returns null if missing/invalid.
   602→ */
   603→function extractProfileFromRaw(raw: string | null): any | null {
   604→  if (!raw) return null;
   605→  try {
   606→    const parsed = JSON.parse(raw);
   607→    const profile = parsed?.state?.userProfile ?? parsed?.userProfile ?? null;
   608→    if (profile && typeof profile === "object") {
   609→      // Must have at least one non-empty field to be considered "real"
   610→      const hasData =
   611→        !!profile.artistName ||
   612→        !!profile.bio ||
   613→        !!profile.email ||
   614→        !!profile.scLink ||
   615→        !!profile.photoUrl ||
   616→        (Array.isArray(profile.links) && profile.links.length > 0) ||
   617→        !!profile.cyaniteApiToken ||
   618→        !!profile.supabaseUrl;
   619→      if (hasData) return profile;
   620→    }
   621→  } catch {
   622→    /* ignore */
   623→  }
   624→  return null;
   625→}
   626→
   627→/**
   628→ * Merge two user profiles — non-empty fields win.
   629→ * This is the safety net: if the cloud ever sends a profile with empty
   630→ * artistName, but the sidecar has the real artistName, we keep the real one.
   631→ */
   632→export function mergeProfiles(base: any | null, incoming: any | null): any | null {
   633→  if (!base && !incoming) return null;
   634→  if (!base) return incoming;
   635→  if (!incoming) return base;
   636→  const merged: any = { ...base };
   637→  for (const k of Object.keys(incoming)) {
   638→    const v = incoming[k];
   639→    if (k === "links") {
   640→      // Merge link arrays (dedupe by type|value)
   641→      const a = Array.isArray(base.links) ? base.links : [];
   642→      const b = Array.isArray(v) ? v : [];
   643→      const seen = new Set<string>();
   644→      const out: { type: string; value: string }[] = [];
   645→      for (const link of [...a, ...b]) {
   646→        if (!link || typeof link !== "object") continue;
   647→        const key = `${link.type}|${link.value}`;
   648→        if (seen.has(key)) continue;
   649→        seen.add(key);
   650→        out.push(link);
   651→      }
   652→      merged.links = out;
   653→    } else if (typeof v === "string") {
   654→      // Non-empty string wins over empty
   655→      if (v && !merged[k]) merged[k] = v;
   656→    } else {
   657→      // For any other type, prefer defined over undefined
   658→      if (v !== undefined && v !== null && !merged[k]) merged[k] = v;
   659→    }
   660→  }
   661→  return merged;
   662→}

   710→// ==================== ARTISTS SIDE CAR ====================
   711→// Like PROFILE_BACKUP_KEY but for the artists array. A full Beatport scrape
   712→// produces ~3400 artists / ~9MB. localStorage has a 5-10MB quota on most
   713→// browsers, so we TRY to write artists here but FAIL GRACEFULLY if quota
   714→// exceeded — the IndexedDB + cloud are the primary stores, this is just
   715→// an emergency backup for when both fail.
   716→//
   717→// Strategy:
   718→//  - writeArtistsSidecar: best-effort write, truncate if needed (keep first
   719→//    N artists rather than fail entirely). Skip silently if quota exceeded.
   720→//  - readArtistsSidecar: returns [] if missing/corrupt
   721→//  - restoreArtistsFromSidecar: splice into live store if local is empty
   722→const ARTISTS_SIDECAR_KEY = "labelpulse-artists-backup";
   723→const ARTISTS_SIDECAR_MAX = 200; // cap to first 200 artists (~600KB) to fit quota
   724→
   725→export function writeArtistsSidecar(artists: any[]): void {
   726→  if (typeof window === "undefined") return;
   727→  if (!Array.isArray(artists) || artists.length === 0) return;
   728→  try {
   729→    // If small enough, write all. Otherwise, write first N + a marker.
   730→    const capped = artists.length > ARTISTS_SIDECAR_MAX
   731→      ? artists.slice(0, ARTISTS_SIDECAR_MAX)
   732→      : artists;
   733→    const payload = JSON.stringify({
   734→      savedAt: new Date().toISOString(),
   735→      count: artists.length,
   736→      capped: artists.length > ARTISTS_SIDECAR_MAX,
   737→      artists: capped,
   738→    });
   739→    safeLocalStorageSet(ARTISTS_SIDECAR_KEY, payload);
   740→  } catch (e) {
   741→    // Quota exceeded — try with even fewer artists
   742→    try {
   743→      const tiny = artists.slice(0, 50);
   744→      const payload = JSON.stringify({
   745→        savedAt: new Date().toISOString(),
   746→        count: artists.length,
   747→        capped: true,
   748→        artists: tiny,
   749→      });
   750→      safeLocalStorageSet(ARTISTS_SIDECAR_KEY, payload);
   751→    } catch {
   752→      // Give up silently — IDB + cloud are the primary stores
   753→    }
   754→  }
   755→}
   756→
   757→export function readArtistsSidecar(): any[] {
   758→  if (typeof window === "undefined") return [];
   759→  const raw = safeLocalStorageGet(ARTISTS_SIDECAR_KEY);
   760→  if (!raw) return [];
   761→  try {
   762→    const parsed = JSON.parse(raw);
   763→    const arr = parsed?.artists;
   764→    return Array.isArray(arr) ? arr : [];
   765→  } catch {
   766→    return [];
   767→  }
   768→}

   771→ * Restore artists from sidecar IF the live store has 0 artists.

   773→ * Does NOT replace live artists if live already has data — sidecar is
   774→ * only an emergency recovery, not a primary source.

   776→export function restoreArtistsFromSidecar(): number {

   877→/**
   878→ * Merge incoming artists (from a fresh scrape) with existing artists
   879→ * (already in the store from previous scrapes).
   880→ *
   881→ * Strategy:
   882→ * - Key by artist.id (e.g. "bp_6824"). Same Beatport id = same artist.
   883→ * - For each artist that exists: union genres, union labelsPublishedOn,
   884→ *   MERGE tracksByGenre by track.id (preserve tracks from older scrapes,
   885→ *   update with new position info), sum totalPoints, min bestPosition.
   886→ * - firstSeenAt preserved from existing; lastSeenAt = now.
   887→ * - Trending recomputed fresh from current data.
   888→ * - New artists: appended with firstSeenAt = lastSeenAt = now.
   889→ *
   890→ * This is what makes historical data accumulate: each scrape adds new
   891→ * tracks to artists' tracksByGenre without losing previously-seen tracks.
   892→ */
   893→function mergeArtists(existing: Artist[], incoming: Artist[], now: string): Artist[] {
   894→  const map = new Map<string, Artist>();
   895→
   896→  // Seed with existing
   897→  for (const a of existing) {
   898→    map.set(a.id, { ...a });
   899→  }
   900→
   901→  for (const inc of incoming) {
   902→    const ex = map.get(inc.id);
   903→    if (!ex) {
   904→      // New artist — first time we see them
   905→      map.set(inc.id, {
   906→        ...inc,
   907→        firstSeenAt: now,
   908→        lastSeenAt: now,
   909→      });
   910→      continue;
   911→    }
   912→
   913→    // Existing artist — merge fields
   914→    // Union genres
   915→    const genres = Array.from(new Set([...(ex.genres || []), ...(inc.genres || [])]));
   916→
   917→    // Union labelsPublishedOn
   918→    const labelsPublishedOn = Array.from(
   919→      new Set([...(ex.labelsPublishedOn || []), ...(inc.labelsPublishedOn || [])])
   920→    );
   921→
   922→    // Merge tracksByGenre — preserve old tracks, add/update from incoming
   923→    const tracksByGenre: Record<string, ArtistTrack[]> = { ...(ex.tracksByGenre || {}) };
   924→    for (const genre of Object.keys(inc.tracksByGenre || {})) {
   925→      const incTracks = inc.tracksByGenre[genre] || [];
   926→      const exTracks = tracksByGenre[genre] || [];
   927→      const byId = new Map<number, ArtistTrack>();
   928→      // Seed with existing
   929→      for (const t of exTracks) byId.set(t.id, t);
   930→      // Update/add with incoming (incoming overrides — fresh positions)
   931→      for (const t of incTracks) byId.set(t.id, t);
   932→      tracksByGenre[genre] = Array.from(byId.values()).sort((a, b) => a.position - b.position);
   933→    }
   934→
   935→    // Recompute totalPoints & bestPosition from current incoming data
   936→    // (the existing totals were from older scrapes — we want the snapshot
   937→    // from THIS import to drive the points, otherwise the totals just
   938→    // accumulate forever and "points" loses meaning).
   939→    // BUT: keep max totalPoints seen (artists who peaked before should
   940→    // keep credit). Actually no — let's use the LATEST totalPoints,
   941→    // since that's the current ranking signal. Old tracks are still in
   942→    // tracksByGenre for browsing, but the points reflect "right now".
   943→    const totalPoints = inc.totalPoints;
   944→    const bestPosition = Math.min(ex.bestPosition, inc.bestPosition);
   945→
   946→    // Trending: use incoming (fresh computation)
   947→    const trending = inc.trending;
   948→    const trendingRankByGenre = inc.trendingRankByGenre;
   949→    const trendingPointsByGenre = inc.trendingPointsByGenre;
   950→
   951→    // isRemixerOnly: false if EITHER snapshot has them as primary
   952→    const isRemixerOnly = ex.isRemixerOnly && inc.isRemixerOnly;
   953→
   954→    map.set(inc.id, {
   955→      ...ex,
   956→      ...inc,
   957→      genres,
   958→      labelsPublishedOn,
   959→      tracksByGenre,
   960→      totalPoints,
   961→      bestPosition,
   962→      trending,
   963→      trendingRankByGenre,
   964→      trendingPointsByGenre,
   965→      isRemixerOnly,
   966→      firstSeenAt: ex.firstSeenAt || now,
   967→      lastSeenAt: now,
   968→    });
   969→  }
   970→
   971→  return Array.from(map.values());
   972→}

  1036→/**
  1037→ * Repair corrupted label data.
  1038→ * Detects and fixes: same email appearing on too many labels (likely corruption).
  1039→ */
  1040→function repairLabelData(labels: Label[]): Label[] {
  1041→  const emailLabelCount = new Map<string, number>();
  1042→  for (const l of labels) {
  1043→    if (l.emails && Array.isArray(l.emails)) {
  1044→      for (const e of l.emails) {
  1045→        if (e && e.trim()) {
  1046→          const key = e.toLowerCase().trim();
  1047→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
  1048→        }
  1049→      }
  1050→    }
  1051→  }
  1052→
  1053→  const corruptedEmails = new Set<string>();
  1054→  for (const [email, count] of emailLabelCount) {
  1055→    if (count > 5) {
  1056→      corruptedEmails.add(email);
  1057→      console.warn(
  1058→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
  1059→      );
  1060→    }
  1061→  }
  1062→
  1063→  if (corruptedEmails.size === 0) {
  1064→    return labels;
  1065→  }
  1066→
  1067→  const emailOwner = new Map<string, string>();
  1068→  for (const email of corruptedEmails) {
  1069→    for (const l of labels) {
  1070→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
  1071→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
  1072→        if (domain && l.name.toLowerCase().includes(domain)) {
  1073→          emailOwner.set(email, l.id);
  1074→          break;
  1075→        }
  1076→      }
  1077→    }
  1078→    if (!emailOwner.has(email)) {
  1079→      for (const l of labels) {
  1080→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
  1081→          emailOwner.set(email, l.id);
  1082→          break;
  1083→        }
  1084→      }
  1085→    }
  1086→  }
  1087→
  1088→  return labels.map(l => {
  1089→    if (!l.emails || !Array.isArray(l.emails)) return l;
  1090→    const cleanedEmails = l.emails.filter(e => {
  1091→      const key = e?.toLowerCase().trim();
  1092→      if (!key) return false;
  1093→      if (!corruptedEmails.has(key)) return true;
  1094→      return emailOwner.get(key) === l.id;
  1095→    });
  1096→    const cleanedContactInfo = cleanedEmails[0] || "";
  1097→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
  1098→      console.log(
  1099→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
  1100→      );
  1101→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
  1102→    }
  1103→    return l;
  1104→  });
  1105→}
