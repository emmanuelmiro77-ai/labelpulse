    90→// ==================== ROBUST STORAGE ====================
    91→// Custom storage with data protection:
    92→// 1. Primary localStorage key
    93→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)
    94→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
    95→// 4. Rehydration guard: blocks ALL writes until Zustand persist loads data
    96→
    97→const PRIMARY_KEY = "labelpulse-storage";
    98→const BACKUP_KEY = "labelpulse-storage-backup";
    99→const SEED_LABEL_COUNT = labelData.labels.length;
   100→
   101→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   102→// Before rehydration, the store contains seed data which would overwrite user data.
   103→let _rehydrated = false;
   104→
   105→function safeJsonParse(str: string | null): any | null {
   106→  if (!str) return null;
   107→  try {
   108→    return JSON.parse(str);
   109→  } catch {
   110→    return null;
   111→  }
   112→}
   113→
   114→function safeLocalStorageGet(key: string): string | null {
   115→  try {
   116→    return localStorage.getItem(key);
   117→  } catch {
   118→    return null;
   119→  }
   120→}
   121→
   122→function safeLocalStorageSet(key: string, value: string): boolean {
   123→  try {
   124→    localStorage.setItem(key, value);
   125→    return true;
   126→  } catch (e) {
   127→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   128→    return false;
   129→  }
   130→}
   131→
   132→function safeLocalStorageRemove(key: string): void {
   133→  try {
   134→    localStorage.removeItem(key);
   135→  } catch {
   136→    // Ignore
   137→  }
   138→}

   140→/**
   141→ * Check if persisted data has user edits (emails, notes, links, etc.)
   142→ * This helps us detect if the store was accidentally reset to seed data.
   143→ */
   144→function countUserEditedLabels(labels: any[]): number {
   145→  if (!Array.isArray(labels)) return 0;
   146→  return labels.filter((l) =>
   147→    (l.emails && l.emails.length > 0) ||
   148→    (l.notes && l.notes.trim() !== "") ||
   149→    (l.website && l.website.trim() !== "") ||
   150→    (l.demoLink && l.demoLink.trim() !== "") ||
   151→    (l.socialLink && l.socialLink.trim() !== "") ||
   152→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   153→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   154→    l.isCustom === true
   155→  ).length;
   156→}
   157→
   158→function getTotalLabelCount(data: any): number {
   159→  if (!data) return 0;
   160→  if (Array.isArray(data.labels)) return data.labels.length;
   161→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   162→  return 0;
   163→}
   164→
   165→/**
   166→ * Extract user-edited label count from raw localStorage string.
   167→ * Used by the write guard to check if a write would cause data loss.
   168→ */
   169→function getUserEditCountFromRaw(raw: string | null): number {
   170→  if (!raw) return 0;
   171→  const data = safeJsonParse(raw);
   172→  if (!data) return 0;
   173→  const labels = data.state?.labels || data.labels || [];
   174→  return countUserEditedLabels(labels);
   175→}

   177→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   178→
   179→const robustStorage: StateStorage = {
   180→  getItem: (name: string): string | null => {
   181→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   182→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   183→
   184→    const primaryData = safeJsonParse(primaryRaw);
   185→    const backupData = safeJsonParse(backupRaw);
   186→
   187→    let result: string | null = null;
   188→
   189→    // If primary is missing/corrupt, try backup
   190→    if (!primaryData) {
   191→      if (backupData) {
   192→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   193→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   194→        result = backupRaw;
   195→      }
   196→      // Both empty — first visit, result stays null
   197→    } else if (backupData) {
   198→      // Both exist — check for data loss
   199→      const primaryLabelCount = getTotalLabelCount(primaryData);
   200→      const backupLabelCount = getTotalLabelCount(backupData);
   201→      const primaryUserEdits = primaryData.state?.labels
   202→        ? countUserEditedLabels(primaryData.state.labels)
   203→        : countUserEditedLabels(primaryData.labels || []);
   204→      const backupUserEdits = backupData.state?.labels
   205→        ? countUserEditedLabels(backupData.state.labels)
   206→        : countUserEditedLabels(backupData.labels || []);
   207→
   208→      // If primary has fewer labels than backup AND backup has more or equal user data
   209→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   210→        console.warn(
   211→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   212→        );
   213→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   214→        result = backupRaw;
   215→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   216→        // If primary has fewer user edits than backup, use backup
   217→        console.warn(
   218→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   219→        );
   220→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   221→        result = backupRaw;
   222→      } else {
   223→        result = primaryRaw;
   224→      }
   225→    } else {
   226→      // Primary exists, no backup — normal case
   227→      result = primaryRaw;
   228→    }
   229→
   230→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   231→    // merged data immediately after this getItem returns, and we MUST allow that write.
   232→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   233→    _rehydrated = true;
   234→
   235→    return result;
   236→  },
   237→
   238→  setItem: (name: string, value: string): void => {
   239→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   240→    // Before rehydration, the store only has seed data — writing it would overwrite
   241→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   242→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   243→    if (!_rehydrated) {
   244→      console.warn(
   245→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   246→      );
   247→      return;
   248→    }
   249→
   250→    // Write to primary immediately
   251→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   252→
   253→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   254→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   255→    safeLocalStorageSet(BACKUP_KEY, value);
   256→
   257→    if (!primaryOk) {
   258→      console.error("[LabelPulse Storage] Primary write failed!");
   259→    }
   260→  },
   261→
   262→  removeItem: (name: string): void => {
   263→    safeLocalStorageRemove(PRIMARY_KEY);
   264→    safeLocalStorageRemove(BACKUP_KEY);
   265→  },
   266→};
   267→
   268→/**
   269→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   270→ * Since backup is now written on every setItem, this is a safety net.
   271→ */
   272→export function forceBackupNow(): void {
   273→  if (typeof window === "undefined") return;
   274→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   275→  if (primaryRaw) {
   276→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   277→  }
   278→}

   343→/**
   344→ * Merge helper: preserves ALL user-editable fields from existing label
   345→ * when imported label has empty/default values for those fields.
   346→ */
   347→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
   348→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
   349→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
   350→  const hasNewRankData = Object.keys(imported.rankByGenre || {}).length > 0;
   351→  const currentRank = hasNewRankData && Object.keys(existing.rankByGenre || {}).length > 0
   352→    ? { ...existing.rankByGenre }
   353→    : existing.prevRankByGenre || {};
   354→
   355→  return {
   356→    // User-editable data — ALWAYS prefer existing if it has real data
   357→    emails: imported.emails?.length ? imported.emails : existing.emails,
   358→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
   359→    website: imported.website?.trim() || existing.website,
   360→    demoLink: imported.demoLink?.trim() || existing.demoLink,
   361→    socialLink: imported.socialLink?.trim() || existing.socialLink,
   362→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   363→    notes: imported.notes?.trim() || existing.notes,
   364→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
   365→
   366→    // Beatport/ranking data — prefer imported (it's the fresh data)
   367→    genres: imported.genres?.length ? imported.genres : existing.genres,
   368→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
   369→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
   370→    trending: imported.trending || existing.trending,
   371→    trendingRankByGenre: Object.keys(imported.trendingRankByGenre || {}).length ? imported.trendingRankByGenre : existing.trendingRankByGenre,
   372→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
   373→
   374→    // Ranking history — save current rank as "previous" when new data arrives
   375→    prevRankByGenre: currentRank,
   376→  };
   377→}

   403→/**
   404→ * Repair corrupted label data.
   405→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   406→ */
   407→function repairLabelData(labels: Label[]): Label[] {
   408→  const emailLabelCount = new Map<string, number>();
   409→  for (const l of labels) {
   410→    if (l.emails && Array.isArray(l.emails)) {
   411→      for (const e of l.emails) {
   412→        if (e && e.trim()) {
   413→          const key = e.toLowerCase().trim();
   414→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   415→        }
   416→      }
   417→    }
   418→  }
   419→
   420→  const corruptedEmails = new Set<string>();
   421→  for (const [email, count] of emailLabelCount) {
   422→    if (count > 5) {
   423→      corruptedEmails.add(email);
   424→      console.warn(
   425→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   426→      );
   427→    }
   428→  }
   429→
   430→  if (corruptedEmails.size === 0) {
   431→    return labels;
   432→  }
   433→
   434→  const emailOwner = new Map<string, string>();
   435→  for (const email of corruptedEmails) {
   436→    for (const l of labels) {
   437→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   438→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   439→        if (domain && l.name.toLowerCase().includes(domain)) {
   440→          emailOwner.set(email, l.id);
   441→          break;
   442→        }
   443→      }
   444→    }
   445→    if (!emailOwner.has(email)) {
   446→      for (const l of labels) {
   447→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   448→          emailOwner.set(email, l.id);
   449→          break;
   450→        }
   451→      }
   452→    }
   453→  }
   454→
   455→  return labels.map(l => {
   456→    if (!l.emails || !Array.isArray(l.emails)) return l;
   457→    const cleanedEmails = l.emails.filter(e => {
   458→      const key = e?.toLowerCase().trim();
   459→      if (!key) return false;
   460→      if (!corruptedEmails.has(key)) return true;
   461→      return emailOwner.get(key) === l.id;
   462→    });
   463→    const cleanedContactInfo = cleanedEmails[0] || "";
   464→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   465→      console.log(
   466→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   467→      );
   468→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   469→    }
   470→    return l;
   471→  });
   472→}

   745→    {

   772→            }));

   774→          if (persisted.demos) {
   775→            persisted.demos = persisted.demos.map((d: any) => ({
   776→              ...d,

   779→              notes: d.notes || "",

   782→            }));

   786→          }
   787→          if (!persisted.userProfile) {

   789→          }
   790→        }
   791→        if (version < 6) {
   792→          if (!persisted.gmailAuth) {

   816→      partialize: (state) => ({
   817→        labels: state.labels,
   818→        demos: state.demos,

   820→        locale: state.locale,

   826→      merge: (persistedState: any, currentState: any) => {
   827→        // SIMPLE & SAFE merge: preserve persisted data, add defaults, append new seed labels
   828→        // CRITICAL: We must NEVER modify persisted user data during merge.
   829→
   830→        if (!persistedState) {
   831→          return currentState;
   832→        }
   833→
   834→        const merged = { ...currentState, ...persistedState };
   835→

   837→          // Step 1: Add safe defaults to ALL persisted labels (never overwrite existing values)

   839→            ...LABEL_DEFAULTS,
   840→            ...l,
   841→          }));

   844→          const persistedIds = new Set(persistedState.labels.map((l: any) => l.id));
   845→          const persistedNames = new Set(
   846→            persistedState.labels
   847→              .map((l: any) => l.name?.toLowerCase().trim())
   848→              .filter(Boolean)
   849→          );
   850→          const seedLabels = buildLabelsFromData();
   851→          for (const seed of seedLabels) {
   852→            if (!persistedIds.has(seed.id) && !persistedNames.has(seed.name.toLowerCase().trim())) {
   853→              merged.labels.push(seed);
   854→            }
   855→          }
   856→
   857→          // Step 3: Repair corrupted data (e.g., same email on too many labels)
   858→          merged.labels = repairLabelData(merged.labels);
   859→        }
   860→
   861→        return merged;
   862→      },
   863→      onRehydrateStorage: () => (state, error) => {

   883→          }, 0);
   884→        }
   885→
   886→        // Force an immediate backup after rehydration

   888→      },
   889→    }

   893→// Helper: get tier for a label in a given genre
   894→export function getLabelTier(
   895→  label: Label,
   896→  genre?: string
   897→): "top" | "mid" | "emerging" | null {
   898→  if (!genre) genre = label.genres?.[0];
   899→  if (!genre) return null;
   900→  const rank = label.rankByGenre?.[genre];
   901→  if (!rank) return label.trending ? "emerging" : null;
   902→  if (rank <= 20) return "top";
   903→  if (rank <= 50) return "mid";
   904→  return label.trending ? "emerging" : null;
   905→}

   907→// ==================== CLOUD SYNC ====================
   908→// Sistema di sincronizzazione con Supabase.
   909→// - Ogni modifica ai dati viene salvata su Supabase (debounced)
   910→// - All'avvio, i dati vengono caricati dal cloud se più recenti
   911→// - Se Supabase non è configurato, funziona solo con localStorage
   912→
   913→let _cloudSyncTimer: ReturnType<typeof setTimeout> | null = null;
   914→const CLOUD_SYNC_DEBOUNCE_MS = 3000; // 3 secondi di debounce
   915→
   916→/**
   917→ * Sincronizza lo stato corrente con Supabase (debounced).
   918→ * Chiamata dopo ogni azione utente che modifica i dati.
   919→ */
   920→export function syncToCloud(): void {
   921→  if (!isSupabaseConfigured) return;
   922→
   923→  if (_cloudSyncTimer) {
   924→    clearTimeout(_cloudSyncTimer);
   925→  }
   926→
   927→  _cloudSyncTimer = setTimeout(async () => {
   928→    _cloudSyncTimer = null;
   929→    const state = useAppStore.getState();
   930→    const dataToSync = {
   931→      labels: state.labels,
   932→      demos: state.demos,
   933→      activeTab: state.activeTab,
   934→      locale: state.locale,
   935→      userProfile: state.userProfile,
   936→      gmailAuth: state.gmailAuth,
   937→      rankingsUpdatedAt: state.rankingsUpdatedAt,
   938→      lastSavedAt: state.lastSavedAt,
   939→    };
   940→    await saveStateToCloud(dataToSync);
   941→  }, CLOUD_SYNC_DEBOUNCE_MS);
   942→}

   945→ * Forza la sincronizzazione immediata con Supabae (senza debounce).
   946→ * Usata quando la pagina sta per essere chiusa o nascosta.
   947→ */
   948→export async function forceCloudSync(): Promise<void> {
   949→  if (!isSupabaseConfigured) return;
   950→
   951→  if (_cloudSyncTimer) {
   952→    clearTimeout(_cloudSyncTimer);
   953→    _cloudSyncTimer = null;
   954→  }
   955→
   956→  const state = useAppStore.getState();
   957→  const dataToSync = {
   958→    labels: state.labels,
   959→    demos: state.demos,
   960→    activeTab: state.activeTab,
   961→    locale: state.locale,
   962→    userProfile: state.userProfile,
   963→    gmailAuth: state.gmailAuth,
   964→    rankingsUpdatedAt: state.rankingsUpdatedAt,
   965→    lastSavedAt: state.lastSavedAt,
   966→  };
   967→  await saveStateToCloud(dataToSync);
   968→}

   970→/**
   971→ * Carica i dati dal cloud e aggiorna lo store se i dati cloud sono più recenti.
   972→ * Chiamata una volta all'avvio dell'app, dopo la reidratazione da localStorage.
   973→ */
   974→export async function loadFromCloud(): Promise<void> {
   975→  if (!isSupabaseConfigured) {
   976→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
   977→    useAppStore.setState({ hasCloudSynced: true });
   978→    return;
   979→  }
   980→
   981→  try {
   982→    const cloudData = await loadStateFromCloud();
   983→    if (!cloudData) {
   984→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
   985→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
   986→      await forceCloudSync();
   987→      useAppStore.setState({ hasCloudSynced: true });
   988→      return;
   989→    }
   990→
   991→    const cloud = cloudData as any;
   992→    const localState = useAppStore.getState();
   993→
   994→    // Confronta i timestamp per decidere quali dati usare
   995→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
   996→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
   997→
   998→    if (cloudSavedAt > localSavedAt) {
   999→      // I dati cloud sono più recenti — aggiorna lo store locale
  1000→      console.log(
  1001→        `[LabelPulse Cloud] Cloud data is newer (cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}). Updating local store.`
  1002→      );
  1003→
  1004→      // Usa la stessa logica di merge della reidratazione
  1005→      const merged = mergeCloudData(cloud, localState);
  1006→      useAppStore.setState({
  1007→        ...merged,
  1008→        hasCloudSynced: true,
  1009→      });
  1010→    } else {
  1011→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1012→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1013→      useAppStore.setState({ hasCloudSynced: true });
  1014→    }
  1015→  } catch (err) {
  1016→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1017→    useAppStore.setState({ hasCloudSynced: true });
  1018→  }
  1019→}

  1021→/**
  1022→ * Merge dei dati cloud con quelli locali, preservando i dati utente.
  1023→ * Usa la stessa logica della funzione merge di Zustand persist.
  1024→ */
  1025→function mergeCloudData(cloudData: any, localState: any): Partial<AppState> {
  1026→  const merged: Partial<AppState> = {};
  1027→
  1028→  // Labels: applica defaults e merge con seed data, come nella reidratazione
  1029→  if (cloudData.labels && Array.isArray(cloudData.labels) && cloudData.labels.length > 0) {
  1030→    merged.labels = cloudData.labels.map((l: any) => ({
  1031→      ...LABEL_DEFAULTS,
  1032→      ...l,
  1033→    }));
  1034→
  1035→    // Aggiungi seed labels mancanti
  1036→    const cloudIds = new Set(cloudData.labels.map((l: any) => l.id));
  1037→    const cloudNames = new Set(
  1038→      cloudData.labels.map((l: any) => l.name?.toLowerCase().trim()).filter(Boolean)
  1039→    );
  1040→    const seedLabels = buildLabelsFromData();
  1041→    for (const seed of seedLabels) {
  1042→      if (!cloudIds.has(seed.id) && !cloudNames.has(seed.name.toLowerCase().trim())) {
  1043→        (merged.labels as Label[]).push(seed);
  1044→      }
  1045→    }
  1046→
  1047→    // Repair corrupted data
  1048→    merged.labels = repairLabelData(merged.labels);
  1049→  } else {
  1050→    merged.labels = localState.labels;
  1051→  }
  1052→
  1053→  // Demos
  1054→  merged.demos = Array.isArray(cloudData.demos) ? cloudData.demos : localState.demos;
  1055→
  1056→  // Simple fields
  1057→  merged.activeTab = cloudData.activeTab || localState.activeTab;
  1058→  merged.locale = cloudData.locale || localState.locale;
  1059→  merged.userProfile = cloudData.userProfile || localState.userProfile;
  1060→  merged.gmailAuth = cloudData.gmailAuth || localState.gmailAuth;
  1061→  merged.rankingsUpdatedAt = cloudData.rankingsUpdatedAt ?? localState.rankingsUpdatedAt;
  1062→  merged.lastSavedAt = cloudData.lastSavedAt ?? localState.lastSavedAt;
  1063→
  1064→  return merged;
  1065→}
