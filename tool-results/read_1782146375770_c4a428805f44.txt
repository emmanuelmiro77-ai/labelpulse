    54→    54→function setStatus(newStatus: CloudSyncStatus, errorMsg: string | null = null) {
    55→    55→  _status = newStatus;
    56→    56→  if (errorMsg !== null) _lastError = errorMsg;

    68→    68→/**
    69→    69→ * Read Supabase credentials from the Zustand store (userProfile).
    70→    70→ * Falls back to env vars if set (for backward compatibility).
    71→    71→ *
    72→    72→ * IMPORTANT: The URL is normalized via normalizeSupabaseUrl() before use.
    73→    73→ * This handles common user mistakes like pasting the dashboard URL
    74→    74→ * (https://supabase.com/dashboard/project/<ref>) instead of the API URL
    75→    75→ * (https://<ref>.supabase.co).
    76→    76→ */
    77→    77→function readCredentials(): { url: string; anonKey: string } {
    78→    78→  if (typeof window !== "undefined") {
    79→    79→    try {
    80→    80→      const state = useAppStore.getState();
    81→    81→      const rawUrl = (state.userProfile as any)?.supabaseUrl?.trim() || "";
    82→    82→      const anonKey = (state.userProfile as any)?.supabaseAnonKey?.trim() || "";
    83→    83→      const url = normalizeSupabaseUrl(rawUrl);
    84→    84→      if (url && anonKey) return { url, anonKey };
    85→    85→      // If only URL is present (no anon key), still return normalized url
    86→    86→      // so we can produce a clearer "missing anon key" error downstream.
    87→    87→      if (url && !anonKey) return { url, anonKey: "" };
    88→    88→    } catch {
    89→    89→      // store not ready yet
    90→    90→    }
    91→    91→  }
    92→    92→  // Fallback to env vars (legacy)
    93→    93→  const envUrl = normalizeSupabaseUrl(
    94→    94→    process.env.NEXT_PUBLIC_SUPABASE_URL || ""
    95→    95→  );
    96→    96→  const envKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || "";
    97→    97→  return { url: envUrl, anonKey: envKey };
    98→    98→}

   100→   100→/**
   101→   101→ * Normalize a user-supplied Supabase URL into the API format
   102→   102→ * (https://<project-ref>.supabase.co).
   103→   103→ *
   104→   104→ * Handles:
   105→   105→ *   - Dashboard URLs: https://supabase.com/dashboard/project/<ref>/...
   106→   106→ *     → https://<ref>.supabase.co
   107→   107→ *   - URLs with trailing slash or path: stripped
   108→   108→ *   - URLs without protocol: https:// prefix added
   109→   109→ *   - URLs that are already correct: returned unchanged
   110→   110→ *
   111→   111→ * Returns "" if the input is empty or doesn't look like a Supabase URL.
   112→   112→ */
   113→   113→export function normalizeSupabaseUrl(input: string): string {
   114→   114→  if (!input) return "";
   115→   115→  let url = input.trim();
   116→   116→
   117→   117→  // Add protocol if missing
   118→   118→  if (!/^https?:\/\//i.test(url)) {
   119→   119→    url = "https://" + url;
   120→   120→  }
   121→   121→
   122→   122→  // Case 1: Dashboard URL → extract project ref
   123→   123→  //   https://supabase.com/dashboard/project/<ref>
   124→   124→  //   https://supabase.com/dashboard/project/<ref>/settings/api
   125→   125→  const dashMatch = url.match(
   126→   126→    /supabase\.com\/dashboard\/project\/([a-z0-9]+)/i
   127→   127→  );
   128→   128→  if (dashMatch) {
   129→   129→    return `https://${dashMatch[1].toLowerCase()}.supabase.co`;
   130→   130→  }
   131→   131→
   132→   132→  // Case 2: Already in API format (https://<ref>.supabase.co[/...])
   133→   133→  const apiMatch = url.match(/^(https?:\/\/)([a-z0-9]+)\.supabase\.(co|com|in)/i);
   134→   134→  if (apiMatch) {
   135→   135→    // Strip any trailing path/slash, keep just the host
   136→   136→    try {
   137→   137→      const u = new URL(url);
   138→   138→      return `${u.protocol}//${u.hostname.toLowerCase()}`;
   139→   139→    } catch {
   140→   140→      return url;
   141→   141→    }
   142→   142→  }
   143→   143→
   144→   144→  // Case 3: Unknown format — return as-is, will fail at fetch time
   145→   145→  // (better to let the user see the error than silently mangle it)
   146→   146→  return url;
   147→   147→}

   149→   149→/**
   150→   150→ * Returns a human-readable validation error for the current credentials,
   151→   151→ * or null if credentials look valid. Used to give the user actionable
   152→   152→ * feedback in the Profilo UI before they even try to sync.
   153→   153→ */
   154→   154→export function validateSupabaseCredentials(): string | null {
   155→   155→  if (typeof window === "undefined") return null;
   156→   156→  const state = useAppStore.getState();
   157→   157→  const rawUrl = (state.userProfile as any)?.supabaseUrl?.trim() || "";
   158→   158→  const anonKey = (state.userProfile as any)?.supabaseAnonKey?.trim() || "";
   159→   159→
   160→   160→  if (!rawUrl && !anonKey) return null; // nothing configured yet, no error
   161→   161→  if (!rawUrl) return "Manca il Project URL.";
   162→   162→  if (!anonKey) return "Manca la anon key — copiala da Supabase → Project Settings → API.";
   163→   163→
   164→   164→  // Check if user pasted a dashboard URL (we'll auto-fix it, but warn)
   165→   165→  if (/supabase\.com\/dashboard\/project\//i.test(rawUrl)) {
   166→   166→    // Already auto-normalized — no error needed, just inform
   167→   167→    return null;
   168→   168→  }
   169→   169→
   170→   170→  // Check URL format
   171→   171→  const normalized = normalizeSupabaseUrl(rawUrl);
   172→   172→  if (!/^https:\/\/[a-z0-9]+\.supabase\.(co|com|in)/i.test(normalized)) {
   173→   173→    return "L'URL non è nel formato corretto. Deve essere https://<project-ref>.supabase.co";
   174→   174→  }
   175→   175→
   176→   176→  // Check anon key length (typical Supabase anon keys are JWT, ~200+ chars)
   177→   177→  if (anonKey.length < 30) {
   178→   178→    return "La anon key sembra troppo corta. Assicurati di aver copiato la 'anon public' key intera.";
   179→   179→  }
   180→   180→
   181→   181→  return null;
   182→   182→}

   187→   187→export function isSupabaseConfigured(): boolean {

   192→   192→/**
   193→   193→ * Translate a low-level fetch/supabase error into an actionable Italian
   194→   194→ * message. The browser surfaces network errors (CORS, DNS, refused
   195→   195→ * connection) as "TypeError: Failed to fetch" which is useless to the
   196→   196→ * user — we replace it with a hint about URL/key correctness.
   197→   197→ */
   198→   198→function humanizeCloudError(err: any): string {
   199→   199→  const raw = err?.message || err?.toString?.() || String(err);
   200→   200→  const lower = raw.toLowerCase();
   201→   201→
   202→   202→  if (lower.includes("failed to fetch") || lower.includes("networkerror")) {
   203→   203→    return (
   204→   204→      "Impossibile contattare Supabase. Cause possibili:\n" +
   205→   205→      "• URL non corretto (deve essere https://<project-ref>.supabase.co, NON l'URL del dashboard)\n" +
   206→   206→      "• Anon key mancante o sbagliata\n" +
   207→   207→      "• Il progetto Supabase è in pausa (riattivalo dal dashboard)"
   208→   208→    );
   209→   209→  }
   210→   210→  if (lower.includes("cors")) {
   211→   211→    return "Blocco CORS: verifica che l'URL sia https://<project-ref>.supabase.co e che la anon key sia corretta.";
   212→   212→  }
   213→   213→  if (lower.includes("jwt") || lower.includes("invalid api key")) {
   214→   214→    return "Anon key non valida. Copia da Supabase → Project Settings → API → 'anon public' (NON la service_role key).";
   215→   215→  }
   216→   216→  if (lower.includes("relation") && lower.includes("does not exist")) {
   217→   217→    return (
   218→   218→      "La tabella 'app_state' non esiste su Supabase. Vai in Supabase → SQL Editor " +
   219→   219→      "→ incolla il contenuto di supabase-schema.sql → esegui."
   220→   220→    );
   221→   221→  }
   222→   222→  if (lower.includes("permission denied") || lower.includes("rls")) {
   223→   223→    return (
   224→   224→      "Permessi insufficienti sulla tabella 'app_state'. Esegui di nuovo " +
   225→   225→      "supabase-schema.sql in Supabase → SQL Editor (contiene i GRANT necessari)."
   226→   226→    );
   227→   227→  }
   228→   228→  return raw;
   229→   229→}

   231→   231→/**
   232→   232→ * Get or create the Supabase client. If the credentials have changed
   233→   233→ * since the last call, a new client is created and the old realtime
   234→   234→ * channel is torn down.
   235→   235→ */
   236→   236→export function getSupabase(): SupabaseClient | null {
   237→   237→  if (typeof window === "undefined") return null;
   238→   238→
   239→   239→  const { url, anonKey } = readCredentials();
   240→   240→  if (!url || !anonKey) {
   241→   241→    // No credentials — reset everything
   242→   242→    if (_supabase) {
   243→   243→      teardownRealtime();
   244→   244→      _supabase = null;
   245→   245→      _currentCredsKey = "";
   246→   246→    }
   247→   247→    // Surface a clear error if URL is present but anon key is missing
   248→   248→    if (url && !anonKey) {
   249→   249→      setStatus(
   250→   250→        "error",
   251→   251→        "Manca la anon key — copiala da Supabase → Project Settings → API → 'anon public'"
   252→   252→      );
   253→   253→    } else if (_status !== "unconfigured" && _status !== "error") {
   254→   254→      setStatus("unconfigured");
   255→   255→    }
   256→   256→    return null;
   257→   257→  }
   258→   258→
   259→   259→  const credsKey = `${url}|${anonKey}`;
   260→   260→  if (_supabase && _currentCredsKey === credsKey) {
   261→   261→    return _supabase;
   262→   262→  }
   263→   263→
   264→   264→  // Credentials changed (or first time) — create a fresh client
   265→   265→  try {
   266→   266→    if (_supabase) {
   267→   267→      teardownRealtime();
   268→   268→      _supabase = null;
   269→   269→    }
   270→   270→    _supabase = createClient(url, anonKey, {
   271→   271→      auth: { persistSession: false },
   272→   272→    });
   273→   273→    _currentCredsKey = credsKey;
   274→   274→    return _supabase;
   275→   275→  } catch (err) {
   276→   276→    console.error("[LabelPulse Cloud] Failed to create client:", err);
   277→   277→    setStatus("error", err instanceof Error ? err.message : String(err));
   278→   278→    return null;
   279→   279→  }
   280→   280→}

   282→   282→// ==================== CLOUD SYNC FUNCTIONS ====================
   283→   283→
   284→   284→const CLOUD_TABLE = "app_state";
   285→   285→// Backward-compat default for users who have not logged in with Google yet.
   286→   286→// Once they do, the row id becomes their lowercased email — so each Google
   287→   287→// account maps to its own isolated row in app_state (multi-profile support).
   288→   288→const DEFAULT_CLOUD_ROW_ID = "default";
   289→   289→
   290→   290→// ⚠️ Artists are stored in a SEPARATE cloud row (suffixed "_artists") because:
   291→   291→// 1. A full Beatport scrape produces ~3400 artists / ~3000 tracks totaling ~9MB.
   292→   292→//    Putting it in the main app_state row alongside labels/demos/profile would
   293→   293→//    risk hitting Supabase's JSONB column size limit and slow down EVERY sync
   294→   294→//    (even tiny profile updates would re-upload 9MB).
   295→   295→// 2. Artists change infrequently (only when user re-scrapes), while labels/
   296→   296→//    profile change often (notes, tier, status). Splitting them keeps the
   297→   297→//    frequent-sync path lightweight.
   298→   298→// 3. On a new device, the user gets labels/profile/snapshots from the main
   299→   299→//    row, then this function pulls artists separately. Without this, a user
   300→   300→//    logging in on a new phone would see 0 artists even with perfect cloud sync.
   301→   301→function getArtistsCloudRowId(): string {
   302→   302→  return `${getCloudRowId()}_artists`;
   303→   303→}
   304→   304→
   305→   305→// Module-level holder for the current user's email. Populated by
   306→   306→// useAuthEffect() (see use-auth.ts) — must be set BEFORE any cloud sync
   307→   307→// operation, otherwise we'd fall back to "default" and mix profiles.
   308→   308→let _currentUserEmail: string | null = null;
   309→   309→
   310→   310→/**
   311→   311→ * Set the current user's email (called from useAuthEffect on session change).
   312→   312→ * Pass null on logout. When null, cloud sync falls back to the legacy "default"
   313→   313→ * row id (so existing users without Google login keep their data).
   314→   314→ */
   315→   315→export function setCurrentUserEmail(email: string | null): void {
   316→   316→  const normalized = email?.trim().toLowerCase() || null;
   317→   317→  if (_currentUserEmail !== normalized) {
   318→   318→    _currentUserEmail = normalized;
   319→   319→    // Invalidate the cached Supabase client so the next call re-evaluates
   320→   320→    // credentials + row id. Not strictly required (creds don't depend on email)
   321→   321→    // but keeps the state consistent.
   322→   322→  }
   323→   323→}
   324→   324→
   325→   325→/**
   326→   326→ * Returns the row id to use for cloud sync. Uses the logged-in user's email
   327→   327→ * if available, falls back to "default" for legacy / unauthenticated users.
   328→   328→ */
   329→   329→function getCloudRowId(): string {
   330→   330→  return _currentUserEmail || DEFAULT_CLOUD_ROW_ID;
   331→   331→}

   333→   333→/**
   334→   334→ * Salva lo stato completo dell'app su Supabase.
   335→   335→ * Usa upsert per creare o aggiornare il record.
   336→   336→ */
   337→   337→export async function saveStateToCloud(data: object): Promise<boolean> {
   338→   338→  const supabase = getSupabase();
   339→   339→  if (!supabase) return false;
   340→   340→
   341→   341→  setStatus("syncing");
   342→   342→
   343→   343→  try {
   344→   344→    const { error } = await supabase.from(CLOUD_TABLE).upsert(
   345→   345→      {
   346→   346→        id: getCloudRowId(),
   347→   347→        data: data,
   348→   348→        updated_at: new Date().toISOString(),
   349→   349→      },
   350→   350→      { onConflict: "id" }
   351→   351→    );
   352→   352→
   353→   353→    if (error) {
   354→   354→      console.error("[LabelPulse Cloud] Save error:", error.message);
   355→   355→      setStatus("error", humanizeCloudError(error));
   356→   356→      return false;
   357→   357→    }
   358→   358→
   359→   359→    setStatus("synced");
   360→   360→    return true;
   361→   361→  } catch (err) {
   362→   362→    console.error("[LabelPulse Cloud] Save exception:", err);
   363→   363→    setStatus("error", humanizeCloudError(err));
   364→   364→    return false;
   365→   365→  }
   366→   366→}
   367→   367→
   368→   368→/**
   369→   369→ * Carica lo stato completo dell'app da Supabase.
   370→   370→ * Ritorna null se non ci sono dati o se Supabase non è configurato.
   371→   371→ */
   372→   372→export async function loadStateFromCloud(): Promise<object | null> {
   373→   373→  const supabase = getSupabase();
   374→   374→  if (!supabase) return null;
   375→   375→
   376→   376→  try {
   377→   377→    const { data, error } = await supabase
   378→   378→      .from(CLOUD_TABLE)
   379→   379→      .select("data, updated_at")
   380→   380→      .eq("id", getCloudRowId())
   381→   381→      .single();
   382→   382→
   383→   383→    if (error) {
   384→   384→      if (error.code === "PGRST116") {
   385→   385→        // No rows returned — first time, no data in cloud yet
   386→   386→        return null;
   387→   387→      }
   388→   388→      console.error("[LabelPulse Cloud] Load error:", error.message);
   389→   389→      setStatus("error", humanizeCloudError(error));
   390→   390→      return null;
   391→   391→    }
   392→   392→
   393→   393→    if (!data?.data || Object.keys(data.data).length === 0) {
   394→   394→      return null;
   395→   395→    }
   396→   396→
   397→   397→    setStatus("synced");
   398→   398→    return data.data as object;
   399→   399→  } catch (err) {
   400→   400→    console.error("[LabelPulse Cloud] Load exception:", err);
   401→   401→    setStatus("error", humanizeCloudError(err));
   402→   402→    return null;
   403→   403→  }
   404→   404→}

   406→   406→// ==================== REALTIME SUBSCRIPTION ====================
   407→   407→
   408→   408→/**
   409→   409→ * Subscribe to changes on the app_state row so that updates from
   410→   410→ * other devices (PC ↔ phone) are reflected in near-real-time.
   411→   411→ *
   412→   412→ * When a remote UPDATE arrives, we:
   413→   413→ *   1. set _isApplyingRemoteUpdate = true (so we don't bounce the change back)
   414→   414→ *   2. fetch the latest data from cloud
   415→   415→ *   3. merge it into the local store
   416→   416→ *   4. set _isApplyingRemoteUpdate = false
   417→   417→ */
   418→   418→export function setupRealtimeSubscription(): () => void {
   419→   419→  const supabase = getSupabase();
   420→   420→  if (!supabase) return () => {};
   421→   421→
   422→   422→  // Tear down any existing channel first
   423→   423→  teardownRealtime();
   424→   424→
   425→   425→  try {
   426→   426→    const channel = supabase
   427→   427→      .channel("app_state_changes")
   428→   428→      .on(
   429→   429→        "postgres_changes",
   430→   430→        {
   431→   431→          event: "UPDATE",
   432→   432→          schema: "public",
   433→   433→          table: CLOUD_TABLE,
   434→   434→          filter: `id=eq.${getCloudRowId()}`,
   435→   435→        },
   436→   436→        async (payload: any) => {
   437→   437→          // Skip our own updates to avoid feedback loops
   438→   438→          if (_isApplyingRemoteUpdate) return;
   439→   439→
   440→   440→          const newData = payload?.new?.data;
   441→   441→          const newUpdatedAt = payload?.new?.updated_at;
   442→   442→          if (!newData) return;
   443→   443→
   444→   444→          console.log(
   445→   445→            "[LabelPulse Cloud] Realtime update received, updated_at:",
   446→   446→            newUpdatedAt
   447→   447→          );
   448→   448→
   449→   449→          // Apply the remote data to the local store
   450→   450→          _isApplyingRemoteUpdate = true;
   451→   451→          try {
   452→   452→            await applyRemoteData(newData);
   453→   453→            setStatus("synced");
   454→   454→          } finally {
   455→   455→            _isApplyingRemoteUpdate = false;
   456→   456→          }
   457→   457→        }
   458→   458→      )
   459→   459→      .subscribe((status: string) => {
   460→   460→        if (status === "SUBSCRIBED") {
   461→   461→          console.log("[LabelPulse Cloud] Realtime subscription active");
   462→   462→        } else if (status === "CHANNEL_ERROR" || status === "TIMED_OUT") {
   463→   463→          console.warn("[LabelPulse Cloud] Realtime subscription issue:", status);
   464→   464→        }
   465→   465→      });
   466→   466→
   467→   467→    _realtimeChannel = channel;
   468→   468→    return teardownRealtime;
   469→   469→  } catch (err) {

   499→   499→ * as loadFromCloud().
   500→   500→ *
   501→   501→ * CRITICAL SAFETY RULE: same as mergeCloudData in store.ts — never
   502→   502→ * overwrite local non-empty arrays with empty cloud arrays. This is
   503→   503→ * the realtime-update path, and we want the same protection here.

   505→   505→async function applyRemoteData(cloudData: any): Promise<void> {
   506→   506→  // Lazy require to avoid circular import
   507→   507→  const store = useAppStore.getState();
   508→   508→  const localLastSavedAt = store.lastSavedAt
   509→   509→    ? new Date(store.lastSavedAt).getTime()
   510→   510→    : 0;
   511→   511→  const cloudLastSavedAt = cloudData?.lastSavedAt
   512→   512→    ? new Date(cloudData.lastSavedAt).getTime()
   513→   513→    : 0;
   514→   514→
   515→   515→  // ⚠️ CRITICAL: Content-aware merge decision (same as loadFromCloud).
   516→   516→  // The old code did `if (cloudLastSavedAt <= localLastSavedAt) return;`
   517→   517→  // which would silently DROP realtime updates if the local timestamp was
   518→   518→  // newer or equal — even if the cloud had profile data local didn't have.
   519→   519→  //
   520→   520→  // This was the realtime-update variant of the same bug that wiped user
   521→   521→  // data on re-login. Always apply if cloud brings new profile data, even
   522→   522→  // if its timestamp looks "older".
   523→   523→  const cloudProfileHasData =
   524→   524→    !!cloudData?.userProfile?.artistName ||
   525→   525→    !!cloudData?.userProfile?.bio ||
   526→   526→    !!cloudData?.userProfile?.email ||
   527→   527→    !!cloudData?.userProfile?.scLink ||
   528→   528→    !!cloudData?.userProfile?.photoUrl ||
   529→   529→    (Array.isArray(cloudData?.userProfile?.links) && cloudData.userProfile.links.length > 0);
   530→   530→  const localProfileHasData =
   531→   531→    !!store.userProfile?.artistName ||
   532→   532→    !!store.userProfile?.bio ||
   533→   533→    !!store.userProfile?.email ||
   534→   534→    !!store.userProfile?.scLink ||
   535→   535→    !!store.userProfile?.photoUrl ||
   536→   536→    (Array.isArray(store.userProfile?.links) && store.userProfile.links.length > 0);
   537→   537→
   538→   538→  const cloudBringsNewProfile = cloudProfileHasData && !localProfileHasData;
   539→   539→  const cloudHasLabels = Array.isArray(cloudData?.labels) && cloudData.labels.length > 0;
   540→   540→  const cloudHasDemos = Array.isArray(cloudData?.demos) && cloudData.demos.length > 0;
   541→   541→  const cloudHasSnapshots = Array.isArray(cloudData?.rankingSnapshots) && cloudData.rankingSnapshots.length > 0;
   542→   542→  // ⚠️ CRITICAL FIX (data-loss bug 2026-06-22, post-login "no charts" v2):
   543→   543→  // Mirror the fix in store.ts:loadFromCloud — ALWAYS run merge if cloud has
   544→   544→  // ANY content, not just when local is empty. The merge function is content-
   545→   545→  // aware (unions by id, preserves Beatport fields per-genre with local wins),
   546→   546→  // so running it unconditionally only adds data, never removes it.
   547→   547→  const cloudHasAnyContent = cloudHasLabels || cloudHasDemos || cloudHasSnapshots || cloudProfileHasData;
   548→   548→  const cloudIsNewerByTimestamp = cloudLastSavedAt > localLastSavedAt;
   549→   549→  const shouldApply =
   550→   550→    cloudBringsNewProfile ||
   551→   551→    cloudHasAnyContent ||
   552→   552→    cloudIsNewerByTimestamp;
   553→   553→
   554→   554→  if (!shouldApply) {
   555→   555→    return;
   556→   556→  }
   557→   557→
   558→   558→  const merged: any = {};
   559→   559→
   560→   560→  // ⚠️ CRITICAL FIX (data-loss bug 2026-06-22):
   561→   561→  // Mirror the fix in store.ts:mergeCloudData — do UNION BY ID instead of
   562→   562→  // REPLACE for arrays. This is the realtime-update path; same protection.
   563→   563→  // Lazy-import mergeSnapshots from store to avoid circular dep at module
   564→   564→  // load time. (We already import mergeProfiles at top of file.)
   565→   565→  let mergeSnapshotsFn: any = null;
   566→   566→  try {
   567→   567→    // Re-require lazily — store.ts imports from supabase.ts at top level
   568→   568→    const storeMod = await import("./store");
   569→   569→    mergeSnapshotsFn = (storeMod as any).mergeSnapshotsPublic || null;
   570→   570→  } catch {
   571→   571→    // fall through — we'll handle snapshots below
   572→   572→  }
   573→   573→
   574→   574→  // ---------- LABELS: union by id, preserve user-edit fields from local ----------
   575→   575→  const cloudLabels = Array.isArray(cloudData.labels) ? cloudData.labels : [];
   576→   576→  const localLabels = Array.isArray(store.labels) ? store.labels : [];
   577→   577→  const labelUserEditFields = [
   578→   578→    "emails", "notes", "website", "demoLink", "socialLink",
   579→   579→    "soundcloudLink", "status", "tier", "instagramLink", "facebookLink",
   580→   580→    "bandcampLink", "beatstatsLink",
   581→   581→  ];

   583→   583→  const labelsByName = new Map<string, any>();

   585→   585→    if (!cl || typeof cl !== "object") continue;
   586→   586→    labelsById.set(cl.id, { ...cl });
   587→   587→    const nm = cl.name?.toLowerCase().trim();

   589→   589→  }

   591→   591→    if (!ll || typeof ll !== "object") continue;
   592→   592→    const nm = ll.name?.toLowerCase().trim();
   593→   593→    const existing = labelsById.get(ll.id) || (nm ? labelsByName.get(nm) : undefined);
   594→   594→    if (existing) {
   595→   595→      for (const f of labelUserEditFields) {
   596→   596→        const lv = (ll as any)[f];
   597→   597→        if (Array.isArray(lv) ? lv.length > 0 : (lv && String(lv).trim() !== "")) {
   598→   598→          (existing as any)[f] = lv;
   599→   599→        }
   600→   600→      }
   601→   601→      // ⚠️ CRITICAL FIX (data-loss bug 2026-06-22, post-login "no charts"):
   602→   602→      // Also preserve Beatport data fields (rankByGenre, pointsByGenre, etc.)
   603→   603→      // from local. Without this, a realtime echo of a partial cloud row
   604→   604→      // (e.g., one that lost rankByGenre during a previous bad sync) would
   605→   605→      // wipe the user's just-restored scraped data.
   606→   606→      // Same logic as mergeCloudData in store.ts:
   607→   607→      //   - Object fields: union by genre key, local wins on conflict
   608→   608→      //   - Array field (genres): union (dedupe)
   609→   609→      //   - Boolean (trending): true wins
   610→   610→      for (const f of ["rankByGenre", "pointsByGenre", "trendingRankByGenre", "trendingPointsByGenre", "prevRankByGenre"]) {
   611→   611→        const lv = (ll as any)[f];
   612→   612→        if (lv && typeof lv === "object" && !Array.isArray(lv)) {
   613→   613→          const cv = (existing as any)[f];
   614→   614→          const baseObj = (cv && typeof cv === "object" && !Array.isArray(cv)) ? cv : {};
   615→   615→          (existing as any)[f] = { ...baseObj, ...lv };
   616→   616→        }
   617→   617→      }
   618→   618→      if (Array.isArray(ll.genres) && ll.genres.length > 0) {
   619→   619→        const cv = Array.isArray(existing.genres) ? existing.genres : [];
   620→   620→        const seen = new Set(cv.map((g: any) => String(g).toLowerCase().trim()));
   621→   621→        const mergedGenres = [...cv];
   622→   622→        for (const g of ll.genres) {
   623→   623→          const key = String(g).toLowerCase().trim();
   624→   624→          if (key && !seen.has(key)) {
   625→   625→            seen.add(key);
   626→   626→            mergedGenres.push(g);
   627→   627→          }
   628→   628→        }
   629→   629→        existing.genres = mergedGenres;
   630→   630→      }
   631→   631→      if (ll.trending === true) {
   632→   632→        existing.trending = true;
   633→   633→      }
   634→   634→    } else {
   635→   635→      labelsById.set(ll.id, { ...ll });
   636→   636→      if (nm) labelsByName.set(nm, ll);
   637→   637→    }
   638→   638→  }
   639→   639→  if (labelsById.size > 0) {
   640→   640→    merged.labels = Array.from(labelsById.values());
   641→   641→  }
   642→   642→
   643→   643→  // ---------- DEMOS: union by id, prefer newest createdAt ----------
   644→   644→  const cloudDemos = Array.isArray(cloudData.demos) ? cloudData.demos : [];
   645→   645→  const localDemos = Array.isArray(store.demos) ? store.demos : [];
   646→   646→  const demosById = new Map<string, any>();
   647→   647→  for (const d of cloudDemos) {
   648→   648→    if (!d || typeof d !== "object" || !d.id) continue;
   649→   649→    demosById.set(d.id, d);
   650→   650→  }
   651→   651→  for (const d of localDemos) {
   652→   652→    if (!d || typeof d !== "object" || !d.id) continue;
   653→   653→    const existing = demosById.get(d.id);
   654→   654→    if (!existing) {
   655→   655→      demosById.set(d.id, d);
   656→   656→    } else {
   657→   657→      const exTs = new Date(existing.createdAt || 0).getTime();
   658→   658→      const loTs = new Date(d.createdAt || 0).getTime();
   659→   659→      if (loTs > exTs) {
   660→   660→        demosById.set(d.id, d);
   661→   661→      } else if (loTs === exTs && ((d.notes && d.notes.trim()) || (d.pitchText && d.pitchText.trim()))) {
   662→   662→        demosById.set(d.id, d);
   663→   663→      }
   664→   664→    }
   665→   665→  }
   666→   666→  merged.demos = Array.from(demosById.values());
   667→   667→
   668→   668→  if (cloudData.userProfile) {
   669→   669→    // ⚠️ CRITICAL: Use mergeProfiles so that non-empty fields from BOTH
   670→   670→    // local and cloud win. Previously this was a spread that OVERWROTE
   671→   671→    // local with cloud — so if cloud had artistName="" and local had
   672→   672→    // artistName="Emmanuel Miro", the local data was silently lost.
   673→   673→    // mergeProfiles ensures non-empty wins in both directions.
   674→   674→    const mergedProfile = mergeProfiles(store.userProfile, cloudData.userProfile);
   675→   675→    merged.userProfile = {
   676→   676→      ...(mergedProfile || cloudData.userProfile),
   677→   677→      // Preserve BYOK credentials — never overwrite with cloud versions
   678→   678→      supabaseUrl: store.userProfile.supabaseUrl,
   679→   679→      supabaseAnonKey: store.userProfile.supabaseAnonKey,
   680→   680→      cyaniteApiToken: store.userProfile.cyaniteApiToken,
   681→   681→    };
   682→   682→  }
   683→   683→  if (cloudData.gmailAuth) merged.gmailAuth = cloudData.gmailAuth;
   684→   684→  if (cloudData.rankingsUpdatedAt !== undefined) {
   685→   685→    merged.rankingsUpdatedAt = cloudData.rankingsUpdatedAt;
   686→   686→  }
   687→   687→  if (cloudData.lastSavedAt) merged.lastSavedAt = cloudData.lastSavedAt;
   688→   688→
   689→   689→  // ---------- RANKING SNAPSHOTS: union by id via mergeSnapshots ----------
   690→   690→  // mergeSnapshots is defined in store.ts. We try to use the lazy-loaded
   691→   691→  // copy; if that fails (e.g., circular dep issue), fall back to local
   692→   692→  // snapshots so we don't lose data.
   693→   693→  const cloudSnaps = Array.isArray(cloudData.rankingSnapshots) ? cloudData.rankingSnapshots : [];
   694→   694→  const localSnaps = Array.isArray(store.rankingSnapshots) ? store.rankingSnapshots : [];
   695→   695→  if (mergeSnapshotsFn && typeof mergeSnapshotsFn === "function") {
   696→   696→    merged.rankingSnapshots = mergeSnapshotsFn(localSnaps, cloudSnaps);
   697→   697→  } else {
   698→   698→    // Manual union by id as fallback
   699→   699→    const seen = new Set<string>();
   700→   700→    const out: any[] = [];
   701→   701→    for (const s of [...localSnaps, ...cloudSnaps]) {
   702→   702→      if (!s || typeof s !== "object") continue;
   703→   703→      const key = s.id || `${s.timestamp}|${s.source}`;
   704→   704→      if (seen.has(key)) continue;
   705→   705→      seen.add(key);
   706→   706→      out.push(s);
   707→   707→    }
   708→   708→    out.sort(
   709→   709→      (x, y) =>
   710→   710→        new Date(x.timestamp || 0).getTime() - new Date(y.timestamp || 0).getTime()
   711→   711→    );
   712→   712→    merged.rankingSnapshots = out;
   713→   713→  }
   714→   714→  if (cloudData.locale) merged.locale = cloudData.locale;
   715→   715→
   716→   716→  useAppStore.setState(merged);
   717→   717→
   718→   718→  // ⚠️ CRITICAL FIX (post-login "no profile / no charts" bug 2026-06-22):
   719→   719→  // After applying the remote data, immediately re-restore profile and
   720→   720→  // snapshots from sidecar backups if the live store is now empty. The
   721→   721→  // merge above should preserve local data, but in edge cases (partial
   722→   722→  // cloud row, schema drift, an older app version) the merged result can
   723→   723→  // still be missing data the user definitely had. The sidecar backups
   724→   724→  // are the safety net — splice them back in. Runs on next tick so
   725→   725→  // setState has committed first.
   726→   726→  setTimeout(() => {
   727→   727→    try {
   728→   728→      // Lazy import to avoid circular dep at module load
   729→   729→      import("./store").then((storeMod: any) => {
   730→   730→        const profileRestored = storeMod.restoreProfileFromSidecar?.();
   731→   731→        const snapsRestored = storeMod.restoreSnapshotsFromSidecar?.();
   732→   732→        if (profileRestored || snapsRestored > 0) {
   733→   733→          console.info(
   734→   734→            `[LabelPulse Cloud] Post-realtime sidecar restore: profile=${profileRestored ? "OK" : "niente"}, snaps=${snapsRestored} recuperati.`
   735→   735→          );
   736→   736→        }
   737→   737→      });
   738→   738→    } catch (e) {
   739→   739→      console.warn("[LabelPulse Cloud] Post-realtime sidecar restore failed:", e);
   740→   740→    }
   741→   741→  }, 0);
   742→   742→}

   748→   748→/**
   749→   749→ * Save artists array to cloud (separate row).
   750→   750→ * Called when artists are imported (scrape) or updated.
   751→   751→ * Uses upsert with onConflict:"id" so it replaces the previous artists row.
   752→   752→ *
   753→   753→ * NOTE: artists array can be large (~9MB for 3400 artists). The Supabase
   754→   754→ * JSONB column supports up to 1GB, so this is fine size-wise. The main
   755→   755→ * concern is upload time on slow connections — we don't block UI on this.

   757→   757→export async function saveArtistsToCloud(artists: any[]): Promise<boolean> {

   780→   780→    console.info(
   781→   781→      `[LabelPulse Cloud] Artists saved to cloud: ${artists.length} artists`

   831→   831→/**
   832→   832→ * Merge two artist arrays by id (union). When the same id appears in both,
   833→   833→ * keep the version with the most-recent `scrapedAt` (or the local one on tie).
   834→   834→ * This is the artist equivalent of mergeCloudData: safe, additive, never
   835→   835→ * removes data.
   836→   836→ *
   837→   837→ * Used by:
   838→   838→ *  - loadArtistsOnBoot (when both IDB and cloud have artists)
   839→   839→ *  - explicitMergeArtistsCloud (user-triggered from CloudRecovery panel)
   840→   840→ */
   841→   841→export function mergeArtistsArrays(local: any[], cloud: any[]): any[] {
   842→   842→  const a = Array.isArray(local) ? local : [];
   843→   843→  const b = Array.isArray(cloud) ? cloud : [];
   844→   844→  const byId = new Map<string, any>();
   845→   845→  const byName = new Map<string, any>();
   846→   846→
   847→   847→  // Insert cloud first (so local can override on conflict below)
   848→   848→  for (const c of b) {
   849→   849→    if (!c || typeof c !== "object") continue;
   850→   850→    const id = c.id || c.artistId || `_${c.name?.toLowerCase?.() || ""}`;
   851→   851→    byId.set(id, { ...c });
   852→   852→    const nm = (c.name || c.artistName || "").toString().toLowerCase().trim();
   853→   853→    if (nm) byName.set(nm, { ...c });
   854→   854→  }
   855→   855→
   856→   856→  for (const l of a) {
   857→   857→    if (!l || typeof l !== "object") continue;
   858→   858→    const id = l.id || l.artistId || `_${(l.name || l.artistName || "").toLowerCase()}`;
   859→   859→    const nm = (l.name || l.artistName || "").toString().toLowerCase().trim();
   860→   860→    const existing = byId.get(id) || (nm ? byName.get(nm) : undefined);
   861→   861→
   862→   862→    if (!existing) {
   863→   863→      byId.set(id, { ...l });
   864→   864→      if (nm) byName.set(nm, { ...l });
   865→   865→      continue;
   866→   866→    }
   867→   867→
   868→   868→    // Both have this artist — keep the one with most tracks / most recent scrape
   869→   869→    const exTracks = Array.isArray(existing.tracksByGenre)
   870→   870→      ? Object.values(existing.tracksByGenre).flat().length
   871→   871→      : (Array.isArray(existing.tracks) ? existing.tracks.length : 0);
   872→   872→    const loTracks = Array.isArray(l.tracksByGenre)
   873→   873→      ? Object.values(l.tracksByGenre).flat().length
   874→   874→      : (Array.isArray(l.tracks) ? l.tracks.length : 0);
   875→   875→
   876→   876→    const exTs = new Date(existing.scrapedAt || existing.updatedAt || 0).getTime();
   877→   877→    const loTs = new Date(l.scrapedAt || l.updatedAt || 0).getTime();
   878→   878→
   879→   879→    // Local wins if it has more tracks OR more recent scrape OR has bio when existing doesn't
   880→   880→    const localHasBio = !!((l as any).bio || (l as any).bioSummary);
   881→   881→    const exHasBio = !!((existing as any).bio || (existing as any).bioSummary);
   882→   882→
   883→   883→    let pickLocal = false;
   884→   884→    if (loTracks > exTracks) pickLocal = true;
   885→   885→    else if (loTracks === exTracks && loTs > exTs) pickLocal = true;
   886→   886→    else if (localHasBio && !exHasBio) pickLocal = true;
   887→   887→
   888→   888→    if (pickLocal) {
   889→   889→      const merged = { ...existing, ...l };
   890→   890→      // Preserve any extra fields from cloud that local doesn't have
   891→   891→      for (const k of Object.keys(existing)) {
   892→   892→        if (!(k in l) && merged[k] === undefined) merged[k] = existing[k];
   893→   893→      }
   894→   894→      byId.set(id, merged);
   895→   895→      if (nm) byName.set(nm, merged);
   896→   896→    } else {
   897→   897→      // Keep cloud's version but ensure local-only fields are preserved
   898→   898→      const merged = { ...l, ...existing };
   899→   899→      for (const k of Object.keys(l)) {
   900→   900→        if (!(k in existing) && merged[k] === undefined) merged[k] = l[k];
   901→   901→      }
   902→   902→      byId.set(id, merged);
   903→   903→      if (nm) byName.set(nm, merged);
   904→   904→    }
   905→   905→  }
   906→   906→
   907→   907→  return Array.from(byId.values());
   908→   908→}
