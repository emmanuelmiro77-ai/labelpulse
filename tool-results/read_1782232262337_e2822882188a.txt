   220→function safeJsonParse(str: string | null): any | null {
   221→  if (!str) return null;
   222→  try {
   223→    return JSON.parse(str);
   224→  } catch {
   225→    return null;
   226→  }
   227→}

   229→function safeLocalStorageGet(key: string): string | null {
   230→  try {
   231→    return localStorage.getItem(key);
   232→  } catch {
   233→    return null;
   234→  }
   235→}

   237→function safeLocalStorageSet(key: string, value: string): boolean {
   238→  try {
   239→    localStorage.setItem(key, value);
   240→    return true;
   241→  } catch (e) {
   242→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   243→    return false;
   244→  }

   255→/**
   256→ * Extract rankingSnapshots array from a raw localStorage JSON string.
   257→ * Returns [] if not found / invalid.
   258→ */
   259→function extractSnapshotsFromRaw(raw: string | null): RankingSnapshot[] {
   260→  if (!raw) return [];
   261→  const data = safeJsonParse(raw);
   262→  if (!data) return [];
   263→  const snaps = data.state?.rankingSnapshots ?? data.rankingSnapshots;
   264→  return Array.isArray(snaps) ? snaps : [];
   265→}

   267→/**
   268→ * Merge two snapshot arrays, deduping by `id` first, then by `timestamp+source`.
   269→ * Always returns ascending-by-timestamp order.
   270→ * Used by: importData (backup file), robustStorage.getItem (safety-net restore),
   271→ * and loadFromCloud (cloud merge).
   272→ */
   273→function mergeSnapshots(
   274→  base: RankingSnapshot[] | null | undefined,
   275→  incoming: RankingSnapshot[] | null | undefined
   276→): RankingSnapshot[] {
   277→  const a = Array.isArray(base) ? base : [];
   278→  const b = Array.isArray(incoming) ? incoming : [];
   279→  const seen = new Set<string>();
   280→  const out: RankingSnapshot[] = [];
   281→  for (const s of [...a, ...b]) {
   282→    if (!s || typeof s !== "object") continue;
   283→    const key = s.id || `${s.timestamp}|${s.source}`;
   284→    if (seen.has(key)) continue;
   285→    seen.add(key);
   286→    out.push(s);
   287→  }
   288→  out.sort(
   289→    (x, y) =>
   290→      new Date(x.timestamp || 0).getTime() - new Date(y.timestamp || 0).getTime()
   291→  );
   292→  return out;
   293→}

   308→/**
   309→ * Check if persisted data has user edits (emails, notes, links, etc.)

   312→function countUserEditedLabels(labels: any[]): number {
   313→  if (!Array.isArray(labels)) return 0;
   314→  return labels.filter((l) =>
   315→    (l.emails && l.emails.length > 0) ||
   316→    (l.notes && l.notes.trim() !== "") ||
   317→    (l.website && l.website.trim() !== "") ||
   318→    (l.demoLink && l.demoLink.trim() !== "") ||
   319→    (l.socialLink && l.socialLink.trim() !== "") ||
   320→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   321→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   322→    l.isCustom === true
   323→  ).length;
   324→}

   326→function getTotalLabelCount(data: any): number {
   327→  if (!data) return 0;
   328→  if (Array.isArray(data.labels)) return data.labels.length;
   329→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   330→  return 0;
   331→}

   333→/**
   334→ * Extract user-edited label count from raw localStorage string.
   335→ * Used by the write guard to check if a write would cause data loss.
   336→ */
   337→function getUserEditCountFromRaw(raw: string | null): number {
   338→  if (!raw) return 0;
   339→  const data = safeJsonParse(raw);
   340→  if (!data) return 0;
   341→  const labels = data.state?.labels || data.labels || [];
   342→  return countUserEditedLabels(labels);

   347→const robustStorage: StateStorage = {
   348→  getItem: (name: string): string | null => {
   349→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   350→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   351→
   352→    const primaryData = safeJsonParse(primaryRaw);
   353→    const backupData = safeJsonParse(backupRaw);
   354→
   355→    let result: string | null = null;
   356→    let usedSidecarSnapshots = false;
   357→
   358→    // If primary is missing/corrupt, try backup
   359→    if (!primaryData) {
   360→      if (backupData) {
   361→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   362→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   363→        result = backupRaw;
   364→      }
   365→      // Both empty — first visit, result stays null
   366→    } else if (backupData) {
   367→      // Both exist — check for data loss
   368→      const primaryLabelCount = getTotalLabelCount(primaryData);
   369→      const backupLabelCount = getTotalLabelCount(backupData);
   370→      const primaryUserEdits = primaryData.state?.labels
   371→        ? countUserEditedLabels(primaryData.state.labels)
   372→        : countUserEditedLabels(primaryData.labels || []);
   373→      const backupUserEdits = backupData.state?.labels
   374→        ? countUserEditedLabels(backupData.state.labels)
   375→        : countUserEditedLabels(backupData.labels || []);
   376→
   377→      // If primary has fewer labels than backup AND backup has more or equal user data
   378→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   379→        console.warn(
   380→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   381→        );
   382→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   383→        result = backupRaw;
   384→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   385→        // If primary has fewer user edits than backup, use backup
   386→        console.warn(
   387→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   388→        );
   389→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   390→        result = backupRaw;
   391→      } else {
   392→        result = primaryRaw;
   393→      }
   394→    } else {
   395→      // Primary exists, no backup — normal case
   396→      result = primaryRaw;
   397→    }
   398→
   399→    // CRITICAL SAFETY NET for rankingSnapshots:
   400→    // If the chosen result has 0 snapshots but the dedicated SNAPSHOTS_BACKUP_KEY
   401→    // has snapshots, splice them in. This handles the scenario where the main
   402→    // store got wiped (bad cloud sync, migration bug, partial overwrite) but the
   403→    // snapshots sidecar survived. Without this, the user loses months of
   404→    // "storico caricamenti" and the Classifiche page resets to "first time".
   405→    try {
   406→      const currentSnaps = extractSnapshotsFromRaw(result);
   407→      const sidecarRaw = safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY);
   408→      const sidecarSnaps = extractSnapshotsFromRaw(sidecarRaw);
   409→      if (sidecarSnaps.length > 0 && sidecarSnaps.length > currentSnaps.length) {
   410→        console.warn(
   411→          `[LabelPulse Storage] SNAPSHOTS RECOVERY: result has ${currentSnaps.length} snapshots, sidecar has ${sidecarSnaps.length}. Merging sidecar into result.`
   412→        );
   413→        const mergedSnaps = mergeSnapshots(currentSnaps, sidecarSnaps);
   414→        const parsed = result ? safeJsonParse(result) : {};
   415→        if (parsed && typeof parsed === "object") {
   416→          if (parsed.state && typeof parsed.state === "object") {
   417→            parsed.state.rankingSnapshots = mergedSnaps;
   418→          } else {
   419→            parsed.rankingSnapshots = mergedSnaps;
   420→          }
   421→          result = JSON.stringify(parsed);
   422→          usedSidecarSnapshots = true;
   423→        }
   424→      }
   425→    } catch (e) {
   426→      console.warn("[LabelPulse Storage] Snapshots sidecar recovery failed:", e);
   427→    }
   428→
   429→    // CRITICAL SAFETY NET for userProfile:
   430→    // Same logic as snapshots. If the chosen result has an empty/seed profile
   431→    // but PROFILE_BACKUP_KEY has real data, splice it in. This is the user's
   432→    // identity card — losing artistName, bio, email, links is catastrophic.
   433→    try {
   434→      const currentProfile = extractProfileFromRaw(result);
   435→      const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   436→      if (sidecarProfile && !currentProfile) {
   437→        console.warn(
   438→          `[LabelPulse Storage] PROFILE RECOVERY: result has empty profile, sidecar has data (artistName=${!!sidecarProfile.artistName}, bio=${!!sidecarProfile.bio}). Merging sidecar into result.`
   439→        );
   440→        const parsed = result ? safeJsonParse(result) : {};
   441→        if (parsed && typeof parsed === "object") {
   442→          if (parsed.state && typeof parsed.state === "object") {
   443→            parsed.state.userProfile = sidecarProfile;
   444→          } else {
   445→            parsed.userProfile = sidecarProfile;
   446→          }
   447→          result = JSON.stringify(parsed);
   448→        }
   449→      } else if (sidecarProfile && currentProfile) {
   450→        // Both have data — merge (sidecar fills empty fields in current)
   451→        const mergedProfile = mergeProfiles(sidecarProfile, currentProfile);
   452→        const parsed = result ? safeJsonParse(result) : {};
   453→        if (parsed && typeof parsed === "object" && mergedProfile) {
   454→          if (parsed.state && typeof parsed.state === "object") {
   455→            parsed.state.userProfile = mergedProfile;
   456→          } else {
   457→            parsed.userProfile = mergedProfile;
   458→          }
   459→          result = JSON.stringify(parsed);
   460→        }
   461→      }
   462→    } catch (e) {
   463→      console.warn("[LabelPulse Storage] Profile sidecar recovery failed:", e);
   464→    }
   465→
   466→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   467→    // merged data immediately after this getItem returns, and we MUST allow that write.
   468→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   469→    _rehydrated = true;
   470→
   471→    if (usedSidecarSnapshots) {
   472→      console.info("[LabelPulse Storage] rankingSnapshots restored from sidecar backup.");
   473→    }
   474→
   475→    return result;
   476→  },
   477→
   478→  setItem: (name: string, value: string): void => {
   479→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   480→    // Before rehydration, the store only has seed data — writing it would overwrite
   481→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   482→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   483→    if (!_rehydrated) {
   484→      console.warn(
   485→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   486→      );
   487→      return;
   488→    }
   489→
   490→    // Write to primary immediately
   491→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   492→
   493→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   494→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   495→    safeLocalStorageSet(BACKUP_KEY, value);
   496→
   497→    // CRITICAL: also persist rankingSnapshots to a dedicated slot.
   498→    // This is the user's "storico caricamenti" — months of imports that cannot
   499→    // be regenerated. Keeping it in a separate key means even a catastrophic
   500→    // main-store wipe (bad cloud sync, migration bug, quota error) leaves the
   501→    // snapshots recoverable.
   502→    try {
   503→      const snaps = extractSnapshotsFromRaw(value);
   504→      if (snaps.length > 0) {
   505→        // Merge with whatever's already in the snapshots slot — NEVER overwrite
   506→        // existing snapshots with fewer snapshots (would happen if the current
   507→        // write had a partial/empty snapshot array due to some upstream bug).
   508→        const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   509→        const merged = mergeSnapshots(existing, snaps);
   510→        safeLocalStorageSet(
   511→          SNAPSHOTS_BACKUP_KEY,
   512→          JSON.stringify(merged)
   513→        );
   514→      }
   515→    } catch (e) {
   516→      console.warn("[LabelPulse Storage] Snapshots-sidecar write failed:", e);
   517→    }
   518→
   519→    // CRITICAL: also persist userProfile to a dedicated slot.
   520→    // Same pattern as snapshots. The profile is the user's identity card —
   521→    // artistName, bio, email, social links, photo. We can NEVER lose this,
   522→    // even if cloud sync wipes the main store.
   523→    try {
   524→      const profile = extractProfileFromRaw(value);
   525→      if (profile) {
   526→        const existing = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   527→        const merged = mergeProfiles(existing, profile);
   528→        if (merged) {
   529→          safeLocalStorageSet(
   530→            PROFILE_BACKUP_KEY,
   531→            JSON.stringify({ userProfile: merged })
   532→          );
   533→        }
   534→      }
   535→    } catch (e) {
   536→      console.warn("[LabelPulse Storage] Profile-sidecar write failed:", e);
   537→    }
   538→
   539→    if (!primaryOk) {
   540→      console.error("[LabelPulse Storage] Primary write failed!");
   541→    }
   542→  },
   543→
   544→  removeItem: (name: string): void => {
   545→    safeLocalStorageRemove(PRIMARY_KEY);
   546→    safeLocalStorageRemove(BACKUP_KEY);
   547→    // NOTE: do NOT remove SNAPSHOTS_BACKUP_KEY or PROFILE_BACKUP_KEY here.
   548→    // These are the user's permanent records (months of ranking history +
   549→    // artist identity) — clearing the main store should NEVER nuke them.
   550→    // The only way to clear them is an explicit user action ("reset all data"
   551→    // button) which calls safeLocalStorageRemove directly.
   552→  },
   553→};

   607→/**
   608→ * Extract userProfile from a raw JSON string (main store JSON, backup JSON,
   609→ * or sidecar JSON). Returns null if missing/invalid.
   610→ */
   611→function extractProfileFromRaw(raw: string | null): any | null {
   612→  if (!raw) return null;
   613→  try {
   614→    const parsed = JSON.parse(raw);
   615→    const profile = parsed?.state?.userProfile ?? parsed?.userProfile ?? null;
   616→    if (profile && typeof profile === "object") {
   617→      // Must have at least one non-empty field to be considered "real"
   618→      const hasData =
   619→        !!profile.artistName ||
   620→        !!profile.bio ||
   621→        !!profile.email ||
   622→        !!profile.scLink ||
   623→        !!profile.photoUrl ||
   624→        (Array.isArray(profile.links) && profile.links.length > 0) ||
   625→        !!profile.cyaniteApiToken ||
   626→        !!profile.supabaseUrl;
   627→      if (hasData) return profile;
   628→    }
   629→  } catch {
   630→    /* ignore */
   631→  }
   632→  return null;
   633→}
   634→
   635→/**
   636→ * Merge two user profiles — non-empty fields win.
   637→ * This is the safety net: if the cloud ever sends a profile with empty
   638→ * artistName, but the sidecar has the real artistName, we keep the real one.
   639→ */
   640→export function mergeProfiles(base: any | null, incoming: any | null): any | null {
   641→  if (!base && !incoming) return null;
   642→  if (!base) return incoming;
   643→  if (!incoming) return base;
   644→  const merged: any = { ...base };
   645→  for (const k of Object.keys(incoming)) {
   646→    const v = incoming[k];
   647→    if (k === "links") {
   648→      // Merge link arrays (dedupe by type|value)
   649→      const a = Array.isArray(base.links) ? base.links : [];
   650→      const b = Array.isArray(v) ? v : [];
   651→      const seen = new Set<string>();
   652→      const out: { type: string; value: string }[] = [];
   653→      for (const link of [...a, ...b]) {
   654→        if (!link || typeof link !== "object") continue;
   655→        const key = `${link.type}|${link.value}`;
   656→        if (seen.has(key)) continue;
   657→        seen.add(key);
   658→        out.push(link);
   659→      }
   660→      merged.links = out;
   661→    } else if (typeof v === "string") {
   662→      // Non-empty string wins over empty
   663→      if (v && !merged[k]) merged[k] = v;
   664→    } else {
   665→      // For any other type, prefer defined over undefined
   666→      if (v !== undefined && v !== null && !merged[k]) merged[k] = v;
   667→    }
   668→  }
   669→  return merged;
   670→}

   682→ * current live profile is empty/seed AND the sidecar has real data.

   684→ * Returns true if a restore happened, false otherwise.

   686→export function restoreProfileFromSidecar(): boolean {
   687→  if (typeof window === "undefined") return false;
   688→  const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   689→  if (!sidecarProfile) return false;
   690→
   691→  const current = useAppStore.getState().userProfile;
   692→  const currentHasData =
   693→    !!current.artistName ||
   694→    !!current.bio ||
   695→    !!current.email ||
   696→    !!current.scLink ||
   697→    !!current.photoUrl ||
   698→    (Array.isArray(current.links) && current.links.length > 0) ||
   699→    !!current.cyaniteApiToken ||
   700→    !!current.supabaseUrl;
   701→
   702→  // If current profile has data, merge (sidecar can fill empty fields)
   703→  // If current profile is empty, replace with sidecar

   709→    useAppStore.setState({ userProfile: merged });

   711→      `[LabelPulse] Restored user profile from sidecar backup (had: ${currentHasData ? "partial" : "empty"}, now has: artistName=${!!merged.artistName}, bio=${!!merged.bio}, email=${!!merged.email}).`
   712→    );
   713→    return true;
   714→  }
   715→  return false;

   718→// ==================== ARTISTS SIDE CAR ====================
   719→// Like PROFILE_BACKUP_KEY but for the artists array. A full Beatport scrape
   720→// produces ~3400 artists / ~9MB. localStorage has a 5-10MB quota on most
   721→// browsers, so we TRY to write artists here but FAIL GRACEFULLY if quota
   722→// exceeded — the IndexedDB + cloud are the primary stores, this is just
   723→// an emergency backup for when both fail.
   724→//
   725→// Strategy:
   726→//  - writeArtistsSidecar: best-effort write, truncate if needed (keep first
   727→//    N artists rather than fail entirely). Skip silently if quota exceeded.
   728→//  - readArtistsSidecar: returns [] if missing/corrupt
   729→//  - restoreArtistsFromSidecar: splice into live store if local is empty
   730→const ARTISTS_SIDECAR_KEY = "labelpulse-artists-backup";
   731→const ARTISTS_SIDECAR_MAX = 200; // cap to first 200 artists (~600KB) to fit quota
   732→
   733→export function writeArtistsSidecar(artists: any[]): void {
   734→  if (typeof window === "undefined") return;
   735→  if (!Array.isArray(artists) || artists.length === 0) return;
   736→  try {
   737→    // If small enough, write all. Otherwise, write first N + a marker.
   738→    const capped = artists.length > ARTISTS_SIDECAR_MAX
   739→      ? artists.slice(0, ARTISTS_SIDECAR_MAX)
   740→      : artists;
   741→    const payload = JSON.stringify({
   742→      savedAt: new Date().toISOString(),
   743→      count: artists.length,
   744→      capped: artists.length > ARTISTS_SIDECAR_MAX,
   745→      artists: capped,
   746→    });
   747→    safeLocalStorageSet(ARTISTS_SIDECAR_KEY, payload);
   748→  } catch (e) {
   749→    // Quota exceeded — try with even fewer artists
   750→    try {
   751→      const tiny = artists.slice(0, 50);
   752→      const payload = JSON.stringify({
   753→        savedAt: new Date().toISOString(),
   754→        count: artists.length,
   755→        capped: true,
   756→        artists: tiny,
   757→      });
   758→      safeLocalStorageSet(ARTISTS_SIDECAR_KEY, payload);
   759→    } catch {
   760→      // Give up silently — IDB + cloud are the primary stores
   761→    }
   762→  }
   763→}
   764→
   765→export function readArtistsSidecar(): any[] {
   766→  if (typeof window === "undefined") return [];
   767→  const raw = safeLocalStorageGet(ARTISTS_SIDECAR_KEY);
   768→  if (!raw) return [];
   769→  try {
   770→    const parsed = JSON.parse(raw);
   771→    const arr = parsed?.artists;
   772→    return Array.isArray(arr) ? arr : [];
   773→  } catch {
   774→    return [];
   775→  }
   776→}

   778→/**
   779→ * Restore artists from sidecar IF the live store has 0 artists.
   780→ * Returns the number of artists restored (0 if nothing happened).
   781→ * Does NOT replace live artists if live already has data — sidecar is
   782→ * only an emergency recovery, not a primary source.
   783→ */
   784→export function restoreArtistsFromSidecar(): number {
   785→  if (typeof window === "undefined") return 0;
   786→  const sidecar = readArtistsSidecar();
   787→  if (sidecar.length === 0) return 0;
   788→
   789→  const current = useAppStore.getState().artists || [];
   790→  if (current.length > 0) return 0;
   791→
   792→  useAppStore.setState({ artists: sidecar });
   793→  console.info(
   794→    `[LabelPulse] Restored ${sidecar.length} artists from sidecar backup (live store was empty).`
   795→  );
   796→  // Also persist to IDB so future boots are fast
   797→  saveArtistsToIDB(sidecar).catch(() => {});
   798→  return sidecar.length;
   799→}

   899→/**
   900→ * Merge incoming artists (from a fresh scrape) with existing artists
   901→ * (already in the store from previous scrapes).
   902→ *
   903→ * Strategy:
   904→ * - Key by artist.id (e.g. "bp_6824"). Same Beatport id = same artist.
   905→ * - For each artist that exists: union genres, union labelsPublishedOn,
   906→ *   MERGE tracksByGenre by track.id (preserve tracks from older scrapes,
   907→ *   update with new position info), sum totalPoints, min bestPosition.
   908→ * - firstSeenAt preserved from existing; lastSeenAt = now.
   909→ * - Trending recomputed fresh from current data.
   910→ * - New artists: appended with firstSeenAt = lastSeenAt = now.
   911→ *
   912→ * This is what makes historical data accumulate: each scrape adds new
   913→ * tracks to artists' tracksByGenre without losing previously-seen tracks.
   914→ */
   915→function mergeArtists(existing: Artist[], incoming: Artist[], now: string): Artist[] {
   916→  const map = new Map<string, Artist>();
   917→
   918→  // Seed with existing
   919→  for (const a of existing) {
   920→    map.set(a.id, { ...a });
   921→  }
   922→
   923→  for (const inc of incoming) {
   924→    const ex = map.get(inc.id);
   925→    if (!ex) {
   926→      // New artist — first time we see them
   927→      map.set(inc.id, {
   928→        ...inc,
   929→        firstSeenAt: now,
   930→        lastSeenAt: now,
   931→      });
   932→      continue;
   933→    }
   934→
   935→    // Existing artist — merge fields
   936→    // Union genres
   937→    const genres = Array.from(new Set([...(ex.genres || []), ...(inc.genres || [])]));
   938→
   939→    // Union labelsPublishedOn
   940→    const labelsPublishedOn = Array.from(
   941→      new Set([...(ex.labelsPublishedOn || []), ...(inc.labelsPublishedOn || [])])
   942→    );
   943→
   944→    // Merge tracksByGenre — preserve old tracks, add/update from incoming
   945→    const tracksByGenre: Record<string, ArtistTrack[]> = { ...(ex.tracksByGenre || {}) };
   946→    for (const genre of Object.keys(inc.tracksByGenre || {})) {
   947→      const incTracks = inc.tracksByGenre[genre] || [];
   948→      const exTracks = tracksByGenre[genre] || [];
   949→      const byId = new Map<number, ArtistTrack>();
   950→      // Seed with existing
   951→      for (const t of exTracks) byId.set(t.id, t);
   952→      // Update/add with incoming (incoming overrides — fresh positions)
   953→      for (const t of incTracks) byId.set(t.id, t);
   954→      tracksByGenre[genre] = Array.from(byId.values()).sort((a, b) => a.position - b.position);
   955→    }
   956→
   957→    // Recompute totalPoints & bestPosition from current incoming data
   958→    // (the existing totals were from older scrapes — we want the snapshot
   959→    // from THIS import to drive the points, otherwise the totals just
   960→    // accumulate forever and "points" loses meaning).
   961→    // BUT: keep max totalPoints seen (artists who peaked before should
   962→    // keep credit). Actually no — let's use the LATEST totalPoints,
   963→    // since that's the current ranking signal. Old tracks are still in
   964→    // tracksByGenre for browsing, but the points reflect "right now".
   965→    const totalPoints = inc.totalPoints;
   966→    const bestPosition = Math.min(ex.bestPosition, inc.bestPosition);
   967→
   968→    // Trending: use incoming (fresh computation)
   969→    const trending = inc.trending;
   970→    const trendingRankByGenre = inc.trendingRankByGenre;
   971→    const trendingPointsByGenre = inc.trendingPointsByGenre;
   972→
   973→    // isRemixerOnly: false if EITHER snapshot has them as primary
   974→    const isRemixerOnly = ex.isRemixerOnly && inc.isRemixerOnly;
   975→
   976→    map.set(inc.id, {
   977→      ...ex,
   978→      ...inc,
   979→      genres,
   980→      labelsPublishedOn,
   981→      tracksByGenre,
   982→      totalPoints,
   983→      bestPosition,
   984→      trending,
   985→      trendingRankByGenre,
   986→      trendingPointsByGenre,
   987→      isRemixerOnly,
   988→      firstSeenAt: ex.firstSeenAt || now,
   989→      lastSeenAt: now,
   990→    });
   991→  }
   992→
   993→  return Array.from(map.values());
   994→}

   996→/**
   997→ * Merge helper: preserves ALL user-editable fields from existing label
   998→ * when imported label has empty/default values for those fields.
   999→ */
  1000→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
  1001→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
  1002→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
  1003→  //
  1004→  // IDEMPOTENT IMPORT FIX: previously, every import — even an identical re-import —
  1005→  // would clobber prevRankByGenre with a copy of the current rankByGenre. After
  1006→  // an identical re-import, prevRank == currentRank for every genre, so all
  1007→  // movements became 0 and Spotlight risers disappeared. This broke the UX:
  1008→  // "if I re-import the same scrape, I should see what I saw before".
  1009→  //
  1010→  // The fix: only snapshot the existing rank as "previous" for genres where the
  1011→  // rank has actually CHANGED. For genres where imported rank === existing rank,
  1012→  // preserve the existing prevRankByGenre[genre] so users keep seeing the risers
  1013→  // they had before the (identical) re-import.
  1014→  const importedRank = imported.rankByGenre || {};
  1015→  const existingRank = existing.rankByGenre || {};
  1016→  const existingPrev = existing.prevRankByGenre || {};
  1017→  const hasNewRankData = Object.keys(importedRank).length > 0;
  1018→
  1019→  let newPrevRankByGenre: Record<string, number>;
  1020→  if (hasNewRankData) {
  1021→    // Start from the existing prev snapshot — we only overwrite entries for
  1022→    // genres where the rank actually changed.
  1023→    newPrevRankByGenre = { ...existingPrev };
  1024→    for (const [genre, newRank] of Object.entries(importedRank)) {
  1025→      const oldRank = existingRank[genre];
  1026→      if (oldRank !== undefined && oldRank !== newRank) {
  1027→        // Rank changed for this genre → snapshot the old rank as "previous".
  1028→        newPrevRankByGenre[genre] = oldRank;
  1029→      }
  1030→      // If oldRank === newRank → leave existing prevRankByGenre[genre] untouched
  1031→      // (preserves the snapshot from the last actual change → risers persist).
  1032→      // If oldRank is undefined (genre new for this label) → don't set prev
  1033→      // (label is a new entry for this genre, no previous rank to compare to).
  1034→    }
  1035→  } else {
  1036→    // No new rank data → keep existing prevRankByGenre unchanged.
  1037→    newPrevRankByGenre = existingPrev;
  1038→  }
  1039→
  1040→  return {
  1041→    // User-editable data — ALWAYS prefer existing if it has real data
  1042→    emails: imported.emails?.length ? imported.emails : existing.emails,
  1043→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
  1044→    website: imported.website?.trim() || existing.website,
  1045→    demoLink: imported.demoLink?.trim() || existing.demoLink,
  1046→    socialLink: imported.socialLink?.trim() || existing.socialLink,
  1047→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
  1048→    notes: imported.notes?.trim() || existing.notes,
  1049→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
  1050→
  1051→    // Beatport/ranking data — prefer imported (it's the fresh data)
  1052→    genres: imported.genres?.length ? imported.genres : existing.genres,
  1053→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
  1054→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
  1055→    trending: imported.trending || existing.trending,
  1056→    trendingRankByGenre: Object.keys(imported.trendingRankByGenre || {}).length ? imported.trendingRankByGenre : existing.trendingRankByGenre,
  1057→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
  1058→
  1059→    // Ranking history — idempotent: only update prev when rank actually changed

  1090→/**
  1091→ * Repair corrupted label data.
  1092→ * Detects and fixes: same email appearing on too many labels (likely corruption).
  1093→ */
  1094→function repairLabelData(labels: Label[]): Label[] {
  1095→  const emailLabelCount = new Map<string, number>();
  1096→  for (const l of labels) {
  1097→    if (l.emails && Array.isArray(l.emails)) {
  1098→      for (const e of l.emails) {
  1099→        if (e && e.trim()) {
  1100→          const key = e.toLowerCase().trim();
  1101→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
  1102→        }
  1103→      }
  1104→    }
  1105→  }
  1106→
  1107→  const corruptedEmails = new Set<string>();
  1108→  for (const [email, count] of emailLabelCount) {
  1109→    if (count > 5) {
  1110→      corruptedEmails.add(email);
  1111→      console.warn(
  1112→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
  1113→      );
  1114→    }
  1115→  }
  1116→
  1117→  if (corruptedEmails.size === 0) {
  1118→    return labels;
  1119→  }
  1120→
  1121→  const emailOwner = new Map<string, string>();
  1122→  for (const email of corruptedEmails) {
  1123→    for (const l of labels) {
  1124→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
  1125→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
  1126→        if (domain && l.name.toLowerCase().includes(domain)) {
  1127→          emailOwner.set(email, l.id);
  1128→          break;
  1129→        }
  1130→      }
  1131→    }
  1132→    if (!emailOwner.has(email)) {
  1133→      for (const l of labels) {
  1134→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
  1135→          emailOwner.set(email, l.id);
  1136→          break;
  1137→        }
  1138→      }
  1139→    }
  1140→  }
  1141→
  1142→  return labels.map(l => {
  1143→    if (!l.emails || !Array.isArray(l.emails)) return l;
  1144→    const cleanedEmails = l.emails.filter(e => {
  1145→      const key = e?.toLowerCase().trim();
  1146→      if (!key) return false;
  1147→      if (!corruptedEmails.has(key)) return true;
  1148→      return emailOwner.get(key) === l.id;
  1149→    });
  1150→    const cleanedContactInfo = cleanedEmails[0] || "";
  1151→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
  1152→      console.log(
  1153→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
  1154→      );
  1155→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
  1156→    }
  1157→    return l;
  1158→  });
  1159→}
