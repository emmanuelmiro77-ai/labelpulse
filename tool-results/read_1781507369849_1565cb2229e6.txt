    73→// ==================== SMART URL HELPER ====================
    74→// Converts partial usernames/handles into full clickable URLs.
    75→// E.g. "spectrummusicnl" → "https://instagram.com/spectrummusicnl"
    76→// If already a full URL, returns as-is.
    77→
    78→function toClickableUrl(value: string, field: "website" | "demoLink" | "socialLink" | "soundcloudLink"): string | null {
    79→  if (!value || !value.trim()) return null;
    80→  const v = value.trim();
    81→
    82→  // Already a full URL
    83→  if (/^https?:\/\//i.test(v)) return v;
    84→
    85→  // Email
    86→  if (v.includes("@") && field !== "socialLink" && field !== "soundcloudLink") {
    87→    return `mailto:${v}`;
    88→  }
    89→
    90→  // Field-specific smart defaults
    91→  switch (field) {
    92→    case "socialLink":
    93→      // Could be Instagram, Facebook, X/Twitter, etc.
    94→      // Heuristic: if it looks like a username (no dots, no slashes), assume Instagram
    95→      if (/^[a-zA-Z0-9._]{1,30}$/.test(v)) {
    96→        return `https://instagram.com/${v.replace(/^@/, "")}`;
    97→      }
    98→      // If it has dots but no slashes, might be a domain-like handle — try as-is with https
    99→      if (!/\//.test(v)) return `https://${v}`;
   100→      return `https://${v}`;
   101→    case "soundcloudLink":
   102→      if (/^[a-zA-Z0-9._-]{1,30}$/.test(v)) {
   103→        return `https://soundcloud.com/${v.replace(/^@/, "")}`;
   104→      }
   105→      return `https://${v}`;
   106→    case "website":
   107→    case "demoLink":
   108→      return `https://${v}`;
   109→  }
   110→}
   111→
   112→// Get display text for a link (strips protocol for cleaner display)
   113→function getLinkDisplay(value: string): string {
   114→  return value
   115→    .replace(/^https?:\/\//, "")
   116→    .replace(/^mailto:/, "")
   117→    .replace(/\/$/, "");
   118→}
