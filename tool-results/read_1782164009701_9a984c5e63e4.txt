   228→/**
   229→ * Get the time cutoff for a period (in milliseconds from now).
   230→ */
   231→function getPeriodCutoff(period: RankingTimePeriod): number {
   232→  const now = Date.now();
   233→  switch (period) {
   234→    case "current": return now; // Only current data
   235→    case "1m": return now - 30 * 24 * 60 * 60 * 1000;
   236→    case "3m": return now - 90 * 24 * 60 * 60 * 1000;
   237→    case "1y": return now - 365 * 24 * 60 * 60 * 1000;
   238→    case "all": return 0; // All time
   239→  }
   240→}

   242→// ==================== AGGREGATION LOGIC ====================
   243→
   244→/**
   245→ * Aggregate ranking snapshots for a given genre and time period.
   246→ * Returns a map of labelName -> { totalPoints, bestRank, snapshotCount, averageRank }
   247→ */
   248→function aggregateSnapshots(
   249→  snapshots: RankingSnapshot[],
   250→  genre: string,
   251→  period: RankingTimePeriod
   252→): Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }> {
   253→  const cutoff = getPeriodCutoff(period);
   254→  const result = new Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }>();
   255→
   256→  // Helper to get or create entry with defaults
   257→  function getOrCreate(name: string): { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number } {
   258→    let entry = result.get(name);
   259→    if (!entry) {
   260→      entry = { totalPoints: 0, bestRank: 999, snapshotCount: 0, averageRank: 0, rankSum: 0 };
   261→      result.set(name, entry);
   262→    }
   263→    return entry;
   264→  }
   265→
   266→  for (const snapshot of snapshots) {
   267→    if (new Date(snapshot.timestamp).getTime() < cutoff) continue;
   268→    if (period === "current") continue; // Current period doesn't use snapshots
   269→
   270→    const genreData = snapshot.genres[genre];
   271→    if (!genreData) continue;
   272→
   273→    for (const [labelName, data] of Object.entries(genreData)) {
   274→      const existing = getOrCreate(labelName);
   275→      existing.totalPoints += data.points;
   276→      existing.rankSum += data.rank;
   277→      existing.snapshotCount++;
   278→      if (data.rank < existing.bestRank) existing.bestRank = data.rank;
   279→    }
   280→  }
   281→
   282→  // Calculate average rank
   283→  for (const [, val] of result) {
   284→    (val as any).averageRank = Math.round(val.rankSum / val.snapshotCount);
   285→  }
   286→
   287→  return result;
   288→}

   290→/**
   291→ * Build ranked list for a genre, considering the time period.
   292→ * - "current": uses live label data (rankByGenre)
   293→ * - Other periods: aggregates snapshots and re-ranks by cumulative points
   294→ */
   295→function buildRankedList(
   296→  labels: Label[],
   297→  snapshots: RankingSnapshot[],
   298→  genre: string,
   299→  period: RankingTimePeriod
   300→): RankedLabel[] {
   301→  if (period === "current") {
   302→    // Current view: use live label data
   303→    const ranked: RankedLabel[] = [];
   304→    for (const label of labels) {
   305→      const rank = label.rankByGenre?.[genre];
   306→      if (rank === undefined || rank === null) continue;
   307→
   308→      const prevRank = label.prevRankByGenre?.[genre] ?? null;
   309→      const points = label.pointsByGenre?.[genre] ?? 0;
   310→      let movement: number | null = null;
   311→
   312→      if (prevRank !== null && prevRank !== undefined) {
   313→        movement = prevRank - rank;
   314→      } else if (Object.keys(label.prevRankByGenre || {}).length > 0) {
   315→        movement = null;
   316→      }
   317→
   318→      ranked.push({ label, rank, prevRank, points, movement, snapshotCount: 1 });
   319→    }
   320→    return ranked;
   321→  }
   322→
   323→  // Historical/cumulative view: aggregate snapshots
   324→  const aggregated = aggregateSnapshots(snapshots, genre, period);
   325→  if (aggregated.size === 0) return [];
   326→
   327→  // Sort by total points descending to determine cumulative rank
   328→  const sorted = Array.from(aggregated.entries()).sort((a, b) => b[1].totalPoints - a[1].totalPoints);
   329→
   330→  // For movement in cumulative view, compare with the current live data
   331→  const ranked: RankedLabel[] = [];
   332→  for (let i = 0; i < sorted.length; i++) {
   333→    const [labelName, data] = sorted[i];
   334→    const cumulativeRank = i + 1;
   335→
   336→    // Find matching label in current data
   337→    const matchingLabel = labels.find(l => l.name === labelName);
   338→    const currentRank = matchingLabel?.rankByGenre?.[genre] ?? null;
   339→
   340→    // Movement: compare current live rank with cumulative rank
   341→    let movement: number | null = null;
   342→    if (currentRank !== null) {
   343→      movement = currentRank - cumulativeRank; // positive = label is better live than cumulative (hot right now)
   344→    }
   345→
   346→    // Use the label if found, otherwise create a minimal reference
   347→    const label = matchingLabel || {
   348→      id: `snapshot_${labelName.toLowerCase().replace(/[^a-z0-9]+/g, '_')}`,
   349→      name: labelName,
   350→      genre: genre,
   351→      submissionType: "email" as const,
   352→      contactInfo: "",
   353→      status: "open" as const,
   354→      notes: "",
   355→      createdAt: new Date().toISOString(),
   356→      emails: [],
   357→      website: "",
   358→      demoLink: "",
   359→      socialLink: "",
   360→      soundcloudLink: "",
   361→      beatportLink: "",
   362→      customLinks: [],
   363→      genres: [genre],
   364→      rankByGenre: { [genre]: currentRank || 999 },
   365→      pointsByGenre: {},
   366→      trending: false,
   367→      trendingRankByGenre: {},
   368→      trendingPointsByGenre: {},
   369→      prevRankByGenre: {},
   370→    };
   371→
   372→    ranked.push({
   373→      label,
   374→      rank: cumulativeRank,
   375→      prevRank: currentRank,
   376→      points: data.totalPoints,
   377→      movement,
   378→      snapshotCount: data.snapshotCount,
   379→    });
   380→  }
   381→
   382→  return ranked;
   383→}

   387→export function RankingsPage() {
   388→  const { labels, locale, rankingsUpdatedAt, rankingSnapshots, setActiveTab, setSelectedLabelId } = useAppStore();
   389→  const snapshots = rankingSnapshots || []; // defensive: might be undefined for existing users
   390→  const [selectedGenre, setSelectedGenre] = useState<string | null>(null);
   391→  const [sortMode, setSortMode] = useState<SortMode>("rank");
   392→  const [movementFilter, setMovementFilter] = useState<MovementFilter>("all");
   393→  const [timePeriod, setTimePeriod] = useState<RankingTimePeriod>("current");
   394→  const [showGenreDropdown, setShowGenreDropdown] = useState(false);
   395→  const [spotlightShowAll, setSpotlightShowAll] = useState(false); // when true + genre selected, show global risers instead of genre-filtered
   396→
   397→  // Click on a label name → navigate to the LabelFinder tab and auto-open
   398→  // the detail dialog for that label. Matches by id when possible, falls
   399→  // back to name (LabelFinder's useEffect handles both).
   400→  const handleOpenLabel = useCallback(
   401→    (label: Label) => {
   402→      setSelectedLabelId?.(label.id || label.name);
   403→      setActiveTab("labels");
   404→    },
   405→    [setActiveTab, setSelectedLabelId]
   406→  );
   407→
   408→  // Get all genres from labels, sorted alphabetically
   409→  const allGenres = useMemo(() => {
   410→    const genreSet = new Set<string>();
   411→    for (const label of labels) {
   412→      if (label.genres) {
   413→        for (const g of label.genres) {
   414→          genreSet.add(g);
   415→        }
   416→      }
   417→    }
   418→    return Array.from(genreSet).sort((a, b) => a.localeCompare(b));
   419→  }, [labels]);
   420→
   421→  // Build ranked list for selected genre and period
   422→  const rankedList = useMemo((): RankedLabel[] => {
   423→    if (!selectedGenre) return [];
   424→    return buildRankedList(labels, snapshots, selectedGenre, timePeriod);
   425→  }, [labels, snapshots, selectedGenre, timePeriod]);
   426→
   427→  // Apply filters
   428→  const filteredList = useMemo(() => {
   429→    let result = [...rankedList];
   430→
   431→    // Movement filter
   432→    switch (movementFilter) {
   433→      case "rising":
   434→        result = result.filter((l) => l.movement !== null && l.movement > 0);
   435→        break;
   436→      case "falling":
   437→        result = result.filter((l) => l.movement !== null && l.movement < 0);
   438→        break;
   439→      case "new":
   440→        result = result.filter((l) => l.movement === null);
   441→        break;
   442→      case "stable":
   443→        result = result.filter((l) => l.movement === 0);
   444→        break;
   445→    }
   446→
   447→    // Sort
   448→    switch (sortMode) {
   449→      case "rank":
   450→        result.sort((a, b) => a.rank - b.rank);
   451→        break;
   452→      case "movement":
   453→        result.sort((a, b) => {
   454→          if (a.movement === null && b.movement === null) return a.rank - b.rank;
   455→          if (a.movement === null) return -1;
   456→          if (b.movement === null) return 1;
   457→          return b.movement - a.movement;
   458→        });
   459→        break;
   460→      case "points":
   461→        result.sort((a, b) => b.points - a.points);
   462→        break;
   463→    }
   464→
   465→    return result;
   466→  }, [rankedList, movementFilter, sortMode]);
   467→
   468→  // Stats summary
   469→  const stats = useMemo(() => {
   470→    const total = rankedList.length;
   471→    const rising = rankedList.filter((l) => l.movement !== null && l.movement > 0).length;
   472→    const falling = rankedList.filter((l) => l.movement !== null && l.movement < 0).length;
   473→    const newEntries = rankedList.filter((l) => l.movement === null).length;

   485→        const currentRank = label.rankByGenre?.[genre];
   486→        const prevRank = label.prevRankByGenre?.[genre];
   487→        if (currentRank && prevRank) {
   488→          const movement = prevRank - currentRank;
   489→          if (movement > 0) {
   490→            risers.push({ label, genre, movement, rank: currentRank });
   491→          }
   492→        }
   493→      }
   494→    }
   495→
   496→    return risers.sort((a, b) => b.movement - a.movement).slice(0, 10);

   506→      const prevRank = label.prevRankByGenre?.[selectedGenre];
   507→      if (currentRank && prevRank) {
   508→        const movement = prevRank - currentRank;
   509→        if (movement > 0) {
   510→          risers.push({ label, genre: selectedGenre, movement, rank: currentRank });
   511→        }
   512→      }
   513→    }
   514→
   515→    return risers.sort((a, b) => b.movement - a.movement).slice(0, 10);

   518→  const hasPreviousData = useMemo(() => {

   522→  // ⚠️ FIX (2026-06-22): Don't show the "no history" alert until cloud sync

   526→  // have history in the cloud but haven't received it yet.
   527→  // We wait for either:
   528→  //   1. hasCloudSynced = true (cloud sync finished, data is final), OR
   529→  //   2. hasPreviousData = true (already has prev ranks, no need to wait)
   530→  const { hasCloudSynced } = useAppStore();
   531→  const hideNoHistoryAlert = hasPreviousData || !hasCloudSynced;
   532→
   533→  // Snapshot stats for info display
   534→  const snapshotStats = useMemo(() => {
   535→    if (!snapshots || snapshots.length === 0) return { count: 0, oldest: null as string | null, newest: null as string | null };

   537→    return {
   538→      count: snapshots.length,
   539→      oldest: sorted[0].timestamp,
   540→      newest: sorted[sorted.length - 1].timestamp,

   542→  }, [snapshots]);

   548→    return isGenreFiltered ? topRisersForGenre : topRisers;
