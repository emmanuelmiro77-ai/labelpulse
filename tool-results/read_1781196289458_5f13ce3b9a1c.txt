     1→"use client";
     2→
     3→import { useAppStore, getLabelTier, type Label } from "@/lib/store";
     4→import { t } from "@/lib/i18n";
     5→import { useState, useMemo, useCallback } from "react";
     6→import {
     7→  Zap,
     8→  Mail,
     9→  Check,
    10→  Languages,
    11→  Music2,
    12→  AlertCircle,
    13→  ChevronDown,
    14→  ChevronUp,
    15→  Send,
    16→  Target,
    17→  Copy,
    18→  ExternalLink,
    19→} from "lucide-react";
    20→import { Button } from "@/components/ui/button";
    21→import { Card, CardContent } from "@/components/ui/card";
    22→import { Input } from "@/components/ui/input";
    23→import { Badge } from "@/components/ui/badge";
    24→import {
    25→  Select,
    26→  SelectContent,
    27→  SelectItem,
    28→  SelectTrigger,
    29→  SelectValue,
    30→} from "@/components/ui/select";
    31→import { Label as UILabel } from "@/components/ui/label";
    32→import { Textarea } from "@/components/ui/textarea";
    33→import { Checkbox } from "@/components/ui/checkbox";
    34→import {
    35→  generateSubject,
    36→  generatePitchBody,
    37→  generateGmailLink,
    38→  generateMailtoLink,
    39→  PITCH_LANGUAGES,
    40→  type PitchTone,
    41→  type PitchLanguage,
    42→} from "@/lib/pitch-utils";
    43→import { useToast } from "@/hooks/use-toast";
    44→
    45→export function PitchGenerator() {
    46→  const { labels, demos, addDemo, locale, userProfile, setUserProfile, getGenres } = useAppStore();
    47→  const { toast } = useToast();
    48→  const genres = getGenres();
    49→
    50→  // Pre-fill from user profile
    51→  const initialArtistName = userProfile.artistName || "";
    52→  const initialScLink = userProfile.scLink || "";
    53→
    54→  // Track setup state
    55→  const [trackName, setTrackName] = useState("");
    56→  const [artistName, setArtistName] = useState(initialArtistName);
    57→  const [scLink, setScLink] = useState(initialScLink);
    58→  const [tone, setTone] = useState<PitchTone>("professional");
    59→  const [language, setLanguage] = useState<PitchLanguage>("en");
    60→  const [customNote, setCustomNote] = useState("");
    61→  const [selectedGenre, setSelectedGenre] = useState("");
    62→
    63→  // Target labels state
    64→  const [selectedLabelIds, setSelectedLabelIds] = useState<Set<string>>(new Set());
    65→  const [onlyWithEmail, setOnlyWithEmail] = useState(false);
    66→  const [expandedTiers, setExpandedTiers] = useState<Record<string, boolean>>({ top: true, mid: true, emerging: true });
    67→
    68→  // Campaign state
    69→  const [sending, setSending] = useState(false);
    70→  const [sendProgress, setSendProgress] = useState({ current: 0, total: 0 });
    71→  const [campaignComplete, setCampaignComplete] = useState(false);
    72→  const [campaignResults, setCampaignResults] = useState({ sent: 0, skipped: 0 });
    73→
    74→  // Save profile fields on blur
    75→  const handleArtistBlur = () => {
    76→    if (artistName.trim() && artistName.trim() !== userProfile.artistName) {
    77→      setUserProfile({ artistName: artistName.trim() });
    78→    }
    79→  };
    80→
    81→  const handleScLinkBlur = () => {
    82→    if (scLink.trim() && scLink.trim() !== userProfile.scLink) {
    83→      setUserProfile({ scLink: scLink.trim() });
    84→    }
    85→  };
    86→
    87→  // Get matching labels by genre, grouped by tier
    88→  const matchingLabels = useMemo(() => {
    89→    if (!selectedGenre) return { top: [], mid: [], emerging: [] };
    90→    const genreLabels = labels.filter((l) =>
    91→      l.genres.includes(selectedGenre) && l.status === "open"
    92→    );
    93→    const top = genreLabels
    94→      .filter((l) => getLabelTier(l, selectedGenre) === "top")
    95→      .sort((a, b) => (a.rankByGenre?.[selectedGenre] || 999) - (b.rankByGenre?.[selectedGenre] || 999));
    96→    const mid = genreLabels
    97→      .filter((l) => getLabelTier(l, selectedGenre) === "mid")
    98→      .sort((a, b) => (a.rankByGenre?.[selectedGenre] || 999) - (b.rankByGenre?.[selectedGenre] || 999));
    99→    const emerging = genreLabels
   100→      .filter((l) => getLabelTier(l, selectedGenre) === "emerging")
   101→      .sort((a, b) => (b.trendingPointsByGenre?.[selectedGenre] || 0) - (a.trendingPointsByGenre?.[selectedGenre] || 0));
   102→    return { top, mid, emerging };
   103→  }, [labels, selectedGenre]);
   104→
   105→  const allMatchingLabels = useMemo(
   106→    () => [...matchingLabels.top, ...matchingLabels.mid, ...matchingLabels.emerging],
   107→    [matchingLabels]
   108→  );
   109→
   110→  // Get emails for a label
   111→  const getLabelEmails = useCallback((label: Label): string[] => {
   112→    if (label.emails?.length) return label.emails;
   113→    if (label.contactInfo) return [label.contactInfo];
   114→    return [];
   115→  }, []);
   116→
   117→  // Filtered labels (only with email toggle)
   118→  const filteredLabels = useMemo(() => {
   119→    if (!onlyWithEmail) return allMatchingLabels;
   120→    return allMatchingLabels.filter((l) => getLabelEmails(l).length > 0);
   121→  }, [allMatchingLabels, onlyWithEmail, getLabelEmails]);
   122→
   123→  const filteredByTier = useMemo(() => {
   124→    const top = filteredLabels.filter((l) => getLabelTier(l, selectedGenre) === "top");
   125→    const mid = filteredLabels.filter((l) => getLabelTier(l, selectedGenre) === "mid");
   126→    const emerging = filteredLabels.filter((l) => getLabelTier(l, selectedGenre) === "emerging");
   127→    return { top, mid, emerging };
   128→  }, [filteredLabels, selectedGenre]);
   129→
   130→  // Selection management
   131→  const toggleLabel = useCallback((id: string) => {
   132→    setSelectedLabelIds((prev) => {
   133→      const next = new Set(prev);
   134→      if (next.has(id)) next.delete(id);
   135→      else next.add(id);
   136→      return next;
   137→    });
   138→  }, []);
   139→
   140→  const selectAll = useCallback(() => {
   141→    const ids = new Set(filteredLabels.map((l) => l.id));
   142→    setSelectedLabelIds(ids);
   143→  }, [filteredLabels]);
   144→
   145→  const deselectAll = useCallback(() => {
   146→    setSelectedLabelIds(new Set());
   147→  }, []);
   148→
   149→  // Selected label stats
   150→  const selectedLabels = useMemo(
   151→    () => allMatchingLabels.filter((l) => selectedLabelIds.has(l.id)),
   152→    [allMatchingLabels, selectedLabelIds]
   153→  );
   154→
   155→  const selectedWithEmail = useMemo(
   156→    () => selectedLabels.filter((l) => getLabelEmails(l).length > 0),
   157→    [selectedLabels, getLabelEmails]
   158→  );
   159→
   160→  const selectedWithoutEmail = useMemo(
   161→    () => selectedLabels.filter((l) => getLabelEmails(l).length === 0),
   162→    [selectedLabels, getLabelEmails]
   163→  );
   164→
   165→  // Generate personalized pitch preview for a label
   166→  const getPitchForLabel = useCallback(
   167→    (label: Label) => {
   168→      if (!trackName.trim()) return { subject: "", body: "" };
   169→      const subject = generateSubject(trackName.trim(), artistName, language);
   170→      const body = generatePitchBody(label.name, trackName.trim(), artistName, scLink, tone, customNote, language);
   171→      return { subject, body };
   172→    },
   173→    [trackName, artistName, scLink, tone, customNote, language]
   174→  );
   175→
   176→  // Tier badge component
   177→  const tierBadge = (tier: "top" | "mid" | "emerging") => {
   178→    if (tier === "top")
   179→      return <Badge className="bg-purple-500/20 text-purple-300 border-purple-500/30 text-[10px] px-1.5 py-0">🟣 {t(locale, "labels.tierTop")}</Badge>;
   180→    if (tier === "mid")
   181→      return <Badge className="bg-blue-500/20 text-blue-300 border-blue-500/30 text-[10px] px-1.5 py-0">🔵 {t(locale, "labels.tierMid")}</Badge>;
   182→    return <Badge className="bg-emerald-500/20 text-emerald-300 border-emerald-500/30 text-[10px] px-1.5 py-0">🟢 {t(locale, "labels.tierEmerging")}</Badge>;
   183→  };
   184→
   185→  // Toggle tier expand
   186→  const toggleTier = (tier: string) => {
   187→    setExpandedTiers((prev) => ({ ...prev, [tier]: !prev[tier] }));
   188→  };
   189→
   190→  // Bulk send campaign via Gmail Web Compose
   191→  const handleSendCampaign = useCallback(async () => {
   192→    if (!selectedWithEmail.length || !trackName.trim()) return;
   193→    setSending(true);
   194→    setCampaignComplete(false);
   195→    setCampaignResults({ sent: 0, skipped: 0 });
   196→
   197→    const toSend = selectedWithEmail.filter((l) => getLabelEmails(l).length > 0);
   198→    const skipped = selectedWithoutEmail.length;
   199→    let sentCount = 0;
   200→
   201→    setSendProgress({ current: 0, total: toSend.length });
   202→
   203→    for (let i = 0; i < toSend.length; i++) {
   204→      const label = toSend[i];
   205→      const emails = getLabelEmails(label);
   206→      if (!emails.length) continue;
   207→
   208→      const { subject, body } = getPitchForLabel(label);
   209→      if (!subject || !body) continue;
   210→
   211→      setSendProgress({ current: i + 1, total: toSend.length });
   212→
   213→      // Open Gmail compose in new tab with pre-filled content
   214→      const gmailUrl = generateGmailLink(emails, subject, body);
   215→      window.open(gmailUrl, `_blank`, "noopener");
   216→
   217→      sentCount++;
   218→
   219→      // Auto-create demo entry
   220→      const demoAlreadyExists = demos.some(
   221→        (d) => d.labelId === label.id && d.trackName.toLowerCase() === trackName.trim().toLowerCase()
   222→      );
   223→      if (!demoAlreadyExists) {
   224→        addDemo({
   225→          trackName: trackName.trim(),
   226→          labelId: label.id,
   227→          status: "sent",
   228→          sentDate: new Date().toISOString().split("T")[0],
   229→          link: scLink.trim(),
   230→          notes: customNote.trim() ? `${customNote.trim()} (Campaign)` : "Campaign",
   231→          pitchText: `Subject: ${subject}\n\n${body}`,
   232→          artistName: artistName.trim(),
   233→        });
   234→      }
   235→
   236→      // Small delay between opens to avoid popup blocking
   237→      if (i < toSend.length - 1) {
   238→        await new Promise((r) => setTimeout(r, 800));
   239→      }
   240→    }
   241→
   242→    setCampaignResults({ sent: sentCount, skipped });
   243→    setCampaignComplete(true);
   244→    setSending(false);
   245→
   246→    toast({
   247→      title: t(locale, "campaign.complete"),
   248→      description: t(locale, "campaign.reviewAndSend").replace("{count}", String(sentCount)),
   249→    });
   250→  }, [selectedWithEmail, selectedWithoutEmail, getLabelEmails, getPitchForLabel, demos, trackName, scLink, customNote, artistName, addDemo, locale, toast]);
   251→
   252→  // Copy all emails to clipboard
   253→  const handleCopyEmails = useCallback(async () => {
   254→    const emails = selectedWithEmail.flatMap((l) => getLabelEmails(l));
   255→    const emailText = emails.join(", ");
   256→    try {
   257→      await navigator.clipboard.writeText(emailText);
   258→      toast({ title: t(locale, "campaign.emailsCopied") });
   259→    } catch {
   260→      const textarea = document.createElement("textarea");
   261→      textarea.value = emailText;
   262→      document.body.appendChild(textarea);
   263→      textarea.select();
   264→      document.execCommand("copy");
   265→      document.body.removeChild(textarea);
   266→      toast({ title: t(locale, "campaign.emailsCopied") });
   267→    }
   268→  }, [selectedWithEmail, getLabelEmails, locale, toast]);
   269→
   270→  // Track if campaign params changed (to reset completion state)
   271→  const campaignParamsKey = `${selectedLabelIds.size}-${trackName}-${tone}-${language}-${customNote}`;
   272→  const [lastCampaignKey, setLastCampaignKey] = useState(campaignParamsKey);
   273→  if (campaignParamsKey !== lastCampaignKey) {
   274→    setLastCampaignKey(campaignParamsKey);
   275→    if (campaignComplete) setCampaignComplete(false);
   276→  }
   277→
   278→  return (
   279→    <div className="space-y-6">
   280→      {/* ===== TOP SECTION: Track Setup ===== */}
   281→      <Card className="bg-card/60 border-border/30">
   282→        <CardContent className="p-4 sm:p-6">
   283→          <div className="flex items-center gap-2 mb-4">
   284→            <Target className="h-4 w-4 text-primary" />
   285→            <h3 className="text-sm font-mono uppercase tracking-wider text-muted-foreground">
   286→              {t(locale, "campaign.trackSetup")}
   287→            </h3>
   288→          </div>
   289→
   290→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   291→            <div className="space-y-1.5">
   292→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.trackName")}</UILabel>
   293→              <Input
   294→                value={trackName}
   295→                onChange={(e) => setTrackName(e.target.value)}
   296→                placeholder="e.g. Midnight Drive"
   297→                className="bg-secondary/50 border-border/50"
   298→              />
   299→            </div>
   300→            <div className="space-y-1.5">
   301→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.artistName")}</UILabel>
   302→              <Input
   303→                value={artistName}
   304→                onChange={(e) => setArtistName(e.target.value)}
   305→                onBlur={handleArtistBlur}
   306→                placeholder="e.g. Traacia"
   307→                className="bg-secondary/50 border-border/50"
   308→              />
   309→            </div>
   310→          </div>
   311→
   312→          <div className="space-y-1.5 mt-3">
   313→            <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.scLink")}</UILabel>
   314→            <Input
   315→              value={scLink}
   316→              onChange={(e) => setScLink(e.target.value)}
   317→              onBlur={handleScLinkBlur}
   318→              placeholder="https://soundcloud.com/..."
   319→              className="bg-secondary/50 border-border/50"
   320→            />
   321→          </div>
   322→
   323→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3 mt-3">
   324→            <div className="space-y-1.5">
   325→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.tone")}</UILabel>
   326→              <Select value={tone} onValueChange={(v) => setTone(v as PitchTone)}>
   327→                <SelectTrigger className="bg-secondary/50 border-border/50">
   328→                  <SelectValue />
   329→                </SelectTrigger>
   330→                <SelectContent>
   331→                  <SelectItem value="professional">{t(locale, "pitch.toneProfessional")}</SelectItem>
   332→                  <SelectItem value="confident">{t(locale, "pitch.toneConfident")}</SelectItem>
   333→                  <SelectItem value="friendly">{t(locale, "pitch.toneFriendly")}</SelectItem>
   334→                  <SelectItem value="storytelling">{t(locale, "pitch.toneStorytelling")}</SelectItem>
   335→                </SelectContent>
   336→              </Select>
   337→            </div>
   338→            <div className="space-y-1.5">
   339→              <UILabel className="text-xs font-mono uppercase text-muted-foreground flex items-center gap-1">
   340→                <Languages className="h-3 w-3" /> {t(locale, "pitch.emailLanguage")}
   341→              </UILabel>
   342→              <Select value={language} onValueChange={(v) => setLanguage(v as PitchLanguage)}>
   343→                <SelectTrigger className="bg-secondary/50 border-border/50">
   344→                  <SelectValue />
   345→                </SelectTrigger>
   346→                <SelectContent>
   347→                  {(Object.keys(PITCH_LANGUAGES) as PitchLanguage[]).map((lang) => (
   348→                    <SelectItem key={lang} value={lang}>{PITCH_LANGUAGES[lang]}</SelectItem>
   349→                  ))}
   350→                </SelectContent>
   351→              </Select>
   352→            </div>
   353→            <div className="space-y-1.5 sm:col-span-2">
   354→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">
   355→                {t(locale, "labels.selectGenre")} 🎯
   356→              </UILabel>
   357→              <Select value={selectedGenre} onValueChange={(v) => { setSelectedGenre(v); setSelectedLabelIds(new Set()); setCampaignComplete(false); }}>
   358→                <SelectTrigger className="bg-secondary/50 border-border/50">
   359→                  <SelectValue placeholder={t(locale, "campaign.selectGenreFirst")} />
   360→                </SelectTrigger>
   361→                <SelectContent>
   362→                  {genres.map((g) => (
   363→                    <SelectItem key={g} value={g}>{g}</SelectItem>
   364→                  ))}
   365→                </SelectContent>
   366→              </Select>
   367→            </div>
   368→          </div>
   369→
   370→          <div className="space-y-1.5 mt-3">
   371→            <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.additionalNote")}</UILabel>
   372→            <Textarea
   373→              value={customNote}
   374→              onChange={(e) => setCustomNote(e.target.value)}
   375→              placeholder="..."
   376→              rows={2}
   377→              className="bg-secondary/50 border-border/50 resize-none"
   378→            />
   379→          </div>
   380→        </CardContent>
   381→      </Card>
   382→
   383→      {/* ===== MIDDLE SECTION: Target Labels ===== */}
   384→      <Card className="bg-card/60 border-border/30">
   385→        <CardContent className="p-4 sm:p-6">
   386→          <div className="flex items-center justify-between mb-4">
   387→            <div className="flex items-center gap-2">
   388→              <Music2 className="h-4 w-4 text-primary" />
   389→              <h3 className="text-sm font-mono uppercase tracking-wider text-muted-foreground">
   390→                {t(locale, "campaign.targetLabels")}
   391→              </h3>
   392→            </div>
   393→            <div className="flex items-center gap-2">
   394→              <label className="flex items-center gap-1.5 text-xs text-muted-foreground cursor-pointer">
   395→                <Checkbox
   396→                  checked={onlyWithEmail}
   397→                  onCheckedChange={(checked) => setOnlyWithEmail(!!checked)}
   398→                />
   399→                {t(locale, "campaign.onlyWithEmail")}
   400→              </label>
   401→            </div>
   402→          </div>
   403→
   404→          {/* Selection stats */}
   405→          {selectedLabelIds.size > 0 && (
   406→            <div className="mb-3 flex items-center gap-3 flex-wrap">
   407→              <span className="text-sm font-medium text-foreground">
   408→                {t(locale, "campaign.selectedCount").replace("{count}", String(selectedLabelIds.size))}
   409→              </span>
   410→              <span className="text-xs text-emerald-400 flex items-center gap-1">
   411→                <Mail className="h-3 w-3" />
   412→                {selectedWithEmail.length} {t(locale, "campaign.withEmail")}
   413→              </span>
   414→              {selectedWithoutEmail.length > 0 && (
   415→                <span className="text-xs text-amber-400 flex items-center gap-1">
   416→                  <AlertCircle className="h-3 w-3" />
   417→                  {selectedWithoutEmail.length} {t(locale, "campaign.noEmail")}
   418→                </span>
   419→              )}
   420→            </div>
   421→          )}
   422→
   423→          {/* Select all / Deselect all */}
   424→          {allMatchingLabels.length > 0 && (
   425→            <div className="flex items-center gap-2 mb-3">
   426→              <Button variant="outline" size="sm" className="text-xs border-border/50" onClick={selectAll}>
   427→                {t(locale, "campaign.selectAll")}
   428→              </Button>
   429→              <Button variant="outline" size="sm" className="text-xs border-border/50" onClick={deselectAll}>
   430→                {t(locale, "campaign.deselectAll")}
   431→              </Button>
   432→            </div>
   433→          )}
   434→
   435→          {/* No genre selected */}
   436→          {!selectedGenre && (
   437→            <div className="flex flex-col items-center justify-center py-10 text-muted-foreground/40">
   438→              <Target className="h-10 w-10 mb-2" />
   439→              <p className="text-sm">{t(locale, "campaign.selectGenreFirst")}</p>
   440→            </div>
   441→          )}
   442→
   443→          {/* No matching labels */}
   444→          {selectedGenre && allMatchingLabels.length === 0 && (
   445→            <div className="flex flex-col items-center justify-center py-10 text-muted-foreground/40">
   446→              <Music2 className="h-10 w-10 mb-2" />
   447→              <p className="text-sm">{t(locale, "campaign.noMatchingLabels")}</p>
   448→            </div>
   449→          )}
   450→
   451→          {/* Labels grouped by tier */}
   452→          {selectedGenre && allMatchingLabels.length > 0 && (
   453→            <div className="space-y-2">
   454→              {(["top", "mid", "emerging"] as const).map((tier) => {
   455→                const tierLabels = filteredByTier[tier];
   456→                if (tierLabels.length === 0) return null;
   457→                const tierLabel = tier === "top" ? t(locale, "labels.topTier") : tier === "mid" ? t(locale, "labels.midTier") : t(locale, "labels.emerging");
   458→                const isExpanded = expandedTiers[tier] !== false;
   459→
   460→                return (
   461→                  <div key={tier} className="border border-border/20 rounded-lg overflow-hidden">
   462→                    <button
   463→                      className="w-full flex items-center justify-between px-3 py-2 bg-secondary/30 hover:bg-secondary/50 transition-colors"
   464→                      onClick={() => toggleTier(tier)}
   465→                    >
   466→                      <div className="flex items-center gap-2">
   467→                        {tierBadge(tier)}
   468→                        <span className="text-xs text-muted-foreground">{tierLabel}</span>
   469→                        <Badge variant="secondary" className="text-[10px] px-1.5 py-0">{tierLabels.length}</Badge>
   470→                      </div>
   471→                      {isExpanded ? <ChevronUp className="h-3.5 w-3.5 text-muted-foreground" /> : <ChevronDown className="h-3.5 w-3.5 text-muted-foreground" />}
   472→                    </button>
   473→                    {isExpanded && (
   474→                      <div className="divide-y divide-border/10 max-h-72 overflow-y-auto">
   475→                        {tierLabels.map((label) => {
   476→                          const emails = getLabelEmails(label);
   477→                          const hasEmail = emails.length > 0;
   478→                          const isSelected = selectedLabelIds.has(label.id);
   479→                          const rank = label.rankByGenre?.[selectedGenre];
   480→
   481→                          return (
   482→                            <label
   483→                              key={label.id}
   484→                              className={`flex items-center gap-3 px-3 py-2 cursor-pointer hover:bg-secondary/20 transition-colors ${
   485→                                !hasEmail ? "opacity-60" : ""
   486→                              }`}
   487→                            >
   488→                              <Checkbox
   489→                                checked={isSelected}
   490→                                onCheckedChange={() => toggleLabel(label.id)}
   491→                              />
   492→                              <div className="flex-1 min-w-0 flex items-center gap-2 flex-wrap">
   493→                                <span className="text-sm font-medium text-foreground truncate">{label.name}</span>
   494→                                {rank && (
