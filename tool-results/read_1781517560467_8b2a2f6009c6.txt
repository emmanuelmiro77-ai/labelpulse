     1→"use client";
     2→
     3→import { useAppStore, type Demo, type DemoStatus } from "@/lib/store";
     4→import { t, type Locale } from "@/lib/i18n";
     5→import { useState, useMemo } from "react";
     6→import {
     7→  Plus,
     8→  Pencil,
     9→  Trash2,
    10→  Clock,
    11→  ChevronRight,
    12→  CheckCircle2,
    13→  XCircle,
    14→  Headphones,
    15→  Send,
    16→  Disc3,
    17→  AlertTriangle,
    18→  Search,
    19→  Music,
    20→  Eye,
    21→  Mail,
    22→  Link2,
    23→  Calendar,
    24→  User,
    25→  FileText,
    26→  Music2,
    27→  ArrowLeft,
    28→} from "lucide-react";
    29→import { Button } from "@/components/ui/button";
    30→import { Badge } from "@/components/ui/badge";
    31→import { Card, CardContent } from "@/components/ui/card";
    32→import { Input } from "@/components/ui/input";
    33→import {
    34→  Dialog,
    35→  DialogContent,
    36→  DialogHeader,
    37→  DialogTitle,
    38→  DialogFooter,
    39→} from "@/components/ui/dialog";
    40→import {
    41→  Select,
    42→  SelectContent,
    43→  SelectItem,
    44→  SelectTrigger,
    45→  SelectValue,
    46→} from "@/components/ui/select";
    47→import { Label as UILabel } from "@/components/ui/label";
    48→import { Textarea } from "@/components/ui/textarea";
    49→
    50→const STATUS_KEYS: DemoStatus[] = [
    51→  "ready",
    52→  "sent",
    53→  "reviewing",
    54→  "accepted",
    55→  "rejected",
    56→];
    57→
    58→const STATUS_ICONS: Record<DemoStatus, React.ReactNode> = {
    59→  ready: <Disc3 className="h-3.5 w-3.5" />,
    60→  sent: <Send className="h-3.5 w-3.5" />,
    61→  reviewing: <Headphones className="h-3.5 w-3.5" />,
    62→  accepted: <CheckCircle2 className="h-3.5 w-3.5" />,
    63→  rejected: <XCircle className="h-3.5 w-3.5" />,
    64→};
    65→
    66→const STATUS_COLORS: Record<
    67→  DemoStatus,
    68→  { color: string; bgColor: string; borderColor: string }
    69→> = {
    70→  ready: { color: "text-gray-400", bgColor: "bg-gray-500/10", borderColor: "border-gray-500/20" },
    71→  sent: { color: "text-purple-400", bgColor: "bg-purple-500/10", borderColor: "border-purple-500/20" },
    72→  reviewing: { color: "text-cyan-400", bgColor: "bg-cyan-500/10", borderColor: "border-cyan-500/20" },
    73→  accepted: { color: "text-emerald-400", bgColor: "bg-emerald-500/10", borderColor: "border-emerald-500/20" },
    74→  rejected: { color: "text-red-400", bgColor: "bg-red-500/10", borderColor: "border-red-500/20" },
    75→};
    76→
    77→const STATUS_FLOW: DemoStatus[] = ["ready", "sent", "reviewing", "accepted"];
    78→
    79→function getDaysSince(dateStr: string | null): number | null {
    80→  if (!dateStr) return null;
    81→  return Math.floor((Date.now() - new Date(dateStr).getTime()) / 86400000);
    82→}
    83→
    84→const STATUS_TKEYS: Record<DemoStatus, "demos.ready" | "demos.sent" | "demos.reviewing" | "demos.accepted" | "demos.rejected"> = {
    85→  ready: "demos.ready",
    86→  sent: "demos.sent",
    87→  reviewing: "demos.reviewing",
    88→  accepted: "demos.accepted",
    89→  rejected: "demos.rejected",
    90→};
    91→
    92→export function DemoTracker() {
    93→  const { labels, demos, addDemo, updateDemo, deleteDemo, advanceDemoStatus, locale: _locale, getGenres } =
    94→    useAppStore();
    95→  const locale = _locale as Locale;
    96→  const genres = getGenres();
    97→  const [viewMode, setViewMode] = useState<"kanban" | "table">("kanban");
    98→  const [search, setSearch] = useState("");
    99→  const [statusFilter, setStatusFilter] = useState("all");
   100→  const [showAddDialog, setShowAddDialog] = useState(false);
   101→  const [editingDemo, setEditingDemo] = useState<Demo | null>(null);
   102→  const [deleteConfirmId, setDeleteConfirmId] = useState<string | null>(null);
   103→
   104→  // Detail dialog
   105→  const [detailDemo, setDetailDemo] = useState<Demo | null>(null);
   106→
   107→  // Label history view
   108→  const [labelHistoryId, setLabelHistoryId] = useState<string | null>(null);
   109→
   110→  const [formTrackName, setFormTrackName] = useState("");
   111→  const [formLabelId, setFormLabelId] = useState("");
   112→  const [formStatus, setFormStatus] = useState<DemoStatus>("ready");
   113→  const [formSentDate, setFormSentDate] = useState("");
   114→  const [formLink, setFormLink] = useState("");
   115→  const [formNotes, setFormNotes] = useState("");
   116→
   117→  const filteredDemos = useMemo(() => {
   118→    return demos.filter((d) => {
   119→      const matchSearch =
   120→        !search ||
   121→        d.trackName.toLowerCase().includes(search.toLowerCase()) ||
   122→        d.notes.toLowerCase().includes(search.toLowerCase());
   123→      const matchStatus = statusFilter === "all" || d.status === statusFilter;
   124→      return matchSearch && matchStatus;
   125→    });
   126→  }, [demos, search, statusFilter]);
   127→
   128→  // Label history: all demos for a specific label
   129→  const labelHistoryDemos = useMemo(() => {
   130→    if (!labelHistoryId) return [];
   131→    return demos
   132→      .filter((d) => d.labelId === labelHistoryId)
   133→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   134→  }, [demos, labelHistoryId]);
   135→
   136→  const labelHistoryName = useMemo(() => {
   137→    if (!labelHistoryId) return "";
   138→    return labels.find((l) => l.id === labelHistoryId)?.name ?? "";
   139→  }, [labels, labelHistoryId]);
   140→
   141→  const getLabelName = (labelId: string) => {
   142→    return labels.find((l) => l.id === labelId)?.name ?? "—";
   143→  };
   144→
   145→  const resetForm = () => {
   146→    setFormTrackName("");
   147→    setFormLabelId("");
   148→    setFormStatus("ready");
   149→    setFormSentDate("");
   150→    setFormLink("");
   151→    setFormNotes("");
   152→  };
   153→
   154→  const openAdd = () => { resetForm(); setEditingDemo(null); setShowAddDialog(true); };
   155→
   156→  const openEdit = (demo: Demo) => {
   157→    setFormTrackName(demo.trackName);
   158→    setFormLabelId(demo.labelId);
   159→    setFormStatus(demo.status);
   160→    setFormSentDate(demo.sentDate ?? "");
   161→    setFormLink(demo.link);
   162→    setFormNotes(demo.notes);
   163→    setEditingDemo(demo);
   164→    setShowAddDialog(true);
   165→  };
   166→
   167→  const handleSave = () => {
   168→    if (!formTrackName.trim() || !formLabelId) return;
   169→    const data = {
   170→      trackName: formTrackName.trim(),
   171→      labelId: formLabelId,
   172→      status: formStatus,
   173→      sentDate: formSentDate || null,
   174→      link: formLink.trim(),
   175→      notes: formNotes.trim(),
   176→      pitchText: editingDemo?.pitchText || "",
   177→      artistName: editingDemo?.artistName || "",
   178→    };
   179→    if (editingDemo) { updateDemo(editingDemo.id, data); }
   180→    else { addDemo(data); }
   181→    setShowAddDialog(false);
   182→    resetForm();
   183→  };
   184→
   185→  const canAdvance = (status: DemoStatus) => STATUS_FLOW.indexOf(status) < STATUS_FLOW.length - 1;
   186→
   187→  // If viewing label history, show that instead
   188→  if (labelHistoryId) {
   189→    return (
   190→      <div className="space-y-4">
   191→        <div className="flex items-center gap-3">
   192→          <Button variant="ghost" size="sm" onClick={() => setLabelHistoryId(null)} className="text-muted-foreground hover:text-foreground">
   193→            <ArrowLeft className="h-4 w-4 mr-1.5" />
   194→            {t(locale, "demos.backToTracker")}
   195→          </Button>
   196→        </div>
   197→
   198→        <div className="flex items-center gap-2">
   199→          <Music2 className="h-5 w-5 text-primary" />
   200→          <h3 className="text-lg font-bold text-foreground">{labelHistoryName}</h3>
   201→          <Badge variant="secondary" className="text-[10px]">{labelHistoryDemos.length} {t(locale, "demos.submissions")}</Badge>
   202→        </div>
   203→
   204→        {labelHistoryDemos.length === 0 ? (
   205→          <div className="text-center py-12 text-muted-foreground">
   206→            <Music className="h-10 w-10 mx-auto mb-2 opacity-30" />
   207→            <p className="text-sm">{t(locale, "demos.noDemos")}</p>
   208→          </div>
   209→        ) : (
   210→          <div className="space-y-2">
   211→            {labelHistoryDemos.map((demo) => {
   212→              const config = STATUS_COLORS[demo.status];
   213→              const daysSince = getDaysSince(demo.sentDate);
   214→              return (
   215→                <Card
   216→                  key={demo.id}
   217→                  className="bg-card/80 border-border/30 hover:border-primary/30 transition-all cursor-pointer"
   218→                  onClick={() => setDetailDemo(demo)}
   219→                >
   220→                  <CardContent className="p-4">
   221→                    <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   222→                      <div className="flex-1 min-w-0">
   223→                        <div className="flex items-center gap-2 flex-wrap">
   224→                          <h4 className="text-sm font-semibold text-foreground">{demo.trackName}</h4>
   225→                          <Badge variant="outline" className={`${config.bgColor} ${config.color} ${config.borderColor} text-[10px]`}>
   226→                            {STATUS_ICONS[demo.status]}
   227→                            <span className="ml-1">{t(locale, STATUS_TKEYS[demo.status])}</span>
   228→                          </Badge>
   229→                          {demo.artistName && (
   230→                            <span className="text-[10px] text-muted-foreground font-mono flex items-center gap-0.5">
   231→                              <User className="h-2.5 w-2.5" /> {demo.artistName}
   232→                            </span>
   233→                          )}
   234→                        </div>
   235→                        <div className="flex flex-wrap items-center gap-x-3 gap-y-0.5 mt-1 text-[11px] text-muted-foreground">
   236→                          {demo.sentDate && (
   237→                            <span className="flex items-center gap-1">
   238→                              <Calendar className="h-3 w-3" /> {demo.sentDate}
   239→                              {daysSince !== null && <span className="ml-0.5">({daysSince}{t(locale, "dash.daysAgo")})</span>}
   240→                            </span>
   241→                          )}
   242→                          {demo.link && (
   243→                            <span className="flex items-center gap-1 font-mono truncate max-w-[200px]">
   244→                              <Link2 className="h-3 w-3" /> {demo.link}
   245→                            </span>
   246→                          )}
   247→                          {demo.pitchText && (
   248→                            <span className="flex items-center gap-1 text-primary/50">
   249→                              <FileText className="h-3 w-3" /> {t(locale, "demos.hasPitch")}
   250→                            </span>
   251→                          )}
   252→                        </div>
   253→                      </div>
   254→                      <div className="flex items-center gap-1 shrink-0">
   255→                        <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); setDetailDemo(demo); }} title={t(locale, "demos.viewDetail")}>
   256→                          <Eye className="h-3.5 w-3.5" />
   257→                        </Button>
   258→                        {canAdvance(demo.status) && (
   259→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); advanceDemoStatus(demo.id); }} title={t(locale, "demos.advance")}>
   260→                            <ChevronRight className="h-3.5 w-3.5" />
   261→                          </Button>
   262→                        )}
   263→                        <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); openEdit(demo); }}>
   264→                          <Pencil className="h-3.5 w-3.5" />
   265→                        </Button>
   266→                      </div>
   267→                    </div>
   268→                  </CardContent>
   269→                </Card>
   270→              );
   271→            })}
   272→          </div>
   273→        )}
   274→
   275→        {/* Detail Dialog (reused) */}
   276→        <DemoDetailDialog demo={detailDemo} onClose={() => setDetailDemo(null)} locale={locale} getLabelName={getLabelName} />
   277→      </div>
   278→    );
   279→  }
   280→
   281→  return (
   282→    <div className="space-y-4">
   283→      {/* Controls */}
   284→      <div className="flex flex-col sm:flex-row gap-3">
   285→        <div className="relative flex-1">
   286→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   287→          <Input
   288→            placeholder={t(locale, "demos.search")}
   289→            value={search}
   290→            onChange={(e) => setSearch(e.target.value)}
   291→            className="pl-9 bg-secondary/50 border-border/50"
   292→          />
   293→        </div>
   294→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   295→          <SelectTrigger className="w-full sm:w-[170px] bg-secondary/50 border-border/50">
   296→            <SelectValue placeholder={t(locale, "demos.allStatuses")} />
   297→          </SelectTrigger>
   298→          <SelectContent>
   299→            <SelectItem value="all">{t(locale, "demos.allStatuses")}</SelectItem>
   300→            {STATUS_KEYS.map((s) => (
   301→              <SelectItem key={s} value={s}>
   302→                {t(locale, STATUS_TKEYS[s])}
   303→              </SelectItem>
   304→            ))}
   305→          </SelectContent>
   306→        </Select>
   307→        <div className="flex gap-2">
   308→          <Button variant={viewMode === "kanban" ? "default" : "outline"} size="sm" onClick={() => setViewMode("kanban")} className="text-xs">
   309→            {t(locale, "demos.kanban")}
   310→          </Button>
   311→          <Button variant={viewMode === "table" ? "default" : "outline"} size="sm" onClick={() => setViewMode("table")} className="text-xs">
   312→            {t(locale, "demos.table")}
   313→          </Button>
   314→        </div>
   315→        <Button onClick={openAdd} className="glow-purple shrink-0">
   316→          <Plus className="h-4 w-4 mr-1.5" />
   317→          {t(locale, "demos.addDemo")}
   318→        </Button>
   319→      </div>
   320→
   321→      {/* Kanban View */}
   322→      {viewMode === "kanban" && (
   323→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-3 min-h-[400px]">
   324→          {STATUS_KEYS.map((status) => {
   325→            const config = STATUS_COLORS[status];
   326→            const columnDemos = filteredDemos.filter((d) => d.status === status);
   327→            return (
   328→              <div key={status} className={`rounded-xl border ${config.borderColor} ${config.bgColor} p-3 space-y-2`}>
   329→                <div className="flex items-center gap-2 mb-1">
   330→                  <span className={config.color}>{STATUS_ICONS[status]}</span>
   331→                  <span className={`text-xs font-semibold uppercase tracking-wider ${config.color}`}>
   332→                    {t(locale, STATUS_TKEYS[status])}
   333→                  </span>
   334→                  <span className="ml-auto text-[10px] font-mono text-muted-foreground bg-secondary/50 px-1.5 py-0.5 rounded">
   335→                    {columnDemos.length}
   336→                  </span>
   337→                </div>
   338→                {columnDemos.map((demo) => {
   339→                  const daysSince = getDaysSince(demo.sentDate);
   340→                  const isOverdue = (demo.status === "sent" || demo.status === "reviewing") && daysSince !== null && daysSince > 14;
   341→                  return (
   342→                    <Card
   343→                      key={demo.id}
   344→                      className={`bg-card/80 border-border/30 hover:border-primary/20 transition-all group cursor-pointer ${isOverdue ? "ring-1 ring-amber-500/40" : ""}`}
   345→                      onClick={() => setDetailDemo(demo)}
   346→                    >
   347→                      <CardContent className="p-3 space-y-1.5">
   348→                        <div className="flex items-start justify-between gap-1">
   349→                          <h4 className="text-sm font-semibold text-foreground leading-tight">{demo.trackName}</h4>
   350→                          <div className="flex items-center gap-0.5 opacity-0 group-hover:opacity-100 transition-opacity shrink-0">
   351→                            <Button variant="ghost" size="icon" className="h-6 w-6" onClick={(e) => { e.stopPropagation(); setDetailDemo(demo); }} title={t(locale, "demos.viewDetail")}>
   352→                              <Eye className="h-3 w-3" />
   353→                            </Button>
   354→                            {canAdvance(demo.status) && (
   355→                              <Button variant="ghost" size="icon" className="h-6 w-6" onClick={(e) => { e.stopPropagation(); advanceDemoStatus(demo.id); }} title={t(locale, "demos.advance")}>
   356→                                <ChevronRight className="h-3 w-3" />
   357→                              </Button>
   358→                            )}
   359→                            <Button variant="ghost" size="icon" className="h-6 w-6" onClick={(e) => { e.stopPropagation(); openEdit(demo); }}>
   360→                              <Pencil className="h-3 w-3" />
   361→                            </Button>
   362→                            <Button variant="ghost" size="icon" className="h-6 w-6 hover:text-destructive" onClick={(e) => { e.stopPropagation(); setDeleteConfirmId(demo.id); }}>
   363→                              <Trash2 className="h-3 w-3" />
   364→                            </Button>
   365→                          </div>
   366→                        </div>
   367→                        <p
   368→                          className="text-xs text-primary/70 font-medium cursor-pointer hover:underline"
   369→                          onClick={(e) => { e.stopPropagation(); setLabelHistoryId(demo.labelId); }}
   370→                        >
   371→                          {getLabelName(demo.labelId)}
   372→                        </p>
   373→                        {demo.sentDate && (
   374→                          <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
   375→                            <Clock className="h-3 w-3" />
   376→                            <span>{demo.sentDate}{daysSince !== null && <span className={isOverdue ? "text-amber-400 font-semibold ml-1" : "ml-1"}>({daysSince}{t(locale, "dash.daysAgo")})</span>}</span>
   377→                          </div>
   378→                        )}
   379→                        {demo.pitchText && (
   380→                          <div className="flex items-center gap-1 text-[10px] text-primary/50">
   381→                            <FileText className="h-2.5 w-2.5" /> {t(locale, "demos.hasPitch")}
   382→                          </div>
   383→                        )}
   384→                        {isOverdue && (
   385→                          <div className="flex items-center gap-1 text-[11px] text-amber-400 font-medium">
   386→                            <AlertTriangle className="h-3 w-3" />
   387→                            {t(locale, "demos.followUp")}
   388→                          </div>
   389→                        )}
   390→                      </CardContent>
   391→                    </Card>
   392→                  );
   393→                })}
   394→                {columnDemos.length === 0 && <div className="text-center py-6 text-muted-foreground/40 text-xs">—</div>}
   395→              </div>
   396→            );
   397→          })}
   398→        </div>
   399→      )}
   400→
   401→      {/* Table View */}
   402→      {viewMode === "table" && (
   403→        <div className="rounded-xl border border-border/40 overflow-hidden">
   404→          <div className="overflow-x-auto">
   405→            <table className="w-full text-sm">
   406→              <thead className="bg-secondary/30">
   407→                <tr className="text-left text-xs font-mono uppercase tracking-wider text-muted-foreground">
   408→                  <th className="px-4 py-3">{t(locale, "demos.trackName")}</th>
   409→                  <th className="px-4 py-3">{t(locale, "demos.targetLabel")}</th>
   410→                  <th className="px-4 py-3">{t(locale, "labels.status")}</th>
   411→                  <th className="px-4 py-3">{t(locale, "demos.sentDate")}</th>
   412→                  <th className="px-4 py-3">{t(locale, "demos.daysSince")}</th>
   413→                  <th className="px-4 py-3 text-right">{t(locale, "demos.actions")}</th>
   414→                </tr>
   415→              </thead>
   416→              <tbody className="divide-y divide-border/20">
   417→                {filteredDemos.map((demo) => {
   418→                  const config = STATUS_COLORS[demo.status];
   419→                  const daysSince = getDaysSince(demo.sentDate);
   420→                  const isOverdue = (demo.status === "sent" || demo.status === "reviewing") && daysSince !== null && daysSince > 14;
   421→                  return (
   422→                    <tr
   423→                      key={demo.id}
   424→                      className="hover:bg-secondary/20 transition-colors cursor-pointer"
   425→                      onClick={() => setDetailDemo(demo)}
   426→                    >
   427→                      <td className="px-4 py-3">
   428→                        <div className="font-medium text-foreground">{demo.trackName}</div>
   429→                        {demo.artistName && <div className="text-[10px] text-muted-foreground/60 mt-0.5">{demo.artistName}</div>}
   430→                        {demo.notes && <div className="text-[11px] text-muted-foreground/60 mt-0.5 line-clamp-1">{demo.notes}</div>}
   431→                      </td>
   432→                      <td className="px-4 py-3">
   433→                        <span
   434→                          className="text-primary/80 text-xs font-medium cursor-pointer hover:underline"
   435→                          onClick={(e) => { e.stopPropagation(); setLabelHistoryId(demo.labelId); }}
   436→                        >
   437→                          {getLabelName(demo.labelId)}
   438→                        </span>
   439→                      </td>
   440→                      <td className="px-4 py-3">
   441→                        <Badge variant="outline" className={`${config.bgColor} ${config.color} ${config.borderColor}`}>
   442→                          {STATUS_ICONS[demo.status]}
   443→                          <span className="ml-1">{t(locale, STATUS_TKEYS[demo.status])}</span>
   444→                        </Badge>
   445→                      </td>
   446→                      <td className="px-4 py-3 text-xs text-muted-foreground font-mono">{demo.sentDate ?? "—"}</td>
   447→                      <td className="px-4 py-3 text-xs font-mono">
   448→                        {daysSince !== null ? (
   449→                          <span className={isOverdue ? "text-amber-400 font-semibold" : "text-muted-foreground"}>
   450→                            {daysSince}{t(locale, "dash.daysAgo")}
   451→                          </span>
   452→                        ) : <span className="text-muted-foreground">—</span>}
   453→                      </td>
   454→                      <td className="px-4 py-3 text-right">
   455→                        <div className="flex items-center justify-end gap-1">
   456→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); setDetailDemo(demo); }} title={t(locale, "demos.viewDetail")}>
   457→                            <Eye className="h-3.5 w-3.5" />
   458→                          </Button>
   459→                          {canAdvance(demo.status) && (
   460→                            <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); advanceDemoStatus(demo.id); }} title={t(locale, "demos.advance")}>
   461→                              <ChevronRight className="h-3.5 w-3.5" />
   462→                            </Button>
   463→                          )}
   464→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); openEdit(demo); }}>
   465→                            <Pencil className="h-3.5 w-3.5" />
   466→                          </Button>
   467→                          <Button variant="ghost" size="icon" className="h-7 w-7 hover:text-destructive" onClick={(e) => { e.stopPropagation(); setDeleteConfirmId(demo.id); }}>
   468→                            <Trash2 className="h-3.5 w-3.5" />
   469→                          </Button>
   470→                        </div>
   471→                      </td>
   472→                    </tr>
   473→                  );
   474→                })}
   475→                {filteredDemos.length === 0 && (
   476→                  <tr><td colSpan={6} className="px-4 py-12 text-center text-muted-foreground">
   477→                    <Music className="h-10 w-10 mx-auto mb-2 opacity-30" />
   478→                    <p className="text-sm">{t(locale, "demos.noDemos")}</p>
   479→                  </td></tr>
   480→                )}
   481→              </tbody>
   482→            </table>
   483→          </div>
   484→        </div>
   485→      )}
   486→
   487→      {/* Detail Dialog */}
   488→      <DemoDetailDialog demo={detailDemo} onClose={() => setDetailDemo(null)} locale={locale} getLabelName={getLabelName} />
   489→
   490→      {/* Add/Edit Dialog */}
   491→      <Dialog open={showAddDialog} onOpenChange={setShowAddDialog}>
   492→        <DialogContent className="sm:max-w-md bg-card border-border/50">
   493→          <DialogHeader>
   494→            <DialogTitle>{editingDemo ? t(locale, "demos.editDemo") : t(locale, "demos.addDemo")}</DialogTitle>
   495→          </DialogHeader>
   496→          <div className="space-y-4 py-2">
   497→            <div className="space-y-1.5">
   498→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "demos.trackName")}</UILabel>
   499→              <Input value={formTrackName} onChange={(e) => setFormTrackName(e.target.value)} placeholder="e.g. Midnight Drive" className="bg-secondary/50" />
   500→            </div>
   501→            <div className="grid grid-cols-2 gap-3">
   502→              <div className="space-y-1.5">
   503→                <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "demos.targetLabel")}</UILabel>
   504→                <Select value={formLabelId} onValueChange={setFormLabelId}>
   505→                  <SelectTrigger className="bg-secondary/50"><SelectValue placeholder={t(locale, "demos.selectLabel")} /></SelectTrigger>
   506→                  <SelectContent>
   507→                    {labels.filter((l) => l.status === "open").slice(0, 50).map((l) => (
   508→                      <SelectItem key={l.id} value={l.id}>{l.name}</SelectItem>
   509→                    ))}
   510→                  </SelectContent>
   511→                </Select>
   512→              </div>
   513→              <div className="space-y-1.5">
   514→                <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "labels.status")}</UILabel>
   515→                <Select value={formStatus} onValueChange={(v) => setFormStatus(v as DemoStatus)}>
   516→                  <SelectTrigger className="bg-secondary/50"><SelectValue /></SelectTrigger>
   517→                  <SelectContent>
   518→                    {STATUS_KEYS.map((s) => (<SelectItem key={s} value={s}>{t(locale, STATUS_TKEYS[s])}</SelectItem>))}
   519→                  </SelectContent>
   520→                </Select>
   521→              </div>
   522→            </div>
   523→            <div className="space-y-1.5">
   524→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "demos.scLink")}</UILabel>
   525→              <Input value={formLink} onChange={(e) => setFormLink(e.target.value)} placeholder="https://soundcloud.com/..." className="bg-secondary/50" />
   526→            </div>
   527→            <div className="grid grid-cols-2 gap-3">
   528→              <div className="space-y-1.5">
   529→                <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "demos.dateSent")}</UILabel>
   530→                <Input type="date" value={formSentDate} onChange={(e) => setFormSentDate(e.target.value)} className="bg-secondary/50" />
   531→              </div>
   532→            </div>
   533→            <div className="space-y-1.5">
   534→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "labels.notes")}</UILabel>
   535→              <Textarea value={formNotes} onChange={(e) => setFormNotes(e.target.value)} placeholder="BPM, key..." rows={2} className="bg-secondary/50 resize-none" />
   536→            </div>
   537→          </div>
   538→          <DialogFooter>
   539→            <Button variant="ghost" onClick={() => setShowAddDialog(false)}>{t(locale, "labels.cancel")}</Button>
   540→            <Button onClick={handleSave} disabled={!formTrackName.trim() || !formLabelId}>
   541→              {editingDemo ? t(locale, "labels.update") : t(locale, "labels.save")}
   542→            </Button>
   543→          </DialogFooter>
   544→        </DialogContent>
   545→      </Dialog>
   546→
   547→      {/* Delete Confirm */}
   548→      <Dialog open={!!deleteConfirmId} onOpenChange={() => setDeleteConfirmId(null)}>
   549→        <DialogContent className="sm:max-w-sm bg-card border-border/50">
   550→          <DialogHeader><DialogTitle>{t(locale, "demos.deleteConfirm")}</DialogTitle></DialogHeader>
   551→          <p className="text-sm text-muted-foreground">{t(locale, "demos.deleteConfirmMsg")}</p>
   552→          <DialogFooter>
   553→            <Button variant="ghost" onClick={() => setDeleteConfirmId(null)}>{t(locale, "labels.cancel")}</Button>
   554→            <Button variant="destructive" onClick={() => { if (deleteConfirmId) { deleteDemo(deleteConfirmId); setDeleteConfirmId(null); } }}>
   555→              {t(locale, "labels.delete")}
   556→            </Button>
   557→          </DialogFooter>
   558→        </DialogContent>
   559→      </Dialog>
   560→    </div>
   561→  );
   562→}
