   231→function safeJsonParse(str: string | null): any | null {
   232→  if (!str) return null;
   233→  try {
   234→    return JSON.parse(str);
   235→  } catch {
   236→    return null;
   237→  }
   238→}

   240→function safeLocalStorageGet(key: string): string | null {
   241→  try {
   242→    return localStorage.getItem(key);
   243→  } catch {
   244→    return null;
   245→  }
   246→}

   248→function safeLocalStorageSet(key: string, value: string): boolean {
   249→  try {
   250→    localStorage.setItem(key, value);
   251→    return true;
   252→  } catch (e) {
   253→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   254→    return false;
   255→  }
   256→}

   259→  try {
   260→    localStorage.removeItem(key);

   262→    // Ignore
   263→  }

   266→/**
   267→ * Extract rankingSnapshots array from a raw localStorage JSON string.
   268→ * Returns [] if not found / invalid.

   270→function extractSnapshotsFromRaw(raw: string | null): RankingSnapshot[] {
   271→  if (!raw) return [];
   272→  const data = safeJsonParse(raw);
   273→  if (!data) return [];

   275→  return Array.isArray(snaps) ? snaps : [];

   278→/**
   279→ * Merge two snapshot arrays, deduping by `id` first, then by `timestamp+source`.
   280→ * Always returns ascending-by-timestamp order.
   281→ * Used by: importData (backup file), robustStorage.getItem (safety-net restore),
   282→ * and loadFromCloud (cloud merge).
   283→ */
   284→function mergeSnapshots(
   285→  base: RankingSnapshot[] | null | undefined,
   286→  incoming: RankingSnapshot[] | null | undefined
   287→): RankingSnapshot[] {
   288→  const a = Array.isArray(base) ? base : [];
   289→  const b = Array.isArray(incoming) ? incoming : [];
   290→  const seen = new Set<string>();
   291→  const out: RankingSnapshot[] = [];
   292→  for (const s of [...a, ...b]) {
   293→    if (!s || typeof s !== "object") continue;
   294→    const key = s.id || `${s.timestamp}|${s.source}`;
   295→    if (seen.has(key)) continue;
   296→    seen.add(key);
   297→    out.push(s);
   298→  }
   299→  out.sort(
   300→    (x, y) =>
   301→      new Date(x.timestamp || 0).getTime() - new Date(y.timestamp || 0).getTime()
   302→  );
   303→  return out;
   304→}

   319→/**
   320→ * Check if persisted data has user edits (emails, notes, links, etc.)

   323→function countUserEditedLabels(labels: any[]): number {
   324→  if (!Array.isArray(labels)) return 0;
   325→  return labels.filter((l) =>
   326→    (l.emails && l.emails.length > 0) ||
   327→    (l.notes && l.notes.trim() !== "") ||
   328→    (l.website && l.website.trim() !== "") ||
   329→    (l.demoLink && l.demoLink.trim() !== "") ||
   330→    (l.socialLink && l.socialLink.trim() !== "") ||
   331→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   332→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   333→    l.isCustom === true
   334→  ).length;
   335→}

   337→function getTotalLabelCount(data: any): number {
   338→  if (!data) return 0;
   339→  if (Array.isArray(data.labels)) return data.labels.length;
   340→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   341→  return 0;

   344→/**
   345→ * Extract user-edited label count from raw localStorage string.
   346→ * Used by the write guard to check if a write would cause data loss.
   347→ */
   348→function getUserEditCountFromRaw(raw: string | null): number {
   349→  if (!raw) return 0;
   350→  const data = safeJsonParse(raw);
   351→  if (!data) return 0;
   352→  const labels = data.state?.labels || data.labels || [];
   353→  return countUserEditedLabels(labels);

   358→const robustStorage: StateStorage = {
   359→  getItem: (name: string): string | null => {
   360→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   361→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   362→
   363→    const primaryData = safeJsonParse(primaryRaw);
   364→    const backupData = safeJsonParse(backupRaw);
   365→
   366→    let result: string | null = null;
   367→    let usedSidecarSnapshots = false;
   368→
   369→    // If primary is missing/corrupt, try backup
   370→    if (!primaryData) {
   371→      if (backupData) {
   372→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   373→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   374→        result = backupRaw;
   375→      }
   376→      // Both empty — first visit, result stays null
   377→    } else if (backupData) {
   378→      // Both exist — check for data loss
   379→      const primaryLabelCount = getTotalLabelCount(primaryData);
   380→      const backupLabelCount = getTotalLabelCount(backupData);
   381→      const primaryUserEdits = primaryData.state?.labels
   382→        ? countUserEditedLabels(primaryData.state.labels)
   383→        : countUserEditedLabels(primaryData.labels || []);
   384→      const backupUserEdits = backupData.state?.labels
   385→        ? countUserEditedLabels(backupData.state.labels)
   386→        : countUserEditedLabels(backupData.labels || []);
   387→
   388→      // If primary has fewer labels than backup AND backup has more or equal user data
   389→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   390→        console.warn(
   391→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   392→        );
   393→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   394→        result = backupRaw;
   395→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   396→        // If primary has fewer user edits than backup, use backup
   397→        console.warn(
   398→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   399→        );
   400→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   401→        result = backupRaw;
   402→      } else {
   403→        result = primaryRaw;
   404→      }
   405→    } else {
   406→      // Primary exists, no backup — normal case
   407→      result = primaryRaw;
   408→    }
   409→
   410→    // CRITICAL SAFETY NET for rankingSnapshots:
   411→    // If the chosen result has 0 snapshots but the dedicated SNAPSHOTS_BACKUP_KEY
   412→    // has snapshots, splice them in. This handles the scenario where the main
   413→    // store got wiped (bad cloud sync, migration bug, partial overwrite) but the
   414→    // snapshots sidecar survived. Without this, the user loses months of
   415→    // "storico caricamenti" and the Classifiche page resets to "first time".
   416→    try {
   417→      const currentSnaps = extractSnapshotsFromRaw(result);
   418→      const sidecarRaw = safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY);
   419→      const sidecarSnaps = extractSnapshotsFromRaw(sidecarRaw);
   420→      if (sidecarSnaps.length > 0 && sidecarSnaps.length > currentSnaps.length) {
   421→        console.warn(
   422→          `[LabelPulse Storage] SNAPSHOTS RECOVERY: result has ${currentSnaps.length} snapshots, sidecar has ${sidecarSnaps.length}. Merging sidecar into result.`
   423→        );
   424→        const mergedSnaps = mergeSnapshots(currentSnaps, sidecarSnaps);
   425→        const parsed = result ? safeJsonParse(result) : {};
   426→        if (parsed && typeof parsed === "object") {
   427→          if (parsed.state && typeof parsed.state === "object") {
   428→            parsed.state.rankingSnapshots = mergedSnaps;
   429→          } else {
   430→            parsed.rankingSnapshots = mergedSnaps;
   431→          }
   432→          result = JSON.stringify(parsed);
   433→          usedSidecarSnapshots = true;
   434→        }
   435→      }
   436→    } catch (e) {
   437→      console.warn("[LabelPulse Storage] Snapshots sidecar recovery failed:", e);
   438→    }
   439→
   440→    // CRITICAL SAFETY NET for userProfile:
   441→    // Same logic as snapshots. If the chosen result has an empty/seed profile
   442→    // but PROFILE_BACKUP_KEY has real data, splice it in. This is the user's
   443→    // identity card — losing artistName, bio, email, links is catastrophic.
   444→    try {
   445→      const currentProfile = extractProfileFromRaw(result);
   446→      const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   447→      if (sidecarProfile && !currentProfile) {
   448→        console.warn(
   449→          `[LabelPulse Storage] PROFILE RECOVERY: result has empty profile, sidecar has data (artistName=${!!sidecarProfile.artistName}, bio=${!!sidecarProfile.bio}). Merging sidecar into result.`
   450→        );
   451→        const parsed = result ? safeJsonParse(result) : {};
   452→        if (parsed && typeof parsed === "object") {
   453→          if (parsed.state && typeof parsed.state === "object") {
   454→            parsed.state.userProfile = sidecarProfile;
   455→          } else {
   456→            parsed.userProfile = sidecarProfile;
   457→          }
   458→          result = JSON.stringify(parsed);
   459→        }
   460→      } else if (sidecarProfile && currentProfile) {
   461→        // Both have data — merge (sidecar fills empty fields in current)
   462→        const mergedProfile = mergeProfiles(sidecarProfile, currentProfile);
   463→        const parsed = result ? safeJsonParse(result) : {};
   464→        if (parsed && typeof parsed === "object" && mergedProfile) {
   465→          if (parsed.state && typeof parsed.state === "object") {
   466→            parsed.state.userProfile = mergedProfile;
   467→          } else {
   468→            parsed.userProfile = mergedProfile;
   469→          }
   470→          result = JSON.stringify(parsed);
   471→        }
   472→      }
   473→    } catch (e) {
   474→      console.warn("[LabelPulse Storage] Profile sidecar recovery failed:", e);
   475→    }
   476→
   477→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   478→    // merged data immediately after this getItem returns, and we MUST allow that write.
   479→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   480→    _rehydrated = true;
   481→
   482→    if (usedSidecarSnapshots) {
   483→      console.info("[LabelPulse Storage] rankingSnapshots restored from sidecar backup.");
   484→    }
   485→
   486→    return result;
   487→  },
   488→
   489→  setItem: (name: string, value: string): void => {
   490→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   491→    // Before rehydration, the store only has seed data — writing it would overwrite
   492→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   493→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   494→    if (!_rehydrated) {
   495→      console.warn(
   496→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   497→      );
   498→      return;
   499→    }
   500→
   501→    // Write to primary immediately
   502→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   503→
   504→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   505→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   506→    safeLocalStorageSet(BACKUP_KEY, value);
   507→
   508→    // CRITICAL: also persist rankingSnapshots to a dedicated slot.
   509→    // This is the user's "storico caricamenti" — months of imports that cannot
   510→    // be regenerated. Keeping it in a separate key means even a catastrophic
   511→    // main-store wipe (bad cloud sync, migration bug, quota error) leaves the
   512→    // snapshots recoverable.
   513→    try {
   514→      const snaps = extractSnapshotsFromRaw(value);
   515→      if (snaps.length > 0) {
   516→        // Merge with whatever's already in the snapshots slot — NEVER overwrite
   517→        // existing snapshots with fewer snapshots (would happen if the current
   518→        // write had a partial/empty snapshot array due to some upstream bug).
   519→        const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   520→        const merged = mergeSnapshots(existing, snaps);
   521→        safeLocalStorageSet(
   522→          SNAPSHOTS_BACKUP_KEY,
   523→          JSON.stringify(merged)
   524→        );
   525→      }
   526→    } catch (e) {
   527→      console.warn("[LabelPulse Storage] Snapshots-sidecar write failed:", e);
   528→    }
   529→
   530→    // CRITICAL: also persist userProfile to a dedicated slot.
   531→    // Same pattern as snapshots. The profile is the user's identity card —
   532→    // artistName, bio, email, social links, photo. We can NEVER lose this,
   533→    // even if cloud sync wipes the main store.
   534→    try {
   535→      const profile = extractProfileFromRaw(value);
   536→      if (profile) {
   537→        const existing = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   538→        const merged = mergeProfiles(existing, profile);
   539→        if (merged) {
   540→          safeLocalStorageSet(
   541→            PROFILE_BACKUP_KEY,
   542→            JSON.stringify({ userProfile: merged })
   543→          );
   544→        }
   545→      }
   546→    } catch (e) {
   547→      console.warn("[LabelPulse Storage] Profile-sidecar write failed:", e);
   548→    }
   549→
   550→    if (!primaryOk) {
   551→      console.error("[LabelPulse Storage] Primary write failed!");
   552→    }
   553→  },
   554→
   555→  removeItem: (name: string): void => {
   556→    safeLocalStorageRemove(PRIMARY_KEY);
   557→    safeLocalStorageRemove(BACKUP_KEY);
   558→    // NOTE: do NOT remove SNAPSHOTS_BACKUP_KEY or PROFILE_BACKUP_KEY here.
   559→    // These are the user's permanent records (months of ranking history +
   560→    // artist identity) — clearing the main store should NEVER nuke them.
   561→    // The only way to clear them is an explicit user action ("reset all data"
   562→    // button) which calls safeLocalStorageRemove directly.
   563→  },
   564→};

   571→  if (typeof window === "undefined") return;

   618→/**
   619→ * Extract userProfile from a raw JSON string (main store JSON, backup JSON,
   620→ * or sidecar JSON). Returns null if missing/invalid.
   621→ */
   622→function extractProfileFromRaw(raw: string | null): any | null {
   623→  if (!raw) return null;
   624→  try {
   625→    const parsed = JSON.parse(raw);
   626→    const profile = parsed?.state?.userProfile ?? parsed?.userProfile ?? null;
   627→    if (profile && typeof profile === "object") {
   628→      // Must have at least one non-empty field to be considered "real"
   629→      const hasData =
   630→        !!profile.artistName ||
   631→        !!profile.bio ||
   632→        !!profile.email ||
   633→        !!profile.scLink ||
   634→        !!profile.photoUrl ||
   635→        (Array.isArray(profile.links) && profile.links.length > 0) ||
   636→        !!profile.cyaniteApiToken ||
   637→        !!profile.supabaseUrl;
   638→      if (hasData) return profile;
   639→    }
   640→  } catch {
   641→    /* ignore */
   642→  }
   643→  return null;
   644→}
   645→
   646→/**
   647→ * Merge two user profiles — non-empty fields win.
   648→ * This is the safety net: if the cloud ever sends a profile with empty
   649→ * artistName, but the sidecar has the real artistName, we keep the real one.
   650→ */
   651→export function mergeProfiles(base: any | null, incoming: any | null): any | null {
   652→  if (!base && !incoming) return null;
   653→  if (!base) return incoming;
   654→  if (!incoming) return base;
   655→  const merged: any = { ...base };
   656→  for (const k of Object.keys(incoming)) {
   657→    const v = incoming[k];
   658→    if (k === "links") {
   659→      // Merge link arrays (dedupe by type|value)
   660→      const a = Array.isArray(base.links) ? base.links : [];
   661→      const b = Array.isArray(v) ? v : [];
   662→      const seen = new Set<string>();
   663→      const out: { type: string; value: string }[] = [];
   664→      for (const link of [...a, ...b]) {
   665→        if (!link || typeof link !== "object") continue;
   666→        const key = `${link.type}|${link.value}`;
   667→        if (seen.has(key)) continue;
   668→        seen.add(key);
   669→        out.push(link);
   670→      }
   671→      merged.links = out;
   672→    } else if (typeof v === "string") {
   673→      // Non-empty string wins over empty
   674→      if (v && !merged[k]) merged[k] = v;
   675→    } else {
   676→      // For any other type, prefer defined over undefined
   677→      if (v !== undefined && v !== null && !merged[k]) merged[k] = v;
   678→    }
   679→  }
   680→  return merged;
   681→}

   686→export function readProfileSidecar(): any | null {

   692→ * Force-restore userProfile from the sidecar into the live store, IF the
   693→ * current live profile is empty/seed AND the sidecar has real data.
   694→ *
   695→ * Returns true if a restore happened, false otherwise.
   696→ */
   697→export function restoreProfileFromSidecar(): boolean {
   698→  if (typeof window === "undefined") return false;
   699→  const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   700→  if (!sidecarProfile) return false;
   701→
   702→  const current = useAppStore.getState().userProfile;
   703→  const currentHasData =
   704→    !!current.artistName ||
   705→    !!current.bio ||
   706→    !!current.email ||
   707→    !!current.scLink ||
   708→    !!current.photoUrl ||
   709→    (Array.isArray(current.links) && current.links.length > 0) ||
   710→    !!current.cyaniteApiToken ||
   711→    !!current.supabaseUrl;
   712→
   713→  // If current profile has data, merge (sidecar can fill empty fields)
   714→  // If current profile is empty, replace with sidecar
   715→  const merged = currentHasData
   716→    ? mergeProfiles(sidecarProfile, current)
   717→    : sidecarProfile;
   718→
   719→  if (merged && merged !== current) {
   720→    useAppStore.setState({ userProfile: merged });
   721→    console.info(
   722→      `[LabelPulse] Restored user profile from sidecar backup (had: ${currentHasData ? "partial" : "empty"}, now has: artistName=${!!merged.artistName}, bio=${!!merged.bio}, email=${!!merged.email}).`
   723→    );
   724→    return true;
   725→  }
   726→  return false;

   729→// ==================== ARTISTS SIDE CAR ====================
   730→// Like PROFILE_BACKUP_KEY but for the artists array. A full Beatport scrape
   731→// produces ~3400 artists / ~9MB. localStorage has a 5-10MB quota on most
   732→// browsers, so we TRY to write artists here but FAIL GRACEFULLY if quota
   733→// exceeded — the IndexedDB + cloud are the primary stores, this is just
   734→// an emergency backup for when both fail.
   735→//
   736→// Strategy:
   737→//  - writeArtistsSidecar: best-effort write, truncate if needed (keep first
   738→//    N artists rather than fail entirely). Skip silently if quota exceeded.
   739→//  - readArtistsSidecar: returns [] if missing/corrupt
   740→//  - restoreArtistsFromSidecar: splice into live store if local is empty
   741→const ARTISTS_SIDECAR_KEY = "labelpulse-artists-backup";
   742→const ARTISTS_SIDECAR_MAX = 200; // cap to first 200 artists (~600KB) to fit quota
   743→
   744→export function writeArtistsSidecar(artists: any[]): void {
   745→  if (typeof window === "undefined") return;
   746→  if (!Array.isArray(artists) || artists.length === 0) return;
   747→  try {
   748→    // If small enough, write all. Otherwise, write first N + a marker.
   749→    const capped = artists.length > ARTISTS_SIDECAR_MAX
   750→      ? artists.slice(0, ARTISTS_SIDECAR_MAX)
   751→      : artists;
   752→    const payload = JSON.stringify({
   753→      savedAt: new Date().toISOString(),
   754→      count: artists.length,
   755→      capped: artists.length > ARTISTS_SIDECAR_MAX,
   756→      artists: capped,
   757→    });
   758→    safeLocalStorageSet(ARTISTS_SIDECAR_KEY, payload);
   759→  } catch (e) {
   760→    // Quota exceeded — try with even fewer artists
   761→    try {
   762→      const tiny = artists.slice(0, 50);
   763→      const payload = JSON.stringify({
   764→        savedAt: new Date().toISOString(),
   765→        count: artists.length,
   766→        capped: true,
   767→        artists: tiny,
   768→      });
   769→      safeLocalStorageSet(ARTISTS_SIDECAR_KEY, payload);
   770→    } catch {
   771→      // Give up silently — IDB + cloud are the primary stores
   772→    }
   773→  }
   774→}
   775→
   776→export function readArtistsSidecar(): any[] {
   777→  if (typeof window === "undefined") return [];
   778→  const raw = safeLocalStorageGet(ARTISTS_SIDECAR_KEY);
   779→  if (!raw) return [];
   780→  try {
   781→    const parsed = JSON.parse(raw);
   782→    const arr = parsed?.artists;
   783→    return Array.isArray(arr) ? arr : [];
   784→  } catch {
   785→    return [];
   786→  }
   787→}

   789→/**
   790→ * Restore artists from sidecar IF the live store has 0 artists.
   791→ * Returns the number of artists restored (0 if nothing happened).
   792→ * Does NOT replace live artists if live already has data — sidecar is
   793→ * only an emergency recovery, not a primary source.
   794→ */
   795→export function restoreArtistsFromSidecar(): number {
   796→  if (typeof window === "undefined") return 0;
   797→  const sidecar = readArtistsSidecar();
   798→  if (sidecar.length === 0) return 0;
   799→
   800→  const current = useAppStore.getState().artists || [];
   801→  if (current.length > 0) return 0;

   806→  );
   807→  // Also persist to IDB so future boots are fast
   808→  saveArtistsToIDB(sidecar).catch(() => {});
   809→  return sidecar.length;

   943→/**
   944→ * Merge incoming artists (from a fresh scrape) with existing artists
   945→ * (already in the store from previous scrapes).
   946→ *
   947→ * Strategy:
   948→ * - Key by artist.id (e.g. "bp_6824"). Same Beatport id = same artist.
   949→ * - For each artist that exists: union genres, union labelsPublishedOn,
   950→ *   MERGE tracksByGenre by track.id (preserve tracks from older scrapes,
   951→ *   update with new position info), sum totalPoints, min bestPosition.
   952→ * - firstSeenAt preserved from existing; lastSeenAt = now.
   953→ * - Trending recomputed fresh from current data.
   954→ * - New artists: appended with firstSeenAt = lastSeenAt = now.
   955→ *
   956→ * This is what makes historical data accumulate: each scrape adds new
   957→ * tracks to artists' tracksByGenre without losing previously-seen tracks.
   958→ */
   959→function mergeArtists(existing: Artist[], incoming: Artist[], now: string): Artist[] {
   960→  const map = new Map<string, Artist>();
   961→
   962→  // Seed with existing
   963→  for (const a of existing) {
   964→    map.set(a.id, { ...a });
   965→  }
   966→
   967→  for (const inc of incoming) {
   968→    const ex = map.get(inc.id);
   969→    if (!ex) {
   970→      // New artist — first time we see them
   971→      map.set(inc.id, {
   972→        ...inc,
   973→        firstSeenAt: now,
   974→        lastSeenAt: now,
   975→      });
   976→      continue;
   977→    }
   978→
   979→    // Existing artist — merge fields
   980→    // Union genres
   981→    const genres = Array.from(new Set([...(ex.genres || []), ...(inc.genres || [])]));
   982→
   983→    // Union labelsPublishedOn
   984→    const labelsPublishedOn = Array.from(
   985→      new Set([...(ex.labelsPublishedOn || []), ...(inc.labelsPublishedOn || [])])
   986→    );
   987→
   988→    // Merge tracksByGenre — preserve old tracks, add/update from incoming
   989→    const tracksByGenre: Record<string, ArtistTrack[]> = { ...(ex.tracksByGenre || {}) };
   990→    for (const genre of Object.keys(inc.tracksByGenre || {})) {
   991→      const incTracks = inc.tracksByGenre[genre] || [];
   992→      const exTracks = tracksByGenre[genre] || [];
   993→      const byId = new Map<number, ArtistTrack>();
   994→      // Seed with existing
   995→      for (const t of exTracks) byId.set(t.id, t);
   996→      // Update/add with incoming (incoming overrides — fresh positions)
   997→      for (const t of incTracks) byId.set(t.id, t);
   998→      tracksByGenre[genre] = Array.from(byId.values()).sort((a, b) => a.position - b.position);
   999→    }
  1000→
  1001→    // Recompute totalPoints & bestPosition from current incoming data
  1002→    // (the existing totals were from older scrapes — we want the snapshot
  1003→    // from THIS import to drive the points, otherwise the totals just
  1004→    // accumulate forever and "points" loses meaning).
  1005→    // BUT: keep max totalPoints seen (artists who peaked before should
  1006→    // keep credit). Actually no — let's use the LATEST totalPoints,
  1007→    // since that's the current ranking signal. Old tracks are still in
  1008→    // tracksByGenre for browsing, but the points reflect "right now".
  1009→    const totalPoints = inc.totalPoints;
  1010→    const bestPosition = Math.min(ex.bestPosition, inc.bestPosition);
  1011→
  1012→    // Trending: use incoming (fresh computation)
  1013→    const trending = inc.trending;
  1014→    const trendingRankByGenre = inc.trendingRankByGenre;
  1015→    const trendingPointsByGenre = inc.trendingPointsByGenre;
  1016→
  1017→    // isRemixerOnly: false if EITHER snapshot has them as primary
  1018→    const isRemixerOnly = ex.isRemixerOnly && inc.isRemixerOnly;
  1019→
  1020→    map.set(inc.id, {
  1021→      ...ex,
  1022→      ...inc,
  1023→      genres,
  1024→      labelsPublishedOn,
  1025→      tracksByGenre,
  1026→      totalPoints,
  1027→      bestPosition,
  1028→      trending,
  1029→      trendingRankByGenre,
  1030→      trendingPointsByGenre,
  1031→      isRemixerOnly,
  1032→      firstSeenAt: ex.firstSeenAt || now,
  1033→      lastSeenAt: now,
  1034→    });
  1035→  }
  1036→
  1037→  return Array.from(map.values());
  1038→}

  1040→/**
  1041→ * Merge helper: preserves ALL user-editable fields from existing label
  1042→ * when imported label has empty/default values for those fields.
  1043→ */
  1044→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
  1045→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
  1046→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
  1047→  //
  1048→  // IDEMPOTENT IMPORT FIX: previously, every import — even an identical re-import —
  1049→  // would clobber prevRankByGenre with a copy of the current rankByGenre. After
  1050→  // an identical re-import, prevRank == currentRank for every genre, so all
  1051→  // movements became 0 and Spotlight risers disappeared. This broke the UX:
  1052→  // "if I re-import the same scrape, I should see what I saw before".
  1053→  //
  1054→  // The fix: only snapshot the existing rank as "previous" for genres where the
  1055→  // rank has actually CHANGED. For genres where imported rank === existing rank,
  1056→  // preserve the existing prevRankByGenre[genre] so users keep seeing the risers
  1057→  // they had before the (identical) re-import.
  1058→  const importedRank = imported.rankByGenre || {};
  1059→  const existingRank = existing.rankByGenre || {};
  1060→  const existingPrev = existing.prevRankByGenre || {};
  1061→  const hasNewRankData = Object.keys(importedRank).length > 0;
  1062→
  1063→  let newPrevRankByGenre: Record<string, number>;
  1064→  if (hasNewRankData) {
  1065→    // Start from the existing prev snapshot — we only overwrite entries for
  1066→    // genres where the rank actually changed.
  1067→    newPrevRankByGenre = { ...existingPrev };
  1068→    for (const [genre, newRank] of Object.entries(importedRank)) {
  1069→      const oldRank = existingRank[genre];
  1070→      if (oldRank !== undefined && oldRank !== newRank) {

  1134→/**
  1135→ * Repair corrupted label data.
  1136→ * Detects and fixes: same email appearing on too many labels (likely corruption).
  1137→ */
  1138→function repairLabelData(labels: Label[]): Label[] {
  1139→  const emailLabelCount = new Map<string, number>();
  1140→  for (const l of labels) {
  1141→    if (l.emails && Array.isArray(l.emails)) {
  1142→      for (const e of l.emails) {
  1143→        if (e && e.trim()) {
  1144→          const key = e.toLowerCase().trim();
  1145→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
  1146→        }
  1147→      }
  1148→    }
  1149→  }
  1150→
  1151→  const corruptedEmails = new Set<string>();
  1152→  for (const [email, count] of emailLabelCount) {
  1153→    if (count > 5) {
  1154→      corruptedEmails.add(email);
  1155→      console.warn(
  1156→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
  1157→      );
  1158→    }
  1159→  }
  1160→
  1161→  if (corruptedEmails.size === 0) {
  1162→    return labels;
  1163→  }
  1164→
  1165→  const emailOwner = new Map<string, string>();
  1166→  for (const email of corruptedEmails) {
  1167→    for (const l of labels) {
  1168→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
  1169→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
  1170→        if (domain && l.name.toLowerCase().includes(domain)) {
  1171→          emailOwner.set(email, l.id);
  1172→          break;
  1173→        }
  1174→      }
  1175→    }
  1176→    if (!emailOwner.has(email)) {
  1177→      for (const l of labels) {
  1178→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
  1179→          emailOwner.set(email, l.id);
  1180→          break;
  1181→        }
  1182→      }
  1183→    }
  1184→  }
  1185→
  1186→  return labels.map(l => {
  1187→    if (!l.emails || !Array.isArray(l.emails)) return l;
  1188→    const cleanedEmails = l.emails.filter(e => {
  1189→      const key = e?.toLowerCase().trim();
  1190→      if (!key) return false;
  1191→      if (!corruptedEmails.has(key)) return true;
  1192→      return emailOwner.get(key) === l.id;
  1193→    });
  1194→    const cleanedContactInfo = cleanedEmails[0] || "";
  1195→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
  1196→      console.log(
  1197→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
  1198→      );
  1199→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
  1200→    }
  1201→    return l;
  1202→  });
  1203→}
