   100→   100→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   101→   101→// Before rehydration, the store contains seed data which would overwrite user data.
   102→   102→let _rehydrated = false;

   104→   104→function safeJsonParse(str: string | null): any | null {
   105→   105→  if (!str) return null;
   106→   106→  try {
   107→   107→    return JSON.parse(str);
   108→   108→  } catch {
   109→   109→    return null;
   110→   110→  }
   111→   111→}
   112→   112→
   113→   113→function safeLocalStorageGet(key: string): string | null {
   114→   114→  try {
   115→   115→    return localStorage.getItem(key);
   116→   116→  } catch {
   117→   117→    return null;
   118→   118→  }
   119→   119→}
   120→   120→
   121→   121→function safeLocalStorageSet(key: string, value: string): boolean {
   122→   122→  try {
   123→   123→    localStorage.setItem(key, value);
   124→   124→    return true;
   125→   125→  } catch (e) {
   126→   126→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   127→   127→    return false;
   128→   128→  }
   129→   129→}
   130→   130→
   131→   131→function safeLocalStorageRemove(key: string): void {
   132→   132→  try {
   133→   133→    localStorage.removeItem(key);
   134→   134→  } catch {
   135→   135→    // Ignore
   136→   136→  }
   137→   137→}

   143→   143→function countUserEditedLabels(labels: any[]): number {
   144→   144→  if (!Array.isArray(labels)) return 0;
   145→   145→  return labels.filter((l) =>
   146→   146→    (l.emails && l.emails.length > 0) ||
   147→   147→    (l.notes && l.notes.trim() !== "") ||
   148→   148→    (l.website && l.website.trim() !== "") ||
   149→   149→    (l.demoLink && l.demoLink.trim() !== "") ||
   150→   150→    (l.socialLink && l.socialLink.trim() !== "") ||
   151→   151→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   152→   152→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   153→   153→    l.isCustom === true
   154→   154→  ).length;
   155→   155→}
   156→   156→
   157→   157→function getTotalLabelCount(data: any): number {
   158→   158→  if (!data) return 0;
   159→   159→  if (Array.isArray(data.labels)) return data.labels.length;
   160→   160→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   161→   161→  return 0;
   162→   162→}
   163→   163→
   164→   164→/**
   165→   165→ * Extract user-edited label count from raw localStorage string.
   166→   166→ * Used by the write guard to check if a write would cause data loss.
   167→   167→ */
   168→   168→function getUserEditCountFromRaw(raw: string | null): number {
   169→   169→  if (!raw) return 0;
   170→   170→  const data = safeJsonParse(raw);
   171→   171→  if (!data) return 0;
   172→   172→  const labels = data.state?.labels || data.labels || [];
   173→   173→  return countUserEditedLabels(labels);
   174→   174→}

   176→   176→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   177→   177→
   178→   178→const robustStorage: StateStorage = {
   179→   179→  getItem: (name: string): string | null => {
   180→   180→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   181→   181→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   182→   182→
   183→   183→    const primaryData = safeJsonParse(primaryRaw);
   184→   184→    const backupData = safeJsonParse(backupRaw);
   185→   185→
   186→   186→    let result: string | null = null;
   187→   187→
   188→   188→    // If primary is missing/corrupt, try backup
   189→   189→    if (!primaryData) {
   190→   190→      if (backupData) {
   191→   191→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   192→   192→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   193→   193→        result = backupRaw;
   194→   194→      }
   195→   195→      // Both empty — first visit, result stays null
   196→   196→    } else if (backupData) {
   197→   197→      // Both exist — check for data loss
   198→   198→      const primaryLabelCount = getTotalLabelCount(primaryData);
   199→   199→      const backupLabelCount = getTotalLabelCount(backupData);
   200→   200→      const primaryUserEdits = primaryData.state?.labels
   201→   201→        ? countUserEditedLabels(primaryData.state.labels)
   202→   202→        : countUserEditedLabels(primaryData.labels || []);
   203→   203→      const backupUserEdits = backupData.state?.labels
   204→   204→        ? countUserEditedLabels(backupData.state.labels)
   205→   205→        : countUserEditedLabels(backupData.labels || []);
   206→   206→
   207→   207→      // If primary has fewer labels than backup AND backup has more or equal user data
   208→   208→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   209→   209→        console.warn(
   210→   210→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   211→   211→        );
   212→   212→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   213→   213→        result = backupRaw;
   214→   214→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   215→   215→        // If primary has fewer user edits than backup, use backup
   216→   216→        console.warn(
   217→   217→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   218→   218→        );
   219→   219→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   220→   220→        result = backupRaw;
   221→   221→      } else {
   222→   222→        result = primaryRaw;
   223→   223→      }
   224→   224→    } else {
   225→   225→      // Primary exists, no backup — normal case
   226→   226→      result = primaryRaw;
   227→   227→    }
   228→   228→
   229→   229→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   230→   230→    // merged data immediately after this getItem returns, and we MUST allow that write.
   231→   231→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   232→   232→    _rehydrated = true;
   233→   233→
   234→   234→    return result;
   235→   235→  },
   236→   236→
   237→   237→  setItem: (name: string, value: string): void => {
   238→   238→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   239→   239→    // Before rehydration, the store only has seed data — writing it would overwrite
   240→   240→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   241→   241→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   242→   242→    if (!_rehydrated) {
   243→   243→      console.warn(
   244→   244→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   245→   245→      );
   246→   246→      return;
   247→   247→    }
   248→   248→
   249→   249→    // Write to primary immediately
   250→   250→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   251→   251→
   252→   252→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   253→   253→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   254→   254→    safeLocalStorageSet(BACKUP_KEY, value);
   255→   255→
   256→   256→    if (!primaryOk) {
   257→   257→      console.error("[LabelPulse Storage] Primary write failed!");
   258→   258→    }
   259→   259→  },
   260→   260→
   261→   261→  removeItem: (name: string): void => {
   262→   262→    safeLocalStorageRemove(PRIMARY_KEY);
   263→   263→    safeLocalStorageRemove(BACKUP_KEY);
   264→   264→  },
   265→   265→};
   266→   266→
   267→   267→/**
   268→   268→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   269→   269→ * Since backup is now written on every setItem, this is a safety net.
   270→   270→ */
   271→   271→export function forceBackupNow(): void {
   272→   272→  if (typeof window === "undefined") return;
   273→   273→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   274→   274→  if (primaryRaw) {
   275→   275→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   276→   276→  }
   277→   277→}

   391→   391→/**
   392→   392→ * Repair corrupted label data.
   393→   393→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   394→   394→ */
   395→   395→function repairLabelData(labels: Label[]): Label[] {
   396→   396→  const emailLabelCount = new Map<string, number>();
   397→   397→  for (const l of labels) {
   398→   398→    if (l.emails && Array.isArray(l.emails)) {
   399→   399→      for (const e of l.emails) {
   400→   400→        if (e && e.trim()) {
   401→   401→          const key = e.toLowerCase().trim();
   402→   402→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   403→   403→        }
   404→   404→      }
   405→   405→    }
   406→   406→  }
   407→   407→
   408→   408→  const corruptedEmails = new Set<string>();
   409→   409→  for (const [email, count] of emailLabelCount) {
   410→   410→    if (count > 5) {
   411→   411→      corruptedEmails.add(email);
   412→   412→      console.warn(
   413→   413→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   414→   414→      );
   415→   415→    }
   416→   416→  }
   417→   417→
   418→   418→  if (corruptedEmails.size === 0) {
   419→   419→    return labels;
   420→   420→  }
   421→   421→
   422→   422→  const emailOwner = new Map<string, string>();
   423→   423→  for (const email of corruptedEmails) {
   424→   424→    for (const l of labels) {
   425→   425→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   426→   426→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   427→   427→        if (domain && l.name.toLowerCase().includes(domain)) {
   428→   428→          emailOwner.set(email, l.id);
   429→   429→          break;
   430→   430→        }
   431→   431→      }
   432→   432→    }
   433→   433→    if (!emailOwner.has(email)) {
   434→   434→      for (const l of labels) {
   435→   435→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   436→   436→          emailOwner.set(email, l.id);
   437→   437→          break;
   438→   438→        }
   439→   439→      }
   440→   440→    }
   441→   441→  }
   442→   442→
   443→   443→  return labels.map(l => {
   444→   444→    if (!l.emails || !Array.isArray(l.emails)) return l;
   445→   445→    const cleanedEmails = l.emails.filter(e => {
   446→   446→      const key = e?.toLowerCase().trim();
   447→   447→      if (!key) return false;
   448→   448→      if (!corruptedEmails.has(key)) return true;
   449→   449→      return emailOwner.get(key) === l.id;
   450→   450→    });
   451→   451→    const cleanedContactInfo = cleanedEmails[0] || "";
   452→   452→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   453→   453→      console.log(
   454→   454→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   455→   455→      );
   456→   456→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   457→   457→    }
   458→   458→    return l;
   459→   459→  });
   460→   460→}

   883→   883→  label: Label,
   884→   884→  genre?: string
   885→   885→): "top" | "mid" | "emerging" | null {
   886→   886→  if (!genre) genre = label.genres?.[0];
   887→   887→  if (!genre) return null;
   888→   888→  const rank = label.rankByGenre?.[genre];
   889→   889→  if (!rank) return label.trending ? "emerging" : null;
   890→   890→  if (rank <= 20) return "top";
   891→   891→  if (rank <= 50) return "mid";
   892→   892→  return label.trending ? "emerging" : null;

   896→   896→// Sistema di sincronizzazione con Supabase.

   903→   903→
   904→   904→/**
   905→   905→ * Sincronizza lo stato corrente con Supabase (debounced).
   906→   906→ * Chiamata dopo ogni azione utente che modifica i dati.
   907→   907→ */
   908→   908→export function syncToCloud(): void {
   909→   909→  if (!isSupabaseConfigured) return;
   910→   910→
   911→   911→  if (_cloudSyncTimer) {
   912→   912→    clearTimeout(_cloudSyncTimer);
   913→   913→  }
   914→   914→
   915→   915→  _cloudSyncTimer = setTimeout(async () => {
   916→   916→    _cloudSyncTimer = null;
   917→   917→    const state = useAppStore.getState();
   918→   918→    const dataToSync = {
   919→   919→      labels: state.labels,
   920→   920→      demos: state.demos,
   921→   921→      activeTab: state.activeTab,
   922→   922→      locale: state.locale,
   923→   923→      userProfile: state.userProfile,
   924→   924→      gmailAuth: state.gmailAuth,
   925→   925→      rankingsUpdatedAt: state.rankingsUpdatedAt,
   926→   926→      lastSavedAt: state.lastSavedAt,
   927→   927→    };
   928→   928→    await saveStateToCloud(dataToSync);
   929→   929→  }, CLOUD_SYNC_DEBOUNCE_MS);
   930→   930→}

   958→   958→/**
   959→   959→ * Carica i dati dal cloud e aggiorna lo store se i dati cloud sono più recenti.
   960→   960→ * Chiamata una volta all'avvio dell'app, dopo la reidratazione da localStorage.
   961→   961→ */
   962→   962→export async function loadFromCloud(): Promise<void> {
   963→   963→  if (!isSupabaseConfigured) {
   964→   964→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
   965→   965→    useAppStore.setState({ hasCloudSynced: true });
   966→   966→    return;
   967→   967→  }
   968→   968→
   969→   969→  try {
   970→   970→    const cloudData = await loadStateFromCloud();
   971→   971→    if (!cloudData) {
   972→   972→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
   973→   973→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
   974→   974→      await forceCloudSync();
   975→   975→      useAppStore.setState({ hasCloudSynced: true });
   976→   976→      return;
   977→   977→    }
   978→   978→
   979→   979→    const cloud = cloudData as any;
   980→   980→    const localState = useAppStore.getState();
   981→   981→
   982→   982→    // Confronta i timestamp per decidere quali dati usare
   983→   983→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
   984→   984→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
   985→   985→
   986→   986→    if (cloudSavedAt > localSavedAt) {
   987→   987→      // I dati cloud sono più recenti — aggiorna lo store locale
   988→   988→      console.log(
   989→   989→        `[LabelPulse Cloud] Cloud data is newer (cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}). Updating local store.`
   990→   990→      );
   991→   991→
   992→   992→      // Usa la stessa logica di merge della reidratazione
   993→   993→      const merged = mergeCloudData(cloud, localState);
   994→   994→      useAppStore.setState({
   995→   995→        ...merged,
   996→   996→        hasCloudSynced: true,
   997→   997→      });
   998→   998→    } else {
   999→   999→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1000→  1000→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1001→  1001→      useAppStore.setState({ hasCloudSynced: true });
  1002→  1002→    }
  1003→  1003→  } catch (err) {
  1004→  1004→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1005→  1005→    useAppStore.setState({ hasCloudSynced: true });
  1006→  1006→  }
  1007→  1007→}

  1009→  1009→/**
  1010→  1010→ * Merge dei dati cloud con quelli locali, preservando i dati utente.
  1011→  1011→ * Usa la stessa logica della funzione merge di Zustand persist.
  1012→  1012→ */
  1013→  1013→function mergeCloudData(cloudData: any, localState: any): Partial<AppState> {
  1014→  1014→  const merged: Partial<AppState> = {};
  1015→  1015→
  1016→  1016→  // Labels: applica defaults e merge con seed data, come nella reidratazione
  1017→  1017→  if (cloudData.labels && Array.isArray(cloudData.labels) && cloudData.labels.length > 0) {
  1018→  1018→    merged.labels = cloudData.labels.map((l: any) => ({
  1019→  1019→      ...LABEL_DEFAULTS,
  1020→  1020→      ...l,
  1021→  1021→    }));
  1022→  1022→
  1023→  1023→    // Aggiungi seed labels mancanti
  1024→  1024→    const cloudIds = new Set(cloudData.labels.map((l: any) => l.id));
  1025→  1025→    const cloudNames = new Set(
  1026→  1026→      cloudData.labels.map((l: any) => l.name?.toLowerCase().trim()).filter(Boolean)
  1027→  1027→    );
  1028→  1028→    const seedLabels = buildLabelsFromData();
  1029→  1029→    for (const seed of seedLabels) {
  1030→  1030→      if (!cloudIds.has(seed.id) && !cloudNames.has(seed.name.toLowerCase().trim())) {
  1031→  1031→        (merged.labels as Label[]).push(seed);
  1032→  1032→      }
  1033→  1033→    }
  1034→  1034→
  1035→  1035→    // Repair corrupted data
  1036→  1036→    merged.labels = repairLabelData(merged.labels);
  1037→  1037→  } else {
  1038→  1038→    merged.labels = localState.labels;
  1039→  1039→  }
  1040→  1040→
  1041→  1041→  // Demos
  1042→  1042→  merged.demos = Array.isArray(cloudData.demos) ? cloudData.demos : localState.demos;
  1043→  1043→
  1044→  1044→  // Simple fields
  1045→  1045→  merged.activeTab = cloudData.activeTab || localState.activeTab;
  1046→  1046→  merged.locale = cloudData.locale || localState.locale;
  1047→  1047→  merged.userProfile = cloudData.userProfile || localState.userProfile;
  1048→  1048→  merged.gmailAuth = cloudData.gmailAuth || localState.gmailAuth;
  1049→  1049→  merged.rankingsUpdatedAt = cloudData.rankingsUpdatedAt ?? localState.rankingsUpdatedAt;
  1050→  1050→  merged.lastSavedAt = cloudData.lastSavedAt ?? localState.lastSavedAt;
  1051→  1051→
  1052→  1052→  return merged;
  1053→  1053→}
