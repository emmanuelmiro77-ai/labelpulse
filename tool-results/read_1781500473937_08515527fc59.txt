    98→    98→function getPeriodCutoff(period: RankingTimePeriod): number {
    99→    99→  const now = Date.now();
   100→   100→  switch (period) {
   101→   101→    case "current": return now; // Only current data
   102→   102→    case "1m": return now - 30 * 24 * 60 * 60 * 1000;
   103→   103→    case "3m": return now - 90 * 24 * 60 * 60 * 1000;
   104→   104→    case "1y": return now - 365 * 24 * 60 * 60 * 1000;
   105→   105→    case "all": return 0; // All time
   106→   106→  }
   107→   107→}

   109→   109→// ==================== AGGREGATION LOGIC ====================
   110→   110→
   111→   111→/**
   112→   112→ * Aggregate ranking snapshots for a given genre and time period.
   113→   113→ * Returns a map of labelName -> { totalPoints, bestRank, snapshotCount, averageRank }
   114→   114→ */
   115→   115→function aggregateSnapshots(
   116→   116→  snapshots: RankingSnapshot[],
   117→   117→  genre: string,
   118→   118→  period: RankingTimePeriod
   119→   119→): Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number }> {
   120→   120→  const cutoff = getPeriodCutoff(period);
   121→   121→  const result = new Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }>();
   122→   122→
   123→   123→  for (const snapshot of snapshots) {
   124→   124→    if (new Date(snapshot.timestamp).getTime() < cutoff) continue;
   125→   125→    if (period === "current") continue; // Current period doesn't use snapshots
   126→   126→
   127→   127→    const genreData = snapshot.genres[genre];
   128→   128→    if (!genreData) continue;
   129→   129→
   130→   130→    for (const [labelName, data] of Object.entries(genreData)) {
   131→   131→      const existing = result.get(labelName);
   132→   132→      if (existing) {
   133→   133→        existing.totalPoints += data.points;
   134→   134→        existing.rankSum += data.rank;
   135→   135→        existing.snapshotCount++;
   136→   136→        if (data.rank < existing.bestRank) existing.bestRank = data.rank;
   137→   137→      } else {
   138→   138→        result.set(labelName, {
   139→   139→          totalPoints: data.points,
   140→   140→          bestRank: data.rank,
   141→   141→          snapshotCount: 1,
   142→   142→          rankSum: data.rank,
   143→   143→        });
   144→   144→      }
   145→   145→    }
   146→   146→  }
   147→   147→
   148→   148→  // Calculate average rank
   149→   149→  for (const [, val] of result) {
   150→   150→    (val as any).averageRank = Math.round(val.rankSum / val.snapshotCount);
   151→   151→  }
   152→   152→
   153→   153→  return result;
   154→   154→}

   156→   156→/**
   157→   157→ * Build ranked list for a genre, considering the time period.
   158→   158→ * - "current": uses live label data (rankByGenre)
   159→   159→ * - Other periods: aggregates snapshots and re-ranks by cumulative points
   160→   160→ */
   161→   161→function buildRankedList(
   162→   162→  labels: Label[],
   163→   163→  snapshots: RankingSnapshot[],
   164→   164→  genre: string,
   165→   165→  period: RankingTimePeriod
   166→   166→): RankedLabel[] {
   167→   167→  if (period === "current") {
   168→   168→    // Current view: use live label data
   169→   169→    const ranked: RankedLabel[] = [];
   170→   170→    for (const label of labels) {
   171→   171→      const rank = label.rankByGenre?.[genre];
   172→   172→      if (rank === undefined || rank === null) continue;
   173→   173→
   174→   174→      const prevRank = label.prevRankByGenre?.[genre] ?? null;
   175→   175→      const points = label.pointsByGenre?.[genre] ?? 0;
   176→   176→      let movement: number | null = null;
   177→   177→
   178→   178→      if (prevRank !== null && prevRank !== undefined) {
   179→   179→        movement = prevRank - rank;
   180→   180→      } else if (Object.keys(label.prevRankByGenre || {}).length > 0) {
   181→   181→        movement = null;
   182→   182→      }
   183→   183→
   184→   184→      ranked.push({ label, rank, prevRank, points, movement, snapshotCount: 1 });
   185→   185→    }
   186→   186→    return ranked;
   187→   187→  }
   188→   188→
   189→   189→  // Historical/cumulative view: aggregate snapshots
   190→   190→  const aggregated = aggregateSnapshots(snapshots, genre, period);
   191→   191→  if (aggregated.size === 0) return [];
   192→   192→
   193→   193→  // Sort by total points descending to determine cumulative rank
   194→   194→  const sorted = Array.from(aggregated.entries()).sort((a, b) => b[1].totalPoints - a[1].totalPoints);
   195→   195→
   196→   196→  // For movement in cumulative view, compare with the current live data
   197→   197→  const ranked: RankedLabel[] = [];
   198→   198→  for (let i = 0; i < sorted.length; i++) {
   199→   199→    const [labelName, data] = sorted[i];
   200→   200→    const cumulativeRank = i + 1;
   201→   201→
   202→   202→    // Find matching label in current data
   203→   203→    const matchingLabel = labels.find(l => l.name === labelName);
   204→   204→    const currentRank = matchingLabel?.rankByGenre?.[genre] ?? null;
   205→   205→
   206→   206→    // Movement: compare current live rank with cumulative rank
   207→   207→    let movement: number | null = null;
   208→   208→    if (currentRank !== null) {
   209→   209→      movement = currentRank - cumulativeRank; // positive = label is better live than cumulative (hot right now)
   210→   210→    }
   211→   211→
   212→   212→    // Use the label if found, otherwise create a minimal reference
   213→   213→    const label = matchingLabel || {
   214→   214→      id: `snapshot_${labelName.toLowerCase().replace(/[^a-z0-9]+/g, '_')}`,
   215→   215→      name: labelName,
   216→   216→      genre: genre,
   217→   217→      submissionType: "email" as const,
   218→   218→      contactInfo: "",
   219→   219→      status: "open" as const,
   220→   220→      notes: "",
   221→   221→      createdAt: new Date().toISOString(),
   222→   222→      emails: [],
   223→   223→      website: "",
   224→   224→      demoLink: "",
   225→   225→      socialLink: "",
   226→   226→      soundcloudLink: "",
   227→   227→      genres: [genre],
   228→   228→      rankByGenre: { [genre]: currentRank || 999 },
   229→   229→      pointsByGenre: {},
   230→   230→      trending: false,
   231→   231→      trendingRankByGenre: {},
   232→   232→      trendingPointsByGenre: {},
   233→   233→      prevRankByGenre: {},
   234→   234→    };
   235→   235→
   236→   236→    ranked.push({
   237→   237→      label,
   238→   238→      rank: cumulativeRank,
   239→   239→      prevRank: currentRank,
   240→   240→      points: data.totalPoints,
   241→   241→      movement,
   242→   242→      snapshotCount: data.snapshotCount,
   243→   243→    });
   244→   244→  }
   245→   245→
   246→   246→  return ranked;
   247→   247→}

   286→   286→        result = result.filter((l) => l.movement !== null && l.movement > 0);
   287→   287→        break;
   288→   288→      case "falling":
   289→   289→        result = result.filter((l) => l.movement !== null && l.movement < 0);
   290→   290→        break;
   291→   291→      case "new":
   292→   292→        result = result.filter((l) => l.movement === null);
   293→   293→        break;
   294→   294→      case "stable":
   295→   295→        result = result.filter((l) => l.movement === 0);
   296→   296→        break;
   297→   297→    }
   298→   298→
   299→   299→    // Sort
   300→   300→    switch (sortMode) {
   301→   301→      case "rank":
   302→   302→        result.sort((a, b) => a.rank - b.rank);
   303→   303→        break;
   304→   304→      case "movement":
   305→   305→        result.sort((a, b) => {
   306→   306→          if (a.movement === null && b.movement === null) return a.rank - b.rank;
   307→   307→          if (a.movement === null) return -1;
   308→   308→          if (b.movement === null) return 1;
   309→   309→          return b.movement - a.movement;
   310→   310→        });
   311→   311→        break;
   312→   312→      case "points":
   313→   313→        result.sort((a, b) => b.points - a.points);
   314→   314→        break;
   315→   315→    }
   316→   316→
