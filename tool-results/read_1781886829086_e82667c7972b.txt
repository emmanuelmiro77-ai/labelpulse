   321→function getDaysSince(dateStr: string | null): number | null {
   322→  if (!dateStr) return null;
   323→  const diff = Date.now() - new Date(dateStr).getTime();
   324→  return Math.floor(diff / (1000 * 60 * 60 * 24));
   325→}

   327→function formatDate(dateStr: string | null, locale: string): string {
   328→  if (!dateStr) return "";
   329→  try {
   330→    return new Date(dateStr).toLocaleDateString(locale === "it" ? "it-IT" : locale === "en" ? "en-US" : locale === "es" ? "es-ES" : locale === "fr" ? "fr-FR" : locale === "de" ? "de-DE" : "pt-BR", {
   331→      day: "numeric",
   332→      month: "long",
   333→      year: "numeric",
   334→    });
   335→  } catch {
   336→    return dateStr.split("T")[0];
   337→  }
