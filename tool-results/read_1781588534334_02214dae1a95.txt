   113→   113→// ==================== ROBUST STORAGE ====================
   114→   114→// Custom storage with data protection:
   115→   115→// 1. Primary localStorage key
   116→   116→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)

   119→   119→
   120→   120→const PRIMARY_KEY = "labelpulse-storage";
   121→   121→const BACKUP_KEY = "labelpulse-storage-backup";
   122→   122→const SEED_LABEL_COUNT = labelData.labels.length;
   123→   123→
   124→   124→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   125→   125→// Before rehydration, the store contains seed data which would overwrite user data.
   126→   126→let _rehydrated = false;
   127→   127→
   128→   128→function safeJsonParse(str: string | null): any | null {
   129→   129→  if (!str) return null;
   130→   130→  try {
   131→   131→    return JSON.parse(str);
   132→   132→  } catch {
   133→   133→    return null;
   134→   134→  }
   135→   135→}
   136→   136→
   137→   137→function safeLocalStorageGet(key: string): string | null {
   138→   138→  try {
   139→   139→    return localStorage.getItem(key);
   140→   140→  } catch {
   141→   141→    return null;
   142→   142→  }
   143→   143→}
   144→   144→
   145→   145→function safeLocalStorageSet(key: string, value: string): boolean {
   146→   146→  try {
   147→   147→    localStorage.setItem(key, value);
   148→   148→    return true;
   149→   149→  } catch (e) {
   150→   150→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   151→   151→    return false;
   152→   152→  }
   153→   153→}
   154→   154→
   155→   155→function safeLocalStorageRemove(key: string): void {
   156→   156→  try {
   157→   157→    localStorage.removeItem(key);
   158→   158→  } catch {
   159→   159→    // Ignore
   160→   160→  }
   161→   161→}

   167→   167→function countUserEditedLabels(labels: any[]): number {
   168→   168→  if (!Array.isArray(labels)) return 0;
   169→   169→  return labels.filter((l) =>
   170→   170→    (l.emails && l.emails.length > 0) ||
   171→   171→    (l.notes && l.notes.trim() !== "") ||
   172→   172→    (l.website && l.website.trim() !== "") ||
   173→   173→    (l.demoLink && l.demoLink.trim() !== "") ||
   174→   174→    (l.socialLink && l.socialLink.trim() !== "") ||

   176→   176→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   177→   177→    l.isCustom === true
   178→   178→  ).length;
   179→   179→}

   181→   181→function getTotalLabelCount(data: any): number {
   182→   182→  if (!data) return 0;
   183→   183→  if (Array.isArray(data.labels)) return data.labels.length;
   184→   184→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   185→   185→  return 0;
   186→   186→}
   187→   187→
   188→   188→/**
   189→   189→ * Extract user-edited label count from raw localStorage string.
   190→   190→ * Used by the write guard to check if a write would cause data loss.
   191→   191→ */
   192→   192→function getUserEditCountFromRaw(raw: string | null): number {
   193→   193→  if (!raw) return 0;
   194→   194→  const data = safeJsonParse(raw);
   195→   195→  if (!data) return 0;
   196→   196→  const labels = data.state?.labels || data.labels || [];
   197→   197→  return countUserEditedLabels(labels);
   198→   198→}

   200→   200→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   201→   201→
   202→   202→const robustStorage: StateStorage = {
   203→   203→  getItem: (name: string): string | null => {
   204→   204→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   205→   205→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   206→   206→
   207→   207→    const primaryData = safeJsonParse(primaryRaw);
   208→   208→    const backupData = safeJsonParse(backupRaw);
   209→   209→
   210→   210→    let result: string | null = null;
   211→   211→
   212→   212→    // If primary is missing/corrupt, try backup
   213→   213→    if (!primaryData) {
   214→   214→      if (backupData) {
   215→   215→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   216→   216→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   217→   217→        result = backupRaw;
   218→   218→      }
   219→   219→      // Both empty — first visit, result stays null
   220→   220→    } else if (backupData) {
   221→   221→      // Both exist — check for data loss
   222→   222→      const primaryLabelCount = getTotalLabelCount(primaryData);
   223→   223→      const backupLabelCount = getTotalLabelCount(backupData);
   224→   224→      const primaryUserEdits = primaryData.state?.labels
   225→   225→        ? countUserEditedLabels(primaryData.state.labels)
   226→   226→        : countUserEditedLabels(primaryData.labels || []);
   227→   227→      const backupUserEdits = backupData.state?.labels
   228→   228→        ? countUserEditedLabels(backupData.state.labels)
   229→   229→        : countUserEditedLabels(backupData.labels || []);
   230→   230→
   231→   231→      // If primary has fewer labels than backup AND backup has more or equal user data
   232→   232→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   233→   233→        console.warn(
   234→   234→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   235→   235→        );
   236→   236→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   237→   237→        result = backupRaw;
   238→   238→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   239→   239→        // If primary has fewer user edits than backup, use backup
   240→   240→        console.warn(
   241→   241→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   242→   242→        );
   243→   243→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   244→   244→        result = backupRaw;
   245→   245→      } else {
   246→   246→        result = primaryRaw;
   247→   247→      }
   248→   248→    } else {
   249→   249→      // Primary exists, no backup — normal case
   250→   250→      result = primaryRaw;
   251→   251→    }
   252→   252→
   253→   253→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   254→   254→    // merged data immediately after this getItem returns, and we MUST allow that write.
   255→   255→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   256→   256→    _rehydrated = true;
   257→   257→
   258→   258→    return result;
   259→   259→  },
   260→   260→
   261→   261→  setItem: (name: string, value: string): void => {
   262→   262→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   263→   263→    // Before rehydration, the store only has seed data — writing it would overwrite
   264→   264→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   265→   265→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   266→   266→    if (!_rehydrated) {
   267→   267→      console.warn(
   268→   268→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   269→   269→      );
   270→   270→      return;
   271→   271→    }
   272→   272→
   273→   273→    // Write to primary immediately
   274→   274→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   275→   275→
   276→   276→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   277→   277→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   278→   278→    safeLocalStorageSet(BACKUP_KEY, value);
   279→   279→
   280→   280→    if (!primaryOk) {
   281→   281→      console.error("[LabelPulse Storage] Primary write failed!");
   282→   282→    }
   283→   283→  },
   284→   284→
   285→   285→  removeItem: (name: string): void => {
   286→   286→    safeLocalStorageRemove(PRIMARY_KEY);
   287→   287→    safeLocalStorageRemove(BACKUP_KEY);
   288→   288→  },
   289→   289→};

   291→   291→/**
   292→   292→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   293→   293→ * Since backup is now written on every setItem, this is a safety net.
   294→   294→ */
   295→   295→export function forceBackupNow(): void {
   296→   296→  if (typeof window === "undefined") return;
   297→   297→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   298→   298→  if (primaryRaw) {
   299→   299→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   300→   300→  }
   301→   301→}

   436→   436→/**
   437→   437→ * Repair corrupted label data.
   438→   438→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   439→   439→ */
   440→   440→function repairLabelData(labels: Label[]): Label[] {
   441→   441→  const emailLabelCount = new Map<string, number>();
   442→   442→  for (const l of labels) {
   443→   443→    if (l.emails && Array.isArray(l.emails)) {
   444→   444→      for (const e of l.emails) {
   445→   445→        if (e && e.trim()) {
   446→   446→          const key = e.toLowerCase().trim();
   447→   447→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   448→   448→        }
   449→   449→      }
   450→   450→    }
   451→   451→  }
   452→   452→
   453→   453→  const corruptedEmails = new Set<string>();
   454→   454→  for (const [email, count] of emailLabelCount) {
   455→   455→    if (count > 5) {
   456→   456→      corruptedEmails.add(email);
   457→   457→      console.warn(
   458→   458→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   459→   459→      );
   460→   460→    }
   461→   461→  }
   462→   462→
   463→   463→  if (corruptedEmails.size === 0) {
   464→   464→    return labels;
   465→   465→  }
   466→   466→
   467→   467→  const emailOwner = new Map<string, string>();
   468→   468→  for (const email of corruptedEmails) {
   469→   469→    for (const l of labels) {
   470→   470→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   471→   471→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   472→   472→        if (domain && l.name.toLowerCase().includes(domain)) {
   473→   473→          emailOwner.set(email, l.id);

   941→   941→        if (persistedState.labels && Array.isArray(persistedState.labels) && persistedState.labels.length > 0) {
   942→   942→          // Step 1: Add safe defaults to ALL persisted labels (never overwrite existing values)
   943→   943→          merged.labels = persistedState.labels.map((l: any) => ({
   944→   944→            ...LABEL_DEFAULTS,
   945→   945→            ...l,
   946→   946→          }));
   947→   947→
   948→   948→          // Step 2: Append NEW seed labels that don't exist in persisted data
   949→   949→          const persistedIds = new Set(persistedState.labels.map((l: any) => l.id));
   950→   950→          const persistedNames = new Set(
   951→   951→            persistedState.labels
   952→   952→              .map((l: any) => l.name?.toLowerCase().trim())
   953→   953→              .filter(Boolean)
   954→   954→          );
   955→   955→          const seedLabels = buildLabelsFromData();
   956→   956→          for (const seed of seedLabels) {
   957→   957→            if (!persistedIds.has(seed.id) && !persistedNames.has(seed.name.toLowerCase().trim())) {
   958→   958→              merged.labels.push(seed);
   959→   959→            }
   960→   960→          }
   961→   961→
   962→   962→          // Step 3: Repair corrupted data (e.g., same email on too many labels)
   963→   963→          merged.labels = repairLabelData(merged.labels);
   964→   964→        }
   965→   965→
   966→   966→        return merged;
   967→   967→      },
   968→   968→      onRehydrateStorage: () => (state, error) => {
   969→   969→        // ALWAYS mark rehydrated — even on error — to avoid permanently blocking writes
   970→   970→        _rehydrated = true;
   971→   971→
   972→   972→        if (error) {
   973→   973→          console.error("[LabelPulse Storage] Rehydration error:", error);
   974→   974→        } else if (state) {
   975→   975→          const userEditedCount = countUserEditedLabels(state.labels);
   976→   976→          console.log(
   977→   977→            `[LabelPulse Storage] Rehydrated: ${state.labels.length} labels, ${userEditedCount} with user data`
   978→   978→          );
   979→   979→        }
   980→   980→
   981→   981→        // Signal to UI that rehydration is complete.
   982→   982→        // IMPORTANT: onRehydrateStorage can be called synchronously INSIDE create(),
   983→   983→        // before useAppStore is assigned. We use setTimeout to defer setState to the
   984→   984→        // next macrotask, by which time useAppStore will be fully initialized.
   985→   985→        if (typeof window !== "undefined") {
   986→   986→          setTimeout(() => {
   987→   987→            useAppStore.setState({ hasRehydrated: true });
   988→   988→          }, 0);
   989→   989→        }
   990→   990→
   991→   991→        // Force an immediate backup after rehydration
   992→   992→        forceBackupNow();
   993→   993→      },
   994→   994→    }
   995→   995→  )
   996→   996→);

  1095→  1095→export async function loadFromCloud(): Promise<void> {
  1096→  1096→  if (!isSupabaseConfigured) {
  1097→  1097→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
  1098→  1098→    useAppStore.setState({ hasCloudSynced: true });
  1099→  1099→    return;
  1100→  1100→  }
  1101→  1101→
  1102→  1102→  try {
  1103→  1103→    const cloudData = await loadStateFromCloud();
  1104→  1104→    if (!cloudData) {
  1105→  1105→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
  1106→  1106→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");

  1143→  1143→ * se hai fatto modifiche sul PC e poi apri il telefono, qui si sincronizzano.

  1202→  1202→/**

  1204→  1204→ * - Per le labels: merge per-ID, preservando le modifiche locali che il cloud non ha

  1206→  1206→ * - Per i campi semplici: cloud vince se più recente

  1224→  1224→        ...LABEL_DEFAULTS,
  1225→  1225→        ...cloudLabel,
  1226→  1226→      };

  1235→  1235→      // se il cloud ha un valore vuoto ma il locale ha dati, mantieni il locale.

  1291→  1291→  const cloudProfile = cloudData.userProfile || {};
  1292→  1292→  const localProfile = localState.userProfile || {};
  1293→  1293→  merged.userProfile = {
  1294→  1294→    artistName: cloudProfile.artistName?.trim() || localProfile.artistName || "",

  1312→  1312→    ? cloudData.rankingSnapshots
  1313→  1313→    : (localState.rankingSnapshots || []);
  1314→  1314→
  1315→  1315→  return merged;
  1316→  1316→}
