    99→// Smart URL builder for each link type
   100→function toClickableUrl(value: string, linkType: string): string | null {
   101→  if (!value || !value.trim()) return null;
   102→  const v = value.trim();
   103→
   104→  // Already a full URL
   105→  if (/^https?:\/\//i.test(v)) return v;
   106→
   107→  // Email
   108→  if (v.includes("@") && linkType !== "instagram" && linkType !== "soundcloud") {
   109→    return `mailto:${v}`;
   110→  }
   111→
   112→  // Type-specific smart defaults
   113→  switch (linkType) {
   114→    case "instagram":
   115→      if (/^[a-zA-Z0-9._]{1,30}$/.test(v)) {
   116→        return `https://instagram.com/${v.replace(/^@/, "")}`;
   117→      }
   118→      return `https://${v}`;
   119→    case "soundcloud":
   120→      if (/^[a-zA-Z0-9._-]{1,30}$/.test(v)) {
   121→        return `https://soundcloud.com/${v.replace(/^@/, "")}`;
   122→      }
   123→      return `https://${v}`;
   124→    case "beatport":
   125→      if (/^[a-zA-Z0-9._-]{1,50}$/.test(v)) {
   126→        return `https://www.beatport.com/label/${v.replace(/^@/, "")}`;
   127→      }
   128→      return `https://${v}`;
   129→    case "spotify":
   130→      if (/^[a-zA-Z0-9._-]{1,40}$/.test(v)) {
   131→        return `https://open.spotify.com/artist/${v.replace(/^@/, "")}`;
   132→      }
   133→      return `https://${v}`;
   134→    case "youtube":
   135→      if (/^@?[a-zA-Z0-9._-]{1,40}$/.test(v)) {
   136→        return `https://youtube.com/${v.replace(/^@/, "@")}`;
   137→      }
   138→      return `https://${v}`;
   139→    case "bandcamp":
   140→      if (/^[a-zA-Z0-9._-]{1,40}$/.test(v)) {
   141→        return `https://${v}.bandcamp.com`;
   142→      }
   143→      return `https://${v}`;
   144→    case "website":
   145→    case "demoLink":
   146→    case "other":
   147→    default:
   148→      return `https://${v}`;
   149→  }
   150→}

   170→// Convert fixed link fields + customLinks into a unified DetailLink array
   171→function labelToDetailLinks(label: Label): DetailLink[] {
   172→  const links: DetailLink[] = [];
   173→  if (label.website?.trim()) links.push({ type: "website", value: label.website });
   174→  if (label.socialLink?.trim()) links.push({ type: "instagram", value: label.socialLink });
   175→  if (label.soundcloudLink?.trim()) links.push({ type: "soundcloud", value: label.soundcloudLink });
   176→  if (label.beatportLink?.trim()) links.push({ type: "beatport", value: label.beatportLink });
   177→  if (label.demoLink?.trim()) links.push({ type: "demoLink", value: label.demoLink });
   178→  // Add custom links
   179→  if (label.customLinks?.length) {
   180→    for (const cl of label.customLinks) {
   181→      if (cl.value?.trim()) links.push({ type: cl.type, value: cl.value });
   182→    }
   183→  }
   184→  return links;
   185→}
   186→
   187→// Save a DetailLink array back to the fixed fields + customLinks
   188→function detailLinksToFields(links: DetailLink[]): Partial<Label> {
   189→  const fields: Partial<Label> = {
   190→    website: "",
   191→    socialLink: "",
   192→    soundcloudLink: "",
   193→    beatportLink: "",
   194→    demoLink: "",
   195→    customLinks: [],
   196→  };
   197→  for (const link of links) {
   198→    if (!link.value?.trim()) continue;
   199→    switch (link.type) {
   200→      case "website":
   201→        fields.website = link.value;
   202→        break;
   203→      case "instagram":
   204→        fields.socialLink = link.value;
   205→        break;
   206→      case "soundcloud":
   207→        fields.soundcloudLink = link.value;
   208→        break;
   209→      case "beatport":
   210→        fields.beatportLink = link.value;
   211→        break;
   212→      case "demoLink":
   213→        fields.demoLink = link.value;
   214→        break;
   215→      default:
   216→        // Custom type - goes into customLinks
   217→        if (!fields.customLinks) fields.customLinks = [];
   218→        fields.customLinks.push({ type: link.type, value: link.value });
   219→        break;
   220→    }
   221→  }
   222→  return fields;
   223→}

   225→// Get display text for a link (strips protocol for cleaner display)
   226→function getLinkDisplay(value: string): string {
   227→  return value
   228→    .replace(/^https?:\/\//, "")
   229→    .replace(/^mailto:/, "")
   230→    .replace(/\/$/, "");
   231→}
