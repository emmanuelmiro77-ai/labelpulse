     1→"use client";
     2→
     3→import { useAppStore, getLabelTier, type Label } from "@/lib/store";
     4→import { t, type Locale } from "@/lib/i18n";
     5→import { useState, useMemo, useCallback } from "react";
     6→import {
     7→  Zap,
     8→  Mail,
     9→  Check,
    10→  Languages,
    11→  Music2,
    12→  AlertCircle,
    13→  ChevronDown,
    14→  ChevronUp,
    15→  Send,
    16→  Target,
    17→  Copy,
    18→  ExternalLink,
    19→  Activity,
    20→  Upload,
    21→  Loader2,
    22→  Sparkles,
    23→} from "lucide-react";
    24→import { Button } from "@/components/ui/button";
    25→import { Card, CardContent } from "@/components/ui/card";
    26→import { Input } from "@/components/ui/input";
    27→import { Badge } from "@/components/ui/badge";
    28→import {
    29→  Select,
    30→  SelectContent,
    31→  SelectItem,
    32→  SelectTrigger,
    33→  SelectValue,
    34→} from "@/components/ui/select";
    35→import { Label as UILabel } from "@/components/ui/label";
    36→import { Textarea } from "@/components/ui/textarea";
    37→import { Checkbox } from "@/components/ui/checkbox";
    38→import {
    39→  generateSubject,
    40→  generatePitchBody,
    41→  generateGmailLink,
    42→  generateMailtoLink,
    43→  PITCH_LANGUAGES,
    44→  type PitchTone,
    45→  type PitchLanguage,
    46→} from "@/lib/pitch-utils";
    47→import { useToast } from "@/hooks/use-toast";
    48→import { SimilarSuggestions } from "@/components/similar-suggestions";
    49→
    50→export function PitchGenerator() {
    51→  const { labels, demos, addDemo, locale, userProfile, setUserProfile, getGenres, artists, setActiveTab, setSelectedLabelId, setSelectedArtistId } = useAppStore();
    52→  const { toast } = useToast();
    53→  const genres = getGenres();
    54→
    55→  // Pre-fill from user profile
    56→  const initialArtistName = userProfile.artistName || "";
    57→  const initialScLink = userProfile.scLink || "";
    58→
    59→  // Track setup state
    60→  const [trackName, setTrackName] = useState("");
    61→  const [artistName, setArtistName] = useState(initialArtistName);
    62→  const [scLink, setScLink] = useState(initialScLink);
    63→  const [tone, setTone] = useState<PitchTone>("professional");
    64→  const [language, setLanguage] = useState<PitchLanguage>("en");
    65→  const [customNote, setCustomNote] = useState("");
    66→  const [selectedGenre, setSelectedGenre] = useState("");
    67→
    68→  // Track audio profile — user can type BPM/key manually OR run an analysis
    69→  // on the SoundCloud link. Used by SimilarSuggestions to mine the scraped
    70→  // DB for labels/artists that match the user's track.
    71→  const [trackBpm, setTrackBpm] = useState("");
    72→  const [trackKey, setTrackKey] = useState("");
    73→  const [trackAnalysis, setTrackAnalysis] = useState<{
    74→    bpm: number;
    75→    bpmConfidence: number;
    76→    key: { camelot: string; confidence: number };
    77→  } | null>(null);
    78→  const [isAnalyzing, setIsAnalyzing] = useState(false);
    79→  const [analysisError, setAnalysisError] = useState<string | null>(null);
    80→
    81→  // Target labels state
    82→  const [selectedLabelIds, setSelectedLabelIds] = useState<Set<string>>(new Set());
    83→  const [onlyWithEmail, setOnlyWithEmail] = useState(false);
    84→  const [expandedTiers, setExpandedTiers] = useState<Record<string, boolean>>({ top: true, mid: true, emerging: true });
    85→
    86→  // Campaign state
    87→  const [sending, setSending] = useState(false);
    88→  const [sendProgress, setSendProgress] = useState({ current: 0, total: 0 });
    89→  const [campaignComplete, setCampaignComplete] = useState(false);
    90→  const [campaignResults, setCampaignResults] = useState({ sent: 0, skipped: 0 });
    91→
    92→  // Save profile fields on blur
    93→  const handleArtistBlur = () => {
    94→    if (artistName.trim() && artistName.trim() !== userProfile.artistName) {
    95→      setUserProfile({ artistName: artistName.trim() });
    96→    }
    97→  };
    98→
    99→  const handleScLinkBlur = () => {
   100→    if (scLink.trim() && scLink.trim() !== userProfile.scLink) {
   101→      setUserProfile({ scLink: scLink.trim() });
   102→    }
   103→  };
   104→
   105→  // Run audio analysis on the SoundCloud link OR a user-uploaded file.
   106→  // Result fills trackBpm / trackKey + trackAnalysis (used by SimilarSuggestions).
   107→  const handleAnalyzeTrack = async (file?: File) => {
   108→    const audioUrl = scLink.trim();
   109→    if (!file && !audioUrl) {
   110→      setAnalysisError(locale === "it"
   111→        ? "Inserisci un link SoundCloud o carica un file"
   112→        : "Provide a SoundCloud link or upload a file");
   113→      return;
   114→    }
   115→    setIsAnalyzing(true);
   116→    setAnalysisError(null);
   117→    try {
   118→      const { analyzeAudio, analyzeAudioFile } = await import("@/lib/audio-analysis");
   119→      const result = file
   120→        ? await analyzeAudioFile(file, () => {})
   121→        : await analyzeAudio(audioUrl, () => {});
   122→      setTrackAnalysis({
   123→        bpm: result.bpm,
   124→        bpmConfidence: result.bpmConfidence,
   125→        key: {
   126→          camelot: result.key.camelot,
   127→          confidence: result.key.confidence,
   128→        },
   129→      });
   130→      setTrackBpm(String(result.bpm));
   131→      if (result.key.confidence > 0) {
   132→        setTrackKey(result.key.camelot);
   133→      }
   134→    } catch (err: any) {
   135→      console.error("[pitch analyze]", err);
   136→      setAnalysisError(err?.message || (locale === "it" ? "Errore analisi" : "Analysis failed"));
   137→    } finally {
   138→      setIsAnalyzing(false);
   139→    }
   140→  };
   141→
   142→  // Cross-tab navigation from SimilarSuggestions
   143→  const handleOpenLabel = useCallback(
   144→    (label: Label) => {
   145→      setSelectedLabelId?.(label.id || label.name);
   146→      setActiveTab("labels");
   147→    },
   148→    [setActiveTab, setSelectedLabelId]
   149→  );
   150→  const handleOpenArtist = useCallback(
   151→    (artistId: string) => {
   152→      setSelectedArtistId?.(artistId);
   153→      setActiveTab("artists");
   154→      if (typeof window !== "undefined") {
   155→        window.scrollTo({ top: 0, behavior: "smooth" });
   156→      }
   157→    },
   158→    [setActiveTab, setSelectedArtistId]
   159→  );
   160→  // "Use as target" in SimilarSuggestions → toggle the label in the
   161→  // selectedLabelIds set so it gets included in the campaign.
   162→  const handleSelectLabelAsTarget = useCallback(
   163→    (label: Label) => {
   164→      setSelectedLabelIds((prev) => {
   165→        const next = new Set(prev);
   166→        if (next.has(label.id)) {
   167→          next.delete(label.id);
   168→        } else {
   169→          next.add(label.id);
   170→        }
   171→        return next;
   172→      });
   173→    },
   174→    []
   175→  );
   176→
   177→  // Get matching labels by genre, grouped by tier
   178→  const matchingLabels = useMemo(() => {
   179→    if (!selectedGenre) return { top: [], mid: [], emerging: [] };
   180→    const genreLabels = labels.filter((l) =>
   181→      l.genres.includes(selectedGenre) && l.status === "open"
   182→    );
   183→    const top = genreLabels
   184→      .filter((l) => getLabelTier(l, selectedGenre) === "top")
   185→      .sort((a, b) => (a.rankByGenre?.[selectedGenre] || 999) - (b.rankByGenre?.[selectedGenre] || 999));
   186→    const mid = genreLabels
   187→      .filter((l) => getLabelTier(l, selectedGenre) === "mid")
   188→      .sort((a, b) => (a.rankByGenre?.[selectedGenre] || 999) - (b.rankByGenre?.[selectedGenre] || 999));
   189→    const emerging = genreLabels
   190→      .filter((l) => getLabelTier(l, selectedGenre) === "emerging")
   191→      .sort((a, b) => (b.trendingPointsByGenre?.[selectedGenre] || 0) - (a.trendingPointsByGenre?.[selectedGenre] || 0));
   192→    return { top, mid, emerging };
   193→  }, [labels, selectedGenre]);
   194→
   195→  const allMatchingLabels = useMemo(
   196→    () => [...matchingLabels.top, ...matchingLabels.mid, ...matchingLabels.emerging],
   197→    [matchingLabels]
   198→  );
   199→
   200→  // Get emails for a label
   201→  const getLabelEmails = useCallback((label: Label): string[] => {
   202→    if (label.emails?.length) return label.emails;
   203→    if (label.contactInfo) return [label.contactInfo];
   204→    return [];
   205→  }, []);
   206→
   207→  // Filtered labels (only with email toggle)
   208→  const filteredLabels = useMemo(() => {
   209→    if (!onlyWithEmail) return allMatchingLabels;
   210→    return allMatchingLabels.filter((l) => getLabelEmails(l).length > 0);
   211→  }, [allMatchingLabels, onlyWithEmail, getLabelEmails]);
   212→
   213→  const filteredByTier = useMemo(() => {
   214→    const top = filteredLabels.filter((l) => getLabelTier(l, selectedGenre) === "top");
   215→    const mid = filteredLabels.filter((l) => getLabelTier(l, selectedGenre) === "mid");
   216→    const emerging = filteredLabels.filter((l) => getLabelTier(l, selectedGenre) === "emerging");
   217→    return { top, mid, emerging };
   218→  }, [filteredLabels, selectedGenre]);
   219→
   220→  // Selection management
   221→  const toggleLabel = useCallback((id: string) => {
   222→    setSelectedLabelIds((prev) => {
   223→      const next = new Set(prev);
   224→      if (next.has(id)) next.delete(id);
   225→      else next.add(id);
   226→      return next;
   227→    });
   228→  }, []);
   229→
   230→  const selectAll = useCallback(() => {
   231→    const ids = new Set(filteredLabels.map((l) => l.id));
   232→    setSelectedLabelIds(ids);
   233→  }, [filteredLabels]);
   234→
   235→  const deselectAll = useCallback(() => {
   236→    setSelectedLabelIds(new Set());
   237→  }, []);
   238→
   239→  // Selected label stats
   240→  const selectedLabels = useMemo(
   241→    () => allMatchingLabels.filter((l) => selectedLabelIds.has(l.id)),
   242→    [allMatchingLabels, selectedLabelIds]
   243→  );
   244→
   245→  const selectedWithEmail = useMemo(
   246→    () => selectedLabels.filter((l) => getLabelEmails(l).length > 0),
   247→    [selectedLabels, getLabelEmails]
   248→  );
   249→
   250→  const selectedWithoutEmail = useMemo(
   251→    () => selectedLabels.filter((l) => getLabelEmails(l).length === 0),
   252→    [selectedLabels, getLabelEmails]
   253→  );
   254→
   255→  // Generate personalized pitch preview for a label
   256→  const getPitchForLabel = useCallback(
   257→    (label: Label) => {
   258→      if (!trackName.trim()) return { subject: "", body: "" };
   259→      const subject = generateSubject(trackName.trim(), artistName, language);
   260→      const body = generatePitchBody(label.name, trackName.trim(), artistName, scLink, tone, customNote, language);
   261→      return { subject, body };
   262→    },
   263→    [trackName, artistName, scLink, tone, customNote, language]
   264→  );
   265→
   266→  // Tier badge component
   267→  const tierBadge = (tier: "top" | "mid" | "emerging") => {
   268→    if (tier === "top")
   269→      return <Badge className="bg-purple-500/20 text-purple-300 border-purple-500/30 text-[10px] px-1.5 py-0">🟣 {t(locale, "labels.tierTop")}</Badge>;
   270→    if (tier === "mid")
   271→      return <Badge className="bg-blue-500/20 text-blue-300 border-blue-500/30 text-[10px] px-1.5 py-0">🔵 {t(locale, "labels.tierMid")}</Badge>;
   272→    return <Badge className="bg-emerald-500/20 text-emerald-300 border-emerald-500/30 text-[10px] px-1.5 py-0">🟢 {t(locale, "labels.tierEmerging")}</Badge>;
   273→  };
   274→
   275→  // Toggle tier expand
   276→  const toggleTier = (tier: string) => {
   277→    setExpandedTiers((prev) => ({ ...prev, [tier]: !prev[tier] }));
   278→  };
   279→
   280→  // Bulk send campaign via Gmail Web Compose
   281→  const handleSendCampaign = useCallback(async () => {
   282→    if (!selectedWithEmail.length || !trackName.trim()) return;
   283→    setSending(true);
   284→    setCampaignComplete(false);
   285→    setCampaignResults({ sent: 0, skipped: 0 });
   286→
   287→    const toSend = selectedWithEmail.filter((l) => getLabelEmails(l).length > 0);
   288→    const skipped = selectedWithoutEmail.length;
   289→    let sentCount = 0;
   290→
   291→    setSendProgress({ current: 0, total: toSend.length });
   292→
   293→    for (let i = 0; i < toSend.length; i++) {
   294→      const label = toSend[i];
   295→      const emails = getLabelEmails(label);
   296→      if (!emails.length) continue;
   297→
   298→      const { subject, body } = getPitchForLabel(label);
   299→      if (!subject || !body) continue;
   300→
   301→      setSendProgress({ current: i + 1, total: toSend.length });
   302→
   303→      // Open Gmail compose in new tab with pre-filled content
   304→      const gmailUrl = generateGmailLink(emails, subject, body);
   305→      window.open(gmailUrl, `_blank`, "noopener");
   306→
   307→      sentCount++;
   308→
   309→      // Auto-create demo entry
   310→      const demoAlreadyExists = demos.some(
   311→        (d) => d.labelId === label.id && d.trackName.toLowerCase() === trackName.trim().toLowerCase()
   312→      );
   313→      if (!demoAlreadyExists) {
   314→        addDemo({
   315→          trackName: trackName.trim(),
   316→          labelId: label.id,
   317→          status: "sent",
   318→          sentDate: new Date().toISOString().split("T")[0],
   319→          link: scLink.trim(),
   320→          notes: customNote.trim() ? `${customNote.trim()} (Campaign)` : "Campaign",
   321→          pitchText: `Subject: ${subject}\n\n${body}`,
   322→          artistName: artistName.trim(),
   323→        });
   324→      }
   325→
   326→      // Small delay between opens to avoid popup blocking
   327→      if (i < toSend.length - 1) {
   328→        await new Promise((r) => setTimeout(r, 800));
   329→      }
   330→    }
   331→
   332→    setCampaignResults({ sent: sentCount, skipped });
   333→    setCampaignComplete(true);
   334→    setSending(false);
   335→
   336→    toast({
   337→      title: t(locale, "campaign.complete"),
   338→      description: t(locale, "campaign.reviewAndSend").replace("{count}", String(sentCount)),
   339→    });
   340→  }, [selectedWithEmail, selectedWithoutEmail, getLabelEmails, getPitchForLabel, demos, trackName, scLink, customNote, artistName, addDemo, locale, toast]);
   341→
   342→  // Copy all emails to clipboard
   343→  const handleCopyEmails = useCallback(async () => {
   344→    const emails = selectedWithEmail.flatMap((l) => getLabelEmails(l));
   345→    const emailText = emails.join(", ");
   346→    try {
   347→      await navigator.clipboard.writeText(emailText);
   348→      toast({ title: t(locale, "campaign.emailsCopied") });
   349→    } catch {
   350→      const textarea = document.createElement("textarea");
   351→      textarea.value = emailText;
   352→      document.body.appendChild(textarea);
   353→      textarea.select();
   354→      document.execCommand("copy");
   355→      document.body.removeChild(textarea);
   356→      toast({ title: t(locale, "campaign.emailsCopied") });
   357→    }
   358→  }, [selectedWithEmail, getLabelEmails, locale, toast]);
   359→
   360→  // Track if campaign params changed (to reset completion state)
   361→  const campaignParamsKey = `${selectedLabelIds.size}-${trackName}-${tone}-${language}-${customNote}`;
   362→  const [lastCampaignKey, setLastCampaignKey] = useState(campaignParamsKey);
   363→  if (campaignParamsKey !== lastCampaignKey) {
   364→    setLastCampaignKey(campaignParamsKey);
   365→    if (campaignComplete) setCampaignComplete(false);
   366→  }
   367→
   368→  return (
   369→    <div className="space-y-6">
   370→      {/* ===== TOP SECTION: Track Setup ===== */}
   371→      <Card className="bg-card/60 border-border/30">
   372→        <CardContent className="p-4 sm:p-6">
   373→          <div className="flex items-center gap-2 mb-4">
   374→            <Target className="h-4 w-4 text-primary" />
   375→            <h3 className="text-sm font-mono uppercase tracking-wider text-muted-foreground">
   376→              {t(locale, "campaign.trackSetup")}
   377→            </h3>
   378→          </div>
   379→
   380→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   381→            <div className="space-y-1.5">
   382→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.trackName")}</UILabel>
   383→              <Input
   384→                value={trackName}
   385→                onChange={(e) => setTrackName(e.target.value)}
   386→                placeholder="e.g. Midnight Drive"
   387→                className="bg-secondary/50 border-border/50"
   388→              />
   389→            </div>
   390→            <div className="space-y-1.5">
   391→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.artistName")}</UILabel>
   392→              <Input
   393→                value={artistName}
   394→                onChange={(e) => setArtistName(e.target.value)}
   395→                onBlur={handleArtistBlur}
   396→                placeholder="e.g. Traacia"
   397→                className="bg-secondary/50 border-border/50"
   398→              />
   399→            </div>
   400→          </div>
   401→
   402→          <div className="space-y-1.5 mt-3">
   403→            <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.scLink")}</UILabel>
   404→            <Input
   405→              value={scLink}
   406→              onChange={(e) => setScLink(e.target.value)}
   407→              onBlur={handleScLinkBlur}
   408→              placeholder="https://soundcloud.com/..."
   409→              className="bg-secondary/50 border-border/50"
   410→            />
   411→          </div>
   412→
   413→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3 mt-3">
   414→            <div className="space-y-1.5">
   415→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.tone")}</UILabel>
   416→              <Select value={tone} onValueChange={(v) => setTone(v as PitchTone)}>
   417→                <SelectTrigger className="bg-secondary/50 border-border/50">
   418→                  <SelectValue />
   419→                </SelectTrigger>
   420→                <SelectContent>
   421→                  <SelectItem value="professional">{t(locale, "pitch.toneProfessional")}</SelectItem>
   422→                  <SelectItem value="confident">{t(locale, "pitch.toneConfident")}</SelectItem>
   423→                  <SelectItem value="friendly">{t(locale, "pitch.toneFriendly")}</SelectItem>
   424→                  <SelectItem value="storytelling">{t(locale, "pitch.toneStorytelling")}</SelectItem>
   425→                </SelectContent>
   426→              </Select>
   427→            </div>
   428→            <div className="space-y-1.5">
   429→              <UILabel className="text-xs font-mono uppercase text-muted-foreground flex items-center gap-1">
   430→                <Languages className="h-3 w-3" /> {t(locale, "pitch.emailLanguage")}
   431→              </UILabel>
   432→              <Select value={language} onValueChange={(v) => setLanguage(v as PitchLanguage)}>
   433→                <SelectTrigger className="bg-secondary/50 border-border/50">
   434→                  <SelectValue />
   435→                </SelectTrigger>
   436→                <SelectContent>
   437→                  {(Object.keys(PITCH_LANGUAGES) as PitchLanguage[]).map((lang) => (
   438→                    <SelectItem key={lang} value={lang}>{PITCH_LANGUAGES[lang]}</SelectItem>
   439→                  ))}
   440→                </SelectContent>
   441→              </Select>
   442→            </div>
   443→            <div className="space-y-1.5 sm:col-span-2">
   444→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">
   445→                {t(locale, "labels.selectGenre")} 🎯
   446→              </UILabel>
   447→              <Select value={selectedGenre} onValueChange={(v) => { setSelectedGenre(v); setSelectedLabelIds(new Set()); setCampaignComplete(false); }}>
   448→                <SelectTrigger className="bg-secondary/50 border-border/50">
   449→                  <SelectValue placeholder={t(locale, "campaign.selectGenreFirst")} />
   450→                </SelectTrigger>
   451→                <SelectContent>
   452→                  {genres.map((g) => (
   453→                    <SelectItem key={g} value={g}>{g}</SelectItem>
   454→                  ))}
   455→                </SelectContent>
   456→              </Select>
   457→            </div>
   458→          </div>
   459→
   460→          <div className="space-y-1.5 mt-3">
   461→            <UILabel className="text-xs font-mono uppercase text-muted-foreground">{t(locale, "pitch.additionalNote")}</UILabel>
   462→            <Textarea
   463→              value={customNote}
   464→              onChange={(e) => setCustomNote(e.target.value)}
   465→              placeholder="..."
   466→              rows={2}
   467→              className="bg-secondary/50 border-border/50 resize-none"
   468→            />
   469→          </div>
   470→
   471→          {/* ===== Track audio profile (BPM / key + audio analysis) =====
   472→              These power the SimilarSuggestions panel below: by analyzing
   473→              your track (or typing BPM/key manually), the app can mine the
   474→              scraped Beatport DB and surface labels / artists that already
   475→              release tracks like yours. */}
   476→          <div className="mt-3 grid grid-cols-2 gap-3">
   477→            <div className="space-y-1.5">
   478→              <UILabel className="text-xs font-mono uppercase text-muted-foreground flex items-center gap-1">
   479→                <Activity className="h-3 w-3" /> BPM
   480→              </UILabel>
   481→              <Input
   482→                value={trackBpm}
   483→                onChange={(e) => setTrackBpm(e.target.value)}
   484→                placeholder="128"
   485→                className="bg-secondary/50 border-border/50"
   486→              />
   487→            </div>
   488→            <div className="space-y-1.5">
   489→              <UILabel className="text-xs font-mono uppercase text-muted-foreground">Key (Camelot)</UILabel>
   490→              <Input
   491→                value={trackKey}
   492→                onChange={(e) => setTrackKey(e.target.value)}
   493→                placeholder="8A"
   494→                className="bg-secondary/50 border-border/50"
   495→              />
   496→            </div>
   497→          </div>
   498→
   499→          <div className="mt-2 flex items-center gap-1.5">
   500→            <Button
   501→              type="button"
   502→              variant="outline"
   503→              size="sm"
   504→              onClick={() => handleAnalyzeTrack()}
   505→              disabled={isAnalyzing || !scLink.trim()}
   506→              className="h-7 text-xs"
   507→            >
   508→              {isAnalyzing ? (
   509→                <><Loader2 className="h-3 w-3 mr-1 animate-spin" /> {locale === "it" ? "Analisi..." : "Analyzing..."}</>
   510→              ) : (
   511→                <><Activity className="h-3 w-3 mr-1" /> {locale === "it" ? "Analizza link" : "Analyze link"}</>
   512→              )}
   513→            </Button>
   514→            <label className="cursor-pointer">
   515→              <input
