   135→// Custom storage with data protection:
   136→// 1. Primary localStorage key

   142→const BACKUP_KEY = "labelpulse-storage-backup";

   169→function safeJsonParse(str: string | null): any | null {
   170→  if (!str) return null;
   171→  try {
   172→    return JSON.parse(str);
   173→  } catch {
   174→    return null;
   175→  }
   176→}
   177→
   178→function safeLocalStorageGet(key: string): string | null {
   179→  try {
   180→    return localStorage.getItem(key);
   181→  } catch {
   182→    return null;
   183→  }
   184→}
   185→
   186→function safeLocalStorageSet(key: string, value: string): boolean {
   187→  try {
   188→    localStorage.setItem(key, value);
   189→    return true;
   190→  } catch (e) {
   191→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   192→    return false;
   193→  }
   194→}
   195→
   196→function safeLocalStorageRemove(key: string): void {
   197→  try {
   198→    localStorage.removeItem(key);
   199→  } catch {
   200→    // Ignore
   201→  }
   202→}

   204→/**
   205→ * Extract rankingSnapshots array from a raw localStorage JSON string.
   206→ * Returns [] if not found / invalid.
   207→ */
   208→function extractSnapshotsFromRaw(raw: string | null): RankingSnapshot[] {
   209→  if (!raw) return [];
   210→  const data = safeJsonParse(raw);
   211→  if (!data) return [];
   212→  const snaps = data.state?.rankingSnapshots ?? data.rankingSnapshots;
   213→  return Array.isArray(snaps) ? snaps : [];
   214→}

   216→/**
   217→ * Merge two snapshot arrays, deduping by `id` first, then by `timestamp+source`.
   218→ * Always returns ascending-by-timestamp order.
   219→ * Used by: importData (backup file), robustStorage.getItem (safety-net restore),
   220→ * and loadFromCloud (cloud merge).
   221→ */
   222→function mergeSnapshots(
   223→  base: RankingSnapshot[] | null | undefined,
   224→  incoming: RankingSnapshot[] | null | undefined
   225→): RankingSnapshot[] {
   226→  const a = Array.isArray(base) ? base : [];
   227→  const b = Array.isArray(incoming) ? incoming : [];
   228→  const seen = new Set<string>();
   229→  const out: RankingSnapshot[] = [];
   230→  for (const s of [...a, ...b]) {
   231→    if (!s || typeof s !== "object") continue;
   232→    const key = s.id || `${s.timestamp}|${s.source}`;
   233→    if (seen.has(key)) continue;
   234→    seen.add(key);
   235→    out.push(s);
   236→  }
   237→  out.sort(
   238→    (x, y) =>
   239→      new Date(x.timestamp || 0).getTime() - new Date(y.timestamp || 0).getTime()
   240→  );
   241→  return out;
   242→}

   244→/**
   245→ * Check if persisted data has user edits (emails, notes, links, etc.)
   246→ * This helps us detect if the store was accidentally reset to seed data.
   247→ */
   248→function countUserEditedLabels(labels: any[]): number {
   249→  if (!Array.isArray(labels)) return 0;
   250→  return labels.filter((l) =>
   251→    (l.emails && l.emails.length > 0) ||
   252→    (l.notes && l.notes.trim() !== "") ||
   253→    (l.website && l.website.trim() !== "") ||
   254→    (l.demoLink && l.demoLink.trim() !== "") ||
   255→    (l.socialLink && l.socialLink.trim() !== "") ||
   256→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   257→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   258→    l.isCustom === true
   259→  ).length;
   260→}

   262→function getTotalLabelCount(data: any): number {
   263→  if (!data) return 0;
   264→  if (Array.isArray(data.labels)) return data.labels.length;
   265→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   266→  return 0;
   267→}

   269→/**
   270→ * Extract user-edited label count from raw localStorage string.
   271→ * Used by the write guard to check if a write would cause data loss.
   272→ */
   273→function getUserEditCountFromRaw(raw: string | null): number {
   274→  if (!raw) return 0;
   275→  const data = safeJsonParse(raw);
   276→  if (!data) return 0;
   277→  const labels = data.state?.labels || data.labels || [];
   278→  return countUserEditedLabels(labels);

   281→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   282→
   283→const robustStorage: StateStorage = {
   284→  getItem: (name: string): string | null => {
   285→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   286→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   287→
   288→    const primaryData = safeJsonParse(primaryRaw);
   289→    const backupData = safeJsonParse(backupRaw);
   290→
   291→    let result: string | null = null;
   292→    let usedSidecarSnapshots = false;
   293→
   294→    // If primary is missing/corrupt, try backup
   295→    if (!primaryData) {
   296→      if (backupData) {
   297→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   298→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   299→        result = backupRaw;
   300→      }
   301→      // Both empty — first visit, result stays null
   302→    } else if (backupData) {
   303→      // Both exist — check for data loss
   304→      const primaryLabelCount = getTotalLabelCount(primaryData);
   305→      const backupLabelCount = getTotalLabelCount(backupData);
   306→      const primaryUserEdits = primaryData.state?.labels
   307→        ? countUserEditedLabels(primaryData.state.labels)
   308→        : countUserEditedLabels(primaryData.labels || []);
   309→      const backupUserEdits = backupData.state?.labels
   310→        ? countUserEditedLabels(backupData.state.labels)
   311→        : countUserEditedLabels(backupData.labels || []);
   312→
   313→      // If primary has fewer labels than backup AND backup has more or equal user data
   314→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   315→        console.warn(
   316→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   317→        );
   318→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   319→        result = backupRaw;
   320→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   321→        // If primary has fewer user edits than backup, use backup
   322→        console.warn(
   323→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   324→        );
   325→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   326→        result = backupRaw;
   327→      } else {
   328→        result = primaryRaw;
   329→      }
   330→    } else {
   331→      // Primary exists, no backup — normal case
   332→      result = primaryRaw;
   333→    }
   334→
   335→    // CRITICAL SAFETY NET for rankingSnapshots:
   336→    // If the chosen result has 0 snapshots but the dedicated SNAPSHOTS_BACKUP_KEY
   337→    // has snapshots, splice them in. This handles the scenario where the main
   338→    // store got wiped (bad cloud sync, migration bug, partial overwrite) but the
   339→    // snapshots sidecar survived. Without this, the user loses months of
   340→    // "storico caricamenti" and the Classifiche page resets to "first time".
   341→    try {
   342→      const currentSnaps = extractSnapshotsFromRaw(result);
   343→      const sidecarRaw = safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY);
   344→      const sidecarSnaps = extractSnapshotsFromRaw(sidecarRaw);
   345→      if (sidecarSnaps.length > 0 && sidecarSnaps.length > currentSnaps.length) {
   346→        console.warn(
   347→          `[LabelPulse Storage] SNAPSHOTS RECOVERY: result has ${currentSnaps.length} snapshots, sidecar has ${sidecarSnaps.length}. Merging sidecar into result.`
   348→        );
   349→        const mergedSnaps = mergeSnapshots(currentSnaps, sidecarSnaps);
   350→        const parsed = result ? safeJsonParse(result) : {};
   351→        if (parsed && typeof parsed === "object") {
   352→          if (parsed.state && typeof parsed.state === "object") {
   353→            parsed.state.rankingSnapshots = mergedSnaps;
   354→          } else {
   355→            parsed.rankingSnapshots = mergedSnaps;
   356→          }
   357→          result = JSON.stringify(parsed);
   358→          usedSidecarSnapshots = true;
   359→        }
   360→      }
   361→    } catch (e) {
   362→      console.warn("[LabelPulse Storage] Snapshots sidecar recovery failed:", e);
   363→    }
   364→
   365→    // CRITICAL SAFETY NET for userProfile:
   366→    // Same logic as snapshots. If the chosen result has an empty/seed profile
   367→    // but PROFILE_BACKUP_KEY has real data, splice it in. This is the user's
   368→    // identity card — losing artistName, bio, email, links is catastrophic.
   369→    try {
   370→      const currentProfile = extractProfileFromRaw(result);
   371→      const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   372→      if (sidecarProfile && !currentProfile) {
   373→        console.warn(
   374→          `[LabelPulse Storage] PROFILE RECOVERY: result has empty profile, sidecar has data (artistName=${!!sidecarProfile.artistName}, bio=${!!sidecarProfile.bio}). Merging sidecar into result.`
   375→        );
   376→        const parsed = result ? safeJsonParse(result) : {};
   377→        if (parsed && typeof parsed === "object") {
   378→          if (parsed.state && typeof parsed.state === "object") {
   379→            parsed.state.userProfile = sidecarProfile;
   380→          } else {
   381→            parsed.userProfile = sidecarProfile;
   382→          }
   383→          result = JSON.stringify(parsed);
   384→        }
   385→      } else if (sidecarProfile && currentProfile) {
   386→        // Both have data — merge (sidecar fills empty fields in current)
   387→        const mergedProfile = mergeProfiles(sidecarProfile, currentProfile);
   388→        const parsed = result ? safeJsonParse(result) : {};
   389→        if (parsed && typeof parsed === "object" && mergedProfile) {
   390→          if (parsed.state && typeof parsed.state === "object") {
   391→            parsed.state.userProfile = mergedProfile;
   392→          } else {
   393→            parsed.userProfile = mergedProfile;
   394→          }
   395→          result = JSON.stringify(parsed);
   396→        }
   397→      }
   398→    } catch (e) {
   399→      console.warn("[LabelPulse Storage] Profile sidecar recovery failed:", e);
   400→    }
   401→
   402→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   403→    // merged data immediately after this getItem returns, and we MUST allow that write.
   404→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   405→    _rehydrated = true;
   406→
   407→    if (usedSidecarSnapshots) {
   408→      console.info("[LabelPulse Storage] rankingSnapshots restored from sidecar backup.");
   409→    }
   410→
   411→    return result;
   412→  },
   413→
   414→  setItem: (name: string, value: string): void => {
   415→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   416→    // Before rehydration, the store only has seed data — writing it would overwrite
   417→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   418→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   419→    if (!_rehydrated) {
   420→      console.warn(
   421→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   422→      );
   423→      return;
   424→    }
   425→
   426→    // Write to primary immediately
   427→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   428→
   429→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   430→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   431→    safeLocalStorageSet(BACKUP_KEY, value);
   432→
   433→    // CRITICAL: also persist rankingSnapshots to a dedicated slot.
   434→    // This is the user's "storico caricamenti" — months of imports that cannot
   435→    // be regenerated. Keeping it in a separate key means even a catastrophic
   436→    // main-store wipe (bad cloud sync, migration bug, quota error) leaves the
   437→    // snapshots recoverable.
   438→    try {
   439→      const snaps = extractSnapshotsFromRaw(value);
   440→      if (snaps.length > 0) {
   441→        // Merge with whatever's already in the snapshots slot — NEVER overwrite
   442→        // existing snapshots with fewer snapshots (would happen if the current
   443→        // write had a partial/empty snapshot array due to some upstream bug).
   444→        const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   445→        const merged = mergeSnapshots(existing, snaps);
   446→        safeLocalStorageSet(
   447→          SNAPSHOTS_BACKUP_KEY,
   448→          JSON.stringify(merged)
   449→        );
   450→      }
   451→    } catch (e) {
   452→      console.warn("[LabelPulse Storage] Snapshots-sidecar write failed:", e);
   453→    }
   454→
   455→    // CRITICAL: also persist userProfile to a dedicated slot.
   456→    // Same pattern as snapshots. The profile is the user's identity card —
   457→    // artistName, bio, email, social links, photo. We can NEVER lose this,
   458→    // even if cloud sync wipes the main store.
   459→    try {
   460→      const profile = extractProfileFromRaw(value);
   461→      if (profile) {
   462→        const existing = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   463→        const merged = mergeProfiles(existing, profile);
   464→        if (merged) {
   465→          safeLocalStorageSet(
   466→            PROFILE_BACKUP_KEY,
   467→            JSON.stringify({ userProfile: merged })
   468→          );
   469→        }
   470→      }
   471→    } catch (e) {
   472→      console.warn("[LabelPulse Storage] Profile-sidecar write failed:", e);
   473→    }
   474→
   475→    if (!primaryOk) {
   476→      console.error("[LabelPulse Storage] Primary write failed!");
   477→    }
   478→  },
   479→
   480→  removeItem: (name: string): void => {
   481→    safeLocalStorageRemove(PRIMARY_KEY);
   482→    safeLocalStorageRemove(BACKUP_KEY);
   483→    // NOTE: do NOT remove SNAPSHOTS_BACKUP_KEY or PROFILE_BACKUP_KEY here.
   484→    // These are the user's permanent records (months of ranking history +
   485→    // artist identity) — clearing the main store should NEVER nuke them.
   486→    // The only way to clear them is an explicit user action ("reset all data"
   487→    // button) which calls safeLocalStorageRemove directly.
   488→  },
   489→};

   491→/**
   492→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   493→ * Since backup is now written on every setItem, this is a safety net.
   494→ */
   495→export function forceBackupNow(): void {
   496→  if (typeof window === "undefined") return;
   497→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   498→  if (primaryRaw) {
   499→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   500→  }
   501→  // Also touch the snapshots sidecar so it stays warm.
   502→  const snaps = extractSnapshotsFromRaw(primaryRaw);
   503→  if (snaps.length > 0) {
   504→    const existing = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   505→    const merged = mergeSnapshots(existing, snaps);
   506→    safeLocalStorageSet(SNAPSHOTS_BACKUP_KEY, JSON.stringify(merged));
   507→  }
   508→}

   510→/**
   511→ * Returns the current rankingSnapshots from the dedicated sidecar backup.
   512→ * Used by the UI to offer "restore history from emergency backup" if the

   515→export function readSnapshotsSidecar(): RankingSnapshot[] {

   520→/**
   521→ * Force-restore rankingSnapshots from the sidecar into the live store.
   522→ * Returns the number of snapshots restored, or 0 if the sidecar was empty
   523→ * / already in sync.
   524→ */
   525→export function restoreSnapshotsFromSidecar(): number {
   526→  if (typeof window === "undefined") return 0;
   527→  const sidecarSnaps = extractSnapshotsFromRaw(safeLocalStorageGet(SNAPSHOTS_BACKUP_KEY));
   528→  if (sidecarSnaps.length === 0) return 0;
   529→  const current = useAppStore.getState().rankingSnapshots || [];
   530→  const merged = mergeSnapshots(current, sidecarSnaps);
   531→  const added = merged.length - current.length;
   532→  if (added > 0) {
   533→    useAppStore.setState({ rankingSnapshots: merged });
   534→    console.info(
   535→      `[LabelPulse] Restored ${added} snapshot(s) from sidecar backup.`
   536→    );
   537→  }
   538→  return Math.max(added, 0);
   539→}

   543→/**
   544→ * Extract userProfile from a raw JSON string (main store JSON, backup JSON,
   545→ * or sidecar JSON). Returns null if missing/invalid.
   546→ */
   547→function extractProfileFromRaw(raw: string | null): any | null {
   548→  if (!raw) return null;
   549→  try {
   550→    const parsed = JSON.parse(raw);
   551→    const profile = parsed?.state?.userProfile ?? parsed?.userProfile ?? null;
   552→    if (profile && typeof profile === "object") {
   553→      // Must have at least one non-empty field to be considered "real"
   554→      const hasData =
   555→        !!profile.artistName ||
   556→        !!profile.bio ||
   557→        !!profile.email ||
   558→        !!profile.scLink ||
   559→        !!profile.photoUrl ||
   560→        (Array.isArray(profile.links) && profile.links.length > 0) ||
   561→        !!profile.cyaniteApiToken ||
   562→        !!profile.supabaseUrl;
   563→      if (hasData) return profile;
   564→    }
   565→  } catch {
   566→    /* ignore */
   567→  }
   568→  return null;
   569→}

   571→/**
   572→ * Merge two user profiles — non-empty fields win.
   573→ * This is the safety net: if the cloud ever sends a profile with empty
   574→ * artistName, but the sidecar has the real artistName, we keep the real one.
   575→ */
   576→export function mergeProfiles(base: any | null, incoming: any | null): any | null {
   577→  if (!base && !incoming) return null;
   578→  if (!base) return incoming;
   579→  if (!incoming) return base;
   580→  const merged: any = { ...base };
   581→  for (const k of Object.keys(incoming)) {
   582→    const v = incoming[k];
   583→    if (k === "links") {
   584→      // Merge link arrays (dedupe by type|value)
   585→      const a = Array.isArray(base.links) ? base.links : [];
   586→      const b = Array.isArray(v) ? v : [];
   587→      const seen = new Set<string>();
   588→      const out: { type: string; value: string }[] = [];
   589→      for (const link of [...a, ...b]) {
   590→        if (!link || typeof link !== "object") continue;
   591→        const key = `${link.type}|${link.value}`;
   592→        if (seen.has(key)) continue;
   593→        seen.add(key);
   594→        out.push(link);
   595→      }
   596→      merged.links = out;
   597→    } else if (typeof v === "string") {
   598→      // Non-empty string wins over empty
   599→      if (v && !merged[k]) merged[k] = v;
   600→    } else {
   601→      // For any other type, prefer defined over undefined
   602→      if (v !== undefined && v !== null && !merged[k]) merged[k] = v;
   603→    }
   604→  }
   605→  return merged;
   606→}

   617→ * Force-restore userProfile from the sidecar into the live store, IF the
   618→ * current live profile is empty/seed AND the sidecar has real data.
   619→ *
   620→ * Returns true if a restore happened, false otherwise.
   621→ */
   622→export function restoreProfileFromSidecar(): boolean {
   623→  if (typeof window === "undefined") return false;
   624→  const sidecarProfile = extractProfileFromRaw(safeLocalStorageGet(PROFILE_BACKUP_KEY));
   625→  if (!sidecarProfile) return false;
   626→
   627→  const current = useAppStore.getState().userProfile;
   628→  const currentHasData =
   629→    !!current.artistName ||
   630→    !!current.bio ||
   631→    !!current.email ||
   632→    !!current.scLink ||
   633→    !!current.photoUrl ||
   634→    (Array.isArray(current.links) && current.links.length > 0) ||
   635→    !!current.cyaniteApiToken ||
   636→    !!current.supabaseUrl;
   637→
   638→  // If current profile has data, merge (sidecar can fill empty fields)
   639→  // If current profile is empty, replace with sidecar
   640→  const merged = currentHasData
   641→    ? mergeProfiles(sidecarProfile, current)

   648→    );
   649→    return true;
   650→  }
   651→  return false;
   652→}

   747→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   748→    notes: imported.notes?.trim() || existing.notes,

   750→
   751→    // Beatport/ranking data — prefer imported (it's the fresh data)
   752→    genres: imported.genres?.length ? imported.genres : existing.genres,
   753→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,

   790→/**
   791→ * Repair corrupted label data.
   792→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   793→ */
   794→function repairLabelData(labels: Label[]): Label[] {
   795→  const emailLabelCount = new Map<string, number>();
   796→  for (const l of labels) {
   797→    if (l.emails && Array.isArray(l.emails)) {
   798→      for (const e of l.emails) {
   799→        if (e && e.trim()) {
   800→          const key = e.toLowerCase().trim();
   801→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   802→        }
   803→      }
   804→    }
   805→  }
   806→
   807→  const corruptedEmails = new Set<string>();
   808→  for (const [email, count] of emailLabelCount) {
   809→    if (count > 5) {
   810→      corruptedEmails.add(email);
   811→      console.warn(
   812→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   813→      );
   814→    }
   815→  }
   816→
   817→  if (corruptedEmails.size === 0) {
   818→    return labels;
   819→  }
   820→
   821→  const emailOwner = new Map<string, string>();
   822→  for (const email of corruptedEmails) {
   823→    for (const l of labels) {
   824→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   825→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   826→        if (domain && l.name.toLowerCase().includes(domain)) {
   827→          emailOwner.set(email, l.id);
   828→          break;
   829→        }
   830→      }
   831→    }
   832→    if (!emailOwner.has(email)) {
   833→      for (const l of labels) {
   834→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   835→          emailOwner.set(email, l.id);
   836→          break;
   837→        }
   838→      }
   839→    }
   840→  }
   841→
   842→  return labels.map(l => {
   843→    if (!l.emails || !Array.isArray(l.emails)) return l;
   844→    const cleanedEmails = l.emails.filter(e => {
   845→      const key = e?.toLowerCase().trim();
   846→      if (!key) return false;
   847→      if (!corruptedEmails.has(key)) return true;
   848→      return emailOwner.get(key) === l.id;
   849→    });
   850→    const cleanedContactInfo = cleanedEmails[0] || "";
   851→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   852→      console.log(
   853→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   854→      );
   855→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   856→    }
   857→    return l;
   858→  });
   859→}

  1597→
  1598→    // Confronta i timestamp per decidere quali dati usare
  1599→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
  1600→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
  1601→
  1602→    // ⚠️ CRITICAL: Content-aware merge decision.
  1603→    // The old code only used timestamps: `if (cloudSavedAt > localSavedAt) merge else skip`.
  1604→    // This was BRITTLE because:
  1605→    //   1. If both timestamps are 0 (e.g., null on both sides), `0 > 0` is false → SKIP merge
  1606→    //      → cloud data is ignored even when local is empty seed.
  1607→    //   2. If local was the last writer (e.g., same device that just saved), local timestamp
  1608→    //      equals cloud timestamp → SKIP merge → if local was wiped (e.g., fresh incognito
  1609→    //      after a force-sync), cloud data is also ignored.
  1610→    //   3. After the skip, useAuthEffect's forceCloudSync() would push the EMPTY local state
  1611→    //      back to cloud, OVERWRITING the cloud's good data.
  1612→    //
  1613→    // FIX: Always merge if cloud has profile data that local doesn't have, regardless of
  1614→    // timestamps. Timestamps are only used as a tiebreaker when both have data.
  1615→    const cloudProfileHasData =
  1616→      !!cloud.userProfile?.artistName ||
  1617→      !!cloud.userProfile?.bio ||
  1618→      !!cloud.userProfile?.email ||
  1619→      !!cloud.userProfile?.scLink ||
  1620→      !!cloud.userProfile?.photoUrl ||
  1621→      (Array.isArray(cloud.userProfile?.links) && cloud.userProfile.links.length > 0);
  1622→    const localProfileHasData =
  1623→      !!localState.userProfile?.artistName ||
  1624→      !!localState.userProfile?.bio ||
  1625→      !!localState.userProfile?.email ||
  1626→      !!localState.userProfile?.scLink ||
  1627→      !!localState.userProfile?.photoUrl ||
  1628→      (Array.isArray(localState.userProfile?.links) && localState.userProfile.links.length > 0);
  1629→
  1630→    const cloudHasLabels = Array.isArray(cloud.labels) && cloud.labels.length > 0;
  1631→    const cloudHasSnapshots = Array.isArray(cloud.rankingSnapshots) && cloud.rankingSnapshots.length > 0;
  1632→    const cloudHasDemos = Array.isArray(cloud.demos) && cloud.demos.length > 0;
  1633→    const localHasLabels = (localState.labels?.length ?? 0) > 0;
  1634→    const localHasSnapshots = (localState.rankingSnapshots?.length ?? 0) > 0;
  1635→    const localHasDemos = (localState.demos?.length ?? 0) > 0;
  1636→
  1637→    // Force merge if cloud has profile data local doesn't have
  1638→    const cloudBringsNewProfile = cloudProfileHasData && !localProfileHasData;
  1639→    // Force merge if cloud has labels/snapshots/demos that local doesn't have
  1640→    const cloudBringsNewLabels = cloudHasLabels && !localHasLabels;
  1641→    const cloudBringsNewSnapshots = cloudHasSnapshots && !localHasSnapshots;
  1642→    const cloudBringsNewDemos = cloudHasDemos && !localHasDemos;
  1643→
  1644→    const cloudIsNewerByTimestamp = cloudSavedAt > localSavedAt;
  1645→    const shouldMergeFromCloud =
  1646→      cloudBringsNewProfile ||
  1647→      cloudBringsNewLabels ||
  1648→      cloudBringsNewSnapshots ||
  1649→      cloudBringsNewDemos ||
  1650→      cloudIsNewerByTimestamp;
  1651→
  1652→    if (shouldMergeFromCloud) {
  1653→      // I dati cloud sono più recenti O hanno contenuti che il locale non ha — mergia
  1654→      console.log(
  1655→        `[LabelPulse Cloud] Merging from cloud. Reasons: ` +
  1656→        `cloudBringsNewProfile=${cloudBringsNewProfile}, ` +
  1657→        `cloudBringsNewLabels=${cloudBringsNewLabels}, ` +
  1658→        `cloudBringsNewSnapshots=${cloudBringsNewSnapshots}, ` +
  1659→        `cloudBringsNewDemos=${cloudBringsNewDemos}, ` +
  1660→        `cloudIsNewerByTimestamp=${cloudIsNewerByTimestamp} ` +
  1661→        `(cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}).`
  1662→      );
  1663→
  1664→      // SAFETY CHECK: se il cloud è "più recente" ma ha array vuoti dove
  1665→      // il locale ha dati, NON fidarti — è probabile che il cloud sia stato
  1666→      // resettato / corrotto. Mantieni i dati locali e basta.
  1667→      if (
  1668→        (localHasLabels && !cloudHasLabels) ||
  1669→        (localHasSnapshots && !cloudHasSnapshots) ||
  1670→        (localHasDemos && !cloudHasDemos)
  1671→      ) {
  1672→        console.warn(
  1673→          "[LabelPulse Cloud] CLOUD DATA LOSS DETECTED — cloud is newer but has empty arrays where local has data. Skipping merge to preserve local data. Run a manual sync to upload local to cloud."
  1674→        );
  1675→        useAppStore.setState({ hasCloudSynced: true });
  1676→        // Trigger upload of local data so cloud gets repopulated
  1677→        setTimeout(() => forceCloudSync(), 500);
  1678→        setupRealtimeSubscriptionSafe();
  1679→        return;
  1680→      }
  1681→
  1682→      // Usa la stessa logica di merge della reidratazione
  1683→      const merged = mergeCloudData(cloud, localState);
  1684→      console.log(
  1685→        `[LabelPulse Cloud] MERGE RESULT — profile from cloud: artistName=${cloud.userProfile?.artistName || "(empty)"}, bio=${cloud.userProfile?.bio ? "yes" : "no"}, links=${cloud.userProfile?.links?.length || 0}; ` +
  1686→        `local profile: artistName=${localState.userProfile?.artistName || "(empty)"}; ` +
  1687→        `merged profile: artistName=${(merged as any).userProfile?.artistName || "(empty)"}, bio=${(merged as any).userProfile?.bio ? "yes" : "no"}, links=${(merged as any).userProfile?.links?.length || 0}.`
  1688→      );
  1689→      useAppStore.setState({
  1690→        ...merged,
  1691→        hasCloudSynced: true,
  1692→      });
  1693→    } else {
  1694→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1695→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1696→      useAppStore.setState({ hasCloudSynced: true });
  1697→    }
  1698→
  1699→    // Setup realtime subscription to receive future updates from other devices
  1700→    setupRealtimeSubscriptionSafe();
  1701→  } catch (err) {
  1702→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1703→    useAppStore.setState({ hasCloudSynced: true });
  1704→  }
  1705→}

  1707→/**
  1708→ * Setup the realtime subscription safely (only on client, only if configured).
  1709→ * Re-imported lazily to avoid circular deps at module load.
  1710→ */
  1711→function setupRealtimeSubscriptionSafe(): void {
  1712→  if (typeof window === "undefined") return;
  1713→  if (!isSupabaseConfigured()) return;
  1714→  // Defer to next tick so we don't block the initial render
  1715→  setTimeout(() => {
  1716→    try {
  1717→      // Re-import to avoid circular dependency
  1718→      import("./supabase").then(({ setupRealtimeSubscription }) => {
  1719→        setupRealtimeSubscription();
  1720→      });
  1721→    } catch (err) {
  1722→      console.warn("[LabelPulse Cloud] Realtime subscription setup failed:", err);
  1723→    }
  1724→  }, 100);
  1725→}

  1727→/**
  1728→ * Merge dei dati cloud con quelli locali, preservando i dati utente.
  1729→ * Usa la stessa logica della funzione merge di Zustand persist.
  1730→ *
  1731→ * CRITICAL SAFETY RULE: mai sovrascrivere array locali non-vuoti con
  1732→ * array cloud vuoti. Se il cloud ha perso dati (es. tabella corrotta,
  1733→ * sync parziale, progetto Supabase resettato), non vogliamo propagare
  1734→ * la perdita ai dispositivi che hanno ancora i dati.
  1735→ */
  1736→function mergeCloudData(cloudData: any, localState: any): Partial<AppState> {
  1737→  const merged: Partial<AppState> = {};
  1738→
  1739→  // Labels: applica defaults e merge con seed data, come nella reidratazione
  1740→  if (cloudData.labels && Array.isArray(cloudData.labels) && cloudData.labels.length > 0) {
  1741→    merged.labels = cloudData.labels.map((l: any) => ({
  1742→      ...LABEL_DEFAULTS,
  1743→      ...l,
  1744→    }));
  1745→
  1746→    // Aggiungi seed labels mancanti
  1747→    const cloudIds = new Set(cloudData.labels.map((l: any) => l.id));
  1748→    const cloudNames = new Set(
  1749→      cloudData.labels.map((l: any) => l.name?.toLowerCase().trim()).filter(Boolean)
  1750→    );
  1751→    const seedLabels = buildLabelsFromData();
  1752→    for (const seed of seedLabels) {
  1753→      if (!cloudIds.has(seed.id) && !cloudNames.has(seed.name.toLowerCase().trim())) {
  1754→        (merged.labels as Label[]).push(seed);
  1755→      }
  1756→    }
  1757→
  1758→    // Repair corrupted data
  1759→    merged.labels = repairLabelData(merged.labels);
  1760→  } else {
  1761→    // Cloud non ha labels (o array vuoto) → mantieni locali
  1762→    merged.labels = localState.labels;
  1763→  }
  1764→
  1765→  // Demos: SOLO se cloud ha effettivamente demos, mantieni locali altrimenti
  1766→  // (before: `Array.isArray(cloudData.demos) ? cloudData.demos : localState.demos`
  1767→  //  sovrascriveva demos locali con [] se cloud aveva demos:[])
  1768→  if (Array.isArray(cloudData.demos) && cloudData.demos.length > 0) {
  1769→    merged.demos = cloudData.demos;
  1770→  } else if (Array.isArray(cloudData.demos) && (!localState.demos || localState.demos.length === 0)) {
  1771→    // Cloud ha demos:[] e locale è vuoto → ok a usare [] (no-op)
  1772→    merged.demos = cloudData.demos;
  1773→  } else {
  1774→    merged.demos = localState.demos;
  1775→  }
  1776→
  1777→  // Simple fields
  1778→  merged.activeTab = cloudData.activeTab || localState.activeTab;
  1779→  merged.locale = cloudData.locale || localState.locale;
  1780→  // ⚠️ CRITICAL: use mergeProfiles (non-empty fields win) instead of `||`.
  1781→  // If cloud has an empty {} profile (e.g., from a fresh seed upload),
  1782→  // `cloudData.userProfile || localState.userProfile` returns the empty cloud
  1783→  // profile (because {} is truthy), wiping the local profile. mergeProfiles
  1784→  // correctly keeps non-empty fields from BOTH sources.
  1785→  merged.userProfile =
  1786→    mergeProfiles(localState.userProfile, cloudData.userProfile) ||
  1787→    cloudData.userProfile ||
  1788→    localState.userProfile;
  1789→  merged.gmailAuth = cloudData.gmailAuth || localState.gmailAuth;
  1790→  merged.rankingsUpdatedAt = cloudData.rankingsUpdatedAt ?? localState.rankingsUpdatedAt;
  1791→  merged.lastSavedAt = cloudData.lastSavedAt ?? localState.lastSavedAt;
  1792→
  1793→  // rankingSnapshots: STesso principio — non sovrascrivere snapshots locali
  1794→  // con array vuoto dal cloud. Gli snapshots sono il cuore delle classifiche
  1795→  // storiche, perderli significa perdere mesi di import Beatport/Beatstats.
  1796→  if (Array.isArray(cloudData.rankingSnapshots) && cloudData.rankingSnapshots.length > 0) {
  1797→    merged.rankingSnapshots = cloudData.rankingSnapshots;
  1798→  } else if (Array.isArray(cloudData.rankingSnapshots) && (!localState.rankingSnapshots || localState.rankingSnapshots.length === 0)) {
  1799→    // Cloud vuoto e locale vuoto → ok a usare []
  1800→    merged.rankingSnapshots = cloudData.rankingSnapshots;
  1801→  } else {
  1802→    // Locale ha dati, cloud vuoto → mantieni locali
  1803→    merged.rankingSnapshots = localState.rankingSnapshots || [];
  1804→  }
  1805→
  1806→  return merged;
  1807→}
