    72→const STATUS_COLORS: Record<
    73→  DemoStatus,
    74→  { color: string; bgColor: string; borderColor: string }
    75→> = {

    81→};
    82→
    83→const STATUS_FLOW: DemoStatus[] = ["ready", "sent", "reviewing", "accepted"];
    84→
    85→function getDaysSince(dateStr: string | null): number | null {
    86→  if (!dateStr) return null;
    87→  return Math.floor((Date.now() - new Date(dateStr).getTime()) / 86400000);
    88→}
    89→
    90→const STATUS_TKEYS: Record<DemoStatus, "demos.ready" | "demos.sent" | "demos.reviewing" | "demos.accepted" | "demos.rejected"> = {
    91→  ready: "demos.ready",
    92→  sent: "demos.sent",
    93→  reviewing: "demos.reviewing",
    94→  accepted: "demos.accepted",
    95→  rejected: "demos.rejected",
    96→};
    97→
    98→export function DemoTracker() {
    99→  const { labels, demos, addDemo, updateDemo, deleteDemo, advanceDemoStatus, locale: _locale, getGenres, userProfile, artists, setActiveTab, setSelectedLabelId, setSelectedArtistId } =
   100→    useAppStore();
   101→  const locale = _locale as Locale;
   102→  const genres = getGenres();
   103→  const [viewMode, setViewMode] = useState<"kanban" | "table">("kanban");
   104→  const [search, setSearch] = useState("");
   105→  const [statusFilter, setStatusFilter] = useState("all");
   106→  const [showAddDialog, setShowAddDialog] = useState(false);
   107→  const [editingDemo, setEditingDemo] = useState<Demo | null>(null);
   108→  const [deleteConfirmId, setDeleteConfirmId] = useState<string | null>(null);
   109→
   110→  // Detail dialog
   111→  const [detailDemo, setDetailDemo] = useState<Demo | null>(null);
   112→
   113→  // Label history view
   114→  const [labelHistoryId, setLabelHistoryId] = useState<string | null>(null);
   115→
   116→  const [formTrackName, setFormTrackName] = useState("");
   117→  const [formLabelId, setFormLabelId] = useState("");
   118→  const [formStatus, setFormStatus] = useState<DemoStatus>("ready");
   119→  const [formSentDate, setFormSentDate] = useState("");
   120→  const [formLink, setFormLink] = useState("");
   121→  const [formLinks, setFormLinks] = useState<{ type: string; value: string }[]>([]);
   122→  const [formNotes, setFormNotes] = useState("");
   123→  const [formGenre, setFormGenre] = useState("");
   124→  const [formBpm, setFormBpm] = useState("");
   125→  const [formKey, setFormKey] = useState("");
   126→  // Audio analysis state
   127→  const [isAnalyzing, setIsAnalyzing] = useState(false);
   128→  const [analysisProgress, setAnalysisProgress] = useState<{
   129→    stage: string;
   130→    message: string;
   131→    progress: number;
   132→  } | null>(null);
   133→  const [analysisError, setAnalysisError] = useState<string | null>(null);
   134→  const [formAnalysis, setFormAnalysis] = useState<Demo["analysis"] | null>(null);
   135→
   136→  const filteredDemos = useMemo(() => {
   137→    return demos.filter((d) => {
   138→      const matchSearch =
   139→        !search ||
   140→        d.trackName.toLowerCase().includes(search.toLowerCase()) ||
   141→        d.notes.toLowerCase().includes(search.toLowerCase());
   142→      const matchStatus = statusFilter === "all" || d.status === statusFilter;
   143→      return matchSearch && matchStatus;
   144→    });
   145→  }, [demos, search, statusFilter]);
   146→
   147→  // Label history: all demos for a specific label
   148→  const labelHistoryDemos = useMemo(() => {
   149→    if (!labelHistoryId) return [];
   150→    return demos
   151→      .filter((d) => d.labelId === labelHistoryId)
   152→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   153→  }, [demos, labelHistoryId]);
   154→
   155→  const labelHistoryName = useMemo(() => {
   156→    if (!labelHistoryId) return "";
   157→    return labels.find((l) => l.id === labelHistoryId)?.name ?? "";
   158→  }, [labels, labelHistoryId]);
   159→
   160→  const getLabelName = (labelId: string) => {
   161→    return labels.find((l) => l.id === labelId)?.name ?? "—";
   162→  };
   163→
   164→  const resetForm = () => {
   165→    setFormTrackName("");
   166→    setFormLabelId("");
   167→    setFormStatus("ready");
   168→    setFormSentDate("");
   169→    setFormLink("");
   170→    setFormLinks([]);
   171→    setFormNotes("");
   172→    setFormGenre("");
   173→    setFormBpm("");
   174→    setFormKey("");
   175→    setFormAnalysis(null);
   176→    setAnalysisError(null);
   177→    setAnalysisProgress(null);
   178→  };
   179→
   180→  const openAdd = () => { resetForm(); setEditingDemo(null); setShowAddDialog(true); };
   181→
   182→  const openEdit = (demo: Demo) => {
   183→    setFormTrackName(demo.trackName);
   184→    setFormLabelId(demo.labelId);
   185→    setFormStatus(demo.status);
   186→    setFormSentDate(demo.sentDate ?? "");
   187→    setFormLink(demo.link);
   188→    setFormLinks(demo.links || []);
   189→    setFormNotes(demo.notes);
   190→    setFormGenre(demo.genre || "");
   191→    setFormBpm(demo.bpm || "");
   192→    setFormKey(demo.key || "");
   193→    setFormAnalysis(demo.analysis || null);
   194→    setAnalysisError(null);
   195→    setAnalysisProgress(null);
   196→    setEditingDemo(demo);
   197→    setShowAddDialog(true);
   198→  };
   199→
   200→  const handleSave = () => {
   201→    if (!formTrackName.trim() || !formLabelId) return;
   202→    const data = {
   203→      trackName: formTrackName.trim(),
   204→      labelId: formLabelId,
   205→      status: formStatus,
   206→      sentDate: formSentDate || null,
   207→      link: formLink.trim(),
   208→      links: formLinks.filter(l => l.value.trim()),
   209→      notes: formNotes.trim(),
   210→      pitchText: editingDemo?.pitchText || "",
   211→      artistName: editingDemo?.artistName || userProfile.artistName || "",
   212→      genre: formGenre.trim(),
   213→      bpm: formBpm.trim(),
   214→      key: formKey.trim(),
   215→      analysis: formAnalysis || undefined,
   216→    };
   217→    if (editingDemo) { updateDemo(editingDemo.id, data); }
   218→    else { addDemo(data); }
   219→    setShowAddDialog(false);
   220→    resetForm();
   221→  };
   222→
   223→  const handleAnalyze = async () => {
   224→    const audioSourceUrl = formLink.trim() || (formLinks.find(l => l.type === "soundcloud" || l.type === "audio")?.value?.trim() ?? "");
   225→    if (!audioSourceUrl) {
   226→      setAnalysisError("Inserisci un link SoundCloud o URL audio diretto, oppure usa 'Carica file'");
   227→      return;
   228→    }
   229→    setIsAnalyzing(true);
   230→    setAnalysisError(null);
   231→    setAnalysisProgress({ stage: "fetching", message: "Inizio...", progress: 0 });
   232→    try {
   233→      // Dynamic import to keep the initial bundle small
   234→      const { analyzeAudio, analyzeWithCyanite } = await import("@/lib/audio-analysis");
   235→      const cyaniteToken = (userProfile as any)?.cyaniteApiToken?.trim?.() || "";
   236→      // Safety timeout: abort analysis after 120s
   237→      const timeoutMs = 120_000;
   238→      const result = await Promise.race([
   239→        cyaniteToken
   240→          ? analyzeWithCyanite(audioSourceUrl, cyaniteToken, (p) => setAnalysisProgress(p))
   241→          : analyzeAudio(audioSourceUrl, (p) => setAnalysisProgress(p)),
   242→        new Promise<never>((_, reject) =>
   243→          setTimeout(() => reject(new Error("Timeout: analisi troppo lunga (>120s). Prova con 'Carica file' o ricarica la pagina.")), timeoutMs)
   244→        ),
   245→      ]);
   246→      setFormAnalysis(result);
   247→      // Auto-fill BPM and key if empty or always update with analysis values
   248→      setFormBpm(String(result.bpm));
   249→      // Only auto-fill key if it was actually detected (confidence > 0)
   250→      if (result.key.confidence > 0) {
   251→        setFormKey(result.key.name);
   252→      }
   253→      setAnalysisProgress({ stage: "done", message: "Analisi completata!", progress: 1 });
   254→      // Clear progress after 2 seconds
   255→      setTimeout(() => setAnalysisProgress(null), 2000);
   256→    } catch (err: any) {
   257→      console.error("[analyze]", err);
   258→      let userMsg = "Errore durante l'analisi";
   259→      if (err?.message) {
   260→        const m = err.message;
   261→        if (/soundcloud|risolvere/i.test(m)) {
   262→          userMsg = m; // already user-friendly from proxy
   263→        } else if (/format|corrupt|decode/i.test(m)) {
   264→          userMsg = `Il browser non è riuscito a decodificare l'audio scaricato. SoundCloud potrebbe aver restituito un formato non supportato. Prova con 'Carica file'. Dettagli: ${m}`;
   265→        } else if (/wasm|essentia|memory/i.test(m)) {
   266→          userMsg = `Errore del motore di analisi: ${m}. Prova con 'Carica file' (più affidabile).`;
   267→        } else {
   268→          userMsg = m;
   269→        }
   270→      } else if (typeof err === "string") {
   271→        userMsg = err;
   272→      }
   273→      setAnalysisError(userMsg);
   274→      setAnalysisProgress(null);
   275→    } finally {
   276→      setIsAnalyzing(false);
   277→    }
   278→  };
   279→
   280→  const handleAnalyzeFile = async (file: File) => {
   281→    setIsAnalyzing(true);
   282→    setAnalysisError(null);
   283→    setAnalysisProgress({ stage: "fetching", message: `Lettura ${file.name}...`, progress: 0.2 });
   284→    try {
   285→      const { analyzeAudioFile } = await import("@/lib/audio-analysis");
   286→      // Safety timeout: abort analysis after 120s to prevent infinite hang
   287→      const timeoutMs = 120_000;
   288→      const result = await Promise.race([
   289→        analyzeAudioFile(file, (p) => setAnalysisProgress(p)),
   290→        new Promise<never>((_, reject) =>
   291→          setTimeout(() => reject(new Error("Timeout: analisi troppo lunga (>120s). Prova con un file più corto (max 60s) o ricarica la pagina.")), timeoutMs)
   292→        ),
   293→      ]);
   294→      setFormAnalysis(result);
   295→      setFormBpm(String(result.bpm));
   296→      // Only auto-fill key if it was actually detected (confidence > 0)
   297→      if (result.key.confidence > 0) {
   298→        setFormKey(result.key.name);
   299→      }
   300→      setAnalysisProgress({ stage: "done", message: "Analisi completata!", progress: 1 });
   301→      setTimeout(() => setAnalysisProgress(null), 2000);
   302→    } catch (err: any) {
   303→      console.error("[analyze file]", err);
   304→      // Build a clear user-facing message
   305→      let userMsg = "Errore durante l'analisi del file";
   306→      if (err?.message) {
   307→        const m = err.message;
   308→        if (/format|corrupt|decode/i.test(m)) {
   309→          userMsg = `Formato non supportato o file corrotto: ${m}`;
   310→        } else if (/wasm|essentia|memory/i.test(m)) {
   311→          userMsg = `Errore interno del motore di analisi (essentia.js): ${m}. Prova con un file più piccolo o un formato diverso (MP3 consigliato).`;
   312→        } else {
   313→          userMsg = m;
   314→        }
   315→      } else if (typeof err === "string") {
   316→        userMsg = err;
   317→      }
   318→      setAnalysisError(userMsg);
   319→      setAnalysisProgress(null);
   320→    } finally {
   321→      setIsAnalyzing(false);
   322→    }
   323→  };
   324→
   325→  const canAdvance = (status: DemoStatus) => STATUS_FLOW.indexOf(status) < STATUS_FLOW.length - 1;
   326→
   327→  // Cross-tab navigation from SimilarSuggestions panel inside the add/edit
   328→  // dialog. Clicking a label name closes this dialog and opens the label
   329→  // detail dialog in the Labels tab; clicking an artist name closes this
   330→  // dialog and opens the artist detail page in the Artists tab. Promoting
   331→  // a suggested label to the demo's target is in-dialog (no navigation).
   332→  const handleOpenLabelFromSuggestion = useCallback(
   333→    (label: Label) => {
   334→      setShowAddDialog(false);
   335→      setSelectedLabelId?.(label.id || label.name);
   336→      setActiveTab("labels");
   337→    },
   338→    [setActiveTab, setSelectedLabelId]
   339→  );
   340→  const handleOpenArtistFromSuggestion = useCallback(
   341→    (artistId: string) => {
   342→      setShowAddDialog(false);
   343→      setSelectedArtistId?.(artistId);
   344→      setActiveTab("artists");
   345→      if (typeof window !== "undefined") {
   346→        window.scrollTo({ top: 0, behavior: "smooth" });
   347→      }
   348→    },
   349→    [setActiveTab, setSelectedArtistId]
   350→  );
   351→  const handleSelectLabelAsTarget = useCallback(
   352→    (label: Label) => {
   353→      setFormLabelId(label.id);
   354→      // Auto-fill genre if the demo doesn't have one yet and the label has a best genre
   355→      if (!formGenre.trim() && label.genres?.length) {
   356→        setFormGenre(label.genres[0]);
   357→      }
   358→    },
   359→    [formGenre]
   360→  );
   361→
   362→  // If viewing label history, show that instead
   363→  if (labelHistoryId) {
   364→    return (
   365→      <div className="space-y-4">
   366→        <div className="flex items-center gap-3">
   367→          <Button variant="ghost" size="sm" onClick={() => setLabelHistoryId(null)} className="text-muted-foreground hover:text-foreground">
   368→            <ArrowLeft className="h-4 w-4 mr-1.5" />
   369→            {t(locale, "demos.backToTracker")}
   370→          </Button>
   371→        </div>
   372→
   373→        <div className="flex items-center gap-2">
   374→          <Music2 className="h-5 w-5 text-primary" />
   375→          <h3 className="text-lg font-bold text-foreground">{labelHistoryName}</h3>
   376→          <Badge variant="secondary" className="text-[10px]">{labelHistoryDemos.length} {t(locale, "demos.submissions")}</Badge>
   377→        </div>
   378→
   379→        {labelHistoryDemos.length === 0 ? (
   380→          <div className="text-center py-12 text-muted-foreground">
   381→            <Music className="h-10 w-10 mx-auto mb-2 opacity-30" />
   382→            <p className="text-sm">{t(locale, "demos.noDemos")}</p>
   383→          </div>
   384→        ) : (
   385→          <div className="space-y-2">
   386→            {labelHistoryDemos.map((demo) => {
   387→              const config = STATUS_COLORS[demo.status];
   388→              const daysSince = getDaysSince(demo.sentDate);
   389→              return (
   390→                <Card
   391→                  key={demo.id}
   392→                  className="bg-card/80 border-border/30 hover:border-primary/30 transition-all cursor-pointer"
   393→                  onClick={() => setDetailDemo(demo)}
   394→                >
   395→                  <CardContent className="p-4">
   396→                    <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   397→                      <div className="flex-1 min-w-0">
   398→                        <div className="flex items-center gap-2 flex-wrap">
   399→                          <h4 className="text-sm font-semibold text-foreground">{demo.trackName}</h4>
   400→                          <Badge variant="outline" className={`${config.bgColor} ${config.color} ${config.borderColor} text-[10px]`}>
   401→                            {STATUS_ICONS[demo.status]}
   402→                            <span className="ml-1">{t(locale, STATUS_TKEYS[demo.status])}</span>
   403→                          </Badge>
   404→                          {demo.artistName && (
   405→                            <span className="text-[10px] text-muted-foreground font-mono flex items-center gap-0.5">
   406→                              <User className="h-2.5 w-2.5" /> {demo.artistName}
   407→                            </span>
   408→                          )}
   409→                        </div>
   410→                        <div className="flex flex-wrap items-center gap-x-3 gap-y-0.5 mt-1 text-[11px] text-muted-foreground">
   411→                          {demo.sentDate && (
   412→                            <span className="flex items-center gap-1">
   413→                              <Calendar className="h-3 w-3" /> {demo.sentDate}
   414→                              {daysSince !== null && <span className="ml-0.5">({daysSince}{t(locale, "dash.daysAgo")})</span>}
   415→                            </span>
   416→                          )}
   417→                          {demo.link && (
   418→                            <span className="flex items-center gap-1 font-mono truncate max-w-[200px]">
   419→                              <Link2 className="h-3 w-3" /> {demo.link}
   420→                            </span>
   421→                          )}
   422→                          {demo.pitchText && (
   423→                            <span className="flex items-center gap-1 text-primary/50">
   424→                              <FileText className="h-3 w-3" /> {t(locale, "demos.hasPitch")}
   425→                            </span>
   426→                          )}
   427→                        </div>
   428→                      </div>
   429→                      <div className="flex items-center gap-1 shrink-0">
   430→                        <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); setDetailDemo(demo); }} title={t(locale, "demos.viewDetail")}>
   431→                          <Eye className="h-3.5 w-3.5" />
   432→                        </Button>
   433→                        {canAdvance(demo.status) && (
   434→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); advanceDemoStatus(demo.id); }} title={t(locale, "demos.advance")}>
   435→                            <ChevronRight className="h-3.5 w-3.5" />
   436→                          </Button>
   437→                        )}
   438→                        <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); openEdit(demo); }}>
   439→                          <Pencil className="h-3.5 w-3.5" />
   440→                        </Button>
   441→                      </div>
   442→                    </div>
   443→                  </CardContent>
   444→                </Card>
   445→              );
   446→            })}
   447→          </div>
   448→        )}
   449→
   450→        {/* Detail Dialog (reused) */}
   451→        <DemoDetailDialog demo={detailDemo} onClose={() => setDetailDemo(null)} locale={locale} getLabelName={getLabelName} />
   452→      </div>
   453→    );
   454→  }
   455→
   456→  return (
   457→    <div className="space-y-4">
   458→      {/* Controls */}
   459→      <div className="flex flex-col sm:flex-row gap-3">
   460→        <div className="relative flex-1">
   461→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   462→          <Input
   463→            placeholder={t(locale, "demos.search")}
   464→            value={search}
   465→            onChange={(e) => setSearch(e.target.value)}
   466→            className="pl-9 bg-secondary/50 border-border/50"
   467→          />
   468→        </div>
   469→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   470→          <SelectTrigger className="w-full sm:w-[170px] bg-secondary/50 border-border/50">
   471→            <SelectValue placeholder={t(locale, "demos.allStatuses")} />
   472→          </SelectTrigger>
   473→          <SelectContent>
   474→            <SelectItem value="all">{t(locale, "demos.allStatuses")}</SelectItem>
   475→            {STATUS_KEYS.map((s) => (
   476→              <SelectItem key={s} value={s}>
   477→                {t(locale, STATUS_TKEYS[s])}
   478→              </SelectItem>
   479→            ))}
   480→          </SelectContent>
   481→        </Select>
   482→        <div className="flex gap-2">
   483→          <Button variant={viewMode === "kanban" ? "default" : "outline"} size="sm" onClick={() => setViewMode("kanban")} className="text-xs">
   484→            {t(locale, "demos.kanban")}
   485→          </Button>
   486→          <Button variant={viewMode === "table" ? "default" : "outline"} size="sm" onClick={() => setViewMode("table")} className="text-xs">
   487→            {t(locale, "demos.table")}
   488→          </Button>
   489→        </div>
   490→        <Button onClick={openAdd} className="glow-purple shrink-0">
   491→          <Plus className="h-4 w-4 mr-1.5" />
   492→          {t(locale, "demos.addDemo")}
   493→        </Button>
   494→      </div>
   495→
