   221→   221→/**
   222→   222→ * Get the time cutoff for a period (in milliseconds from now).
   223→   223→ */
   224→   224→function getPeriodCutoff(period: RankingTimePeriod): number {
   225→   225→  const now = Date.now();
   226→   226→  switch (period) {
   227→   227→    case "current": return now; // Only current data
   228→   228→    case "1m": return now - 30 * 24 * 60 * 60 * 1000;
   229→   229→    case "3m": return now - 90 * 24 * 60 * 60 * 1000;
   230→   230→    case "1y": return now - 365 * 24 * 60 * 60 * 1000;
   231→   231→    case "all": return 0; // All time
   232→   232→  }

   235→   235→// ==================== AGGREGATION LOGIC ====================
   236→   236→
   237→   237→/**
   238→   238→ * Aggregate ranking snapshots for a given genre and time period.
   239→   239→ * Returns a map of labelName -> { totalPoints, bestRank, snapshotCount, averageRank }
   240→   240→ */
   241→   241→function aggregateSnapshots(
   242→   242→  snapshots: RankingSnapshot[],
   243→   243→  genre: string,
   244→   244→  period: RankingTimePeriod
   245→   245→): Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }> {
   246→   246→  const cutoff = getPeriodCutoff(period);
   247→   247→  const result = new Map<string, { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number }>();
   248→   248→
   249→   249→  // Helper to get or create entry with defaults
   250→   250→  function getOrCreate(name: string): { totalPoints: number; bestRank: number; snapshotCount: number; averageRank: number; rankSum: number } {
   251→   251→    let entry = result.get(name);
   252→   252→    if (!entry) {
   253→   253→      entry = { totalPoints: 0, bestRank: 999, snapshotCount: 0, averageRank: 0, rankSum: 0 };
   254→   254→      result.set(name, entry);
   255→   255→    }
   256→   256→    return entry;
   257→   257→  }
   258→   258→
   259→   259→  for (const snapshot of snapshots) {
   260→   260→    if (new Date(snapshot.timestamp).getTime() < cutoff) continue;
   261→   261→    if (period === "current") continue; // Current period doesn't use snapshots
   262→   262→
   263→   263→    const genreData = snapshot.genres[genre];
   264→   264→    if (!genreData) continue;
   265→   265→
   266→   266→    for (const [labelName, data] of Object.entries(genreData)) {
   267→   267→      const existing = getOrCreate(labelName);
   268→   268→      existing.totalPoints += data.points;
   269→   269→      existing.rankSum += data.rank;
   270→   270→      existing.snapshotCount++;
   271→   271→      if (data.rank < existing.bestRank) existing.bestRank = data.rank;
   272→   272→    }
   273→   273→  }
   274→   274→
   275→   275→  // Calculate average rank
   276→   276→  for (const [, val] of result) {
   277→   277→    (val as any).averageRank = Math.round(val.rankSum / val.snapshotCount);
   278→   278→  }
   279→   279→
   280→   280→  return result;
   281→   281→}
   282→   282→
   283→   283→/**
   284→   284→ * Build ranked list for a genre, considering the time period.
   285→   285→ * - "current": uses live label data (rankByGenre)
   286→   286→ * - Other periods: aggregates snapshots and re-ranks by cumulative points
   287→   287→ */
   288→   288→function buildRankedList(
   289→   289→  labels: Label[],
   290→   290→  snapshots: RankingSnapshot[],
   291→   291→  genre: string,
   292→   292→  period: RankingTimePeriod
   293→   293→): RankedLabel[] {
   294→   294→  if (period === "current") {
   295→   295→    // Current view: use live label data
   296→   296→    const ranked: RankedLabel[] = [];
   297→   297→    for (const label of labels) {
   298→   298→      const rank = label.rankByGenre?.[genre];
   299→   299→      if (rank === undefined || rank === null) continue;
   300→   300→
   301→   301→      const prevRank = label.prevRankByGenre?.[genre] ?? null;
   302→   302→      const points = label.pointsByGenre?.[genre] ?? 0;
   303→   303→      let movement: number | null = null;
   304→   304→
   305→   305→      if (prevRank !== null && prevRank !== undefined) {
   306→   306→        movement = prevRank - rank;
   307→   307→      } else if (Object.keys(label.prevRankByGenre || {}).length > 0) {
   308→   308→        movement = null;
   309→   309→      }
   310→   310→
   311→   311→      ranked.push({ label, rank, prevRank, points, movement, snapshotCount: 1 });
   312→   312→    }
   313→   313→    return ranked;
   314→   314→  }
   315→   315→
   316→   316→  // Historical/cumulative view: aggregate snapshots
   317→   317→  const aggregated = aggregateSnapshots(snapshots, genre, period);
   318→   318→  if (aggregated.size === 0) return [];
   319→   319→
   320→   320→  // Sort by total points descending to determine cumulative rank
   321→   321→  const sorted = Array.from(aggregated.entries()).sort((a, b) => b[1].totalPoints - a[1].totalPoints);
   322→   322→
   323→   323→  // For movement in cumulative view, compare with the current live data
   324→   324→  const ranked: RankedLabel[] = [];
   325→   325→  for (let i = 0; i < sorted.length; i++) {
   326→   326→    const [labelName, data] = sorted[i];
   327→   327→    const cumulativeRank = i + 1;
   328→   328→
   329→   329→    // Find matching label in current data
   330→   330→    const matchingLabel = labels.find(l => l.name === labelName);
   331→   331→    const currentRank = matchingLabel?.rankByGenre?.[genre] ?? null;
   332→   332→
   333→   333→    // Movement: compare current live rank with cumulative rank
   334→   334→    let movement: number | null = null;
   335→   335→    if (currentRank !== null) {
   336→   336→      movement = currentRank - cumulativeRank; // positive = label is better live than cumulative (hot right now)
   337→   337→    }
   338→   338→
   339→   339→    // Use the label if found, otherwise create a minimal reference
   340→   340→    const label = matchingLabel || {
   341→   341→      id: `snapshot_${labelName.toLowerCase().replace(/[^a-z0-9]+/g, '_')}`,
   342→   342→      name: labelName,
   343→   343→      genre: genre,
   344→   344→      submissionType: "email" as const,
   345→   345→      contactInfo: "",
   346→   346→      status: "open" as const,
   347→   347→      notes: "",
   348→   348→      createdAt: new Date().toISOString(),
   349→   349→      emails: [],
   350→   350→      website: "",
   351→   351→      demoLink: "",
   352→   352→      socialLink: "",
   353→   353→      soundcloudLink: "",
   354→   354→      beatportLink: "",
   355→   355→      customLinks: [],
   356→   356→      genres: [genre],
   357→   357→      rankByGenre: { [genre]: currentRank || 999 },
   358→   358→      pointsByGenre: {},
   359→   359→      trending: false,
   360→   360→      trendingRankByGenre: {},
   361→   361→      trendingPointsByGenre: {},
   362→   362→      prevRankByGenre: {},
   363→   363→    };
   364→   364→
   365→   365→    ranked.push({
   366→   366→      label,
   367→   367→      rank: cumulativeRank,
   368→   368→      prevRank: currentRank,
   369→   369→      points: data.totalPoints,
   370→   370→      movement,
   371→   371→      snapshotCount: data.snapshotCount,
   372→   372→    });
   373→   373→  }
   374→   374→
   375→   375→  return ranked;
   376→   376→}

   380→   380→export function RankingsPage() {
   381→   381→  const { labels, locale, rankingsUpdatedAt, rankingSnapshots } = useAppStore();
   382→   382→  const snapshots = rankingSnapshots || []; // defensive: might be undefined for existing users
   383→   383→  const [selectedGenre, setSelectedGenre] = useState<string | null>(null);
   384→   384→  const [sortMode, setSortMode] = useState<SortMode>("rank");
   385→   385→  const [movementFilter, setMovementFilter] = useState<MovementFilter>("all");
   386→   386→  const [timePeriod, setTimePeriod] = useState<RankingTimePeriod>("current");
   387→   387→  const [showGenreDropdown, setShowGenreDropdown] = useState(false);
   388→   388→  const [spotlightShowAll, setSpotlightShowAll] = useState(false); // when true + genre selected, show global risers instead of genre-filtered
   389→   389→
   390→   390→  // Get all genres from labels, sorted alphabetically
   391→   391→  const allGenres = useMemo(() => {
   392→   392→    const genreSet = new Set<string>();
   393→   393→    for (const label of labels) {
   394→   394→      if (label.genres) {
   395→   395→        for (const g of label.genres) {
   396→   396→          genreSet.add(g);
   397→   397→        }
   398→   398→      }
   399→   399→    }
   400→   400→    return Array.from(genreSet).sort((a, b) => a.localeCompare(b));
   401→   401→  }, [labels]);
   402→   402→
   403→   403→  // Build ranked list for selected genre and period
   404→   404→  const rankedList = useMemo((): RankedLabel[] => {
   405→   405→    if (!selectedGenre) return [];
   406→   406→    return buildRankedList(labels, snapshots, selectedGenre, timePeriod);
   407→   407→  }, [labels, snapshots, selectedGenre, timePeriod]);
   408→   408→
   409→   409→  // Apply filters
   410→   410→  const filteredList = useMemo(() => {
   411→   411→    let result = [...rankedList];
   412→   412→
   413→   413→    // Movement filter
   414→   414→    switch (movementFilter) {
   415→   415→      case "rising":
   416→   416→        result = result.filter((l) => l.movement !== null && l.movement > 0);
   417→   417→        break;
   418→   418→      case "falling":
   419→   419→        result = result.filter((l) => l.movement !== null && l.movement < 0);
   420→   420→        break;
   421→   421→      case "new":
   422→   422→        result = result.filter((l) => l.movement === null);
   423→   423→        break;
   424→   424→      case "stable":
   425→   425→        result = result.filter((l) => l.movement === 0);
   426→   426→        break;
   427→   427→    }
   428→   428→
   429→   429→    // Sort
   430→   430→    switch (sortMode) {
   431→   431→      case "rank":
   432→   432→        result.sort((a, b) => a.rank - b.rank);
   433→   433→        break;
   434→   434→      case "movement":
   435→   435→        result.sort((a, b) => {
   436→   436→          if (a.movement === null && b.movement === null) return a.rank - b.rank;
   437→   437→          if (a.movement === null) return -1;
   438→   438→          if (b.movement === null) return 1;
   439→   439→          return b.movement - a.movement;
   440→   440→        });
   441→   441→        break;
   442→   442→      case "points":
   443→   443→        result.sort((a, b) => b.points - a.points);
   444→   444→        break;
   445→   445→    }
   446→   446→
   447→   447→    return result;
   448→   448→  }, [rankedList, movementFilter, sortMode]);
   449→   449→
   450→   450→  // Stats summary
   451→   451→  const stats = useMemo(() => {
   452→   452→    const total = rankedList.length;
   453→   453→    const rising = rankedList.filter((l) => l.movement !== null && l.movement > 0).length;
   454→   454→    const falling = rankedList.filter((l) => l.movement !== null && l.movement < 0).length;
   455→   455→    const newEntries = rankedList.filter((l) => l.movement === null).length;
   456→   456→
   457→   457→    return { total, rising, falling, newEntries };
   458→   458→  }, [rankedList]);
   459→   459→
   460→   460→  // Top rising labels across ALL genres (for the global spotlight section)
   461→   461→  const topRisers = useMemo(() => {
   462→   462→    const risers: { label: Label; genre: string; movement: number; rank: number }[] = [];
   463→   463→
   464→   464→    for (const label of labels) {
   465→   465→      if (!label.prevRankByGenre || Object.keys(label.prevRankByGenre).length === 0) continue;
