   105→// ==================== ROBUST STORAGE ====================
   106→// Custom storage with data protection:
   107→// 1. Primary localStorage key
   108→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)
   109→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
   110→// 4. Rehydration guard: blocks ALL writes until Zustand persist loads data
   111→
   112→const PRIMARY_KEY = "labelpulse-storage";
   113→const BACKUP_KEY = "labelpulse-storage-backup";
   114→const SEED_LABEL_COUNT = labelData.labels.length;
   115→
   116→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   117→// Before rehydration, the store contains seed data which would overwrite user data.
   118→let _rehydrated = false;
   119→
   120→function safeJsonParse(str: string | null): any | null {
   121→  if (!str) return null;
   122→  try {
   123→    return JSON.parse(str);
   124→  } catch {
   125→    return null;
   126→  }
   127→}
   128→
   129→function safeLocalStorageGet(key: string): string | null {
   130→  try {
   131→    return localStorage.getItem(key);
   132→  } catch {
   133→    return null;
   134→  }
   135→}
   136→
   137→function safeLocalStorageSet(key: string, value: string): boolean {
   138→  try {
   139→    localStorage.setItem(key, value);
   140→    return true;
   141→  } catch (e) {
   142→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   143→    return false;
   144→  }
   145→}
   146→
   147→function safeLocalStorageRemove(key: string): void {
   148→  try {
   149→    localStorage.removeItem(key);
   150→  } catch {
   151→    // Ignore
   152→  }
   153→}

   155→/**
   156→ * Check if persisted data has user edits (emails, notes, links, etc.)
   157→ * This helps us detect if the store was accidentally reset to seed data.
   158→ */
   159→function countUserEditedLabels(labels: any[]): number {
   160→  if (!Array.isArray(labels)) return 0;
   161→  return labels.filter((l) =>
   162→    (l.emails && l.emails.length > 0) ||
   163→    (l.notes && l.notes.trim() !== "") ||
   164→    (l.website && l.website.trim() !== "") ||
   165→    (l.demoLink && l.demoLink.trim() !== "") ||
   166→    (l.socialLink && l.socialLink.trim() !== "") ||
   167→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   168→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   169→    l.isCustom === true
   170→  ).length;
   171→}
   172→
   173→function getTotalLabelCount(data: any): number {
   174→  if (!data) return 0;
   175→  if (Array.isArray(data.labels)) return data.labels.length;
   176→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   177→  return 0;
   178→}

   180→/**
   181→ * Extract user-edited label count from raw localStorage string.
   182→ * Used by the write guard to check if a write would cause data loss.
   183→ */
   184→function getUserEditCountFromRaw(raw: string | null): number {
   185→  if (!raw) return 0;
   186→  const data = safeJsonParse(raw);
   187→  if (!data) return 0;
   188→  const labels = data.state?.labels || data.labels || [];
   189→  return countUserEditedLabels(labels);
   190→}

   192→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   193→
   194→const robustStorage: StateStorage = {
   195→  getItem: (name: string): string | null => {
   196→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   197→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   198→
   199→    const primaryData = safeJsonParse(primaryRaw);
   200→    const backupData = safeJsonParse(backupRaw);
   201→
   202→    let result: string | null = null;
   203→
   204→    // If primary is missing/corrupt, try backup
   205→    if (!primaryData) {
   206→      if (backupData) {
   207→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   208→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   209→        result = backupRaw;
   210→      }
   211→      // Both empty — first visit, result stays null
   212→    } else if (backupData) {
   213→      // Both exist — check for data loss
   214→      const primaryLabelCount = getTotalLabelCount(primaryData);
   215→      const backupLabelCount = getTotalLabelCount(backupData);
   216→      const primaryUserEdits = primaryData.state?.labels
   217→        ? countUserEditedLabels(primaryData.state.labels)
   218→        : countUserEditedLabels(primaryData.labels || []);
   219→      const backupUserEdits = backupData.state?.labels
   220→        ? countUserEditedLabels(backupData.state.labels)
   221→        : countUserEditedLabels(backupData.labels || []);
   222→
   223→      // If primary has fewer labels than backup AND backup has more or equal user data
   224→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   225→        console.warn(
   226→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   227→        );
   228→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   229→        result = backupRaw;
   230→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   231→        // If primary has fewer user edits than backup, use backup
   232→        console.warn(
   233→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   234→        );
   235→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   236→        result = backupRaw;
   237→      } else {
   238→        result = primaryRaw;
   239→      }
   240→    } else {
   241→      // Primary exists, no backup — normal case
   242→      result = primaryRaw;
   243→    }
   244→
   245→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   246→    // merged data immediately after this getItem returns, and we MUST allow that write.
   247→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   248→    _rehydrated = true;
   249→
   250→    return result;
   251→  },
   252→
   253→  setItem: (name: string, value: string): void => {
   254→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   255→    // Before rehydration, the store only has seed data — writing it would overwrite
   256→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   257→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   258→    if (!_rehydrated) {
   259→      console.warn(
   260→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   261→      );
   262→      return;
   263→    }
   264→
   265→    // Write to primary immediately
   266→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   267→
   268→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   269→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   270→    safeLocalStorageSet(BACKUP_KEY, value);
   271→
   272→    if (!primaryOk) {
   273→      console.error("[LabelPulse Storage] Primary write failed!");
   274→    }
   275→  },
   276→
   277→  removeItem: (name: string): void => {
   278→    safeLocalStorageRemove(PRIMARY_KEY);
   279→    safeLocalStorageRemove(BACKUP_KEY);
   280→  },
   281→};
   282→
   283→/**
   284→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   285→ * Since backup is now written on every setItem, this is a safety net.
   286→ */
   287→export function forceBackupNow(): void {
   288→  if (typeof window === "undefined") return;
   289→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   290→  if (primaryRaw) {
   291→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   292→  }
   293→}

   362→/**
   363→ * Merge helper: preserves ALL user-editable fields from existing label
   364→ * when imported label has empty/default values for those fields.
   365→ */
   366→function mergePreservingUserData(existing: Label, imported: Label): Partial<Label> {
   367→  // Before overwriting ranking data, save the current rank as "previous" snapshot.
   368→  // This allows the Rankings page to show movement arrows (↑↓) between imports.
   369→  const hasNewRankData = Object.keys(imported.rankByGenre || {}).length > 0;
   370→  const currentRank = hasNewRankData && Object.keys(existing.rankByGenre || {}).length > 0
   371→    ? { ...existing.rankByGenre }
   372→    : existing.prevRankByGenre || {};
   373→
   374→  return {
   375→    // User-editable data — ALWAYS prefer existing if it has real data
   376→    emails: imported.emails?.length ? imported.emails : existing.emails,
   377→    contactInfo: imported.contactInfo?.trim() || existing.contactInfo,
   378→    website: imported.website?.trim() || existing.website,
   379→    demoLink: imported.demoLink?.trim() || existing.demoLink,
   380→    socialLink: imported.socialLink?.trim() || existing.socialLink,
   381→    soundcloudLink: imported.soundcloudLink?.trim() || existing.soundcloudLink,
   382→    notes: imported.notes?.trim() || existing.notes,
   383→    status: (imported.status === "open" || imported.status === "closed") ? imported.status : (existing.status || "open"),
   384→
   385→    // Beatport/ranking data — prefer imported (it's the fresh data)
   386→    genres: imported.genres?.length ? imported.genres : existing.genres,
   387→    rankByGenre: Object.keys(imported.rankByGenre || {}).length ? imported.rankByGenre : existing.rankByGenre,
   388→    pointsByGenre: Object.keys(imported.pointsByGenre || {}).length ? imported.pointsByGenre : existing.pointsByGenre,
   389→    trending: imported.trending || existing.trending,
   390→    trendingRankByGenre: Object.keys(imported.trendingRankByGenre || {}).length ? imported.trendingRankByGenre : existing.trendingRankByGenre,
   391→    trendingPointsByGenre: Object.keys(imported.trendingPointsByGenre || {}).length ? imported.trendingPointsByGenre : existing.trendingPointsByGenre,
   392→
   393→    // Ranking history — save current rank as "previous" when new data arrives
   394→    prevRankByGenre: currentRank,
   395→  };
   396→}

   422→/**
   423→ * Repair corrupted label data.
   424→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   425→ */
   426→function repairLabelData(labels: Label[]): Label[] {
   427→  const emailLabelCount = new Map<string, number>();
   428→  for (const l of labels) {
   429→    if (l.emails && Array.isArray(l.emails)) {
   430→      for (const e of l.emails) {
   431→        if (e && e.trim()) {
   432→          const key = e.toLowerCase().trim();
   433→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   434→        }
   435→      }
   436→    }
   437→  }
   438→
   439→  const corruptedEmails = new Set<string>();
   440→  for (const [email, count] of emailLabelCount) {
   441→    if (count > 5) {
   442→      corruptedEmails.add(email);
   443→      console.warn(
   444→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   445→      );
   446→    }
   447→  }
   448→
   449→  if (corruptedEmails.size === 0) {
   450→    return labels;
   451→  }
   452→
   453→  const emailOwner = new Map<string, string>();
   454→  for (const email of corruptedEmails) {
   455→    for (const l of labels) {
   456→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   457→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   458→        if (domain && l.name.toLowerCase().includes(domain)) {
   459→          emailOwner.set(email, l.id);
   460→          break;
   461→        }
   462→      }
   463→    }
   464→    if (!emailOwner.has(email)) {
   465→      for (const l of labels) {
   466→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   467→          emailOwner.set(email, l.id);
   468→          break;
   469→        }
   470→      }
   471→    }
   472→  }
   473→
   474→  return labels.map(l => {
   475→    if (!l.emails || !Array.isArray(l.emails)) return l;
   476→    const cleanedEmails = l.emails.filter(e => {
   477→      const key = e?.toLowerCase().trim();
   478→      if (!key) return false;
   479→      if (!corruptedEmails.has(key)) return true;
   480→      return emailOwner.get(key) === l.id;
   481→    });
   482→    const cleanedContactInfo = cleanedEmails[0] || "";
   483→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   484→      console.log(
   485→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   486→      );
   487→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   488→    }
   489→    return l;
   490→  });
   491→}

   984→// Helper: get tier for a label in a given genre
   985→export function getLabelTier(
   986→  label: Label,
   987→  genre?: string
   988→): "top" | "mid" | "emerging" | null {
   989→  if (!genre) genre = label.genres?.[0];
   990→  if (!genre) return null;
   991→  const rank = label.rankByGenre?.[genre];
   992→  if (!rank) return label.trending ? "emerging" : null;
   993→  if (rank <= 20) return "top";
   994→  if (rank <= 50) return "mid";
   995→  return label.trending ? "emerging" : null;
   996→}

   998→// ==================== CLOUD SYNC ====================
   999→// Sistema di sincronizzazione con Supabase.

  1007→/**
  1008→ * Sincronizza lo stato corrente con Supabase (debounced).
  1009→ * Chiamata dopo ogni azione utente che modifica i dati.
  1010→ */
  1011→export function syncToCloud(): void {
  1012→  if (!isSupabaseConfigured) return;

  1014→  if (_cloudSyncTimer) {
  1015→    clearTimeout(_cloudSyncTimer);

  1019→    _cloudSyncTimer = null;

  1022→      labels: state.labels,
  1023→      demos: state.demos,

  1029→      lastSavedAt: state.lastSavedAt,
  1030→      rankingSnapshots: state.rankingSnapshots,
  1031→    };
  1032→    await saveStateToCloud(dataToSync);
  1033→  }, CLOUD_SYNC_DEBOUNCE_MS);
  1034→}

  1065→ * Chiamata una volta all'avvio dell'app, dopo la reidratazione da localStorage.
  1066→ */
  1067→export async function loadFromCloud(): Promise<void> {
  1068→  if (!isSupabaseConfigured) {
  1069→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
  1070→    useAppStore.setState({ hasCloudSynced: true });
  1071→    return;

  1078→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
  1079→      await forceCloudSync();
  1080→      useAppStore.setState({ hasCloudSynced: true });
  1081→      return;
  1082→    }
  1083→
  1084→    const cloud = cloudData as any;
  1085→    const localState = useAppStore.getState();
  1086→
  1087→    // Confronta i timestamp per decidere quali dati usare
  1088→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
  1089→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
  1090→
  1091→    if (cloudSavedAt > localSavedAt) {
  1092→      // I dati cloud sono più recenti — aggiorna lo store locale
  1093→      console.log(
  1094→        `[LabelPulse Cloud] Cloud data is newer (cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}). Updating local store.`
  1095→      );
  1096→
  1097→      // Usa la stessa logica di merge della reidratazione
  1098→      const merged = mergeCloudData(cloud, localState);
  1099→      useAppStore.setState({
  1100→        ...merged,
  1101→        hasCloudSynced: true,
  1102→      });
  1103→    } else {
  1104→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1105→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1106→      useAppStore.setState({ hasCloudSynced: true });
  1107→    }
  1108→  } catch (err) {

  1110→    useAppStore.setState({ hasCloudSynced: true });
  1111→  }
  1112→}
