    89→// ==================== ROBUST STORAGE ====================
    90→// Custom storage with data protection:
    91→// 1. Primary localStorage key
    92→// 2. Backup localStorage key (written IMMEDIATELY on every save — no debounce)
    93→// 3. Data integrity check on load (auto-recover from backup if data loss detected)
    94→// 4. Rehydration guard: blocks ALL writes until Zustand persist loads data

    96→const PRIMARY_KEY = "labelpulse-storage";
    97→const BACKUP_KEY = "labelpulse-storage-backup";
    98→const SEED_LABEL_COUNT = labelData.labels.length;

   100→// Rehydration guard: blocks ALL writes until Zustand persist has loaded data from localStorage.
   101→// Before rehydration, the store contains seed data which would overwrite user data.
   102→let _rehydrated = false;
   103→
   104→function safeJsonParse(str: string | null): any | null {
   105→  if (!str) return null;
   106→  try {
   107→    return JSON.parse(str);
   108→  } catch {
   109→    return null;
   110→  }
   111→}
   112→
   113→function safeLocalStorageGet(key: string): string | null {
   114→  try {
   115→    return localStorage.getItem(key);
   116→  } catch {
   117→    return null;
   118→  }
   119→}
   120→
   121→function safeLocalStorageSet(key: string, value: string): boolean {
   122→  try {
   123→    localStorage.setItem(key, value);
   124→    return true;
   125→  } catch (e) {
   126→    console.error(`[LabelPulse Storage] Failed to write ${key}:`, e);
   127→    return false;
   128→  }
   129→}
   130→
   131→function safeLocalStorageRemove(key: string): void {
   132→  try {
   133→    localStorage.removeItem(key);
   134→  } catch {
   135→    // Ignore
   136→  }
   137→}

   139→/**
   140→ * Check if persisted data has user edits (emails, notes, links, etc.)
   141→ * This helps us detect if the store was accidentally reset to seed data.
   142→ */
   143→function countUserEditedLabels(labels: any[]): number {
   144→  if (!Array.isArray(labels)) return 0;
   145→  return labels.filter((l) =>
   146→    (l.emails && l.emails.length > 0) ||
   147→    (l.notes && l.notes.trim() !== "") ||
   148→    (l.website && l.website.trim() !== "") ||
   149→    (l.demoLink && l.demoLink.trim() !== "") ||
   150→    (l.socialLink && l.socialLink.trim() !== "") ||
   151→    (l.soundcloudLink && l.soundcloudLink.trim() !== "") ||
   152→    (l.contactInfo && l.contactInfo.trim() !== "") ||
   153→    l.isCustom === true
   154→  ).length;
   155→}
   156→
   157→function getTotalLabelCount(data: any): number {
   158→  if (!data) return 0;
   159→  if (Array.isArray(data.labels)) return data.labels.length;
   160→  if (data.state && Array.isArray(data.state.labels)) return data.state.labels.length;
   161→  return 0;
   162→}
   163→
   164→/**
   165→ * Extract user-edited label count from raw localStorage string.
   166→ * Used by the write guard to check if a write would cause data loss.
   167→ */
   168→function getUserEditCountFromRaw(raw: string | null): number {
   169→  if (!raw) return 0;
   170→  const data = safeJsonParse(raw);
   171→  if (!data) return 0;
   172→  const labels = data.state?.labels || data.labels || [];
   173→  return countUserEditedLabels(labels);
   174→}

   176→// (debounced backup removed — we now write backup IMMEDIATELY on every write for maximum data safety)
   177→
   178→const robustStorage: StateStorage = {
   179→  getItem: (name: string): string | null => {
   180→    const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   181→    const backupRaw = safeLocalStorageGet(BACKUP_KEY);
   182→
   183→    const primaryData = safeJsonParse(primaryRaw);
   184→    const backupData = safeJsonParse(backupRaw);
   185→
   186→    let result: string | null = null;
   187→
   188→    // If primary is missing/corrupt, try backup
   189→    if (!primaryData) {
   190→      if (backupData) {
   191→        console.warn("[LabelPulse Storage] Primary data missing/corrupt, restoring from backup");
   192→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   193→        result = backupRaw;
   194→      }
   195→      // Both empty — first visit, result stays null
   196→    } else if (backupData) {
   197→      // Both exist — check for data loss
   198→      const primaryLabelCount = getTotalLabelCount(primaryData);
   199→      const backupLabelCount = getTotalLabelCount(backupData);
   200→      const primaryUserEdits = primaryData.state?.labels
   201→        ? countUserEditedLabels(primaryData.state.labels)
   202→        : countUserEditedLabels(primaryData.labels || []);
   203→      const backupUserEdits = backupData.state?.labels
   204→        ? countUserEditedLabels(backupData.state.labels)
   205→        : countUserEditedLabels(backupData.labels || []);
   206→
   207→      // If primary has fewer labels than backup AND backup has more or equal user data
   208→      if (backupLabelCount > primaryLabelCount && backupUserEdits >= primaryUserEdits) {
   209→        console.warn(
   210→          `[LabelPulse Storage] DATA LOSS DETECTED! Primary: ${primaryLabelCount} labels (${primaryUserEdits} edited), Backup: ${backupLabelCount} labels (${backupUserEdits} edited). Restoring from backup.`
   211→        );
   212→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   213→        result = backupRaw;
   214→      } else if (backupUserEdits > primaryUserEdits && backupLabelCount >= primaryLabelCount) {
   215→        // If primary has fewer user edits than backup, use backup
   216→        console.warn(
   217→          `[LabelPulse Storage] More user data in backup (${backupUserEdits} vs ${primaryUserEdits} edited labels). Restoring from backup.`
   218→        );
   219→        if (backupRaw) safeLocalStorageSet(PRIMARY_KEY, backupRaw);
   220→        result = backupRaw;
   221→      } else {
   222→        result = primaryRaw;
   223→      }
   224→    } else {
   225→      // Primary exists, no backup — normal case
   226→      result = primaryRaw;
   227→    }
   228→
   229→    // CRITICAL: Mark rehydrated BEFORE returning — Zustand will call setItem with
   230→    // merged data immediately after this getItem returns, and we MUST allow that write.
   231→    // This ensures seed data (initial state) writes are blocked, but merged data writes pass.
   232→    _rehydrated = true;
   233→
   234→    return result;
   235→  },
   236→
   237→  setItem: (name: string, value: string): void => {
   238→    // REHYDRATION GUARD: Block ALL writes before Zustand persist has loaded data.
   239→    // Before rehydration, the store only has seed data — writing it would overwrite
   240→    // any user data in localStorage. After rehydration, ALL writes are legitimate
   241→    // (user actions, repairs, imports, etc.) and must ALWAYS be saved.
   242→    if (!_rehydrated) {
   243→      console.warn(
   244→        `[LabelPulse Storage] BLOCKED pre-rehydration write — protecting user data in localStorage`
   245→      );
   246→      return;
   247→    }
   248→
   249→    // Write to primary immediately
   250→    const primaryOk = safeLocalStorageSet(PRIMARY_KEY, value);
   251→
   252→    // Write to backup IMMEDIATELY (no debounce) for maximum data safety.
   253→    // Previous 60s debounce caused data loss if browser crashed before backup was written.
   254→    safeLocalStorageSet(BACKUP_KEY, value);
   255→
   256→    if (!primaryOk) {
   257→      console.error("[LabelPulse Storage] Primary write failed!");
   258→    }
   259→  },
   260→
   261→  removeItem: (name: string): void => {
   262→    safeLocalStorageRemove(PRIMARY_KEY);
   263→    safeLocalStorageRemove(BACKUP_KEY);
   264→  },
   265→};
   266→
   267→/**
   268→ * Force-write a backup immediately (e.g., on visibility change or before unload).
   269→ * Since backup is now written on every setItem, this is a safety net.
   270→ */
   271→export function forceBackupNow(): void {
   272→  if (typeof window === "undefined") return;
   273→  const primaryRaw = safeLocalStorageGet(PRIMARY_KEY);
   274→  if (primaryRaw) {
   275→    safeLocalStorageSet(BACKUP_KEY, primaryRaw);
   276→  }
   277→}

   391→/**
   392→ * Repair corrupted label data.
   393→ * Detects and fixes: same email appearing on too many labels (likely corruption).
   394→ */
   395→function repairLabelData(labels: Label[]): Label[] {
   396→  const emailLabelCount = new Map<string, number>();
   397→  for (const l of labels) {
   398→    if (l.emails && Array.isArray(l.emails)) {
   399→      for (const e of l.emails) {
   400→        if (e && e.trim()) {
   401→          const key = e.toLowerCase().trim();
   402→          emailLabelCount.set(key, (emailLabelCount.get(key) || 0) + 1);
   403→        }
   404→      }
   405→    }
   406→  }
   407→
   408→  const corruptedEmails = new Set<string>();
   409→  for (const [email, count] of emailLabelCount) {
   410→    if (count > 5) {
   411→      corruptedEmails.add(email);
   412→      console.warn(
   413→        `[LabelPulse Repair] Email "${email}" found on ${count} labels — likely corruption, will be removed from all but the owner.`
   414→      );
   415→    }
   416→  }
   417→
   418→  if (corruptedEmails.size === 0) {
   419→    return labels;
   420→  }
   421→
   422→  const emailOwner = new Map<string, string>();
   423→  for (const email of corruptedEmails) {
   424→    for (const l of labels) {
   425→      if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   426→        const domain = email.split("@")[1]?.split(".")[0]?.toLowerCase();
   427→        if (domain && l.name.toLowerCase().includes(domain)) {
   428→          emailOwner.set(email, l.id);
   429→          break;
   430→        }
   431→      }
   432→    }
   433→    if (!emailOwner.has(email)) {
   434→      for (const l of labels) {
   435→        if (l.emails?.some(e => e.toLowerCase().trim() === email)) {
   436→          emailOwner.set(email, l.id);
   437→          break;
   438→        }
   439→      }
   440→    }
   441→  }
   442→
   443→  return labels.map(l => {
   444→    if (!l.emails || !Array.isArray(l.emails)) return l;
   445→    const cleanedEmails = l.emails.filter(e => {
   446→      const key = e?.toLowerCase().trim();
   447→      if (!key) return false;
   448→      if (!corruptedEmails.has(key)) return true;
   449→      return emailOwner.get(key) === l.id;
   450→    });
   451→    const cleanedContactInfo = cleanedEmails[0] || "";
   452→    if (cleanedEmails.length !== l.emails.length || cleanedContactInfo !== l.contactInfo) {
   453→      console.log(
   454→        `[LabelPulse Repair] Cleaned label "${l.name}": removed ${l.emails.length - cleanedEmails.length} corrupted email(s)`
   455→      );
   456→      return { ...l, emails: cleanedEmails, contactInfo: cleanedContactInfo };
   457→    }
   458→    return l;
   459→  });
   460→}

   675→          const currentDemoById = new Map(currentDemos.map(d => [d.id, d]));

   728→        } catch {

   733→    {

   760→            }));
   761→          }
   762→          if (persisted.demos) {
   763→            persisted.demos = persisted.demos.map((d: any) => ({

   770→            }));

   774→          }
   775→          if (!persisted.userProfile) {

   777→          }
   778→        }
   779→        if (version < 6) {

   804→      partialize: (state) => ({
   805→        labels: state.labels,
   806→        demos: state.demos,
   807→        activeTab: state.activeTab,
   808→        locale: state.locale,

   814→      merge: (persistedState: any, currentState: any) => {
   815→        // SIMPLE & SAFE merge: preserve persisted data, add defaults, append new seed labels
   816→        // CRITICAL: We must NEVER modify persisted user data during merge.
   817→
   818→        if (!persistedState) {
   819→          return currentState;
   820→        }
   821→
   822→        const merged = { ...currentState, ...persistedState };
   823→
   824→        if (persistedState.labels && Array.isArray(persistedState.labels) && persistedState.labels.length > 0) {
   825→          // Step 1: Add safe defaults to ALL persisted labels (never overwrite existing values)
   826→          merged.labels = persistedState.labels.map((l: any) => ({
   827→            ...LABEL_DEFAULTS,

   829→          }));

   831→          // Step 2: Append NEW seed labels that don't exist in persisted data
   832→          const persistedIds = new Set(persistedState.labels.map((l: any) => l.id));
   833→          const persistedNames = new Set(
   834→            persistedState.labels
   835→              .map((l: any) => l.name?.toLowerCase().trim())
   836→              .filter(Boolean)
   837→          );
   838→          const seedLabels = buildLabelsFromData();
   839→          for (const seed of seedLabels) {
   840→            if (!persistedIds.has(seed.id) && !persistedNames.has(seed.name.toLowerCase().trim())) {
   841→              merged.labels.push(seed);
   842→            }
   843→          }
   844→
   845→          // Step 3: Repair corrupted data (e.g., same email on too many labels)
   846→          merged.labels = repairLabelData(merged.labels);
   847→        }
   848→

   850→      },
   851→      onRehydrateStorage: () => (state, error) => {

   881→// Helper: get tier for a label in a given genre
   882→export function getLabelTier(
   883→  label: Label,
   884→  genre?: string
   885→): "top" | "mid" | "emerging" | null {
   886→  if (!genre) genre = label.genres?.[0];
   887→  if (!genre) return null;
   888→  const rank = label.rankByGenre?.[genre];
   889→  if (!rank) return label.trending ? "emerging" : null;
   890→  if (rank <= 20) return "top";
   891→  if (rank <= 50) return "mid";
   892→  return label.trending ? "emerging" : null;
   893→}

   895→// ==================== CLOUD SYNC ====================
   896→// Sistema di sincronizzazione con Supabase.
   897→// - Ogni modifica ai dati viene salvata su Supabase (debounced)
   898→// - All'avvio, i dati vengono caricati dal cloud se più recenti
   899→// - Se Supabase non è configurato, funziona solo con localStorage
   900→
   901→let _cloudSyncTimer: ReturnType<typeof setTimeout> | null = null;
   902→const CLOUD_SYNC_DEBOUNCE_MS = 3000; // 3 secondi di debounce
   903→
   904→/**
   905→ * Sincronizza lo stato corrente con Supabase (debounced).
   906→ * Chiamata dopo ogni azione utente che modifica i dati.
   907→ */
   908→export function syncToCloud(): void {
   909→  if (!isSupabaseConfigured) return;
   910→
   911→  if (_cloudSyncTimer) {
   912→    clearTimeout(_cloudSyncTimer);
   913→  }
   914→
   915→  _cloudSyncTimer = setTimeout(async () => {
   916→    _cloudSyncTimer = null;
   917→    const state = useAppStore.getState();
   918→    const dataToSync = {
   919→      labels: state.labels,
   920→      demos: state.demos,
   921→      activeTab: state.activeTab,
   922→      locale: state.locale,
   923→      userProfile: state.userProfile,
   924→      gmailAuth: state.gmailAuth,
   925→      rankingsUpdatedAt: state.rankingsUpdatedAt,
   926→      lastSavedAt: state.lastSavedAt,
   927→    };
   928→    await saveStateToCloud(dataToSync);
   929→  }, CLOUD_SYNC_DEBOUNCE_MS);
   930→}

   932→/**
   933→ * Forza la sincronizzazione immediata con Supabae (senza debounce).
   934→ * Usata quando la pagina sta per essere chiusa o nascosta.
   935→ */
   936→export async function forceCloudSync(): Promise<void> {
   937→  if (!isSupabaseConfigured) return;
   938→
   939→  if (_cloudSyncTimer) {
   940→    clearTimeout(_cloudSyncTimer);
   941→    _cloudSyncTimer = null;
   942→  }
   943→
   944→  const state = useAppStore.getState();
   945→  const dataToSync = {
   946→    labels: state.labels,
   947→    demos: state.demos,
   948→    activeTab: state.activeTab,
   949→    locale: state.locale,
   950→    userProfile: state.userProfile,
   951→    gmailAuth: state.gmailAuth,
   952→    rankingsUpdatedAt: state.rankingsUpdatedAt,
   953→    lastSavedAt: state.lastSavedAt,
   954→  };
   955→  await saveStateToCloud(dataToSync);
   956→}

   958→/**
   959→ * Carica i dati dal cloud e aggiorna lo store se i dati cloud sono più recenti.
   960→ * Chiamata una volta all'avvio dell'app, dopo la reidratazione da localStorage.
   961→ */
   962→export async function loadFromCloud(): Promise<void> {
   963→  if (!isSupabaseConfigured) {
   964→    console.log("[LabelPulse Cloud] Supabase not configured, skipping cloud load");
   965→    useAppStore.setState({ hasCloudSynced: true });
   966→    return;
   967→  }
   968→
   969→  try {
   970→    const cloudData = await loadStateFromCloud();
   971→    if (!cloudData) {
   972→      // Nessun dato nel cloud — facciamo il primo upload dei dati locali
   973→      console.log("[LabelPulse Cloud] No cloud data, uploading local data as initial sync");
   974→      await forceCloudSync();
   975→      useAppStore.setState({ hasCloudSynced: true });
   976→      return;
   977→    }
   978→
   979→    const cloud = cloudData as any;
   980→    const localState = useAppStore.getState();
   981→
   982→    // Confronta i timestamp per decidere quali dati usare
   983→    const localSavedAt = localState.lastSavedAt ? new Date(localState.lastSavedAt).getTime() : 0;
   984→    const cloudSavedAt = cloud.lastSavedAt ? new Date(cloud.lastSavedAt).getTime() : 0;
   985→
   986→    if (cloudSavedAt > localSavedAt) {
   987→      // I dati cloud sono più recenti — aggiorna lo store locale
   988→      console.log(
   989→        `[LabelPulse Cloud] Cloud data is newer (cloud: ${cloud.lastSavedAt}, local: ${localState.lastSavedAt}). Updating local store.`
   990→      );
   991→
   992→      // Usa la stessa logica di merge della reidratazione
   993→      const merged = mergeCloudData(cloud, localState);
   994→      useAppStore.setState({
   995→        ...merged,
   996→        hasCloudSynced: true,
   997→      });
   998→    } else {
   999→      // I dati locali sono più recenti o uguali — mantieni i dati locali, aggiorna il cloud
  1000→      console.log("[LabelPulse Cloud] Local data is up to date. Cloud sync complete.");
  1001→      useAppStore.setState({ hasCloudSynced: true });
  1002→    }
  1003→  } catch (err) {
  1004→    console.error("[LabelPulse Cloud] Load from cloud failed:", err);
  1005→    useAppStore.setState({ hasCloudSynced: true });
  1006→  }
  1007→}

  1009→/**
  1010→ * Merge dei dati cloud con quelli locali, preservando i dati utente.
  1011→ * Usa la stessa logica della funzione merge di Zustand persist.
  1012→ */
  1013→function mergeCloudData(cloudData: any, localState: any): Partial<AppState> {
  1014→  const merged: Partial<AppState> = {};
  1015→
  1016→  // Labels: applica defaults e merge con seed data, come nella reidratazione
  1017→  if (cloudData.labels && Array.isArray(cloudData.labels) && cloudData.labels.length > 0) {
  1018→    merged.labels = cloudData.labels.map((l: any) => ({
  1019→      ...LABEL_DEFAULTS,
  1020→      ...l,
  1021→    }));
  1022→
  1023→    // Aggiungi seed labels mancanti
  1024→    const cloudIds = new Set(cloudData.labels.map((l: any) => l.id));
  1025→    const cloudNames = new Set(
  1026→      cloudData.labels.map((l: any) => l.name?.toLowerCase().trim()).filter(Boolean)
  1027→    );
  1028→    const seedLabels = buildLabelsFromData();
  1029→    for (const seed of seedLabels) {
  1030→      if (!cloudIds.has(seed.id) && !cloudNames.has(seed.name.toLowerCase().trim())) {
  1031→        (merged.labels as Label[]).push(seed);
  1032→      }
  1033→    }
  1034→
  1035→    // Repair corrupted data
  1036→    merged.labels = repairLabelData(merged.labels);
  1037→  } else {
  1038→    merged.labels = localState.labels;
  1039→  }
  1040→
  1041→  // Demos
  1042→  merged.demos = Array.isArray(cloudData.demos) ? cloudData.demos : localState.demos;
  1043→
  1044→  // Simple fields
  1045→  merged.activeTab = cloudData.activeTab || localState.activeTab;
  1046→  merged.locale = cloudData.locale || localState.locale;
  1047→  merged.userProfile = cloudData.userProfile || localState.userProfile;
  1048→  merged.gmailAuth = cloudData.gmailAuth || localState.gmailAuth;
  1049→  merged.rankingsUpdatedAt = cloudData.rankingsUpdatedAt ?? localState.rankingsUpdatedAt;
  1050→  merged.lastSavedAt = cloudData.lastSavedAt ?? localState.lastSavedAt;
  1051→
  1052→  return merged;
  1053→}
