    85→    85→function readCredentials(): { url: string; anonKey: string } {
    86→    86→  // ⚠️ CLOUD-FIRST (2026-06-23):
    87→    87→  // Le credenziali Supabase vengono lette SOLO dalle env vars. Il vecchio
    88→    88→  // sistema BYOK (utente inserisce URL+key nel Profilo) è stato rimosso
    89→    89→  // perché era la causa principale della perdita dati: se l'utente non
    90→    90→  // configurava (90% dei casi), niente cloud, e cambiando dispositivo
    91→    91→  // tutto era perso.
    92→    92→  //
    93→    93→  // Ora: configura .env.local una volta, e ogni dispositivo/utente usa
    94→    94→  // le stesse credenziali. Multi-utenza garantita dalla chiave id=email
    95→    95→  // nella tabella app_state (vedi setCurrentUserEmail + schema SQL).
    96→    96→  const envUrl = normalizeSupabaseUrl(
    97→    97→    process.env.NEXT_PUBLIC_SUPABASE_URL || ""
    98→    98→  );
    99→    99→  const envKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || "";
   100→   100→  return { url: envUrl, anonKey: envKey };
   101→   101→}

   103→   103→/**
   104→   104→ * Normalize a user-supplied Supabase URL into the API format
   105→   105→ * (https://<project-ref>.supabase.co).
   106→   106→ *
   107→   107→ * Handles:
   108→   108→ *   - Dashboard URLs: https://supabase.com/dashboard/project/<ref>/...
   109→   109→ *     → https://<ref>.supabase.co
   110→   110→ *   - URLs with trailing slash or path: stripped
   111→   111→ *   - URLs without protocol: https:// prefix added
   112→   112→ *   - URLs that are already correct: returned unchanged
   113→   113→ *
   114→   114→ * Returns "" if the input is empty or doesn't look like a Supabase URL.
   115→   115→ */
   116→   116→export function normalizeSupabaseUrl(input: string): string {
   117→   117→  if (!input) return "";
   118→   118→  let url = input.trim();
   119→   119→
   120→   120→  // Add protocol if missing
   121→   121→  if (!/^https?:\/\//i.test(url)) {
   122→   122→    url = "https://" + url;
   123→   123→  }
   124→   124→
   125→   125→  // Case 1: Dashboard URL → extract project ref
   126→   126→  //   https://supabase.com/dashboard/project/<ref>
   127→   127→  //   https://supabase.com/dashboard/project/<ref>/settings/api
   128→   128→  const dashMatch = url.match(
   129→   129→    /supabase\.com\/dashboard\/project\/([a-z0-9]+)/i
   130→   130→  );
   131→   131→  if (dashMatch) {
   132→   132→    return `https://${dashMatch[1].toLowerCase()}.supabase.co`;
   133→   133→  }
   134→   134→
   135→   135→  // Case 2: Already in API format (https://<ref>.supabase.co[/...])
   136→   136→  const apiMatch = url.match(/^(https?:\/\/)([a-z0-9]+)\.supabase\.(co|com|in)/i);
   137→   137→  if (apiMatch) {
   138→   138→    // Strip any trailing path/slash, keep just the host
   139→   139→    try {
   140→   140→      const u = new URL(url);
   141→   141→      return `${u.protocol}//${u.hostname.toLowerCase()}`;
   142→   142→    } catch {
   143→   143→      return url;
   144→   144→    }
   145→   145→  }
   146→   146→
   147→   147→  // Case 3: Unknown format — return as-is, will fail at fetch time
   148→   148→  // (better to let the user see the error than silently mangle it)
   149→   149→  return url;
   150→   150→}

   152→   152→/**
   153→   153→ * Returns a human-readable validation error for the current credentials,
   154→   154→ * or null if credentials look valid. Used to give the user actionable
   155→   155→ * feedback in the Profilo UI before they even try to sync.
   156→   156→ */
   157→   157→export function validateSupabaseCredentials(): string | null {
   158→   158→  if (typeof window === "undefined") return null;
   159→   159→  // CLOUD-FIRST: legge solo da env vars (vedi readCredentials).
   160→   160→  const { url: rawUrl, anonKey } = readCredentials();
   161→   161→
   162→   162→  if (!rawUrl && !anonKey) {
   163→   163→    return "Configura NEXT_PUBLIC_SUPABASE_URL e NEXT_PUBLIC_SUPABASE_ANON_KEY nel file .env.local (vedi istruzioni nel file).";
   164→   164→  }
   165→   165→  if (!rawUrl) return "Manca NEXT_PUBLIC_SUPABASE_URL nel file .env.local.";
   166→   166→  if (!anonKey) return "Manca NEXT_PUBLIC_SUPABASE_ANON_KEY nel file .env.local.";
   167→   167→
   168→   168→  // Check URL format
   169→   169→  if (!/^https:\/\/[a-z0-9]+\.supabase\.(co|com|in)/i.test(rawUrl)) {
   170→   170→    return "L'URL non è nel formato corretto. Deve essere https://<project-ref>.supabase.co";
   171→   171→  }
   172→   172→
   173→   173→  // Check anon key length (typical Supabase anon keys are JWT, ~200+ chars)
   174→   174→  if (anonKey.length < 30) {
   175→   175→    return "La anon key sembra troppo corta. Assicurati di aver copiato la 'anon public' key intera da Supabase → Project Settings → API.";
   176→   176→  }
   177→   177→
   178→   178→  return null;
   179→   179→}

   191→   191→/**
   192→   192→ * Translate a low-level fetch/supabase error into an actionable Italian
   193→   193→ * message. The browser surfaces network errors (CORS, DNS, refused
   194→   194→ * connection) as "TypeError: Failed to fetch" which is useless to the
   195→   195→ * user — we replace it with a hint about URL/key correctness.
   196→   196→ */
   197→   197→function humanizeCloudError(err: any): string {
   198→   198→  const raw = err?.message || err?.toString?.() || String(err);
   199→   199→  const lower = raw.toLowerCase();
   200→   200→
   201→   201→  if (lower.includes("failed to fetch") || lower.includes("networkerror")) {
   202→   202→    return (
   203→   203→      "Impossibile contattare Supabase. Cause possibili:\n" +
   204→   204→      "• URL non corretto (deve essere https://<project-ref>.supabase.co, NON l'URL del dashboard)\n" +
   205→   205→      "• Anon key mancante o sbagliata\n" +
   206→   206→      "• Il progetto Supabase è in pausa (riattivalo dal dashboard)"
   207→   207→    );
   208→   208→  }
   209→   209→  if (lower.includes("cors")) {
   210→   210→    return "Blocco CORS: verifica che l'URL sia https://<project-ref>.supabase.co e che la anon key sia corretta.";
   211→   211→  }
   212→   212→  if (lower.includes("jwt") || lower.includes("invalid api key")) {
   213→   213→    return "Anon key non valida. Copia da Supabase → Project Settings → API → 'anon public' (NON la service_role key).";
   214→   214→  }
   215→   215→  if (lower.includes("relation") && lower.includes("does not exist")) {
   216→   216→    return (
   217→   217→      "La tabella 'app_state' non esiste su Supabase. Vai in Supabase → SQL Editor " +
   218→   218→      "→ incolla il contenuto di supabase-schema.sql → esegui."
   219→   219→    );
   220→   220→  }
   221→   221→  if (lower.includes("permission denied") || lower.includes("rls")) {
   222→   222→    return (
   223→   223→      "Permessi insufficienti sulla tabella 'app_state'. Esegui di nuovo " +
   224→   224→      "supabase-schema.sql in Supabase → SQL Editor (contiene i GRANT necessari)."
   225→   225→    );
   226→   226→  }
   227→   227→  return raw;
   228→   228→}

   230→   230→/**
   231→   231→ * Get or create the Supabase client. If the credentials have changed
   232→   232→ * since the last call, a new client is created and the old realtime
   233→   233→ * channel is torn down.
   234→   234→ */
   235→   235→export function getSupabase(): SupabaseClient | null {
   236→   236→  if (typeof window === "undefined") return null;
   237→   237→
   238→   238→  const { url, anonKey } = readCredentials();
   239→   239→  if (!url || !anonKey) {
   240→   240→    // No credentials — reset everything
   241→   241→    if (_supabase) {
   242→   242→      teardownRealtime();
   243→   243→      _supabase = null;
   244→   244→      _currentCredsKey = "";
   245→   245→    }
   246→   246→    // Surface a clear error if URL is present but anon key is missing
   247→   247→    if (url && !anonKey) {
   248→   248→      setStatus(
   249→   249→        "error",
   250→   250→        "Manca la anon key — copiala da Supabase → Project Settings → API → 'anon public'"
   251→   251→      );
   252→   252→    } else if (_status !== "unconfigured" && _status !== "error") {
   253→   253→      setStatus("unconfigured");
   254→   254→    }
   255→   255→    return null;
   256→   256→  }
   257→   257→
   258→   258→  const credsKey = `${url}|${anonKey}`;
   259→   259→  if (_supabase && _currentCredsKey === credsKey) {
   260→   260→    return _supabase;
   261→   261→  }
   262→   262→
   263→   263→  // Credentials changed (or first time) — create a fresh client
   264→   264→  try {
   265→   265→    if (_supabase) {
   266→   266→      teardownRealtime();
   267→   267→      _supabase = null;
   268→   268→    }
   269→   269→    _supabase = createClient(url, anonKey, {
   270→   270→      auth: { persistSession: false },
   271→   271→    });
   272→   272→    _currentCredsKey = credsKey;
   273→   273→    return _supabase;
   274→   274→  } catch (err) {
   275→   275→    console.error("[LabelPulse Cloud] Failed to create client:", err);
   276→   276→    setStatus("error", err instanceof Error ? err.message : String(err));
   277→   277→    return null;
   278→   278→  }
   279→   279→}

   283→   283→const CLOUD_TABLE = "app_state";
   284→   284→// Backward-compat default for users who have not logged in with Google yet.
   285→   285→// Once they do, the row id becomes their lowercased email — so each Google
   286→   286→// account maps to its own isolated row in app_state (multi-profile support).
   287→   287→const DEFAULT_CLOUD_ROW_ID = "default";
   288→   288→
   289→   289→// Module-level holder for the current user's email. Populated by
   290→   290→// useAuthEffect() (see use-auth.ts) — must be set BEFORE any cloud sync
   291→   291→// operation, otherwise we'd fall back to "default" and mix profiles.
   292→   292→let _currentUserEmail: string | null = null;
   293→   293→
   294→   294→/**
   295→   295→ * Set the current user's email (called from useAuthEffect on session change).
   296→   296→ * Pass null on logout. When null, cloud sync falls back to the legacy "default"

   299→   299→export function setCurrentUserEmail(email: string | null): void {
   300→   300→  const normalized = email?.trim().toLowerCase() || null;
   301→   301→  if (_currentUserEmail !== normalized) {
   302→   302→    _currentUserEmail = normalized;
   303→   303→    // Invalidate the cached Supabase client so the next call re-evaluates
   304→   304→    // credentials + row id. Not strictly required (creds don't depend on email)
   305→   305→    // but keeps the state consistent.
   306→   306→  }
   307→   307→}

   374→   374→ * Returns the row id for the GLOBAL artists (read by everyone, written by

   379→   379→  return `${GLOBAL_CLOUD_ROW_ID}_artists`;

   381→   381→
   382→   382→/**
   383→   383→ * SPLIT STRATEGY — what goes where:
   384→   384→ *
   385→   385→ * GLOBAL ROW (id='global', written only by admin):
   386→   386→ *   - labels[*].id, name, genres, rankByGenre, pointsByGenre, trending,
   387→   387→ *     trendingRankByGenre, trendingPointsByGenre, beatportLink, isCustom(false)
   388→   388→ *   - rankingSnapshots
   389→   389→ *   - rankingsUpdatedAt
   390→   390→ *   - (artists saved separately via saveArtistsToCloud → global_artists row)
   391→   391→ *
   392→   392→ * PERSONAL ROW (id=email, written by each user):
   393→   393→ *   - userProfile (artist name, email, bio, links, photo, etc.)
   394→   394→ *   - demos (the user's pitch submissions)
   395→   395→ *   - labels[*].PERSONAL fields: emails, contactInfo, website, demoLink,
   396→   396→ *     socialLink, soundcloudLink, customLinks, notes, status, submissionType,
   397→   397→ *     genre (user-set), createdAt
   398→   398→ *   - labels[*] with isCustom=true (user-added labels not from Beatport)
   399→   399→ *   - gmailAuth, locale, lastSavedAt
   400→   400→ *
   401→   401→ * Merge at load time:
   402→   402→ *   - Start from global labels (Beatport truth)
   403→   403→ *   - For each, look up the personal entry by id (or by name fallback)
   404→   404→ *     and splice in user-edit fields if present
   405→   405→ *   - Append personal-only custom labels
   406→   406→ *   - artists: load from global_artists row
   407→   407→ *   - snapshots: from global
   408→   408→ *   - rankingsUpdatedAt: from global
   409→   409→ *   - userProfile, demos, gmailAuth, locale, lastSavedAt: from personal
   410→   410→ */
   411→   411→const LABEL_BEATPORT_FIELDS = [
   412→   412→  "id", "name",
   413→   413→  "genres", "rankByGenre", "pointsByGenre",
   414→   414→  "trending", "trendingRankByGenre", "trendingPointsByGenre",
   415→   415→  "beatportLink", "isCustom",
   416→   416→] as const;
   417→   417→
   418→   418→const LABEL_PERSONAL_FIELDS = [
   419→   419→  "emails", "contactInfo", "website", "demoLink", "socialLink",
   420→   420→  "soundcloudLink", "customLinks", "notes", "status",
   421→   421→  "submissionType", "genre", "createdAt",
   422→   422→] as const;
   423→   423→
   424→   424→/**
   425→   425→ * Build the GLOBAL payload (what admin pushes to id='global').
   426→   426→ * Only Beatport-sourced labels (isCustom !== true) are included; user's
   427→   427→ * custom labels stay in their personal row.
   428→   428→ */
   429→   429→function buildGlobalPayload(state: any): object {
   430→   430→  const labels = Array.isArray(state?.labels) ? state.labels : [];
   431→   431→  const globalLabels = labels
   432→   432→    .filter((l: any) => l && typeof l === "object" && !l.isCustom)
   433→   433→    .map((l: any) => {
   434→   434→      const out: Record<string, any> = {};
   435→   435→      for (const f of LABEL_BEATPORT_FIELDS) {
   436→   436→        if (l[f] !== undefined) out[f] = l[f];
   437→   437→      }
   438→   438→      return out;
   439→   439→    });
   440→   440→
   441→   441→  return {
   442→   442→    labels: globalLabels,
   443→   443→    rankingSnapshots: Array.isArray(state?.rankingSnapshots) ? state.rankingSnapshots : [],
   444→   444→    rankingsUpdatedAt: state?.rankingsUpdatedAt || null,
   445→   445→    lastGlobalUpdate: new Date().toISOString(),
   446→   446→  };
   447→   447→}
   448→   448→
   449→   449→/**
   450→   450→ * Build the PERSONAL payload (what each user pushes to id=email).
   451→   451→ * Includes profile, demos, and per-label personal fields. Custom labels
   452→   452→ * (isCustom=true) are included in full.
   453→   453→ */
   454→   454→function buildPersonalPayload(state: any): object {
   455→   455→  const labels = Array.isArray(state?.labels) ? state.labels : [];
   456→   456→  const personalLabels = labels
   457→   457→    .filter((l: any) => l && typeof l === "object")
   458→   458→    .map((l: any) => {
   459→   459→      if (l.isCustom) {
   460→   460→        // Custom (user-added) labels go in full — they have no Beatport side.
   461→   461→        return { ...l };
   462→   462→      }
   463→   463→      // Beatport labels: only the personal fields. We include id+name so we
   464→   464→      // can match them against the global labels at merge time.
   465→   465→      const out: Record<string, any> = { id: l.id, name: l.name };
   466→   466→      for (const f of LABEL_PERSONAL_FIELDS) {
   467→   467→        // Only include non-empty personal fields to keep the row small.
   468→   468→        const v = l[f];
   469→   469→        if (v === undefined || v === null) continue;
   470→   470→        if (Array.isArray(v) ? v.length > 0 : (String(v).trim() !== "")) {
   471→   471→          out[f] = v;
   472→   472→        }
   473→   473→      }
   474→   474→      return out;
   475→   475→    })
   476→   476→    .filter((l: any) => {
   477→   477→      // Drop labels that have NO personal data and aren't custom — they'd
   478→   478→      // just bloat the personal row for no reason.
   479→   479→      if (l.isCustom) return true;
   480→   480→      const keys = Object.keys(l).filter((k) => k !== "id" && k !== "name");
   481→   481→      return keys.length > 0;
   482→   482→    });
   483→   483→
   484→   484→  return {
   485→   485→    labels: personalLabels,
   486→   486→    demos: Array.isArray(state?.demos) ? state.demos : [],
   487→   487→    userProfile: state?.userProfile || {},
   488→   488→    gmailAuth: state?.gmailAuth || null,
   489→   489→    locale: state?.locale || null,
   490→   490→    lastSavedAt: state?.lastSavedAt || new Date().toISOString(),
   491→   491→  };
   492→   492→}
   493→   493→
   494→   494→/**
   495→   495→ * Merge a global Beatport label with its personal overlay.
   496→   496→ * Personal fields overwrite global ones (user-edited notes/emails/status win).
   497→   497→ */
   498→   498→function mergeGlobalAndPersonalLabel(global: any, personal: any): any {
   499→   499→  const merged: Record<string, any> = { ...global };
   500→   500→  for (const f of LABEL_PERSONAL_FIELDS) {
   501→   501→    const v = personal?.[f];
   502→   502→    if (v === undefined || v === null) continue;
   503→   503→    if (Array.isArray(v) ? v.length > 0 : (String(v).trim() !== "")) {
   504→   504→      merged[f] = v;
   505→   505→    }
   506→   506→  }
   507→   507→  return merged;
   508→   508→}
   509→   509→
   510→   510→/**
   511→   511→ * Merge the global cloud data with the personal cloud data, producing the
   512→   512→ * final state to hydrate the store with.
   513→   513→ *
   514→   514→ * Strategy:
   515→   515→ * - labels: start from global (Beatport truth), splice in personal fields.
   516→   516→ *           Append personal-only custom labels (isCustom=true).
   517→   517→ * - rankingSnapshots, rankingsUpdatedAt: from global.
   518→   518→ * - userProfile, demos, gmailAuth, locale, lastSavedAt: from personal.
   519→   519→ *
   520→   520→ * Falls back gracefully if either side is missing (e.g., new user with no
   521→   521→ * personal row yet, or admin who hasn't pushed global yet).
   522→   522→ */
   523→   523→export function mergeGlobalAndPersonalCloud(global: any, personal: any): any {
   524→   524→  const globalLabels = Array.isArray(global?.labels) ? global.labels : [];
   525→   525→  const personalLabels = Array.isArray(personal?.labels) ? personal.labels : [];
   526→   526→
   527→   527→  const personalById = new Map<string, any>();
   528→   528→  const personalByName = new Map<string, any>();
   529→   529→  for (const pl of personalLabels) {
   530→   530→    if (!pl || typeof pl !== "object") continue;
   531→   531→    if (pl.id) personalById.set(pl.id, pl);
   532→   532→    const nm = pl.name?.toLowerCase().trim();
   533→   533→    if (nm) personalByName.set(nm, pl);
   534→   534→  }
   535→   535→
   536→   536→  // Track which personal labels got merged into a global one
   537→   537→  const consumedPersonalIds = new Set<string>();
   538→   538→  const consumedPersonalNames = new Set<string>();
   539→   539→
   540→   540→  const mergedLabels: any[] = [];
   541→   541→  for (const gl of globalLabels) {
   542→   542→    if (!gl || typeof gl !== "object") continue;
   543→   543→    const personalMatch = (gl.id && personalById.get(gl.id)) ||
   544→   544→      (gl.name && personalByName.get(gl.name.toLowerCase().trim()));
   545→   545→    if (personalMatch) {
   546→   546→      if (personalMatch.id) consumedPersonalIds.add(personalMatch.id);
   547→   547→      const nm = personalMatch.name?.toLowerCase().trim();
   548→   548→      if (nm) consumedPersonalNames.add(nm);
   549→   549→      mergedLabels.push(mergeGlobalAndPersonalLabel(gl, personalMatch));
   550→   550→    } else {
   551→   551→      // No personal overlay — use global as-is, fill defaults for personal fields
   552→   552→      mergedLabels.push({
   553→   553→        emails: [],
   554→   554→        contactInfo: "",
   555→   555→        website: "",
   556→   556→        demoLink: "",
   557→   557→        socialLink: "",
   558→   558→        soundcloudLink: "",
   559→   559→        customLinks: [],
   560→   560→        notes: "",
   561→   561→        status: "open",
   562→   562→        submissionType: "email",
   563→   563→        genre: "",
   564→   564→        createdAt: new Date().toISOString(),
   565→   565→        ...gl,
   566→   566→      });
   567→   567→    }
   568→   568→  }
   569→   569→
   570→   570→  // Append personal-only labels (custom labels not in global)
   571→   571→  for (const pl of personalLabels) {
   572→   572→    if (!pl || typeof pl !== "object") continue;
   573→   573→    const idConsumed = pl.id && consumedPersonalIds.has(pl.id);
   574→   574→    const nameConsumed = pl.name && consumedPersonalNames.has(pl.name.toLowerCase().trim());
   575→   575→    if (idConsumed || nameConsumed) continue;
   576→   576→    // This is a custom label (or one the user added manually) — include in full
   577→   577→    mergedLabels.push({ ...pl });
   578→   578→  }
   579→   579→
   580→   580→  return {
   581→   581→    labels: mergedLabels,
   582→   582→    demos: personal?.demos || [],
   583→   583→    userProfile: personal?.userProfile || {},
   584→   584→    gmailAuth: personal?.gmailAuth || null,
   585→   585→    locale: personal?.locale || null,
   586→   586→    rankingSnapshots: global?.rankingSnapshots || [],
   587→   587→    rankingsUpdatedAt: global?.rankingsUpdatedAt || null,
   588→   588→    lastSavedAt: personal?.lastSavedAt || new Date().toISOString(),
   589→   589→  };
   590→   590→}
   591→   591→
   592→   592→/**
   593→   593→ * Salva lo stato completo dell'app su Supabase.
   594→   594→ * Usa upsert per creare o aggiornare il record.
   595→   595→ *
   596→   596→ * CLOUD-FIRST SPLIT (2026-06-23): this function now writes to TWO rows:
   597→   597→ *   1. id='global' — Beatport rankings, snapshots, rankingsUpdatedAt
   598→   598→ *      (only admin can write here; non-admin saves are silently skipped
   599→   599→ *      for the global row to prevent users from clobbering shared data)
   600→   600→ *   2. id=email — profile, demos, per-label personal fields
   601→   601→ *      (every user writes here)
   602→   602→ *
   603→   603→ * Artists are saved separately via saveArtistsToCloud (id='global_artists'
   604→   604→ * for admin, id='<email>_artists' for users — but loadArtistsFromCloud
   605→   605→ * always reads from global_artists).
   606→   606→ */
   607→   607→export async function saveStateToCloud(data: object): Promise<boolean> {
   608→   608→  const supabase = getSupabase();
   609→   609→  if (!supabase) return false;
   610→   610→
   611→   611→  setStatus("syncing");
   612→   612→
   613→   613→  const now = new Date().toISOString();
   614→   614→  const personalPayload = buildPersonalPayload(data);
   615→   615→  const adminUser = isCurrentUserAdmin();
   616→   616→
   617→   617→  try {
   618→   618→    // Step 1: always write the personal row (every user has one)
   619→   619→    const personalResult = await supabase.from(CLOUD_TABLE).upsert(
   620→   620→      {
   621→   621→        id: getCloudRowId(),
   622→   622→        data: personalPayload,
   623→   623→        updated_at: now,
   624→   624→      },
   625→   625→      { onConflict: "id" }
   626→   626→    );
   627→   627→    if (personalResult.error) {
   628→   628→      console.error("[LabelPulse Cloud] Personal save error:", personalResult.error.message);
   629→   629→      setStatus("error", humanizeCloudError(personalResult.error));
   630→   630→      return false;
   631→   631→    }
   632→   632→
   633→   633→    // Step 2: if admin, ALSO write the global row (Beatport truth)
   634→   634→    if (adminUser) {
   635→   635→      const globalPayload = buildGlobalPayload(data);
   636→   636→      const globalResult = await supabase.from(CLOUD_TABLE).upsert(
   637→   637→        {
   638→   638→          id: getGlobalCloudRowId(),
   639→   639→          data: globalPayload,
   640→   640→          updated_at: now,
   641→   641→        },
   642→   642→        { onConflict: "id" }
   643→   643→      );
   644→   644→      if (globalResult.error) {
   645→   645→        console.error("[LabelPulse Cloud] Global save error:", globalResult.error.message);
   646→   646→        // Don't fail the whole save — personal data was saved successfully
   647→   647→        console.warn("[LabelPulse Cloud] Continuing despite global save error (personal save succeeded)");
   648→   648→      } else {
   649→   649→        console.info("[LabelPulse Cloud] Admin global row updated: labels=" +
   650→   650→          (globalPayload as any).labels?.length + ", snapshots=" +
   651→   651→          (globalPayload as any).rankingSnapshots?.length);
   652→   652→      }
   653→   653→    }
   654→   654→
   655→   655→    setStatus("synced");
   656→   656→    return true;
   657→   657→  } catch (err) {
   658→   658→    console.error("[LabelPulse Cloud] Save exception:", err);
   659→   659→    setStatus("error", humanizeCloudError(err));
   660→   660→    return false;
   661→   661→  }
   662→   662→}
   663→   663→
   664→   664→/**
   665→   665→ * Carica lo stato completo dell'app da Supabase.
   666→   666→ *
   667→   667→ * CLOUD-FIRST SPLIT (2026-06-23): reads from BOTH rows and merges:
   668→   668→ *   - id='global' → Beatport rankings, snapshots, rankingsUpdatedAt
   669→   669→ *   - id=email → profile, demos, per-label personal fields
   670→   670→ *
   671→   671→ * Returns null if BOTH rows are empty (first-time user with no data).
   672→   672→ */
   673→   673→export async function loadStateFromCloud(): Promise<object | null> {
   674→   674→  const supabase = getSupabase();
   675→   675→  if (!supabase) return null;
   676→   676→
   677→   677→  try {
   678→   678→    // Fetch both rows in parallel — they're independent.
   679→   679→    const [personalRes, globalRes] = await Promise.all([
   680→   680→      supabase
   681→   681→        .from(CLOUD_TABLE)
   682→   682→        .select("data, updated_at")
   683→   683→        .eq("id", getCloudRowId())
   684→   684→        .single(),
   685→   685→      supabase
   686→   686→        .from(CLOUD_TABLE)
   687→   687→        .select("data, updated_at")
   688→   688→        .eq("id", getGlobalCloudRowId())
   689→   689→        .single(),
   690→   690→    ]);
   691→   691→
   692→   692→    // Personal row: error other than "no rows" is a real failure
   693→   693→    if (personalRes.error && personalRes.error.code !== "PGRST116") {
   694→   694→      console.error("[LabelPulse Cloud] Personal load error:", personalRes.error.message);
   695→   695→      setStatus("error", humanizeCloudError(personalRes.error));
   696→   696→      return null;
   697→   697→    }
   698→   698→
   699→   699→    // Global row: error other than "no rows" is a real failure
   700→   700→    if (globalRes.error && globalRes.error.code !== "PGRST116") {
   701→   701→      console.error("[LabelPulse Cloud] Global load error:", globalRes.error.message);
   702→   702→      setStatus("error", humanizeCloudError(globalRes.error));
   703→   703→      return null;
   704→   704→    }
   705→   705→
   706→   706→    const personalData = personalRes.data?.data || null;
   707→   707→    const globalData = globalRes.data?.data || null;
   708→   708→
   709→   709→    if (!personalData && !globalData) {
   710→   710→      // First-time user: no data anywhere
   711→   711→      return null;
   712→   712→    }
   713→   713→
   714→   714→    const merged = mergeGlobalAndPersonalCloud(globalData, personalData);
   715→   715→
   716→   716→    setStatus("synced");
   717→   717→    return merged;
   718→   718→  } catch (err) {
   719→   719→    console.error("[LabelPulse Cloud] Load exception:", err);
   720→   720→    setStatus("error", humanizeCloudError(err));
   721→   721→    return null;
   722→   722→  }
   723→   723→}
   724→   724→
   725→   725→// ==================== REALTIME SUBSCRIPTION ====================
   726→   726→
   727→   727→/**
   728→   728→ * Subscribe to changes on the app_state row so that updates from
   729→   729→ * other devices (PC ↔ phone) are reflected in near-real-time.
   730→   730→ *
   731→   731→ * CLOUD-FIRST SPLIT (2026-06-23): sets up TWO subscriptions:
   732→   732→ *   1. PERSONAL row (id=email) — profile/demos/per-label personal fields
   733→   733→ *   2. GLOBAL row (id='global') — admin's Beatport rankings/snapshots/artists
   734→   734→ *
   735→   735→ * When a remote UPDATE arrives:
   736→   736→ *   1. set _isApplyingRemoteUpdate = true (so we don't bounce the change back)
   737→   737→ *   2. fetch the latest data from BOTH cloud rows
   738→   738→ *   3. merge it into the local store via applyRemoteData
   739→   739→ *   4. set _isApplyingRemoteUpdate = false
   740→   740→ */
   741→   741→export function setupRealtimeSubscription(): () => void {
   742→   742→  const supabase = getSupabase();
   743→   743→  if (!supabase) return () => {};
   744→   744→
   745→   745→  // Tear down any existing channel first
   746→   746→  teardownRealtime();
   747→   747→
   748→   748→  try {
   749→   749→    // ---- PERSONAL channel: own row updates (profile edits from other devices) ----
   750→   750→    const personalChannel = supabase
   751→   751→      .channel("app_state_personal")
   752→   752→      .on(
   753→   753→        "postgres_changes",
   754→   754→        {
   755→   755→          event: "UPDATE",
   756→   756→          schema: "public",
   757→   757→          table: CLOUD_TABLE,
   758→   758→          filter: `id=eq.${getCloudRowId()}`,
   759→   759→        },
   760→   760→        async (payload: any) => {
   761→   761→          if (_isApplyingRemoteUpdate) return;
   762→   762→          const newData = payload?.new?.data;
   763→   763→          if (!newData) return;
   764→   764→          console.log("[LabelPulse Cloud] Realtime PERSONAL update, applying personal overlay...");
   765→   765→          _isApplyingRemoteUpdate = true;
   766→   766→          try {
   767→   767→            // Personal update: only the personal overlay changed. We need to
   768→   768→            // re-merge with the global row to produce the final state.
   769→   769→            // Easiest: just reload both from cloud.
   770→   770→            const fresh = await loadStateFromCloud();
   771→   771→            if (fresh) await applyRemoteData(fresh);
   772→   772→            setStatus("synced");
   773→   773→          } finally {
   774→   774→            _isApplyingRemoteUpdate = false;
   775→   775→          }
   776→   776→        }
   777→   777→      )
   778→   778→      .subscribe((status: string) => {
   779→   779→        if (status === "SUBSCRIBED") {
   780→   780→          console.log("[LabelPulse Cloud] Realtime PERSONAL subscription active");
   781→   781→        } else if (status === "CHANNEL_ERROR" || status === "TIMED_OUT") {
   782→   782→          console.warn("[LabelPulse Cloud] Personal subscription issue:", status);
   783→   783→        }
   784→   784→      });
   785→   785→
   786→   786→    // ---- GLOBAL channel: admin pushed a new scrape — refresh rankings ----
   787→   787→    const globalChannel = supabase
   788→   788→      .channel("app_state_global")
   789→   789→      .on(
   790→   790→        "postgres_changes",
   791→   791→        {
   792→   792→          event: "UPDATE",
   793→   793→          schema: "public",
   794→   794→          table: CLOUD_TABLE,
   795→   795→          filter: `id=eq.${getGlobalCloudRowId()}`,
   796→   796→        },
   797→   797→        async (payload: any) => {
   798→   798→          if (_isApplyingRemoteUpdate) return;
   799→   799→          const newData = payload?.new?.data;
   800→   800→          if (!newData) return;
   801→   801→          console.log("[LabelPulse Cloud] Realtime GLOBAL update (admin pushed new rankings), refreshing...");
   802→   802→          _isApplyingRemoteUpdate = true;
   803→   803→          try {
   804→   804→            // Global update: admin pushed new Beatport data. Reload both rows
   805→   805→            // and re-merge so the user sees the new rankings immediately.
   806→   806→            const fresh = await loadStateFromCloud();
   807→   807→            if (fresh) await applyRemoteData(fresh);
   808→   808→            setStatus("synced");
   809→   809→          } finally {
   810→   810→            _isApplyingRemoteUpdate = false;
   811→   811→          }
   812→   812→        }
   813→   813→      )
   814→   814→      .subscribe((status: string) => {
   815→   815→        if (status === "SUBSCRIBED") {
   816→   816→          console.log("[LabelPulse Cloud] Realtime GLOBAL subscription active");
   817→   817→        } else if (status === "CHANNEL_ERROR" || status === "TIMED_OUT") {
   818→   818→          console.warn("[LabelPulse Cloud] Global subscription issue:", status);
   819→   819→        }
   820→   820→      });
   821→   821→
   822→   822→    // Track both channels for teardown
   823→   823→    _realtimeChannel = personalChannel;
   824→   824→    _realtimeGlobalChannel = globalChannel;
   825→   825→    return teardownRealtime;
   826→   826→  } catch (err) {
   827→   827→    console.warn("[LabelPulse Cloud] Realtime setup failed:", err);
   828→   828→    return () => {};
   829→   829→  }
   830→   830→}
   831→   831→
   832→   832→function teardownRealtime() {
   833→   833→  const supabase = _supabase;
   834→   834→  if (_realtimeChannel) {
   835→   835→    try {
   836→   836→      if (supabase) supabase.removeChannel(_realtimeChannel);
   837→   837→    } catch {
   838→   838→      // ignore
   839→   839→    }
   840→   840→    _realtimeChannel = null;
   841→   841→  }
   842→   842→  if (_realtimeGlobalChannel) {
   843→   843→    try {
   844→   844→      if (supabase) supabase.removeChannel(_realtimeGlobalChannel);
   845→   845→    } catch {
   846→   846→      // ignore
   847→   847→    }
   848→   848→    _realtimeGlobalChannel = null;
   849→   849→  }
   850→   850→}
   851→   851→
   852→   852→/**
   853→   853→ * Returns true if the current saveStateToCloud call should be skipped
   854→   854→ * because we're applying a remote update (avoids loop).
   855→   855→ */
   856→   856→export function isApplyingRemoteUpdate(): boolean {
   857→   857→  return _isApplyingRemoteUpdate;
   858→   858→}
   859→   859→
   860→   860→/**
   861→   861→ * Apply remote data to the local store. Uses the same merge logic
   862→   862→ * as loadFromCloud().
   863→   863→ *
   864→   864→ * CRITICAL SAFETY RULE: same as mergeCloudData in store.ts — never
   865→   865→ * overwrite local non-empty arrays with empty cloud arrays. This is
   866→   866→ * the realtime-update path, and we want the same protection here.
   867→   867→ */
   868→   868→async function applyRemoteData(cloudData: any): Promise<void> {
   869→   869→  // Lazy require to avoid circular import
   870→   870→  const store = useAppStore.getState();
   871→   871→  const localLastSavedAt = store.lastSavedAt
   872→   872→    ? new Date(store.lastSavedAt).getTime()
   873→   873→    : 0;
   874→   874→  const cloudLastSavedAt = cloudData?.lastSavedAt
   875→   875→    ? new Date(cloudData.lastSavedAt).getTime()
   876→   876→    : 0;
   877→   877→
   878→   878→  // ⚠️ CRITICAL: Content-aware merge decision (same as loadFromCloud).
   879→   879→  // The old code did `if (cloudLastSavedAt <= localLastSavedAt) return;`
   880→   880→  // which would silently DROP realtime updates if the local timestamp was
   881→   881→  // newer or equal — even if the cloud had profile data local didn't have.
   882→   882→  //
   883→   883→  // This was the realtime-update variant of the same bug that wiped user
   884→   884→  // data on re-login. Always apply if cloud brings new profile data, even
   885→   885→  // if its timestamp looks "older".
   886→   886→  const cloudProfileHasData =
   887→   887→    !!cloudData?.userProfile?.artistName ||
   888→   888→    !!cloudData?.userProfile?.bio ||
   889→   889→    !!cloudData?.userProfile?.email ||
   890→   890→    !!cloudData?.userProfile?.scLink ||
   891→   891→    !!cloudData?.userProfile?.photoUrl ||
   892→   892→    (Array.isArray(cloudData?.userProfile?.links) && cloudData.userProfile.links.length > 0);
   893→   893→  const localProfileHasData =
   894→   894→    !!store.userProfile?.artistName ||
   895→   895→    !!store.userProfile?.bio ||
   896→   896→    !!store.userProfile?.email ||
   897→   897→    !!store.userProfile?.scLink ||
   898→   898→    !!store.userProfile?.photoUrl ||
   899→   899→    (Array.isArray(store.userProfile?.links) && store.userProfile.links.length > 0);
   900→   900→
   901→   901→  const cloudBringsNewProfile = cloudProfileHasData && !localProfileHasData;
   902→   902→  const cloudHasLabels = Array.isArray(cloudData?.labels) && cloudData.labels.length > 0;

  1198→  1198→/**
  1199→  1199→ * Merge two artist arrays by id (union). When the same id appears in both,
  1200→  1200→ * keep the version with the most-recent `scrapedAt` (or the local one on tie).
  1201→  1201→ * This is the artist equivalent of mergeCloudData: safe, additive, never
  1202→  1202→ * removes data.
  1203→  1203→ *
  1204→  1204→ * Used by:
  1205→  1205→ *  - loadArtistsOnBoot (when both IDB and cloud have artists)
  1206→  1206→ *  - explicitMergeArtistsCloud (user-triggered from CloudRecovery panel)
  1207→  1207→ */
  1208→  1208→export function mergeArtistsArrays(local: any[], cloud: any[]): any[] {
  1209→  1209→  const a = Array.isArray(local) ? local : [];
  1210→  1210→  const b = Array.isArray(cloud) ? cloud : [];
  1211→  1211→  const byId = new Map<string, any>();
  1212→  1212→  const byName = new Map<string, any>();
  1213→  1213→
  1214→  1214→  // Insert cloud first (so local can override on conflict below)
  1215→  1215→  for (const c of b) {
  1216→  1216→    if (!c || typeof c !== "object") continue;
  1217→  1217→    const id = c.id || c.artistId || `_${c.name?.toLowerCase?.() || ""}`;
  1218→  1218→    byId.set(id, { ...c });
  1219→  1219→    const nm = (c.name || c.artistName || "").toString().toLowerCase().trim();
  1220→  1220→    if (nm) byName.set(nm, { ...c });
  1221→  1221→  }
  1222→  1222→
  1223→  1223→  for (const l of a) {
  1224→  1224→    if (!l || typeof l !== "object") continue;
  1225→  1225→    const id = l.id || l.artistId || `_${(l.name || l.artistName || "").toLowerCase()}`;
  1226→  1226→    const nm = (l.name || l.artistName || "").toString().toLowerCase().trim();
  1227→  1227→    const existing = byId.get(id) || (nm ? byName.get(nm) : undefined);
  1228→  1228→
  1229→  1229→    if (!existing) {
  1230→  1230→      byId.set(id, { ...l });
  1231→  1231→      if (nm) byName.set(nm, { ...l });
  1232→  1232→      continue;
  1233→  1233→    }
  1234→  1234→
  1235→  1235→    // Both have this artist — keep the one with most tracks / most recent scrape
  1236→  1236→    const exTracks = Array.isArray(existing.tracksByGenre)
  1237→  1237→      ? Object.values(existing.tracksByGenre).flat().length
  1238→  1238→      : (Array.isArray(existing.tracks) ? existing.tracks.length : 0);
  1239→  1239→    const loTracks = Array.isArray(l.tracksByGenre)
  1240→  1240→      ? Object.values(l.tracksByGenre).flat().length
  1241→  1241→      : (Array.isArray(l.tracks) ? l.tracks.length : 0);
  1242→  1242→
  1243→  1243→    const exTs = new Date(existing.scrapedAt || existing.updatedAt || 0).getTime();
  1244→  1244→    const loTs = new Date(l.scrapedAt || l.updatedAt || 0).getTime();
  1245→  1245→
  1246→  1246→    // Local wins if it has more tracks OR more recent scrape OR has bio when existing doesn't
  1247→  1247→    const localHasBio = !!((l as any).bio || (l as any).bioSummary);
  1248→  1248→    const exHasBio = !!((existing as any).bio || (existing as any).bioSummary);
  1249→  1249→
  1250→  1250→    let pickLocal = false;
  1251→  1251→    if (loTracks > exTracks) pickLocal = true;
  1252→  1252→    else if (loTracks === exTracks && loTs > exTs) pickLocal = true;
  1253→  1253→    else if (localHasBio && !exHasBio) pickLocal = true;
  1254→  1254→
  1255→  1255→    if (pickLocal) {
  1256→  1256→      const merged = { ...existing, ...l };
  1257→  1257→      // Preserve any extra fields from cloud that local doesn't have
  1258→  1258→      for (const k of Object.keys(existing)) {
  1259→  1259→        if (!(k in l) && merged[k] === undefined) merged[k] = existing[k];
  1260→  1260→      }
  1261→  1261→      byId.set(id, merged);
  1262→  1262→      if (nm) byName.set(nm, merged);
  1263→  1263→    } else {
  1264→  1264→      // Keep cloud's version but ensure local-only fields are preserved
  1265→  1265→      const merged = { ...l, ...existing };
  1266→  1266→      for (const k of Object.keys(l)) {
  1267→  1267→        if (!(k in existing) && merged[k] === undefined) merged[k] = l[k];
  1268→  1268→      }
  1269→  1269→      byId.set(id, merged);
  1270→  1270→      if (nm) byName.set(nm, merged);
  1271→  1271→    }
  1272→  1272→  }
  1273→  1273→
  1274→  1274→  return Array.from(byId.values());
  1275→  1275→}

  1277→  1277→/**
  1278→  1278→ * Force-push the current local artists to cloud (REPLACING cloud's artists).
  1279→  1279→ * Used by:

  1281→  1281→ *  - loadArtistsOnBoot when local has artists and cloud is empty/has fewer
